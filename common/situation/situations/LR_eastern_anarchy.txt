# The Eastern Anarchy (gen_eastern_anarchy.py): the fractured East of the 6066 starts.
# Started at game start (LR_eastern_anarchy_on_actions.txt) when no one holds an Eastern Roman empire independently;
# ends when someone does (on_yearly).
LR_eastern_anarchy = {
	is_unique = yes

	gui_window_name = "window_situation"
	gui_participation_window_name = "window_situation_participation"
	illustration = "gfx/interface/illustrations/decisions/decision_byz_throne.dds"
	situation_group_type = major
	map_mode = sub_regions

	sub_regions = {
		byzantine_thunderdome = {
			map_color = rgb { 109 18 117 }
			geographical_regions = { custom_eastern_roman_imperial_core }
		}
		byzantine_border = {
			map_color = rgb { 255 200 100 }
			geographical_regions = { custom_eastern_imperial_borderlands }
		}
		roman_italy = {
			map_color = rgb { 255 55 80 }
			geographical_regions = { custom_western_roman_imperial_core }
		}
	}

	# Ends once an Eastern Roman empire is held independently again
	on_yearly = {
		if = {
			limit = { LR_eastern_roman_empire_restored_trigger = yes }
			title:e_eastern_roman_empire.holder ?= {
				if = {
					limit = { is_independent_ruler = yes }
					trigger_event = LR_eastern_anarchy_situation.0014
				}
			}
			title:h_eastern_roman_empire.holder ?= {
				if = {
					limit = { is_independent_ruler = yes }
					trigger_event = LR_eastern_anarchy_situation.0014
				}
			}
			end_situation = yes
		}
	}

	participant_groups = {
		byzantine_court = {
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_pro.dds"
			require_realm_in_sub_region = yes
			is_character_valid = {
				is_landed = yes
				has_title = title:e_byzantium
			}
			on_join = { trigger_event = LR_eastern_anarchy_situation.0013 }
			map_color = hsv { 0.08 0.85 0.75 }
		}
		byzantine_rulers = {
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_emperor.dds"
			require_capital_in_sub_region = yes
			is_character_valid = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_independent_ruler = yes
				capital_province = { geographical_region = custom_eastern_roman_imperial_core }
				NOT = { has_title = title:e_byzantium } # The Byzantine Court
			}
			on_join = { trigger_event = LR_eastern_anarchy_situation.0013 }
			map_color = hsv { 0.43 0.88 0.19 }
		}
		byzantine_marches = {
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_independent.dds"
			require_realm_in_sub_region = yes
			is_character_valid = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_independent_ruler = yes
				capital_province = { geographical_region = custom_eastern_imperial_borderlands }
				NOT = { has_title = title:e_byzantium } # The Byzantine Court
			}
			on_join = { trigger_event = LR_eastern_anarchy_situation.0013 }
			map_color = hsv { 0.8 0.75 0.83 }
		}
		latin_interlopers = {
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_group_expand.dds"
			require_realm_in_sub_region = yes
			is_character_valid = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_independent_ruler = yes
				capital_province = { geographical_region = custom_western_roman_imperial_core }
				NOT = { has_title = title:e_byzantium } # The Byzantine Court
			}
			on_join = { trigger_event = LR_eastern_anarchy_situation.0013 }
			map_color = hsv { 0.00 0.93 0.57 }
		}
	}

	start_phase = situation_phase_eastern_anarchy_peace

	phases = {
		situation_phase_eastern_anarchy_peace = { # Uneasy Peace
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_phase_advancement.dds"
			illustration = "gfx/interface/illustrations/event_story/tgp_dynastic_cycle_stability.dds"

			on_start = {
				scope:situation_sub_region = {
					every_situation_sub_region_participant = {
						trigger_event = LR_eastern_anarchy_situation.0010
					}
				}
			}

			max_duration = { years = 5 }
			max_duration_next_phase = weighted_non_takeover # The engine's spelling (its documentation writes takover)
			future_phases = {
				situation_phase_eastern_anarchy_tension = {
					takeover_type = none
					weight = 3
				}
				situation_phase_eastern_anarchy_war = {
					takeover_type = none
					weight = 1
				}
			}

			modifier_sets = {
				LR_eastern_anarchy_peace_effects = {
					icon = "gfx/interface/icons/situations/situation_modifier_subject.dds"
					byzantine_court = {
						character_modifier = {
							tax_mult = 0.1
							monthly_prestige_gain_mult = 0.1
						}
					}
					byzantine_rulers = {
						character_modifier = {
							tax_mult = 0.05
							monthly_county_control_growth_factor = -0.1
						}
					}
					byzantine_marches = {
						character_modifier = {
							tax_mult = 0.05
						}
					}
				}
			}
		}
		situation_phase_eastern_anarchy_tension = { # Usurpations
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_phase_instability.dds"
			illustration = "gfx/interface/illustrations/event_story/tgp_dynastic_cycle_corruption.dds"

			on_start = {
				scope:situation_sub_region = {
					every_situation_sub_region_participant = {
						trigger_event = LR_eastern_anarchy_situation.0011
					}
				}
			}

			max_duration = { years = 3 }
			max_duration_next_phase = weighted_non_takeover # The engine's spelling (its documentation writes takover)
			future_phases = {
				situation_phase_eastern_anarchy_war = {
					takeover_type = none
					weight = 2
				}
				situation_phase_eastern_anarchy_peace = {
					takeover_type = none
					weight = 1
				}
			}

			modifier_sets = {
				LR_eastern_anarchy_tension_effects = {
					icon = "gfx/interface/icons/situations/situation_modifier_character.dds"
					byzantine_court = {
						character_modifier = {
							vassal_opinion = -15
							monthly_prestige_gain_mult = -0.1
						}
					}
					byzantine_rulers = {
						character_modifier = {
							vassal_opinion = -10
							levy_size = 0.1
							title_creation_cost_mult = -0.25
						}
					}
					byzantine_marches = {
						character_modifier = {
							vassal_opinion = -5
							levy_size = 0.1
						}
					}
					latin_interlopers = {
						character_modifier = {
							monthly_prestige_gain_mult = 0.1
						}
					}
				}
			}
		}
		situation_phase_eastern_anarchy_war = { # Civil War
			icon = "gfx/interface/icons/dynastic_cycle/dynastic_cycle_phase_chaos.dds"
			illustration = "gfx/interface/illustrations/decisions/decision_siege_warfare.dds"

			on_start = {
				scope:situation_sub_region = {
					every_situation_sub_region_participant = {
						trigger_event = LR_eastern_anarchy_situation.0012
					}
				}
			}

			max_duration = { years = 3 }
			max_duration_next_phase = weighted_non_takeover # The engine's spelling (its documentation writes takover)
			future_phases = {
				situation_phase_eastern_anarchy_tension = {
					takeover_type = none
					weight = 2
				}
				situation_phase_eastern_anarchy_peace = {
					takeover_type = none
					weight = 1
				}
			}

			modifier_sets = {
				LR_eastern_anarchy_war_effects = {
					icon = "gfx/interface/icons/situations/situation_modifier_military.dds"
					byzantine_court = {
						character_modifier = {
							levy_size = 0.15
							defender_advantage = 5
						}
					}
					byzantine_rulers = {
						character_modifier = {
							levy_size = 0.2
							men_at_arms_maintenance = -0.15
						}
						county_modifier = {
							development_growth_factor = -0.2
						}
					}
					byzantine_marches = {
						character_modifier = {
							levy_size = 0.1
							defender_advantage = 3
						}
						county_modifier = {
							development_growth_factor = -0.1
						}
					}
					latin_interlopers = {
						character_modifier = {
							monthly_prestige_gain_mult = 0.15
						}
					}
				}
			}
		}
	}
}
