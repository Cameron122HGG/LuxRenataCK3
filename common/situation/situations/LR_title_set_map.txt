# Title Set Map - a world-wide hidden situation used only as a map mode, built like the Trait Map (tmm_trait_map.txt).
# The 'sub_regions' map mode paints eight sub-regions (Brythonic and Galatian added 2026-11-06), filled county by county by LR_tsm_refresh_effect:
# one per title set (LR_title_set_rules.md) and one for land ruled by an outsider under Foreign Administration.
# Two views (situation variable LR_tsm_view, switched in the window):
#   flag:land - each county takes the colour of the set its de jure duchy, kingdom or empire belongs to.
#   flag:rule - a county ruled under a held set title (the title's holder is the county holder or one of its lieges)
#               takes that set's colour if the holder is a native, and the Foreign Administration colour if any such
#               holder is an outsider.
# Set titles are tagged with the title variable LR_tsm_set (generated by gen_title_set_map.py), so the refresh reads one
# variable per tier instead of the long set triggers.
# Refreshed at the start, on a view change, the month after a set title changes hands (LR_tsm_dirty) and yearly.
# Third view, flag:preview: paints one set's whole layout at one tier (LR_tsm_preview_set, LR_tsm_preview_tier) in the
# palette sub-regions LR_tsm_p1..p8, neighbouring titles in different colours, even where the set holds no de jure
# land in this game. Includes the Warlord set (the 1066 de jure map). Generated by tools/gen_title_set_preview.py.
# No modifier: no effect on the game.

	map_mode = sub_regions

	##################################################
	# On Actions
	##################################################
	on_start = {
		set_variable = { name = LR_tsm_layout value = 1 }
		set_variable = { name = LR_tsm_view value = flag:land }
		LR_tsm_tag_set_titles_effect = yes
		LR_tsm_refresh_effect = yes
	}

	on_monthly = {
		if = {
			limit = { has_variable = LR_tsm_layout }
			change_variable = { name = LR_tsm_months add = 1 }
			if = {
				limit = {
					OR = {
						has_variable = LR_tsm_dirty
						var:LR_tsm_months >= 12
					}
					# A preview is a fixed layout: no repaint until the view changes
					NOT = { LR_tsm_view_is_preview_trigger = yes }
				}
				if = {
					limit = { var:LR_tsm_months >= 12 }
					LR_tsm_tag_set_titles_effect = yes
				}
				LR_tsm_refresh_effect = yes
			}
		}
		else = {
			end_situation = yes
		}
	}

	##################################################
	# Groups (participant list of the window)
	##################################################
	# First valid group wins, so an outsider who is also a native of another set is listed as an outsider
	participant_groups = {

			}
		}
	}

	phases = {
