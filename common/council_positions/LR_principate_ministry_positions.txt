# LR 2026-11-19 (user): the Princeps' own ministers, early-empire offices (separate from the Dominate's late-empire ministries in
# LR_roman_ministry_positions.txt). Extra council seats for a Princeps; each minister gains merit on the cursus honorum
# (LR_principate_minister_modifier). Reform into the Dominate / Restore the Principate carry them over (LR_principate_effects.txt).

LR_praefectus_praetorio = {
	skill = martial

	name = LR_praefectus_praetorio

	valid_position = {
		LR_is_princeps_trigger = yes
		highest_held_title_tier >= tier_empire
	}

	tooltip = LR_praefectus_praetorio_desc

	modifier = {
		name = LR_praefectus_praetorio_modifier
		men_at_arms_maintenance = -0.05
		knight_effectiveness_mult = 0.05
		monthly_martial_lifestyle_xp_gain_mult = 0.05
		scale = council_scaled_by_liege_tier
	}

	modifier = {
		name = LR_praefectus_praetorio_modifier
		monthly_prestige = 1
		scale = council_scaled_monthly_income
	}

	valid_character = {
		exists = root.liege_or_court_owner
		LR_can_be_principate_minister_trigger = { COURT_OWNER = root.liege_or_court_owner }
	}

	on_get_position = {
		got_council_position_effect = yes
		if = {
			limit = { NOT = { has_character_modifier = LR_principate_minister_modifier } }
			add_character_modifier = LR_principate_minister_modifier
		}
	}

	on_fired_from_position = {
		if = {
			limit = { has_character_modifier = LR_principate_minister_modifier }
			remove_character_modifier = LR_principate_minister_modifier
		}
		fired_from_council_position_effect = yes
	}

	portrait_animation = happy_teacher

	barbershop_data = {
		position = { 1268.0 172.0 }
		click_to_front = no
	}
}
