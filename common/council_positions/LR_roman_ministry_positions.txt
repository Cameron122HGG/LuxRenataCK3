# Roman imperial ministries, adapted from the celestial ministers (01_ministry_positions.txt).
# Extra council seats for a Roman emperor. Each seat holds a landless ministry title for its half of the empire
# (k_lr_ministry_east_* / k_lr_ministry_west_*), see LR_roman_office_effects.txt.

LR_magister_officiorum = {
	skill = diplomacy

	name = LR_magister_officiorum

	valid_position = {
		government_has_flag = government_is_roman_imperial
		highest_held_title_tier >= tier_empire
	}

	tooltip = LR_magister_officiorum_desc

	modifier = {
		name = LR_magister_officiorum_modifier
		fellow_vassal_opinion = 5
		monthly_diplomacy_lifestyle_xp_gain_mult = 0.05
		scale = council_scaled_by_liege_tier
	}

	modifier = {
		name = LR_magister_officiorum_modifier
		monthly_prestige = 1
		scale = council_scaled_monthly_income
	}

	modifier = {
		name = LR_magister_officiorum_modifier
		monthly_influence = 1
		enemy_political_scheme_success_chance_max_add = -5
		scale = council_scaled_admin_value
	}

	council_owner_modifier = {
		name = learn_on_the_job_modifier
		diplomacy = 1
		scale = chancellor_learn_on_the_job_scale
	}

	council_owner_modifier = {
		name = reliable_house_modifier
		diplomacy = 1
		scale = chancellor_reliable_house_scale
	}

	valid_character = {
		exists = root.liege_or_court_owner
		LR_can_be_roman_minister_trigger = { COURT_OWNER = root.liege_or_court_owner }
	}

	on_get_position = {
		got_council_position_effect = yes
		LR_roman_minister_seat_gained_effect = { EAST = k_lr_ministry_east_officiorum WEST = k_lr_ministry_west_officiorum LAW = LR_roman_minister_appointment_succession_law }
	}

	on_fired_from_position = {
		LR_roman_minister_seat_lost_effect = yes
		fired_from_council_position_effect = yes
	}

	portrait_animation = happy_teacher

	barbershop_data = {
		position = { 918.0 172.0 }
		click_to_front = no
	}
}
