# Roman governor appointments, adapted from meritocratic_governor.txt.
# Family ties count for less than in the Confucian version; citizenship and imperial service count instead.

LR_roman_civic_governor = {
	allowed_candidate_tier = lower_or_equal
	cooldown = yes

	level = merit

	candidate_score = {
		value = {
			add = appointment_score_base
			add = appointment_score_civic
			add = appointment_score_governor
			add = appointment_score_non_military

			# Same dynasty as holder bonus
			if = {
				limit = {
					root.house ?= scope:title.holder.house
				}
				add = {
					value = 25
					desc = closely_related
				}
			}

			if = {
				limit = {
					root.dynasty.dynast ?= {
						has_domicile = yes
					}
					scope:title = {
						is_de_jure_liege_or_above_target = root.dynasty.dynast.domicile.domicile_location.county
					}
				}
				add = {
					value = 25
					desc = ancestral_home
				}
			}

			# Age - With age comes experience. We provide a rather small bonus for the elderly, to exist as a potential tie-breaker.
			add = {
				value = age
				divide = 2
				round = yes
				desc = appointment_investment_cap_age # Reused loc
			}
			if = {
				limit = {
					tgp_is_above_retirement_age_trigger = { REALM_OWNER = root }
				}
				add = {
					value = retirement_age_current_value
					subtract = age
					multiply = 3
					desc = is_above_retirement_age_desc
				}
			}

			# Roman citizenship: shares the emperor's culture
			if = {
				limit = {
					exists = root.top_liege
					root.top_liege != root
					culture = root.top_liege.culture
				}
				add = {
					value = 25
					desc = LR_roman_appointment_citizen
				}
			}

			# Already serves on the emperor's consistory (council)
			if = {
				limit = {
					exists = root.top_liege
					is_councillor_of = root.top_liege
				}
				add = {
					value = 25
					desc = LR_roman_appointment_consistory
				}
			}

			add = appointment_score_final_factors
		}
	}
}
