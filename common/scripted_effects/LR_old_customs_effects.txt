# Character (the ruler): take the decision's effect
LR_restore_own_customs_effect = {
	save_scope_as = LR_customs_ruler
	LR_store_old_customs_government_effect = yes
	# the subjects' reaction, judged by the OLD customs (before the switch)
	every_vassal = {
		limit = { LR_old_customs_aggrieved_trigger = yes }
		add_to_list = LR_old_customs_aggrieved
	}
	every_vassal = {
		limit = {
			NOT = { is_in_list = LR_old_customs_aggrieved }
			LR_old_customs_pleased_trigger = yes
		}
		add_to_list = LR_old_customs_pleased
	}
	if = {
		limit = { LR_old_customs_revolt_trigger = yes }
		save_scope_value_as = { name = LR_old_customs_revolt value = yes }
	}
	# the switch
	LR_apply_culture_government_group_effect = yes
	add_character_flag = { flag = LR_restored_own_customs years = 25 }
	every_in_list = {
		list = LR_old_customs_aggrieved
		add_opinion = { target = scope:LR_customs_ruler modifier = LR_abandoned_old_customs_opinion }
	}
	every_in_list = {
		list = LR_old_customs_pleased
		add_opinion = { target = scope:LR_customs_ruler modifier = LR_restored_own_customs_opinion }
	}
	# the revolt
	if = {
		limit = { exists = scope:LR_old_customs_revolt }
		ordered_in_list = {
			list = LR_old_customs_aggrieved
			order_by = max_military_strength
			save_scope_as = LR_old_customs_leader
		}
		scope:LR_old_customs_leader = {
			start_war = {
				casus_belli = LR_old_customs_war
				target = scope:LR_customs_ruler
			}
			random_character_war = {
				limit = { using_cb = LR_old_customs_war }
				save_scope_as = LR_old_customs_war_scope
			}
		}
		scope:LR_old_customs_war_scope ?= {
			every_in_list = {
				list = LR_old_customs_aggrieved
				limit = { NOT = { this = scope:LR_old_customs_leader } }
				save_temporary_scope_as = LR_old_customs_joiner
				scope:LR_old_customs_war_scope = { add_attacker = scope:LR_old_customs_joiner }
			}
		}
	}
}
