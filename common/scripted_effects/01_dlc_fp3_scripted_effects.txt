	scope:new_title = {
		generate_coa = yes
		set_variable = {
			name = temporary_title
			value = yes
		}
	}
	#Declare the war.
	scope:rebel_leader = {
		start_war = {
			cb = fp3_zanj_rebellion_war

	scope:rebel_leader = {
		spawn_army = {
			levies = zanj_rebellion_strength_score_levy
			men_at_arms = {
				type = pikemen_unit

				type = bowmen
				stacks = zanj_rebellion_strength_score_bowmen
			}

			}
			location = province:6022

		}
	}
}
