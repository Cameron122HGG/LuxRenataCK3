# Scope: a title. If it has a holder but no de jure land left, its holder gets the title of the same tier over its
# capital county (if unheld and theirs to use) and the old title is dissolved. $GRANT$ = yes lets a sitting king
# take the regnum (missatica set, a sub-king); no demotes him (Merovingian set), with a grudge against his liege.
LR_frankish_dissolve_if_empty_effect = {
	if = {
		limit = {
			exists = holder
			NOT = { any_de_jure_county = { always = yes } }
		}
		save_temporary_scope_as = LR_frankish_old
		holder = { save_temporary_scope_as = LR_frankish_old_holder }
		title_capital_county ?= {
			if = {
				limit = { scope:LR_frankish_old.tier = tier_duchy }
				duchy ?= { save_temporary_scope_as = LR_frankish_new }
			}
			else = { kingdom ?= { save_temporary_scope_as = LR_frankish_new } }
		}
		if = {
			limit = {
				exists = scope:LR_frankish_new
				scope:LR_frankish_new = {
					NOT = { this = scope:LR_frankish_old }
					NOT = { exists = holder }
				}
				OR = {
					scope:LR_frankish_old.tier = tier_duchy
					always = $GRANT$
				}
				scope:LR_frankish_old_holder = { LR_may_use_title_set_trigger = { TITLE = scope:LR_frankish_new } }
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = LR_frankish_change
				add_claim_on_loss = no
			}
			scope:LR_frankish_new = {
				change_title_holder = {
					holder = scope:LR_frankish_old_holder
					change = scope:LR_frankish_change
				}
			}
			resolve_title_and_vassal_change = scope:LR_frankish_change
		}
		else_if = {
			limit = { scope:LR_frankish_old.tier = tier_kingdom }
			scope:LR_frankish_old_holder = {
				if = {
					limit = { NOT = { top_liege = this } }
					add_opinion = {
						target = top_liege
						modifier = LR_frankish_crown_taken_opinion
					}
				}
			}
		}
		scope:LR_frankish_old_holder = { destroy_title = scope:LR_frankish_old }
		clear_saved_scope = LR_frankish_new
	}
}

	if = {
		limit = { any_held_title = { LR_frankish_march_title_trigger = yes } }
		if = {
			limit = { NOT = { has_character_modifier = LR_frankish_march_modifier } }
			add_character_modifier = LR_frankish_march_modifier
		}
	}
	else_if = {
		limit = { has_character_modifier = LR_frankish_march_modifier }
		remove_character_modifier = LR_frankish_march_modifier
	}
}
