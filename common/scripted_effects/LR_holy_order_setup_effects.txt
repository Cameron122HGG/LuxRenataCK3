# LR: historical holy orders at game start (fix_holy_orders.py). The vanilla setup assumes vanilla's 1066 holders;
# these let a founding or lease use a barony the engine accepts, and skip it cleanly when none is left.

# Barony scope (saved as scope:LR_ho_capital): make it, or another barony of its county, a valid order capital
LR_holy_order_find_capital_effect = {
	scope:LR_ho_capital = { LR_holy_order_claim_barony_effect = yes }
	if = {
		limit = { NOT = { scope:LR_ho_capital = { LR_holy_order_capital_valid_trigger = yes } } }
		scope:LR_ho_capital.county ?= {
			random_in_de_jure_hierarchy = {
				limit = {
					tier = tier_barony
					is_capital_barony = no
					title_province = { has_holding = yes }
				}
				save_scope_as = LR_ho_capital
				LR_holy_order_claim_barony_effect = yes
			}
		}
	}
}

# Barony scope: a non-capital barony held by someone other than its county's holder passes to the county's holder
LR_holy_order_claim_barony_effect = {
	if = {
		limit = {
			is_capital_barony = no
			title_province = { has_holding = yes }
			exists = county.holder
			NOT = { holder ?= county.holder }
		}
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = LR_ho_change
			add_claim_on_loss = no
		}
		change_title_holder = {
			holder = county.holder
			change = scope:LR_ho_change
		}
		resolve_title_and_vassal_change = scope:LR_ho_change
	}
}

# Barony scope: lease it to the holy order $ORDER$, or another barony of the same county if it can't be leased
LR_lease_to_holy_order_effect = {
	if = {
		limit = { exists = $ORDER$ }
		save_scope_as = LR_ho_capital
		if = {
			limit = { NOT = { LR_holy_order_capital_valid_trigger = yes } }
			LR_holy_order_find_capital_effect = yes
		}
		if = {
			limit = { scope:LR_ho_capital = { LR_holy_order_capital_valid_trigger = yes } }
			scope:LR_ho_capital = { lease_out_to_holy_order = $ORDER$ }
		}
	}
}

# The ruler of the Eastern Roman Empire for the Greek monasteries: e_byzantium, else the restored empire
LR_save_eastern_monastic_founder_effect = {
	if = {
		limit = { exists = title:e_byzantium.holder }
		title:e_byzantium.holder = { save_scope_as = LR_eastern_founder }
	}
	else_if = {
		limit = { exists = title:h_eastern_roman_empire.holder }
		title:h_eastern_roman_empire.holder = { save_scope_as = LR_eastern_founder }
	}
	else_if = {
		limit = { exists = title:e_eastern_roman_empire.holder }
		title:e_eastern_roman_empire.holder = { save_scope_as = LR_eastern_founder }
	}
}
