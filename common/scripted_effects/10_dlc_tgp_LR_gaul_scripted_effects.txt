sacral_establish_house_fief_effect = {
	save_scope_as = house_fiefer
	save_scope_as = new_head
	# Dynasty
	if = {
		limit = { LR_tgp_gaul_cadet_creates_dynasty_trigger = yes }
		if = {
			limit = { exists = var:new_sacral_house_name }
			create_dynasty = {
				name = {
					first_valid = {
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_random_barony }
							desc = sacral_house_random_barony
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_primary_county }
							desc = sacral_house_primary_county
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_domicile_barony }
							desc = sacral_house_domicile_barony
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_domicile_county }
							desc = sacral_house_domicile_county
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_first_name }
							desc = sacral_house_first_name
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:custom }
							desc = sacral_house_name_custom
						}
						desc = sacral_house_name_flag
					}
				}
				save_scope_as = new_dynasty
			}
		}
		else = {
			create_dynasty = { save_scope_as = new_dynasty }
		}
		house = {
			set_house_name = sacral_dynasty_name
			save_scope_as = new_house
		}
	}
	# House
	else_if = {
		limit = { is_house_head = no }
		found_cadet_house_decision_effect = {
			CHARACTER = root
			PRESTIGE = 0
		}
	}
	# Government
	change_government = sacral_feudal_government
	# Old Dynasty
	if = {
		limit = { exists = scope:new_house }
		# Rename old NF title
		if = {
			limit = {
				scope:old_head ?= this
				exists = scope:old_nf_title
			}
			#scope:old_nf_title = { set_title_name_dynamic = sacral_noble_family_name }
		}
		else = {
			create_noble_family_effect = { GOVERNMENT_GIVER = this }
			scope:new_title = {
				#set_title_name_dynamic = sacral_noble_family_name
				set_coa = scope:new_house
			}
		}
		# Upset old Clan Head
		scope:old_head ?= {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					modifier = established_house_fief_old_clan_head_opinion
					target = root
				}
			}
			trigger_event = LR_tgp_gaul_decision.9005
		}
	}
	# Crime to Regent
	top_liege ?= {
		if = {
			limit = {
				scope:new_head = {
					is_landed = yes
					NOR = {
						var:petition_liege_house_fief_allowed_flag ?= scope:new_head.top_liege
						top_liege = {
							government_has_flag = government_is_sacral_feudal
						}
					}
				}
			}
			show_as_tooltip = {
				add_opinion = {
					modifier = established_house_fief_regent_crime
					target = scope:new_head
				}
			}
			hidden_effect = {
				send_interface_message = {
					type = event_toast_effect_bad
					title = LR_tgp_gaul_establish_house_fief_sacral_majordomo_toast
					left_icon = scope:new_head
					right_icon = scope:new_head.primary_title
					show_as_tooltip = {
						scope:new_head = { change_government = sacral_feudal_government }
					}
					add_opinion = {
						modifier = established_house_fief_regent_crime
						target = scope:new_head
					}
				}
			}
		}
	}
	create_character_memory = { type = establish_house_fief_memory }
	ordered_memory = {
		memory_type = establish_house_fief_memory
		order_by = memory_creation_date
		set_variable = {
			name = title
			value = root.primary_title
		}
	}
	add_achievement_flag_effect = { FLAG = achievement_a_house_of_my_own_flag }
}

set_house_of_all_descendants_effect = {
	if = {
		limit = { any_child = { count >= 1 } }
		every_child = { # Children
			limit = { house ?= $OLD_HOUSE$ }
			set_house = $NEW_HOUSE$
			every_child = { # Grandchildren
				limit = { house ?= $OLD_HOUSE$ }
				set_house = $NEW_HOUSE$
				every_child = { # Great-grandchildren
					limit = { house ?= $OLD_HOUSE$ }
					set_house = $NEW_HOUSE$
					every_child = { # Great-grandchildren
						limit = { house ?= $OLD_HOUSE$ }
						set_house = $NEW_HOUSE$
					}
				}
			}
		}
	}
}

		}
	}
}

LR_tgp_restore_palatine_civil_war_effect = {
	$REBEL$ = {
		save_scope_as = actor
		save_scope_value_as = {
			name = war_for_hoken_flavour
			value = yes
		}
		start_war = {
			casus_belli = sacral_refused_palatine_cb
			target = $LIEGE$
		}
	}
	$LIEGE$ = {
		save_scope_as = recipient
		every_vassal = { #Vassals joining the rebel.
			custom = vassals_joining_war_tt
			limit = {
				government_has_flag = government_is_sacral_feudal
				NOR = {
					this = $REBEL$
					is_ai = no
					has_trait = incapable
					is_imprisoned = yes
					dread_modified_ai_boldness = { #Too scared to do anything.
						dreaded_character = $LIEGE$
						value <= -50
					}
					opinion = { # People who really, really like you will still back you
						target = $LIEGE$
						value >= 80
					}
					is_allied_to = $LIEGE$
					is_at_war_with = $REBEL$
					is_at_war_with = $LIEGE$
				}
				highest_held_title_tier > tier_barony
				OR = {
					#Lovers and friends always back rebel (unless also lovers/friends of the liege)
					AND = {
						OR = {
							is_allied_to = $REBEL$
							has_relation_lover = $REBEL$
							has_relation_friend = $REBEL$
						}
						NOR = {
							is_allied_to = $LIEGE$
							has_relation_lover = $LIEGE$
							has_relation_friend = $LIEGE$
						}
					}
					#Vassals that either like recipient or have high honor will rally if the liege is being tyrannical.
					AND = {
						# If they can't join a faction against the liege they should not revolt
						NOR = {
							has_relation_lover = $LIEGE$
							has_relation_friend = $LIEGE$
							is_allied_to = $LIEGE$
							is_primary_heir_of = $LIEGE$
							$LIEGE$ = { has_strong_hook = prev }
							has_dread_level_towards = {
								target = $LIEGE$
								level = 2
							}
						}
						OR = {
							ai_boldness >= 25
							ai_honor >= 25
							opinion = {
								target = $REBEL$
								value >= 40
							}
						}
						NOR = {
							$LIEGE$ = { has_revoke_title_reason = $REBEL$ }
							trigger_if = {
								limit = { exists = scope:landed_title }
								$LIEGE$ = { has_claim_on = scope:landed_title }
							}
							faith = {
								faith_hostility_level = {
									target = $REBEL$.faith
									value >= faith_hostile_level
								}
							}
						}
					}
					#And finally, vassals that really hate their liege will join regardless (provided that they also do not hate recipient).
					AND = {
						opinion = {
							target = $REBEL$
							value >= -25
						}
						OR = {
							#default threshold
							opinion = {
								target = $LIEGE$
								value <= -50
							}
							#If tyrannical, the default threshold will be reached if at this level
							AND = {
								$LIEGE$ = {
									title_revocation_is_tyrannical_trigger = { VASSAL = $REBEL$ }
								}
								opinion = {
									target = $LIEGE$
									value <= -30 #revoke_title_tyranny_gain adds -20
								}
							}
						}
					}
				}
			}
			save_scope_as = joining_vassal
			custom_tooltip = tooltip_fellow_vassal_joins_war #The war has not started when viewing the interaction screen.
			hidden_effect = {
				$REBEL$ = {
					every_character_war = {
						limit = {
							using_cb = sacral_refused_palatine_cb
							casus_belli = {
								primary_attacker = $REBEL$
								primary_defender = $LIEGE$
							}
						}
						add_to_list = war_to_join
					}
				}
				every_in_list = {
					list = war_to_join
					limit = {
						NOT = { is_defender = prev }
					}
					hidden_effect = { set_called_to = prev }
					add_attacker = prev
				}
			}
			hidden_effect = {
				$LIEGE$ = {
					add_opinion = {
						target = prev
						modifier = rebellious_vassal_opinion
					}
				}
			}
		}
		every_vassal = {
			limit = {
				is_ai = no
				NOT = { this = $REBEL$ }
				is_imprisoned = no
			}
			$REBEL$ = { save_scope_as = recipient }
			$LIEGE$ = { save_scope_as = actor }
			trigger_event = LR_tgp_gaul_decision.9913
		}
	}
}

LR_tgp_liege_change_bloc_management_effect = {
	$VASSAL$ = {
		if = { # Leave if not a valid member anymore
			limit = {
				OR = {
					exists = confederation.leading_house
					exists = house.house_confederation
				}
			}
			save_temporary_scope_as = vassal_temp
			house ?= { save_temporary_scope_as = vassal_house_temp }
			liege ?= { save_temporary_scope_as = new_liege }
			confederation ?= {
				save_temporary_scope_as = current_bloc_temp
				leading_house = { save_temporary_scope_as = current_bloc_leader_temp }
			}
			house.house_confederation ?= {
				save_temporary_scope_as = current_house_bloc_temp
				leading_house = { save_temporary_scope_as = current_house_bloc_leader_temp }
			}
			if = {
				limit = {
					exists = scope:current_bloc_temp
					exists =  scope:old_liege
					trigger_if = { # If a vassal
						limit = { is_independent_ruler = no }
						scope:new_liege.liege != scope:old_liege.liege
						scope:new_liege = {
							is_independent_ruler = no
						}
					}
					trigger_else = {
						NOT = { scope:vassal_temp = scope:old_liege.top_liege }
					}
				}
				scope:current_bloc_temp = {
					if = {
						limit = {
							scope:vassal_temp = {
								is_house_head = yes
								house.house_confederation = scope:current_bloc_temp
								scope:new_liege = {
									is_independent_ruler = no
								}
							}
						}
						scope:vassal_house_temp = {
							LR_tgp_leave_house_bloc_effect = {
								OPINION = flag:no
								TRUCE = flag:yes
							}
						}
					}
					else_if = {
						limit = { scope:vassal_temp.confederation = scope:current_bloc_temp }
						remove_confederation_member = scope:vassal_temp
					}
				}
				debug_log = "Bloc liege change member leave"
				debug_log_scopes = yes
			}
			if = { # Join if not a member of valid
				limit = {
					is_house_head = no
					exists = scope:current_house_bloc_temp
					scope:vassal_house_temp.house_head.liege ?= liege
					NOT = { confederation = scope:vassal_house_temp.house_confederation }
				}
				confederation ?= {
					save_temporary_scope_as = current_bloc_temp
					remove_confederation_member = scope:vassal_temp
				}
				house.house_confederation = {
					save_temporary_scope_as = house_bloc_temp
					add_confederation_member = scope:vassal_temp
				}
				debug_log = "Bloc liege change member join"
				debug_log_scopes = yes
			}
		}
	}
}

							recipient = scope:new_player
						}
					}
				}
			}
			else = {
				# pass the artifact if it's binded to NF (should not happen in vanilla tho), take from others title or not binded to a title
				every_character_artifact = {
					limit = {
						trigger_if = {
							limit = { has_variable = artifact_succession_title }
							OR = {
								var:artifact_succession_title.holder ?= { this != scope:retired }
								var:artifact_succession_title ?= { is_noble_family_title = yes }
							}
						}
					}
					if = {
						limit = { is_equipped = yes }
						add_to_list = equipped
					}
					set_owner = {
						target = scope:new_player
						history = {
							type = inherited
							recipient = scope:new_player
						}
					}
				}
				every_held_title = {
					limit = { is_noble_family_title = yes }
					change_title_holder = {
						holder = scope:new_player
						change = scope:title_change
					}
				}
			}
			resolve_title_and_vassal_change = scope:title_change
			if = {
				limit = { is_ruler = yes }
				primary_title.current_heir = { save_scope_as = heir_temp }
				# now handle title succession
				every_character_artifact = {
					limit = { var:artifact_succession_title.holder ?= scope:retired }
					if = {
						limit = { is_equipped = yes }
						add_to_list = equipped
					}
					set_owner = {
						target = scope:heir_temp
						history = {
							type = inherited
							recipient = scope:heir_temp
						}
					}
				}
				create_title_and_vassal_change = {
					type = appointment
					save_scope_as = title_change
					add_claim_on_loss = no
				}
				every_held_title = {
					limit = { tier > tier_barony }
					change_title_holder_include_vassals = {
						holder = scope:heir_temp
						change = scope:title_change
					}
				}
				resolve_title_and_vassal_change = scope:title_change
				every_in_list = {
					list = equipped
					limit = { artifact_owner = { can_equip_artifact = prev } }
					equip_artifact_to_owner = yes
				}
				pay_short_term_gold = {
					target = scope:new_player
					gold = gold
				}
			}
		}
		house = {
			if = {
				limit = { NOT = { house_head = scope:new_player } }
				set_house_head = scope:new_player
			}
		}
		if = {
			limit = { exists = scope:new_human_player_character }
			set_player_character = scope:new_human_player_character
		}
		progress_towards_friend_effect = {
			REASON = friend_retired_from_governorship
			CHARACTER = scope:new_player
			OPINION = default_friend_opinion
		}
	}
	LR_tgp_step_down_title_recipient_tooltip_effect = yes
	set_variable = {
		name = appointment_trait_override
		value = trait:intellect_good_3
	}
}

setup_sacral_house_name_scopes_effect = {
	save_scope_as = new_head
	save_scope_as = new_house_head # don't ask
	house ?= {
		save_scope_as = old_house
		house_head = { save_scope_as = old_head }
	}
	if = {
		limit = { domicile.domicile_location.county.holder ?= scope:new_head }
		domicile.domicile_location.barony = {
			save_scope_as = domicile_barony
			county = { save_scope_as = domicile_county }
		}
	}
	else_if = {
		limit = {
			is_landed = no
			scope:old_head.domicile.domicile_location.county.holder ?= scope:old_head
		}
		scope:old_head.domicile.domicile_location.barony = {
			save_scope_as = domicile_barony
			county = { save_scope_as = domicile_county }
		}
	}
	else_if = {
		limit = { is_landed = yes }
		save_scope_as = land_haver
	}
	else = {
		limit = { house.house_head ?= { is_landed = yes } }
		house.house_head = { save_scope_as = land_haver }
	}
	scope:land_haver.capital_county ?= {
		save_scope_as = primary_county
		random_county_province = {
			limit = {
				has_holding = yes
				NOT = { has_holding_type = church_holding }
			}
			alternative_limit = { has_holding = yes }
			barony = { save_scope_as = random_barony }
		}
	}
	if = {
		limit = { NOT = { exists = scope:random_barony } }
		location.barony = { save_scope_as = random_barony }
	}
}
