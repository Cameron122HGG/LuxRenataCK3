gaul_establish_house_fief_effect = {
	save_scope_as = house_fiefer
	save_scope_as = new_head
	# Dynasty
	if = {
		limit = { LR_tgp_gaul_cadet_creates_dynasty_trigger = yes }
		if = {
			limit = { exists = var:new_sacral_house_name }
			create_dynasty = {
				name = {
					first_valid = {
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_random_barony }
							desc = sacral_house_random_barony
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_primary_county }
							desc = sacral_house_primary_county
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_domicile_barony }
							desc = sacral_house_domicile_barony
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_domicile_county }
							desc = sacral_house_domicile_county
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:sacral_house_first_name }
							desc = sacral_house_first_name
						}
						triggered_desc = {
							trigger = { scope:new_head.var:new_sacral_house_name ?= flag:custom }
							desc = sacral_house_name_custom
						}
						desc = sacral_house_name_flag
					}
				}
				save_scope_as = new_dynasty
			}
		}
		else = {
			create_dynasty = { save_scope_as = new_dynasty }
		}
		house = {
			set_house_name = sacral_dynasty_name
			save_scope_as = new_house
		}
	}
	# House
	else_if = {
		limit = { is_house_head = no }
		if = {
			limit = { exists = var:new_sacral_house_name }
			found_cadet_house_decision_effect = {
				CHARACTER = root
				PRESTIGE = 0
			}
			house = { save_scope_as = new_house }
		}
		else = {
			create_cadet_branch = { save_scope_as = new_house }
		}
	}
	# Government
	change_government = sacral_feudal_government
	# Old Dynasty
	if = {
		limit = { exists = scope:new_house }
		# Rename old NF title
		if = {
			limit = {
				scope:old_head ?= this
				exists = scope:old_nf_title
			}
			#scope:old_nf_title = { set_title_name_dynamic = sacral_noble_family_name }
		}
		else = {
			create_noble_family_effect = { GOVERNMENT_GIVER = this }
			scope:new_title = {
				#set_title_name_dynamic = sacral_noble_family_name
				set_coa = scope:new_house
			}
		}
		# Upset old Clan Head
		scope:old_head ?= {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					modifier = established_house_fief_old_clan_head_opinion
					target = root
				}
			}
			trigger_event = LR_tgp_gaul_decision.9005
		}
	}
	# Crime to Regent
	top_liege ?= {
		if = {
			limit = {
				scope:new_head = {
					is_landed = yes
					NOR = { 
						var:petition_liege_house_fief_allowed_flag ?= scope:new_head.top_liege
						top_liege = {
							government_has_flag = government_is_sacral_feudal
						}
					}
				}
			}
			show_as_tooltip = {
				add_opinion = {
					modifier = established_house_fief_regent_crime
					target = scope:new_head
				}
			}
			hidden_effect = {
				send_interface_message = {
					type = event_toast_effect_bad
					title = LR_tgp_gaul_establish_house_fief_mayor_gaul_toast
					left_icon = scope:new_head
					right_icon = scope:new_head.primary_title
					show_as_tooltip = {
						scope:new_head = { change_government = sacral_feudal_government }
					}
					add_opinion = {
						modifier = established_house_fief_regent_crime
						target = scope:new_head
					}
				}
			}
		}
	}
	create_character_memory = { type = establish_house_fief_memory }
	ordered_memory = {
		memory_type = establish_house_fief_memory
		order_by = memory_creation_date
		set_variable = {
			name = title
			value = root.primary_title
		}
	}
	add_achievement_flag_effect = { FLAG = achievement_a_house_of_my_own_flag }
}

sacral_set_house_of_all_descendants_effect = {
	if = {
		limit = { any_child = { count >= 1 } }
		every_child = { # Children
			limit = { house ?= $OLD_HOUSE$ }
			set_house = $NEW_HOUSE$
			every_child = { # Grandchildren
				limit = { house ?= $OLD_HOUSE$ }
				set_house = $NEW_HOUSE$
				every_child = { # Great-grandchildren
					limit = { house ?= $OLD_HOUSE$ }
					set_house = $NEW_HOUSE$
					every_child = { # Great-grandchildren
						limit = { house ?= $OLD_HOUSE$ }
						set_house = $NEW_HOUSE$
					}
				}
			}
		}
	}
}

gaul_set_house_effect = {
	house = { save_scope_as = old_house }
	$NEW_HOUSE$ = { save_scope_as = new_house }
	set_house = scope:new_house
	custom_tooltip = {
		text = LR_tgp_gaul_establish_house_fief_direct_descendents_effect
		if = {
			limit = { exists = $NEW_HOUSE$ }
			set_house_of_all_descendants_effect = {
				OLD_HOUSE = scope:old_house
				NEW_HOUSE = scope:new_house
			}
		}
	}
}

sacral_faction_demand_regent_transfer_effect = {
	top_liege = {
		save_temporary_scope_as = old_regent_temp
		primary_title = {
			save_temporary_scope_as = primary_title_temp
			title_capital_county = { save_temporary_scope_as = preferred_capital_temp }
		}
	}
	$NEW_REGENT$ = {
		save_temporary_scope_as = new_regent_temp
		if = {
			limit = { government_has_flag = government_is_sacral_feudal }
			save_temporary_scope_as = sacral_feudal
		}
	}
	# COUNTIES
	create_title_and_vassal_change = {
		type = $REASON$
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	scope:old_regent_temp = { # Give landed titles to the new Regent
		show_as_tooltip = {
			primary_title = { # Give top liege title to the new Regent
				change_title_holder_include_vassals = {
					holder = scope:new_regent_temp
					change = scope:title_change
				}
			}
		}
		hidden_effect = {
			every_held_title = {
				limit = {
					tier > tier_barony # we don't want to try and pass off baronies
					is_landless_type_title = no
					is_noble_family_title = no
					NOT = { exists = var:ceremonial_title } # we don't want to hand away the ceremonial throne, which is not landless
				}
				change_title_holder_include_vassals = {
					holder = scope:new_regent_temp
					change = scope:title_change
				}
			}
			change_liege = {
				liege = scope:new_regent_temp
				change = scope:title_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	scope:new_regent_temp = { 
		if = {
			limit = {
				has_title = scope:preferred_capital_temp # do I own the preferred capital?
				capital_county != scope:preferred_capital_temp # and is my capital not the preferred capital?
			}
			set_realm_capital = scope:preferred_capital_temp # now my capital is the preferred capital
		}
		# GOVERNMENT
		if = {
			limit = {
				exists = scope:sacral_feudal

tgp_restore_palatine_civil_war_effect = {
	$REBEL$ = {
		save_scope_as = actor
		save_scope_value_as = {
			name = war_for_hoken_flavour
			value = yes
		}
		start_war = {
			casus_belli = gaul_refused_palatine_cb
			target = $LIEGE$
		}
	}
	$LIEGE$ = {
		save_scope_as = recipient
		every_vassal = { #Vassals joining the rebel.
			custom = vassals_joining_war_tt
			limit = {
				government_has_flag = government_is_sacral_feudal
				NOR = {
					this = $REBEL$
					is_ai = no
					has_trait = incapable
					is_imprisoned = yes
					dread_modified_ai_boldness = { #Too scared to do anything.
						dreaded_character = $LIEGE$
						value <= -50
					}
					opinion = { # People who really, really like you will still back you
						target = $LIEGE$
						value >= 80
					}
					is_allied_to = $LIEGE$ 
					is_at_war_with = $REBEL$
					is_at_war_with = $LIEGE$
				}
				highest_held_title_tier > tier_barony
				OR = {
					#Lovers and friends always back rebel (unless also lovers/friends of the liege)
					AND = {
						OR = {
							is_allied_to = $REBEL$ 
							has_relation_lover = $REBEL$
							has_relation_friend = $REBEL$
						}
						NOR = {
							is_allied_to = $LIEGE$ 
							has_relation_lover = $LIEGE$
							has_relation_friend = $LIEGE$
						}
					}
					#Vassals that either like recipient or have high honor will rally if the liege is being tyrannical.
					AND = {
						# If they can't join a faction against the liege they should not revolt
						NOR = {
							has_relation_lover = $LIEGE$
							has_relation_friend = $LIEGE$
							is_allied_to = $LIEGE$
							is_primary_heir_of = $LIEGE$
							$LIEGE$ = { has_strong_hook = prev }
							has_dread_level_towards = {
								target = $LIEGE$
								level = 2
							}
						}
						OR = {
							ai_boldness >= 25
							ai_honor >= 25
							opinion = {
								target = $REBEL$
								value >= 40
							}
						}
						NOR = {
							$LIEGE$ = { has_revoke_title_reason = $REBEL$ }
							trigger_if = {
								limit = { exists = scope:landed_title }
								$LIEGE$ = { has_claim_on = scope:landed_title }
							}
							faith = {
								faith_hostility_level = {
									target = $REBEL$.faith
									value >= faith_hostile_level
								}
							}
						}
					}
					#And finally, vassals that really hate their liege will join regardless (provided that they also do not hate recipient).
					AND = {
						opinion = {
							target = $REBEL$
							value >= -25
						}
						OR = {
							#default threshold
							opinion = {
								target = $LIEGE$
								value <= -50
							}
							#If tyrannical, the default threshold will be reached if at this level
							AND = {
								$LIEGE$ = {
									title_revocation_is_tyrannical_trigger = { VASSAL = $REBEL$ }
								}
								opinion = {
									target = $LIEGE$
									value <= -30 #revoke_title_tyranny_gain adds -20
								}
							}
						}
					}
				}
			}
			save_scope_as = joining_vassal
			custom_tooltip = tooltip_fellow_vassal_joins_war #The war has not started when viewing the interaction screen.
			hidden_effect = {
				$REBEL$ = {
					every_character_war = {
						limit = {
							using_cb = gaul_refused_palatine_cb
							casus_belli = {
								primary_attacker = $REBEL$
								primary_defender = $LIEGE$
							}
						}
						add_to_list = war_to_join
					}
				}
				every_in_list = {
					list = war_to_join
					limit = {
						NOT = { is_defender = prev }
					}
					hidden_effect = { set_called_to = prev }
					add_attacker = prev
				}
			}
			hidden_effect = {
				$LIEGE$ = {
					add_opinion = {
						target = prev
						modifier = rebellious_vassal_opinion
					}
				}
			}
		}
		every_vassal = {
			limit = {
				is_ai = no
				NOT = { this = $REBEL$ }
				is_imprisoned = no
			}
			$REBEL$ = { save_scope_as = recipient }
			$LIEGE$ = { save_scope_as = actor }
			trigger_event = LR_tgp_gaul_decision.9913
		}
	}
}

						is_allied_to = scope:super_manorial
						government_has_flag = government_is_herder
						trigger_if = {
							limit = {
								exists = scope:super_manorial.confederation
							}
							confederation ?= scope:super_manorial.confederation
						}	
					}
					#They wont join if you're too scary
					has_dread_level_towards = {
						target = scope:super_manorial
						level < 2
					}
					#Must be strong enough to feel like they can fight you
					current_military_strength >= {
						value = scope:super_manorial.current_military_strength
						multiply = 0.5
					}
				}
			}
		}
		hidden_effect = {
			scope:title_scope = {
				every_de_jure_county_holder = {
					limit = {

							NOR = {

								trigger_if = {
									limit = {
										exists = scope:super_manorial.confederation
									}
									confederation ?= scope:super_manorial.confederation
								}	
							}
							
							#They wont join if you're too scary

								target = scope:super_manorial
								level < 2
							}
							#Must be strong enough to feel like they can fight you
							current_military_strength >= {
								value = scope:super_manorial.current_military_strength
								multiply = 0.5
							}
						}
						add_to_list = war_targets
					}
					else = {
						add_to_list = potential_vassals 
					}
					
				}
				ordered_in_list = {
					list = war_targets
					order_by = current_military_strength
					save_scope_as = main_defender
				}
			}
			scope:super_manorial = {
				if = {
					limit = {
						any_truce_target = { this = scope:main_defender }
					}
					cancel_truce_both_ways = scope:main_defender
				}
			}
			start_war = {
				cb = domination_cb
				target = scope:main_defender
				target_title = scope:title_scope
			}
			#find ai for war
			random_character_war = {
				limit = {
					primary_attacker = scope:super_manorial
					primary_defender = scope:main_defender
					using_cb = domination_cb
				}

				every_in_list = {
					list = war_targets
					limit = {
						NOT = { is_participant_in_war = scope:super_manorial_war }

					}

							set_called_to = scope:super_manorial_war_target
						}
						add_defender = scope:super_manorial_war_target
					}
				}
			}

			every_in_list = {
				list = war_targets
				limit = {
					is_ai = no
				}
				trigger_event = LR_tgp_gaul_decision.9963
			}
		}
		if = {
			limit = { exists = scope:war }
			tgp_bloc_members_join_war_effect = { WAR = scope:war }
		}

LR_tgp_gaul_assert_regional_dominance_war_outcome_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
	}
	change_title_holder = {
		holder = root
		change = scope:change
	}
	every_de_jure_county = {
		limit = {
			holder.top_liege = root.top_liege # in same realm
			holder.liege = root.top_liege # is not already a sub-vassal
			NOR = {
				holder = root # held my me
				holder.liege = root # or my vassal
				holder = { is_liege_or_above_of = root } # or my liege
				this = holder.top_liege.capital_county # stop transfer of capital
			}
		}
		if = {
			limit = {
				holder = {
					NOT = { government_has_mechanic = administrative }
					highest_held_title_tier <= tier_duchy
				}
			}
			holder = {
				change_liege = {
					liege = root
					change = scope:change
				}
			}
		}
		else_if = {
			limit = {
				holder = {
					OR = {
						government_has_mechanic = administrative
						this = root.top_liege
					}
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
	}
	resolve_title_and_vassal_change = scope:change
}
