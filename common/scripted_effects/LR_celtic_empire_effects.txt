# Western Celtic Empire (form_western_celtic_empire_decision)
# User 2026-11-06: the Celtic Empire is the hegemony of the Brythonic title set (it replaced Ynys Prydain), and no warlord
# titles are part of the set. Forming it restores every old Celtic kingdom of the set whose lands the founder completely
# controls (Brythonic and Gaelic alike, LR_brythonic_title_set_effects.txt). Every high-kingship of the set (Cymru, Yr Hen
# Ogledd, Lloegr, Dumnonia, Prydyn, Eriu) then goes under the Celtic Empire. The old effect re-laid the warlord kingdoms under
# e_ireland, e_scotland and e_britons; that route is gone.
LR_form_western_celtic_empire_effect = {
	custom_tooltip = LR_form_western_celtic_empire_dejure_tt
	hidden_effect = {
		LR_brython_restore_all_controlled_effect = yes
		LR_brython_set_under_celtic_empire_effect = yes
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:h_western_celtic_empire = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	dynasty = { add_dynasty_prestige = major_dynasty_prestige_gain }

	# Restored kingdoms may leave titles of other sets without land
	hidden_effect = {
		LR_title_set_reorganise_realm_effect = yes
	}
}
