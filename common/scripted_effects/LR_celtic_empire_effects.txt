# Western Celtic Empire (form_western_celtic_empire_decision)
# The Insular Celtic kingdoms leave Roman Britannia / Gaul and are regrouped under the three
# Celtic high kingships, which in turn become de jure parts of h_western_celtic_empire.
LR_form_western_celtic_empire_effect = {
	custom_tooltip = LR_form_western_celtic_empire_dejure_tt
	hidden_effect = {
		# Hibernia
		title:k_ireland = { set_de_jure_liege_title = title:e_ireland }
		# Alba
		title:k_scotland = { set_de_jure_liege_title = title:e_scotland }
		title:k_alba = { set_de_jure_liege_title = title:e_scotland }
		title:k_mann_the_isles = { set_de_jure_liege_title = title:e_scotland }
		# Britons
		title:k_wales = { set_de_jure_liege_title = title:e_britons }
		title:k_petty_dumnonia = { set_de_jure_liege_title = title:e_britons }
		title:k_rheged = { set_de_jure_liege_title = title:e_britons }
		title:k_brittany = { set_de_jure_liege_title = title:e_britons }
		# The empire
		title:e_ireland = { set_de_jure_liege_title = title:h_western_celtic_empire }
		title:e_scotland = { set_de_jure_liege_title = title:h_western_celtic_empire }
		title:e_britons = { set_de_jure_liege_title = title:h_western_celtic_empire }
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:h_western_celtic_empire = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	dynasty = { add_dynasty_prestige = major_dynasty_prestige_gain }
}
