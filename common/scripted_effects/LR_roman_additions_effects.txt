# Called at the end of LR_sort_diocese_into_hegemony_effect: a diocese of a formed magisterium goes under it
# (DIOCESE = k_diocese_x). The magisterium's holder may be the emperor or his vassal.
LR_magisteria_reroute_effect = {
	if = {
		limit = {
			LR_magisterium_italia_diocese_trigger = { DIOCESE = $DIOCESE$ }
			title:e_magisterium_italia = { OR = { holder ?= root holder.top_liege ?= root } }
		}
		title:$DIOCESE$ = { set_de_jure_liege_title = title:e_magisterium_italia }
	}
	if = {
		limit = {
			LR_magisterium_africa_diocese_trigger = { DIOCESE = $DIOCESE$ }
			title:e_magisterium_africa = { OR = { holder ?= root holder.top_liege ?= root } }
		}
		title:$DIOCESE$ = { set_de_jure_liege_title = title:e_magisterium_africa }
	}
}

# Root (a restored Roman emperor) forms the Magisterium per Italiam
LR_form_magisterium_italia_effect = {
	if = { limit = { NOT = { exists = title:e_magisterium_italia.holder } }
		create_title_and_vassal_change = { type = created save_scope_as = LR_radd_change add_claim_on_loss = no }
		title:e_magisterium_italia = { change_title_holder = { holder = root change = scope:LR_radd_change } }
		resolve_title_and_vassal_change = scope:LR_radd_change
	}
	set_global_variable = { name = LR_magisterium_italia_formed value = yes }
	if = { limit = { root.primary_title.tier = tier_hegemony } title:e_magisterium_italia = { set_de_jure_liege_title = root.primary_title } }
	if = { limit = { title:k_diocese_italia_annonaria = { is_titular = no } } title:k_diocese_italia_annonaria = { set_de_jure_liege_title = title:e_magisterium_italia } }
	if = { limit = { title:k_diocese_italia_suburbicaria = { is_titular = no } } title:k_diocese_italia_suburbicaria = { set_de_jure_liege_title = title:e_magisterium_italia } }
	title:e_praetorian_prefecture_italia = { set_variable = { name = LR_merged_into_magisterium value = title:e_magisterium_italia } }
	if = { limit = { has_title = title:e_praetorian_prefecture_italia title:e_praetorian_prefecture_italia = { NOT = { any_de_jure_county = { always = yes } } } } destroy_title = title:e_praetorian_prefecture_italia }
}
