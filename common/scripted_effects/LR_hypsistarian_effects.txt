# LR 2026-11-22: Sons of the Lightning crossover. hypsistarian_rite leaves Hellenic Pagan and becomes the full
# Hypsistian faith (SotL's 'hypsistian'): it takes SotL's name and tenets, and the Pontic emperor takes the new
# landless head title d_theos_hypsistos.
LR_found_hypsistarian_faith_effect = {
	set_global_variable = { name = LR_hypsistarian_faith_founded value = yes }
	create_title_and_vassal_change = {
		type = created
		save_scope_as = LR_hypsistos_change
		add_claim_on_loss = no
	}
	title:d_theos_hypsistos = {
		change_title_holder = {
			holder = root
			change = scope:LR_hypsistos_change
		}
	}
	resolve_title_and_vassal_change = scope:LR_hypsistos_change
	rite:hypsistarian_rite = {
		set_can_convert_to_rite = yes
		set_parent_faith = {
			target = faith:hypsistian
			main = yes
		}
		if = {
			limit = { NOT = { rite_has_doctrine = doctrine_temporal_head } }
			change_rite_doctrine = doctrine_temporal_head
		}
		remove_rite_tenet = tenet:tenet_astrology
		remove_rite_tenet = tenet:tenet_household_gods
		remove_rite_tenet = tenet:tenet_esotericism
		add_rite_tenet = tenet:tenet_pursuit_of_power
		add_rite_tenet = tenet:tenet_adaptive
		add_rite_tenet = tenet:tenet_ritual_celebrations
		set_rite_name = {
			name = hypsistian
			desc = hypsistian_desc
			adjective = hypsistian_adj
			adherent = hypsistian_adherent
			adherent_plural = hypsistian_adherent_plural
		}
	}
	# SotL's other Hypsistarian faiths, LR rites since 2026-11-22, follow as sister rites
	rite:theosebian_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	rite:pantokratic_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	rite:menist_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	rite:gaithasuran_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	rite:decenean_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	rite:sabazaic_rite = {
		set_parent_faith = {
			target = faith:hypsistian
			main = no
		}
	}
	faith:hypsistian = { set_religious_head_title = title:d_theos_hypsistos }
	every_player = {
		limit = { this != root }
		send_interface_toast = {
			title = LR_hypsistarian_faith_founded_toast
			left_icon = root
		}
	}
}
