# Root founds the Strategia of Sērindia
LR_hcomp_found_serindia_effect = {
	if = {
		limit = { NOT = { exists = title:e_strategia_serindia.holder } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = LR_hcomp_change
			add_claim_on_loss = no
		}
		title:e_strategia_serindia = {
			change_title_holder = {
				holder = root
				change = scope:LR_hcomp_change
			}
		}
		resolve_title_and_vassal_change = scope:LR_hcomp_change
	}
	title:e_strategia_serindia = { set_de_jure_liege_title = title:h_hellenistic_empire }
	if = {
		limit = { primary_title = { tier < tier_empire } }
		set_primary_title_to = title:e_strategia_serindia
	}
	if = {
		limit = {
			LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_kasia }
			LR_hcomp_kasia_controlled_trigger = yes
		}
		LR_restore_satrapy_effect = { SATRAPY = k_satrapy_kasia EMPIRE = e_strategia_serindia }
	}
	else_if = {
		limit = { title:k_satrapy_kasia = { is_titular = no } }
		title:k_satrapy_kasia = { set_de_jure_liege_title = title:e_strategia_serindia } # already restored: into the new strategia
	}
	if = {
		limit = { NOT = { title:c_kashgar = { holder.top_liege ?= root } } }
		add_unpressed_claim = title:c_kashgar # TEMP: a claim on each missing satrapy's seat
	}
	if = {
		limit = {
			LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_issedonia }
			LR_hcomp_issedonia_controlled_trigger = yes
		}
		LR_restore_satrapy_effect = { SATRAPY = k_satrapy_issedonia EMPIRE = e_strategia_serindia }
	}
