LR_fix_title_granting_on_start = {
	every_ruler = {
		limit = {
			is_landed = yes
		}
		create_title_and_vassal_change = {
			type = appointment
			save_scope_as = title_to_fix
			add_claim_on_loss = no
		}
		primary_title = {
			change_title_holder = {
				holder = holder
				change = scope:title_to_fix
			}
		}
		resolve_title_and_vassal_change = scope:title_to_fix
		primary_title = { save_scope_as = title_to_fix }
		set_primary_title_to = primary_title
	}
}

		save_temporary_scope_as = overlord_to_grant
		every_held_title = {
			limit = {

				}

		every_held_title = {
			limit = {
				NOT = {

				create_character = {
					gender = male

					culture = scope:province_for_culture.culture

					save_temporary_scope_as =  lux_temp_vassal
				}
			}
			else_if = {
				limit = { scope:overlord_to_grant.faith = { has_doctrine = doctrine_gender_female_dominated } }

						create_character = {

				}
			}			
			
			save_temporary_scope_as = current_duchy
			every_in_list = {
				list = titles_for_granting
				if = {

					create_title_and_vassal_change = {
						type = granted
						save_scope_as = change
						add_claim_on_loss = no
					}				
					change_title_holder_include_vassals = {
						holder = scope:lux_temp_vassal
						change = scope:change
						take_baronies = no
						government_base = scope:overlord_to_grant
					}
					resolve_title_and_vassal_change = scope:change
				}
			}

LR_generate_noble_families_on_start = {
	every_ruler = {
		limit = { 
			is_independent_ruler = yes
			government_has_mechanic = administrative
			NOT = { any_held_title = { is_noble_family_title = yes } }
		}
		save_temporary_scope_as = this_ruler
		create_noble_family_effect = { GOVERNMENT_GIVER = scope:this_ruler }
	}
	every_ruler = {
		limit = { 
			government_has_flag = government_is_sacral_feudal
			NOT = { any_held_title = { is_noble_family_title = yes } }
		}
		save_temporary_scope_as = this_ruler
		create_noble_family_effect = { GOVERNMENT_GIVER = scope:this_ruler }
	}
	every_ruler = {
		limit = { 
			is_independent_ruler = no
			government_has_mechanic = administrative
			NOT = { any_held_title = { is_noble_family_title = yes } }
		}
		save_temporary_scope_as = this_ruler
		create_noble_family_effect = { GOVERNMENT_GIVER = scope:this_ruler.liege }
	}
		
}
