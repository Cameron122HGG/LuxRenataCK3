# Great Moravia (form_great_moravia_decision)
# Nitra and Pannonia leave Hungary for Nitrava (k_moravia); Moravia (k_bohemia), Nitrava and
# Venedia (k_sorbia) leave Sclavinia and become de jure parts of e_great_moravia.
LR_form_great_moravia_effect = {
	custom_tooltip = LR_form_great_moravia_dejure_tt
	hidden_effect = {
		title:d_nyitra = { set_de_jure_liege_title = title:k_moravia }
		title:d_gyor = { set_de_jure_liege_title = title:k_moravia }
		title:d_somogy = { set_de_jure_liege_title = title:k_moravia }
		title:k_bohemia = { set_de_jure_liege_title = title:e_great_moravia }
		title:k_moravia = { set_de_jure_liege_title = title:e_great_moravia }
		title:k_sorbia = { set_de_jure_liege_title = title:e_great_moravia }
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:e_great_moravia = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	if = {
		limit = { NOT = { exists = title:k_moravia.holder } }
		title:k_moravia = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	dynasty = { add_dynasty_prestige = major_dynasty_prestige_gain }
}
