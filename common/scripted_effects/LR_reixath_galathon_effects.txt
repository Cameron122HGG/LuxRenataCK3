# used (the old effect re-laid the warlord k_galatia and k_pontus). Forming it restores every Galatian kingdom the founder

# kingdoms, under the hegemony.

	custom_tooltip = LR_form_reixath_galathon_dejure_tt
	hidden_effect = {
		LR_galat_restore_all_controlled_effect = yes
		LR_galat_set_under_reixath_effect = yes
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:h_reixath_galathon = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	dynasty ?= { add_dynasty_prestige = major_dynasty_prestige_gain }

	hidden_effect = { LR_title_set_reorganise_realm_effect = yes }
}
