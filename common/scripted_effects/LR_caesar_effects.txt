# Root (an Augustus): put scope:college_title into his Imperial College, founding it if he has none
LR_college_add_title_effect = {
	primary_title = { save_scope_as = lr_college_lead } # scope-relative: also run for others than root
	if = {
		limit = { exists = primary_title.title_confederation }
		primary_title.title_confederation = {
			save_scope_as = lr_college
			add_confederation_member_title = scope:college_title
		}
	}
	else = {
		create_confederation = {
			name = LR_COLLEGIUM_IMPERIALE
			type = confederation_type:lr_imperial_college
			leading_title = scope:lr_college_lead
		}
		scope:new_confederation ?= {
			save_scope_as = lr_college
			set_cohesion = 30
			set_coa = scope:lr_college_lead
			add_confederation_member_title = scope:college_title
		}
	}
	scope:lr_college ?= { LR_college_rename_effect = yes }
}

# Confederation: name by membership (Tetrarchia / Concordia Augustorum / Collegium Imperiale)
LR_college_rename_effect = {
	if = {
		limit = {
			any_confederation_member = { count >= 2 LR_caesar_is_augustus_trigger = yes }
			any_confederation_member = { count >= 2 has_trait = caesar }
		}
		set_confederation_name = LR_TETRARCHIA
	}
	else_if = {
		limit = { any_confederation_member = { count >= 2 LR_caesar_is_augustus_trigger = yes } }
		set_confederation_name = LR_CONCORDIA_AUGUSTORUM
	}
	else = { set_confederation_name = LR_COLLEGIUM_IMPERIALE }
}

# Character: take your titles out of their Imperial College (not a leading title)
LR_college_leave_effect = {
	every_held_title = {
		limit = {
			exists = title_confederation
			NOT = { title_confederation.leading_title ?= this }
		}
		save_temporary_scope_as = lr_leaving_title
		title_confederation = { remove_confederation_member_title = scope:lr_leaving_title }
	}
}

# Character: stop being a Caesar (recalled, proclaimed Augustus, or promoted). The dynamic command itself is
# ended (LR_caesar_command_end_effect) or turned usurper (LR_caesar_command_usurp_effect) by the caller; a static
# command from before 2026-11-25 just loses its mark.
LR_caesar_unmake_effect = {
	if = { limit = { has_trait = caesar } remove_trait = caesar }
	if = { limit = { has_character_flag = LR_caesar_flag } remove_character_flag = LR_caesar_flag }
	every_held_title = {
		limit = { has_variable = LR_caesar_of NOT = { has_variable = LR_caesar_region } }
		remove_variable = LR_caesar_of
	}
}

# Root (the Augustus) gives scope:new_holder his counties, duchies and kingdoms in REGION and his direct vassals based
# there. Like LR_cleave_hand_over_half_effect, but empire-tier titles stay: the static Roman titles never move.
LR_caesar_hand_over_effect = {
	if = {
		limit = { exists = domicile domicile.domicile_location = { geographical_region = $REGION$ } }
		domicile = { move_domicile = root.capital_province }
	}
	create_title_and_vassal_change = { type = granted save_scope_as = change add_claim_on_loss = no }
	every_held_title = {
		limit = {
			tier = tier_county
			is_landless_type_title = no
			title_province = { geographical_region = $REGION$ }
			NOT = { this = root.capital_county }
		}
		change_title_holder = { holder = scope:new_holder change = scope:change }
	}
	every_held_title = {
		limit = {
			tier >= tier_duchy
			tier <= tier_kingdom
			is_landless_type_title = no
			NOT = { this = root.primary_title }
			any_de_jure_county = { percent >= 0.51 title_province = { geographical_region = $REGION$ } }
		}
		change_title_holder = { holder = scope:new_holder change = scope:change }
	}
	resolve_title_and_vassal_change = scope:change
	create_title_and_vassal_change = { type = granted save_scope_as = change add_claim_on_loss = no }
	every_vassal = {
		limit = {
			NOT = { this = scope:new_holder }
			highest_held_title_tier >= tier_county

# Title: a dynamic Caesar command ends (recall, promotion, confirmed lapse): out of the lists and the college,
# then destroyed by its holder. The land stays where it is (the caller moves it first if needed).
LR_caesar_command_end_effect = {
	save_scope_as = lr_ending_command
	var:LR_caesar_court ?= { holder ?= { destroy_title = prev } } # 2026-11-25: its Imperial Court goes with it
	remove_list_global_variable = { name = LR_caesar_commands target = scope:lr_ending_command }
	title_confederation ?= { remove_confederation_member_title = scope:lr_ending_command }
	holder ?= { destroy_title = scope:lr_ending_command }
}

# Title: a dynamic Caesar command whose Caesar breaks away (refused recall, Proclaim Yourself Augustus): it stays
# his, renamed as an empire of its own, out of the lists and the college.
LR_caesar_command_usurp_effect = {
	save_scope_as = lr_usurped_command
	remove_list_global_variable = { name = LR_caesar_commands target = scope:lr_usurped_command }
	title_confederation ?= { remove_confederation_member_title = scope:lr_usurped_command }
	remove_variable = LR_caesar_of
	remove_variable = LR_caesar_augustus
	set_variable = LR_caesar_usurper
	LR_caesar_command_name_effect = yes
}

# Character: end or turn usurper every dynamic command root holds (MODE = end / usurp)
LR_caesar_commands_of_root_effect = {
	every_held_title = {
		limit = { has_variable = LR_caesar_region NOT = { has_variable = LR_caesar_usurper } NOT = { has_variable = LR_caesar_court_title } }
		LR_caesar_command_$MODE$_effect = yes
	}
}
