# Root (an Augustus): put scope:college_title into his Imperial College, founding it if he has none
LR_college_add_title_effect = {
	primary_title = { save_scope_as = lr_college_lead } # scope-relative: also run for others than root
	if = {
		limit = { exists = primary_title.title_confederation }
		primary_title.title_confederation = {
			save_scope_as = lr_college
			add_confederation_member_title = scope:college_title
		}
	}
	else = {
		create_confederation = {
			name = LR_COLLEGIUM_IMPERIALE
			type = confederation_type:lr_imperial_college

		scope:new_confederation ?= {
			save_scope_as = lr_college
			set_cohesion = 30

		}

	scope:lr_college ?= { LR_college_rename_effect = yes }
}
