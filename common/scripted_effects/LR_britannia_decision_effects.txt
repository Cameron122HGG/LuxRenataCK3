# Claim the Bretwalda (unify_england_anglo_saxon_decision)
# Ported from Lux Invicta's become_bretwalda: the Heptarchy is folded into a single kingdom (k_hk_england, Englaland)
# and the petty Saxon kingdoms are destroyed and locked via can_create.

# Moves the de jure duchies of a petty kingdom into k_hk_england, then destroys it if the founder holds it
# $KINGDOM$ = title:k_petty_wessex etc.
LR_bretwalda_absorb_petty_kingdom_effect = {
	$KINGDOM$ = {
		every_in_de_jure_hierarchy = {
			limit = { tier = tier_duchy }
			set_de_jure_liege_title = title:k_hk_england
		}
	}
	if = {
		limit = { has_title = $KINGDOM$ }
		destroy_title = $KINGDOM$
	}
}

LR_claim_bretwalda_decision_scripted_effect = {
	save_scope_as = founder
	primary_title = { save_scope_as = old_primary_title }

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:k_hk_england = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	hidden_effect = {
		set_primary_title_to = title:k_hk_england
		title:k_hk_england = {
			set_de_jure_liege_title = title:e_britannia
			copy_title_history = scope:old_primary_title
		}
	}
	custom_tooltip = LR_claim_bretwalda_decision_de_jure_tt

	hidden_effect = {
		# Landed kingdoms of the Heptarchy
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_wessex }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_mercia }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_magonsaete }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_anglia }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_deira }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_bernicia }
		# Titular kingdoms of the Heptarchy
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_east_anglia }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_essex }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_kent }
		LR_bretwalda_absorb_petty_kingdom_effect = { KINGDOM = title:k_petty_sussex }

		set_global_variable = {
			name = bretwalda_claimed
			value = root
		}
	}
	custom_tooltip = LR_claim_bretwalda_decision_locked_tt
}
