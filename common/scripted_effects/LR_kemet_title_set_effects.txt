# Scope: a title left behind by a Kemet layout. If it has a holder but no de jure land left, the holder gets the
# Kemet title of the same tier over its capital county (if unheld and theirs to use) and the old title is dissolved.
LR_kemet_dissolve_if_empty_effect = {
	if = {
		limit = {
			exists = holder
			NOT = { LR_title_set_kemet_title_trigger = yes }
			NOT = { any_de_jure_county = { always = yes } }
		}
		save_temporary_scope_as = LR_kemet_old
		holder = { save_temporary_scope_as = LR_kemet_old_holder }
		title_capital_county ?= {
			if = {
				limit = { scope:LR_kemet_old.tier = tier_duchy }
				duchy ?= { save_temporary_scope_as = LR_kemet_new }
			}
			else_if = {
				limit = { scope:LR_kemet_old.tier = tier_kingdom }
				kingdom ?= { save_temporary_scope_as = LR_kemet_new }
			}
		}
		if = {
			limit = {
				exists = scope:LR_kemet_new
				scope:LR_kemet_new = {
					LR_title_set_kemet_title_trigger = yes
					NOT = { exists = holder }
				}
				scope:LR_kemet_old_holder = { LR_title_set_kemet_native_trigger = yes }
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = LR_kemet_change
				add_claim_on_loss = no
			}
			scope:LR_kemet_new = {
				change_title_holder = {
					holder = scope:LR_kemet_old_holder
					change = scope:LR_kemet_change
				}
			}
			resolve_title_and_vassal_change = scope:LR_kemet_change
		}
		scope:LR_kemet_old_holder = { destroy_title = scope:LR_kemet_old }
		clear_saved_scope = LR_kemet_new
	}
}
