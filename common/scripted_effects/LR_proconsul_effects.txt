# Root = a Proconsul who lays down his command (recalled at the end of his term): the command ends, and he
# swears fealty to the Republic with his lands
LR_proconsul_lay_down_effect = {
	var:LR_caesar_augustus ?= { save_scope_as = lr_proconsul_head }
	LR_caesar_commands_of_root_effect = { MODE = end }
	LR_college_leave_effect = yes
	if = { limit = { has_variable = LR_caesar_augustus } remove_variable = LR_caesar_augustus }
	LR_caesar_unmake_effect = yes
	if = {
		limit = {
			exists = scope:lr_proconsul_head
			scope:lr_proconsul_head = { is_alive = yes }
			is_landed = yes
			is_independent_ruler = yes
		}
		create_title_and_vassal_change = { type = swear_fealty save_scope_as = lr_pc_fealty add_claim_on_loss = no }
		change_liege = { liege = scope:lr_proconsul_head change = scope:lr_pc_fealty }
		resolve_title_and_vassal_change = scope:lr_pc_fealty
	}
	add_prestige = medium_prestige_gain # Cincinnatus went back to his plough
}

	save_scope_as = usurper

		LR_caesar_commands_of_root_effect = { MODE = usurp }

		remove_variable = LR_caesar_augustus
	}
	LR_caesar_unmake_effect = yes
	add_trait = augustus
	add_pressed_claim = scope:usurp_target
	start_war = {
		casus_belli = claim_cb
		target = scope:old_augustus

	}
	scope:old_augustus = { trigger_event = LR_proconsul.0003 }
}

		limit = {

			limit = { is_landed = yes is_independent_ruler = yes }
