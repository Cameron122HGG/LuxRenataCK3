eastern_roman_core_dev_decision_effect = {
	every_county_in_region = {
		region = custom_eastern_roman_imperial_core
		county = { add_to_list = custom_eastern_roman_imperial_core_list }
	}
	every_in_list = {
		list = custom_eastern_roman_imperial_core_list
		limit = {
			holder = {
				NOT = { holder = root }
			} 
		}
		create_title_and_vassal_change = {
            type = usurped
            save_scope_as = change
            add_claim_on_loss = no
        }
		change_title_holder_include_vassals = {
			holder = root
			change = scope:change
		}
		resolve_title_and_vassal_change = scope:change	
	}
}
