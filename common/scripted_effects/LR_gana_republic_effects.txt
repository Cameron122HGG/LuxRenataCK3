# LR 2026-11-25 group republics (tools/group_republics/build_group_republics.py)
# Normalizer (LR_normalize_dynastic_government_effect): a republic of the Indic group (culture keeps Indic Statecraft) becomes the Gana-Sangha (government
# first: its administrations are exclusive to it), then gets its default administration; one that no longer qualifies
# goes back to the plain republic.
LR_gana_republic_government_effect = {
	if = {
		limit = {
			has_government = republic_government
			LR_gana_group_republic_trigger = yes
		}
		change_government = gana_republic_government
		if = {
			limit = { NOT = { has_realm_law_flag = LR_gana_republic_type } }
			add_realm_law_skip_effects = republic_type_gana_kula
		}
	}
	else_if = {
		limit = {
			has_government = gana_republic_government
			LR_gana_group_republic_trigger = no
			gdp_is_assembly_head_trigger = yes # 2026-11-25: vassal republics follow their liege instead (LR_republic_vassal_follows_liege_effect)
			NOT = { has_realm_law_flag = LR_gana_republic_type }
		}
		change_government = republic_government
	}
}

# character scope (the head of the republic): appoint the Uparaja
LR_appoint_uparaja_effect = {
	if = {
		limit = {
			has_government = gana_republic_government
			has_realm_law_flag = LR_has_uparaja
		}
		save_scope_as = LR_chief_raja
		if = {
			limit = { any_vassal = { LR_uparaja_candidate_trigger = yes } }
			ordered_vassal = {
				limit = { LR_uparaja_candidate_trigger = yes }
				order_by = diplomacy
				save_scope_as = LR_group_diarch
			}
		}
		else_if = {
			limit = { any_courtier = { LR_uparaja_candidate_trigger = yes } }
			ordered_courtier = {
				limit = { LR_uparaja_candidate_trigger = yes }
				order_by = diplomacy
				save_scope_as = LR_group_diarch
			}
		}
		if = {
			limit = { exists = scope:LR_group_diarch }
			designate_diarch = scope:LR_group_diarch
			try_start_diarchy = uparaja_diarchy
			set_diarch = scope:LR_group_diarch
			LR_republic_diarch_title_effect = yes
		}
	}
}

LR_end_uparaja_effect = {
	if = {
		limit = { has_diarchy_type = uparaja_diarchy }
		end_diarchy = yes
	}
}

LR_uparaja_upkeep_effect = {
	if = {
		limit = {
			has_government = gana_republic_government
			has_realm_law_flag = LR_has_uparaja
			NOT = { exists = diarch }
		}
		LR_appoint_uparaja_effect = yes
	}
}

# game start, a day late (after the normalizer): every such republic starts on the two-man law, then the diarch
LR_gana_diarchy_start_effect = {
	if = {
		limit = { has_government = gana_republic_government }
		if = {
			limit = { NOT = { has_realm_law = LR_uparaja_elected } }
			add_realm_law_skip_effects = LR_uparaja_elected
		}
		LR_uparaja_upkeep_effect = yes
	}
}
