unite_the_japanese_courts_decision_effects = {
# Create united Japan
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}

	title:e_japan = {
		change_title_holder = {
			holder = scope:japan_restorer
			change = scope:title_change
		}
	}

	resolve_title_and_vassal_change = scope:title_change


	if = {
		limit = { primary_title = { this = title:e_north_japan } }
		title:e_japan = { set_capital_county = title:c_yamashiro }
	}
	else = {
		title:e_japan = { set_capital_county = title:c_yamato }
	}

	scope:japan_restorer = {
		if = {
			limit = {
				completely_controls_region = world_asia_japan # this has to be done via region, since k_domingo won't exist to scope off of if the hispaniola decision was taken
			}
			title:k_hitakami = { set_de_jure_liege_title = title:e_japan }
			title:k_aynumosir = { set_de_jure_liege_title = title:e_japan }
		}
	}
	
	# Determine which claimaint you're pushing.
	if = {
		limit = {
			primary_title = {
				this = title:e_north_japan
			}
			exists = title:k_chrysanthemum_throne_north.holder
		}
		title:k_chrysanthemum_throne = {
			copy_title_history = title:k_chrysanthemum_throne_north
		}
		title:k_chrysanthemum_throne_north.holder = {
			save_scope_as = chrysanthemum_holder
		}
	}
	else_if = {
		limit = {
			primary_title = {
				this = title:e_south_japan
			}
			exists = title:k_chrysanthemum_throne_south.holder
		}
		title:k_chrysanthemum_throne = {
			copy_title_history = title:k_chrysanthemum_throne_south
		}
		title:k_chrysanthemum_throne_south.holder = {
			save_scope_as = chrysanthemum_holder
		}
	}
	if = {
		limit = {
			AND = {
				exists = title:e_japan.holder
				OR ={
					exists = title:e_north_japan.holder
					exists = title:e_south_japan.holder
				}
			} 
		}
		destroy_title = title:e_north_japan
		destroy_title = title:e_south_japan
	}
	# Unite the Courts	
	create_title_and_vassal_change = {
		type = created
		save_scope_as = chrysanthemum_title_change
		add_claim_on_loss = no
	}
	title:k_chrysanthemum_throne ?= {
		change_title_holder = {
			holder = scope:chrysanthemum_holder
			change = scope:chrysanthemum_title_change
		}
	}
	if = {
		limit = {
				scope:chrysanthemum_holder.liege != scope:japan_restorer
				scope:chrysanthemum_holder != scope:japan_restorer
			}
			scope:chrysanthemum_holder ?= {
				change_liege = {
					liege = scope:japan_restorer
					change = scope:chrysanthemum_title_change
				}
			}
		}
	resolve_title_and_vassal_change = scope:chrysanthemum_title_change
	destroy_title = title:k_chrysanthemum_throne_north
	destroy_title = title:k_chrysanthemum_throne_south
	
}
