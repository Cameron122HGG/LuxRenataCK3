# Restore the Roman Empire (restore_roman_empire_decision), reworked 2026-11-06.
# Replaces vanilla create_roman_empire_scripted_effect for this decision: h_roman_empire is founded by the holder
# of a restored hegemony (East or West). The praetorian prefectures, exarchates and other empire-tier titles are
# kept and move de jure under Rome; only the two hegemonies are dissolved into it.
# Run by roman_restoration.0001 (immediate). Character scope (the new Augustus).

LR_restore_roman_empire_effect = {
	primary_title = { save_scope_as = old_primary_title }
	# The hegemony Rome is restored from (the East if you hold both)
	if = {
		limit = { has_title = title:h_eastern_roman_empire }
		title:h_eastern_roman_empire = { save_scope_as = old_hegemony }
	}
	else_if = {
		limit = { has_title = title:h_western_roman_empire }
		title:h_western_roman_empire = { save_scope_as = old_hegemony }
	}

	add_trait = augustus
	if = {
		limit = { has_dlc_feature = legends }
		create_legend_seed = {
			type = legitimizing
			quality = illustrious
			chronicle = new_title
			properties = {
				title = title:h_roman_empire
				founder = root
			}
		}
	}

	# The new title's seat follows yours if it is already Rome or Constantinople
	if = {
		limit = { capital_county = { this = title:c_byzantion } }
		title:h_roman_empire = { set_capital_county = title:c_byzantion }
	}
	else_if = {
		limit = { capital_county = { this = title:c_roma } }
		title:h_roman_empire = { set_capital_county = title:c_roma }
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:h_roman_empire = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
		if = {
			limit = { exists = scope:old_hegemony }
			copy_title_history = scope:old_hegemony
		}
	}
	resolve_title_and_vassal_change = scope:change

	if = { # Move your title men-at-arms to the new title
		limit = {
			government_has_mechanic = administrative
			primary_title = {
				any_title_maa_regiment = { count >= 1 }
			}
		}
		hidden_effect = {
			primary_title = { transfer_title_maa_ownership = title:h_roman_empire }
		}
	}
	hidden_effect = { # Keep the guard bonuses of either half
		if = {
			limit = {
				OR = {
					scope:old_primary_title = { has_variable = founded_varangian_guard }
					scope:old_hegemony ?= { has_variable = founded_varangian_guard }
					title:e_eastern_roman_empire = { has_variable = founded_varangian_guard }
				}
			}
			title:h_roman_empire = { set_variable = founded_varangian_guard }
		}
		if = {
			limit = {
				OR = {
					scope:old_primary_title = { has_variable = refounded_praetorian_guard }
					scope:old_hegemony ?= { has_variable = refounded_praetorian_guard }
					title:e_western_roman_empire = { has_variable = refounded_praetorian_guard }
				}
			}
			title:h_roman_empire = { set_variable = refounded_praetorian_guard }
		}
	}

	hidden_effect = { set_primary_title_to = title:h_roman_empire }

	# The empire tier stays: your landed empire titles (prefectures, exarchates, the Eastern and Western Empires)
	# become de jure vassals of Rome instead of being destroyed
	custom_tooltip = LR_restore_roman_empire_keep_prefectures_tt
	hidden_effect = {
		every_held_title = {
			limit = {
				tier = tier_empire
				is_landless_type_title = no
			}
			set_de_jure_liege_title = title:h_roman_empire
		}
	}

	# Both halves' empire-tier de jure vassals move to Rome, unless the other half is held outside your realm
	custom_tooltip = LR_restore_roman_empire_halves_tt
	hidden_effect = {
		title:h_eastern_roman_empire = {
			if = {
				limit = {
					OR = {
						NOT = { exists = holder }
						holder = root
						holder = { target_is_liege_or_above = root }
					}
				}
				every_in_de_jure_hierarchy = {
					continue = { tier = tier_hegemony }
					limit = { tier = tier_empire }
					set_de_jure_liege_title = title:h_roman_empire
				}
			}
		}
		title:h_western_roman_empire = {
			if = {
				limit = {
					OR = {
						NOT = { exists = holder }
						holder = root
						holder = { target_is_liege_or_above = root }
					}
				}
				every_in_de_jure_hierarchy = {
					continue = { tier = tier_hegemony }
					limit = { tier = tier_empire }
					set_de_jure_liege_title = title:h_roman_empire
				}
			}
		}
	}

	# The hegemonies you hold are dissolved into Rome
	if = {
		limit = { has_title = title:h_eastern_roman_empire }
		destroy_title = title:h_eastern_roman_empire
	}
	if = {
		limit = { has_title = title:h_western_roman_empire }
		destroy_title = title:h_western_roman_empire
	}

	# Move the realm capital to Constantinople (Byzantine heritage) or Rome (everyone else), unless it is already at one
	if = {
		limit = {
			NOT = { capital_county = { this = title:c_roma } }
			NOT = { capital_county = { this = title:c_byzantion } }
			culture = { has_cultural_pillar = heritage_byzantine }
		}
		hidden_effect = {
			if = {
				limit = { NOT = { this = title:c_byzantion.holder } }
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:c_byzantion = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		set_realm_capital = title:c_byzantion
	}
	else_if = {
		limit = {
			NOT = { capital_county = { this = title:c_roma } }
			NOT = { capital_county = { this = title:c_byzantion } }
		}
		hidden_effect = {
			if = {
				limit = { NOT = { this = title:c_roma.holder } }
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:c_roma = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		set_realm_capital = title:c_roma
	}

	hidden_effect = {
		LR_greek_east_change_effect = { ADD = -15 } # LR 2026-11-06: the Greek East looks to Rome again
		add_character_flag = flag_restorer_of_rome # used for Eulogy
		if = {
			limit = {
				exists = scope:old_hegemony
				scope:old_hegemony = title:h_western_roman_empire
			}
			title:h_roman_empire = { set_variable = rome_was_restored_by_spqr }
		}
		else = {
			title:h_roman_empire = { set_variable = rome_was_restored_by_byzantium }
		}
	}
}


############################################################################
# Cleave the Empire (split_roman_empire_decision), reworked 2026-11-06.
# The Augustus keeps the half of their capital (Theodosian line, custom_theodosian_ere/_wre).
# LR_cleave_empire.0001 offers: Two Augusti (two real hegemonies, Rome dissolved and restorable again)
# or A Caesar (Rome stays; the other half's empire title goes to a junior emperor who stays your vassal).
# Character scope (the Augustus). scope:new_holder is saved by LR_cleave_find_new_augustus_effect.
############################################################################

# Picks the other Augustus: your co- or junior emperor, else your strongest direct vassal based in the other half,
# else the head of the strongest other powerful family, else your strongest vassal
LR_cleave_find_new_augustus_effect = {
	if = {
		limit = {
			has_active_diarchy = yes
			OR = {
				has_diarchy_type = co_emperorship
				has_diarchy_type = junior_emperorship
			}
			exists = diarch
		}
		diarch = { save_scope_as = new_holder }
	}
	else_if = {
		limit = {
			any_vassal = {
				highest_held_title_tier >= tier_county
				capital_province = { geographical_region = $OTHER_REGION$ }
			}
		}
		ordered_vassal = {
			limit = {
				highest_held_title_tier >= tier_county
				capital_province = { geographical_region = $OTHER_REGION$ }
			}
			order_by = max_military_strength
			save_scope_as = new_holder
		}
	}
	else_if = {
		limit = {
			any_powerful_family = {
				this != root.house
				exists = house_head
			}
		}
		ordered_powerful_family = {
			limit = {
				this != root.house
				exists = house_head
			}
			order_by = house_power_score
			house_head = { save_scope_as = new_holder }
		}
	}
	else = {
		ordered_vassal = {
			limit = { highest_held_title_tier >= tier_county }
			order_by = max_military_strength
			save_scope_as = new_holder
		}
	}
}

# Hands scope:new_holder your counties, your titles that lie mostly in $REGION$, and your direct vassals based there
LR_cleave_hand_over_half_effect = {
	if = { # Move your domicile out of the other half
		limit = {
			exists = domicile
			domicile.domicile_location = { geographical_region = $REGION$ }
		}
		domicile = { move_domicile = root.capital_province }
	}
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = change
		add_claim_on_loss = no
	}
	every_held_title = {
		limit = {
			tier = tier_county
			is_landless_type_title = no
			title_province = { geographical_region = $REGION$ }
			NOT = { this = root.capital_county }
		}
		change_title_holder = {
			holder = scope:new_holder
			change = scope:change
		}
	}
	every_held_title = {
		limit = {
			tier >= tier_duchy
			tier <= tier_empire
			is_landless_type_title = no
			NOT = { this = root.primary_title }
			any_de_jure_county = {
				percent >= 0.51
				title_province = { geographical_region = $REGION$ }
			}
		}
		change_title_holder = {
			holder = scope:new_holder
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = change
		add_claim_on_loss = no
	}
	every_vassal = {
		limit = {
			NOT = { this = scope:new_holder }
			highest_held_title_tier >= tier_county
			capital_province = { geographical_region = $REGION$ }
		}
		change_liege = {
			liege = scope:new_holder
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
}

# Gives scope:new_holder a title of yours, or an unheld one
LR_cleave_give_title_effect = {
	if = {
		limit = {
			OR = {
				NOT = { exists = title:$TITLE$.holder }
				title:$TITLE$.holder = root
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:$TITLE$ = {
			change_title_holder = {
				holder = scope:new_holder
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}

# Two Augusti
LR_cleave_empire_full_inner_effect = {
	# Every empire under Rome goes to the half that holds most of its counties; the two Roman Empires go to their own half
	hidden_effect = {
		title:h_roman_empire = {
			every_in_de_jure_hierarchy = {
				continue = { tier = tier_hegemony }
				limit = { tier = tier_empire }
				if = {
					limit = {
						OR = {
							this = title:e_eastern_roman_empire
							AND = {
								NOT = { this = title:e_western_roman_empire }
								any_de_jure_county = {
									percent >= 0.5
									title_province = { geographical_region = custom_theodosian_ere }
								}
							}
						}
					}
					set_de_jure_liege_title = title:h_eastern_roman_empire
				}
				else = {
					set_de_jure_liege_title = title:h_western_roman_empire
				}
			}
		}
	}

	# Your half: its hegemony carries on Rome's history
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:$MINE$ = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
		copy_title_history = title:h_roman_empire
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = {
		if = {
			limit = {
				government_has_mechanic = administrative
				title:h_roman_empire = { any_title_maa_regiment = { count >= 1 } }
			}
			title:h_roman_empire = { transfer_title_maa_ownership = title:$MINE$ }
		}
		if = {
			limit = { title:h_roman_empire = { has_variable = founded_varangian_guard } }
			title:$MINE$ = { set_variable = founded_varangian_guard }
		}
		if = {
			limit = { title:h_roman_empire = { has_variable = refounded_praetorian_guard } }
			title:$MINE$ = { set_variable = refounded_praetorian_guard }
		}
		set_primary_title_to = title:$MINE$
	}

	# A co-emperor who leaves cannot stay your diarch
	if = {
		limit = {
			has_active_diarchy = yes
			diarch ?= scope:new_holder
		}
		end_diarchy = yes
	}

	# The other half
	custom_tooltip = LR_cleave_empire_other_half_tt
	LR_cleave_hand_over_half_effect = { REGION = $OTHER_REGION$ }
	LR_cleave_give_title_effect = { TITLE = $OTHER_EMPIRE$ }
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:$OTHER$ = {
		change_title_holder = {
			holder = scope:new_holder
			change = scope:change
		}
		copy_title_history = title:h_roman_empire
	}
	resolve_title_and_vassal_change = scope:change
	scope:new_holder = {
		add_trait = augustus
		set_primary_title_to = title:$OTHER$
		if = {
			limit = { has_title = title:$OTHER_CAPITAL$ }
			set_realm_capital = title:$OTHER_CAPITAL$
		}
		if = {
			limit = { is_independent_ruler = no }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = change
				add_claim_on_loss = no
			}
			becomes_independent = { change = scope:change }
			resolve_title_and_vassal_change = scope:change
		}
	}

	destroy_title = title:h_roman_empire

	add_truce_both_ways = {
		character = scope:new_holder
		years = 10
		name = TRUCE_COLLABORATION
	}
	if = {
		limit = { government_has_mechanic = administrative }
		change_influence = {
			value = monumental_influence_gain
			multiply = 2
		}
	}

	# Rome can be restored again by reuniting the halves
	custom_tooltip = LR_cleave_empire_restorable_tt
	hidden_effect = {
		remove_list_global_variable = {
			name = unavailable_unique_decisions
			target = flag:flag_restored_roman_empire
		}
		remove_global_variable = flag_restored_roman_empire
		set_global_variable = {
			name = LR_roman_empire_cleaved
			value = root
		}
	}
}

LR_cleave_empire_full_effect = {
	if = {
		limit = { LR_cleave_keeps_east_trigger = yes }
		LR_cleave_empire_full_inner_effect = {
			MINE = h_eastern_roman_empire
			OTHER = h_western_roman_empire
			OTHER_EMPIRE = e_western_roman_empire
			OTHER_REGION = custom_theodosian_wre
			OTHER_CAPITAL = c_roma
		}
	}
	else = {
		LR_cleave_empire_full_inner_effect = {
			MINE = h_western_roman_empire
			OTHER = h_eastern_roman_empire
			OTHER_EMPIRE = e_eastern_roman_empire
			OTHER_REGION = custom_theodosian_ere
			OTHER_CAPITAL = c_byzantion
		}
	}
}

# A Caesar: Rome stays yours, the other half's empire title goes to a junior emperor who stays your vassal
LR_cleave_empire_junior_inner_effect = {
	custom_tooltip = LR_cleave_empire_caesar_tt
	LR_cleave_hand_over_half_effect = { REGION = $OTHER_REGION$ }
	LR_cleave_give_title_effect = { TITLE = $OTHER_EMPIRE$ }
	hidden_effect = {
		title:$OTHER_EMPIRE$ = { set_de_jure_liege_title = title:h_roman_empire }
	}
	scope:new_holder = {
		if = {
			limit = { has_title = title:$OTHER_EMPIRE$ }
			set_primary_title_to = title:$OTHER_EMPIRE$
		}
		if = {
			limit = { has_title = title:$OTHER_CAPITAL$ }
			set_realm_capital = title:$OTHER_CAPITAL$
		}
		add_opinion = {
			target = root
			modifier = grateful_opinion
			opinion = 40
		}
	}
	hidden_effect = {
		title:h_roman_empire = {
			set_variable = {
				name = LR_junior_caesar
				value = scope:new_holder
			}
		}
	}
}

LR_cleave_empire_junior_effect = {
	if = {
		limit = { LR_cleave_keeps_east_trigger = yes }
		LR_cleave_empire_junior_inner_effect = {
			OTHER_EMPIRE = e_western_roman_empire
			OTHER_REGION = custom_theodosian_wre
			OTHER_CAPITAL = c_roma
		}
	}
	else = {
		LR_cleave_empire_junior_inner_effect = {
			OTHER_EMPIRE = e_eastern_roman_empire
			OTHER_REGION = custom_theodosian_ere
			OTHER_CAPITAL = c_byzantion
		}
	}
}
