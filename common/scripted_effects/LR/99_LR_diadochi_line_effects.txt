# Root keeps one line and the other lines' titles are destroyed
# Usage: LR_keep_diadochi_line_effect = { LINE = seleukid }
LR_keep_diadochi_line_effect = {
	if = {
		limit = { primary_title = { LR_is_diadochi_line_title_trigger = yes } }
		LR_set_primary_diadochi_line_$LINE$_effect = yes
	}
	LR_destroy_other_diadochi_lines_$LINE$_effect = yes
}

LR_set_primary_diadochi_line_antipatrid_effect = {
	if = {
		limit = { has_title = title:h_antipatrid_hegemony }
		set_primary_title_to = title:h_antipatrid_hegemony
	}
	else_if = {
		limit = { has_title = title:e_antipatrid_empire }
		set_primary_title_to = title:e_antipatrid_empire
	}
	else_if = {
		limit = { has_title = title:k_antipatrid_kingdom }
		set_primary_title_to = title:k_antipatrid_kingdom
	}
}

			limit = {
				OR = { government_has_flag = government_is_hellenistic_government has_government = alexandrian_dynastic_government }

	}
	else_if = {
		limit = {
			has_dlc_feature = all_under_heaven
			LR_holds_diadochi_line_empire_trigger = yes
			NOT = { LR_holds_diadochi_line_hegemony_trigger = yes }

			NOR = { has_government = hellenistic_autocratic_government has_government = hellenistic_imperial_government }

			limit = {
				OR = { government_has_flag = government_is_hellenistic_government has_government = alexandrian_dynastic_government }
				highest_held_title_tier > tier_county
				NOR = { has_government = hellenistic_autocratic_government has_government = hellenistic_imperial_government }
			}

		}
