# Formable kingdoms of post-Alexander warlords (02_LR_diadochi_warlord_decisions.txt)

# Root founds TITLE and makes it their primary title unless they already hold an empire or above
# Usage: LR_create_diadochi_warlord_kingdom_effect = { TITLE = k_attalid_kingdom }
LR_create_diadochi_warlord_kingdom_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:$TITLE$ = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = {
		if = {
			limit = { highest_held_title_tier <= tier_kingdom }
			set_primary_title_to = title:$TITLE$
		}
	}
}

# Root reforms KINGDOM into EMPIRE: the kingdom becomes de jure part of the empire, which becomes the primary title
# unless root holds a hegemony
# Usage: LR_create_diadochi_warlord_empire_effect = { KINGDOM = k_attalid_kingdom EMPIRE = e_attalid_empire }
LR_create_diadochi_warlord_empire_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:$EMPIRE$ = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = {
		title:$KINGDOM$ = { set_de_jure_liege_title = title:$EMPIRE$ }
		if = {
			limit = { highest_held_title_tier <= tier_empire }
			set_primary_title_to = title:$EMPIRE$
		}
	}
}
