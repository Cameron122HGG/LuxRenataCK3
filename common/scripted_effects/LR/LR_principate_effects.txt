# LR 2026-11-19: the Principate (principate_government), designed with the user on 2026-11-19:
#   - Mechanic: republic. The Princeps is the head of the Roman Senate assembly for life. Shadowed vanilla laws
#     (common/laws/04_republic_laws.txt, regenerate with tools/principate/patch_republic_laws.py) lock the term to
#     Indefinite, the assembly to the Roman Senate and the authority to Legislative or Ceremonial.
#   - Succession: principate_succession_law (common/laws/00_succession_laws.txt), an election by the Senate
#     (common/succession_election/LR_principate_elective.txt) that favours the heir the Princeps adopted
#     (LR_principate_adopt_successor_interaction).
#   - Provinces: landed direct vassals of duchy tier or above become Principate governors (LR_principate_governor_trigger)
#     with Senatorial or Imperial province contracts (common/subject_contracts/contracts/principate.txt). Senatorial
#     governors sit in the Senate.
#   - Path: a Roman Republic's consul takes LR_accept_principate_decision (the Senate votes on it); a Princeps with a
#     Ceremonial Senate may take LR_reform_into_dominate_decision (roman_dominate_government, All Under Heaven).

# Root is a ruler who just took the Principate (a republic). Gives a seatless character a patrician family title, as vanilla's
# gdp_found_a_republic_effect (step 4) does for a new republic; characters who already have an estate keep it.
LR_principate_patrician_seat_effect = {
	# LR 2026-11-19: no longer destroys anything. Destroying the administrative noble-family titles (estates) of the Princeps and every
	# new governor inside Restore the Principate is the prime suspect for its crash (vanilla's on_title_destroyed already errors
	# on it), and the realm-window crash this was added for was the window routing (gui/hud.gui). Only a character with no seat
	# at all (e.g. a courtier made legate) gets a patrician family title.
	if = {
		limit = {
			NOT = { exists = domicile }
			NOT = { any_held_title = { is_noble_family_title = yes } }
			exists = house
		}
		save_scope_as = republican_founder # read by the vanilla NEW_PATRICIAN_TITLE_NAME
		create_dynamic_title = {
			name = NEW_PATRICIAN_TITLE_NAME
			tier = county
			type = title_type_merchant_family
			holder = scope:republican_founder
			article = DEFAULT_TITLE_NAME_ARTICLE
			save_scope_as = LR_new_patrician_title
		}
		scope:LR_new_patrician_title ?= {
			add_title_law = noble_family_succession_law
		}
	}
}

# Root is a ruler. Called first in LR_normalize_dynastic_government_effect (so on game start, on government change and
# when a liege gains a vassal): direct vassals of a Princeps become governors; a governor who no longer serves a Princeps
# goes back to feudal, which the normalizer maps to the culture's group.
LR_principate_align_vassal_effect = {
	if = {
		limit = {
			is_ruler = yes
			NOT = { has_government = principate_government }
			LR_principate_governor_trigger = yes
		}
		change_government = principate_government
		LR_principate_patrician_seat_effect = yes # LR 2026-11-19 crash fix
	}
	else_if = {
		limit = {
			is_ruler = yes
			has_government = principate_government
			exists = liege
			LR_principate_governor_trigger = no
			NAND = { # LR 2026-11-19: a Princeps' landless legate keeps the Principate
				LR_is_roman_legion_trigger = yes
				top_liege = { has_government = principate_government }
			}
		}
		change_government = feudal_government # re-normalized by on_government_change
	}
}

# Root is a ruler. LR 2026-11-19 (user): a decayed Roman empire's ruler (LR_is_decayed_roman_realm_trigger: e_byzantium,
# e_spqr, e_tadmor, e_syagria, e_constantinid, with no restored title) cannot keep the Dominate or the Principate (e.g.
# after inheriting the decayed title): the Dominate falls back to Roman Administrative (the decayed empires' government in
# history), the Principate to a plain republic. Called first in LR_normalize_dynastic_government_effect and on title gain.
LR_downgrade_upgraded_roman_government_effect = {
	# LR 2026-11-19 (user): without Roads to Power, Silk & Silver and All Under Heaven there is no Dominate or Principate. The Dominate
	# falls back to Roman Autocratic (All Under Heaven), else Roman Administrative (Roads to Power), else Classical; the Principate to a
	# plain republic (Silk & Silver), else Classical.
	if = {
		limit = {
			has_government = roman_dominate_government
			LR_has_roman_evolution_dlc_trigger = no
		}
		if = {
			limit = { has_tgp_dlc_trigger = yes }
			change_government = roman_autocratic_government
		}
		else_if = {
			limit = { has_ep3_dlc_trigger = yes }
			change_government = roman_administrative_government
		}
		else = {
			change_government = classical_government
		}
	}
	else_if = {
		limit = {
			has_government = principate_government
			LR_has_roman_evolution_dlc_trigger = no
		}
		if = {
			limit = { has_gdp_dlc_trigger = yes }
			change_government = republic_government
		}
		else = {
			change_government = classical_government
		}
	}
	else_if = {
		limit = {
			has_government = roman_dominate_government
			LR_is_decayed_roman_realm_trigger = yes
		}
		change_government = roman_administrative_government
	}
	else_if = {
		limit = {
			has_government = principate_government
			LR_is_decayed_roman_realm_trigger = yes
		}
		change_government = republic_government
	}
}

# Root is the consul of a Roman Republic. Makes root Princeps: the government, the Principate's laws and the governors.
LR_become_princeps_effect = {
	change_government = principate_government
	hidden_effect = { LR_principate_patrician_seat_effect = yes } # LR 2026-11-19 crash fix

	# Laws. LR 2026-11-19: the Empire tier too, for an emperor restoring the Principate (LR_restore_roman_principate_effect)
	if = {
		limit = { NOT = { has_realm_law = republic_tier_empire } }
		add_realm_law_skip_effects = republic_tier_empire
	}
	# The shadowed republic laws would switch most of these on their own (can_keep / should_start_with);
	# set them here so the tooltip shows them and nothing waits for the next law check.
	add_realm_law_skip_effects = principate_succession_law
	if = {
		limit = { NOT = { has_realm_law = republic_type_roman } }
		add_realm_law_skip_effects = republic_type_roman
	}
	if = {
		limit = { NOT = { has_realm_law = republic_assembly_composition_roman_senate } }
		add_realm_law_skip_effects = republic_assembly_composition_roman_senate
	}
	if = {
		limit = { NOT = { has_realm_law = republic_assembly_term_length_indefinite } }
		add_realm_law_skip_effects = republic_assembly_term_length_indefinite
	}
	if = {
		limit = { has_realm_law = republic_assembly_authority_constitutional }
		add_realm_law_skip_effects = republic_assembly_authority_legislative
	}

	# Governors
	custom_tooltip = LR_become_princeps_governors_tt
	hidden_effect = {
		every_vassal = {
			limit = { LR_principate_governor_trigger = yes }
			change_government = principate_government
			LR_principate_patrician_seat_effect = yes # LR 2026-11-19 crash fix
		}
	}

	# LR 2026-11-19 (user test: the Senate had only the Princeps and no Senate Size law). The landed vassals of duchy tier
	# and up take seats (as vanilla gdp_restore_the_roman_republic_effect seats the old assembly), and the Senate Size law
	# is set from how many sit.
	custom_tooltip = LR_become_princeps_senators_tt
	hidden_effect = {
		every_vassal_or_below = {
			limit = {
				is_landed = yes
				highest_held_title_tier >= tier_duchy
				NOT = { has_character_flag = republic_appointed_senator }
			}
			add_character_flag = republic_appointed_senator
		}
		every_vassal_or_below = {
			limit = { has_character_flag = republic_appointed_senator }
			add_to_list = LR_princeps_senators
		}
		if = {
			limit = { NOT = { has_realm_law_flag = republic_senate_size } }
			if = {
				limit = {
					any_in_list = {
						list = LR_princeps_senators
						count > { value = republic_senate_size_2_value multiply = 1.5 }
					}
				}
				add_realm_law_skip_effects = republic_senate_size_3
			}
			else_if = {
				limit = {
					any_in_list = {
						list = LR_princeps_senators
						count > { value = republic_senate_size_1_value multiply = 1.5 }
					}
				}
				add_realm_law_skip_effects = republic_senate_size_2
			}
			else = {
				add_realm_law_skip_effects = republic_senate_size_1
			}
		}
		update_assembly = yes
	}

	# Rewards and costs
	add_prestige = major_prestige_gain
	add_character_modifier = {
		modifier = LR_princeps_auctoritas_modifier
		years = 20
	}
	capital_county = { add_popular_opinion = 15 }
	every_vassal = {
		custom = LR_become_princeps_opinion_custom
		limit = { government_has_flag = government_is_republic }
		add_opinion = {
			target = root
			modifier = LR_became_princeps_opinion
		years = 10
		}
	}
}

# Root is the Princeps. Ends the Principate: the Senate is set aside and root and his governors take the Dominate
# (roman_dominate_government; the decision needs All Under Heaven, as the Dominate's can_get_government).
LR_reform_into_dominate_effect = {
	save_scope_as = dominus
	custom_tooltip = LR_reform_into_dominate_governors_tt
	custom_tooltip = LR_roman_institutions_to_dominate_tt # LR 2026-11-19 (user): institutions carry over
	hidden_effect = {
		every_vassal = {
			limit = { has_government = principate_government }
			add_to_list = LR_principate_governors
			# Senatorial governors keep a seat in the ceremonial Senate
			if = {
				limit = { vassal_contract_has_flag = principate_senatorial_province }
				add_character_flag = republic_appointed_senator
			}
		}
		cp:LR_praefectus_praetorio ?= { save_scope_as = LR_carry_0 }
		cp:LR_principate_praefectus_urbi ?= { save_scope_as = LR_carry_1 }
		cp:LR_a_rationibus ?= { save_scope_as = LR_carry_2 }
		cp:LR_ab_epistulis ?= { save_scope_as = LR_carry_3 }
	}
	change_government = roman_dominate_government
	hidden_effect = {
		every_in_list = {
			list = LR_principate_governors
			change_government = roman_dominate_government
			add_opinion = {
				target = scope:dominus
				modifier = LR_senate_set_aside_opinion
			years = 15
			}
		}
		remove_variable = principate_adopted_heir
		# The Princeps' ministers take the matching Dominate ministries
		if = {
			limit = { exists = scope:LR_carry_0 }
			assign_councillor_type = {
				type = LR_magister_militum
				remove_existing_councillor = yes
				target = scope:LR_carry_0
			}
		}
		if = {
			limit = { exists = scope:LR_carry_1 }
			assign_councillor_type = {
				type = LR_praefectus_urbi
				remove_existing_councillor = yes
				target = scope:LR_carry_1
			}
		}
		if = {
			limit = { exists = scope:LR_carry_2 }
			assign_councillor_type = {
				type = LR_magister_officiorum
				remove_existing_councillor = yes
				target = scope:LR_carry_2
			}
		}
		if = {
			limit = { exists = scope:LR_carry_3 }
			assign_councillor_type = {
				type = LR_magister_memoriae
				remove_existing_councillor = yes
				target = scope:LR_carry_3
			}
		}
		if = {
			limit = { NOT = { has_realm_law = LR_dominate_ceremonial_senate_law } }
			add_realm_law_skip_effects = LR_dominate_ceremonial_senate_law
		}
	}
	add_prestige = major_prestige_gain
	add_character_modifier = {
		modifier = LR_dominus_reform_modifier
		years = 15
	}
	capital_county = { add_popular_opinion = -15 }
}

# Root is the Princeps, scope:new_heir the adopted successor (LR_principate_adopt_successor_interaction)
LR_principate_adopt_successor_effect = {
	if = {
		limit = {
			var:principate_adopted_heir ?= { is_alive = yes }
		}
		var:principate_adopted_heir = {
			save_scope_as = old_heir
			remove_character_modifier = LR_principate_adopted_heir_modifier
			add_opinion = {
				target = root
				modifier = LR_principate_disinherited_opinion
			years = 15
			}
		}
	}
	set_variable = {
		name = principate_adopted_heir
		value = scope:new_heir
	}
	scope:new_heir = {
		add_character_modifier = LR_principate_adopted_heir_modifier
		add_opinion = {
			target = root
			modifier = LR_principate_adopted_opinion
		years = 20
		}
	}
	# Sons and daughters passed over for an outsider resent it
	every_child = {
		limit = {
			is_adult = yes
			NOT = { this = scope:new_heir }
			NOT = { scope:new_heir = { is_child_of = root } }
		}
		custom = LR_principate_passed_over_children_custom
		add_opinion = {
			target = root
			modifier = LR_principate_passed_over_opinion
		years = 10
		}
	}
}

# Root is a Roman Autocratic emperor of hegemony tier (LR_adopt_roman_dominate_decision). Root and the realm's Roman
# Autocratic governors take up the Dominate (LR's realm code keeps same-group administrations, so they are moved here).
LR_adopt_roman_dominate_effect = {
	save_scope_as = dominus
	every_vassal_or_below = {
		limit = { has_government = roman_autocratic_government }
		add_to_list = LR_roman_autocratic_governors
	}
	change_government = roman_dominate_government
	custom_tooltip = LR_adopt_roman_dominate_governors_tt
	hidden_effect = {
		every_in_list = {
			list = LR_roman_autocratic_governors
			limit = { LR_can_have_upgraded_roman_government_trigger = yes }
			change_government = roman_dominate_government
		}
		if = {
			limit = { NOT = { has_realm_law = LR_dominate_ceremonial_senate_law } }
			add_realm_law_skip_effects = LR_dominate_ceremonial_senate_law
		}
	}
	add_prestige = major_prestige_gain
	add_character_modifier = {
		modifier = LR_dominus_reform_modifier
		years = 15
	}
}

# Root is a Roman Autocratic emperor of hegemony tier (LR_restore_roman_principate_decision): the Senate is given back its
# place and root rules as Princeps. The great administrative vassals become Principate governors and the lesser ones go
# feudal (LR_align_admin_vassals_of_top_liege_effect on the government change).
LR_restore_roman_principate_effect = {
	# LR 2026-11-19 (user): institutions carry over. A Dominus' ministers take the matching Principate seats; the ceremonial
	# Senate's members (senators and great governors) sit in the restored Senate.
	if = {
		limit = { has_government = roman_dominate_government }
		custom_tooltip = LR_roman_institutions_to_principate_tt
		hidden_effect = {
			cp:LR_magister_militum ?= { save_scope_as = LR_carry_0 }
			cp:LR_praefectus_urbi ?= { save_scope_as = LR_carry_1 }
			cp:LR_magister_officiorum ?= { save_scope_as = LR_carry_2 }
			cp:LR_magister_memoriae ?= { save_scope_as = LR_carry_3 }
			every_vassal_or_below = {
				limit = {
					is_landed = yes
					highest_held_title_tier >= tier_kingdom
				}
				add_character_flag = republic_appointed_senator
			}
		}
	}
	LR_become_princeps_effect = yes
	hidden_effect = {
		if = {
			limit = { exists = scope:LR_carry_0 }
			assign_councillor_type = {
				type = LR_praefectus_praetorio
				remove_existing_councillor = yes
				target = scope:LR_carry_0
			}
		}
		if = {
			limit = { exists = scope:LR_carry_1 }
			assign_councillor_type = {
				type = LR_principate_praefectus_urbi
				remove_existing_councillor = yes
				target = scope:LR_carry_1
			}
		}
		if = {
			limit = { exists = scope:LR_carry_2 }
			assign_councillor_type = {
				type = LR_a_rationibus
				remove_existing_councillor = yes
				target = scope:LR_carry_2
			}
		}
		if = {
			limit = { exists = scope:LR_carry_3 }
			assign_councillor_type = {
				type = LR_ab_epistulis
				remove_existing_councillor = yes
				target = scope:LR_carry_3
			}
		}
		update_assembly = yes
	}
}
