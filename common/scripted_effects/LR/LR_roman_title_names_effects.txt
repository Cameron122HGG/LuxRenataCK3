# Roman title names (game rules name_of_byzantium / name_of_spqr), 2026-11-19
#
# Every Roman title (LR_roman_named_title_trigger) is named by its holder's culture:
#   latin   - roman, western_roman, or speaks language_latin / language_western_roman
#   greek   - heritage_byzantine / heritage_hellenistic, or speaks language_greek
#   foreign - anyone else (English exonyms)
# The two game rules can fix the style of the imperial titles of their half (LR_roman_named_title_east/west_trigger);
# their default option "dynamic" leaves them dynamic. "default_name_of_*" = the title's own loc.
#
# Variable LR_roman_title_name on a title = the style flag it was named with. The title is renamed only when
# the style changes, or when something else (e.g. vanilla's on_vassal_change) dropped the custom name.
# A Roman title named by LR_dynastic_realm_names is reset first; the dynastic system skips Roman titles.
# Generated by tools/roman_title_names/gen_roman_title_names.py (936 workspace). Edit the generator, not this file.

# scope = a Roman title
LR_roman_title_names_refresh_effect = {
	if = {
		limit = { exists = holder }
		LR_roman_title_names_pick_style_effect = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = LR_roman_title_name }
					NOT = { var:LR_roman_title_name = scope:lr_rtn_style }
					AND = {
						NOT = { scope:lr_rtn_style = flag:vanilla }
						has_custom_title_name = no
					}
				}
			}
			if = {
				limit = { has_variable = LR_dynastic_realm_name }
				LR_dynastic_realm_names_reset_title_effect = yes
			}
			set_variable = {
				name = LR_roman_title_name
				value = scope:lr_rtn_style
			}
			LR_roman_title_names_apply_effect = yes
		}
	}
	else_if = {
		limit = { has_variable = LR_roman_title_name }
		reset_title_name = yes
		remove_variable = LR_roman_title_name
	}
}

# scope = a Roman title that forces a new name next refresh
LR_roman_title_names_force_refresh_effect = {
	if = {
		limit = { has_variable = LR_roman_title_name }
		remove_variable = LR_roman_title_name
	}
	LR_roman_title_names_refresh_effect = yes
}

# root or any scope = a character. Refreshes every Roman title they hold
LR_roman_title_names_refresh_held_effect = {
	every_held_title = {
		limit = { LR_roman_named_title_trigger = yes }
		LR_roman_title_names_refresh_effect = yes
	}
}
