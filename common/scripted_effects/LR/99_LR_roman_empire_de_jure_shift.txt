LR_restore_western_hegemony_decision_effect = {
	save_scope_as = root_scope
	add_trait = augustus
	give_nickname = nick_the_great
	
	change_influence = {
		value = massive_influence_gain
		multiply = 5
	}
	
	custom_tooltip = {
		text = every_county_in_theodosian_wre_region
	}
	
	add_legitimacy = {
		value = massive_legitimacy_gain
		multiply = 2
	}
	
	dynasty ?= {
		add_dynasty_modifier = retaken_western_provinces_modifier
		add_dynasty_prestige = 2500
	}
	
	if = {
		limit = {
			culture = { has_cultural_parameter = holds_triumphs }
		}
		custom_tooltip = {
			text = free_triumph_tooltip
			add_character_flag = free_triumph_flag
		}
	}
	
	gain_legitimizing_legend_seed_tooltip_effect = yes

		legend_seed_great_deed_dynasty_effect = yes 
		
		# Creating the Hegemony of Western Roman Empire
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:h_western_roman_empire = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
			copy_title_history = title:e_western_roman_empire
		}
		resolve_title_and_vassal_change = scope:change

		every_held_title = { #Should destroy all other Empires owned at the time.
			title_tier = empire
			root = { destroy_title = prev }
		}

		# Creating the Praetorian Prefectures

		# Illyricum
		# Shift the Kingdoms
		every_county_in_region = {
			region = custom_roman_prefecture_illyricum
			kingdom = { add_to_list = custom_roman_prefecture_illyricum_list }
		}	
		every_in_list = {
			list = custom_roman_prefecture_illyricum_list
			limit = {
				NOT = { de_jure_liege = title:e_praetorian_prefecture_illyricum }
			}
			set_de_jure_liege_title = title:e_praetorian_prefecture_illyricum
		}
		title:e_praetorian_prefecture_illyricum = { set_de_jure_liege_title = title:h_western_roman_empire }

		#Italia

		every_county_in_region = {
			region = custom_western_roman_imperial_core
			kingdom = { add_to_list = custom_western_roman_imperial_core }
		}	
		every_in_list = {
			list = custom_western_roman_imperial_core
			limit = {
				NOT = { de_jure_liege = title:e_praetorian_prefecture_italia }
			}
			set_de_jure_liege_title = title:e_praetorian_prefecture_italia
		}
		title:e_praetorian_prefecture_italia = { set_de_jure_liege_title = title:h_western_roman_empire }

		#Gallia
		# Shift the Kingdoms
		every_county_in_region = { #Galliarum
			region = custom_roman_lugdunensis
			kingdom = { add_to_list = custom_roman_lugdunensis_list }
		}	
		every_county_in_region = { #Viennensis
			region = custom_roman_viennensis
			kingdom = { add_to_list = custom_roman_lugdunensis_list }
		}
		every_in_list = {
			list = custom_roman_lugdunensis_list
			limit = {
				NOT = { de_jure_liege = title:e_praetorian_prefecture_gallia }
			}
			set_de_jure_liege_title = title:e_praetorian_prefecture_gallia
		}

		limit = { has_title = title:e_western_roman_empire }
		title:k_diocese_illyricum = { set_de_jure_liege_title = title:e_western_roman_empire }
		title:k_diocese_dardania = { set_de_jure_liege_title = title:e_western_roman_empire }

### Region-based diocese restoration
# Makes every duchy in a geographical region de jure part of a diocese kingdom.
# Usage: LR_move_region_duchies_to_diocese_effect = { REGION = custom_roman_macedonia DIOCESE = k_diocese_macedonia }
LR_move_region_duchies_to_diocese_effect = {
	every_county_in_region = {
		region = $REGION$
		duchy = {
			if = {
				limit = {
					NOT = { is_in_list = $REGION$_duchies }
				}
				add_to_temporary_list = $REGION$_duchies
			}
		}
	}
	every_in_list = {
		list = $REGION$_duchies
		limit = {
			NOT = { de_jure_liege = title:$DIOCESE$ }
		}
		set_de_jure_liege_title = title:$DIOCESE$
	}
	# Late-Roman provinces: counties move from the old duchies into the diocese's province duchies (where defined)
	LR_assign_roman_provinces_effect = { DIOCESE = $DIOCESE$ }
}

	LR_move_region_duchies_to_diocese_effect = { REGION = custom_roman_macedonia DIOCESE = k_diocese_macedonia }
