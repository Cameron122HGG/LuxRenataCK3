		# Creating the Hegemony of Western Roman Empire
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:e_western_roman_empire = {
			change_title_holder = {
				holder = root

			}

			title_tier = empire
			root = { destroy_title = prev }

			region = custom_roman_italia_suburbicaria
			duchy = { add_to_list = custom_roman_italia_suburbicaria_list }
		}
		every_in_list = {
			list = custom_roman_italia_suburbicaria_list
			limit = {
				NOT = { de_jure_liege = title:k_diocese_italia_suburbicaria }
			}
			set_de_jure_liege_title = title:k_diocese_italia_suburbicaria
		}
	}

	# Destroy useless kingdoms
	if = {
		limit = { exists = title:k_lombardy.holder }
		destroy_title = title:k_lombardy
	}
	if = {
		limit = { exists = title:k_urbs.holder }
		destroy_title = title:k_urbs
	}
	if = {
		limit = { exists = title:k_sicily.holder }
		destroy_title = title:k_sicily
	}
	if = {
		limit = { exists = title:k_sardinia.holder }
		destroy_title = title:k_sardinia
	}
	if = {
		limit = { exists = title:k_mediterranean_sea.holder }
		destroy_title = title:k_mediterranean_sea
	}
	if = {
		limit = { exists = title:k_genoa.holder }
		destroy_title = title:k_genoa
	}	
	if = {
		limit = { exists = title:k_venice.holder }
		destroy_title = title:k_venice
	}	
	if = {
		limit = { exists = title:k_austria.holder }
		destroy_title = title:k_austria
	}		

	hidden_effect = {
		#Fix. Carinthia. For. Reasons.
		title:d_carinthia = { set_de_jure_liege_title = title:k_diocese_italia_annonaria }

		every_held_title = { #Should destroy all kingdoms owned at the time.
			limit = {

		}
