LR_restore_western_hegemony_decision_effect = {
	save_scope_as = root_scope
	add_trait = augustus
	give_nickname = nick_the_great
	
	change_influence = {
		value = massive_influence_gain
		multiply = 5
	}
	
	custom_tooltip = {
		text = every_county_in_theodosian_wre_region
	}
	
	add_legitimacy = {
		value = massive_legitimacy_gain
		multiply = 2
	}
	
	dynasty ?= {
		add_dynasty_modifier = retaken_western_provinces_modifier
		add_dynasty_prestige = 2500
	}
	
	if = {
		limit = {
			culture = { has_cultural_parameter = holds_triumphs }
		}
		custom_tooltip = {
			text = free_triumph_tooltip
			add_character_flag = free_triumph_flag
		}
	}
	
	gain_legitimizing_legend_seed_tooltip_effect = yes
	
	hidden_effect = { 
		legend_seed_great_deed_dynasty_effect = yes 
		
		# Creating the Hegemony of Western Roman Empire
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:h_western_roman_empire = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
			copy_title_history = title:e_western_roman_empire
		}
		resolve_title_and_vassal_change = scope:change

		every_held_title = { # The old empire titles go; prefectures, exarchates and landless offices stay (LR 2026-11-06)
			limit = {
				tier = tier_empire
				is_landless_type_title = no
				LR_is_roman_prefecture_or_exarchate_title_trigger = no
			}
			root = { destroy_title = prev }
		}

		# LR 2026-11-06 (user): only restored dioceses move, into the three Theodosian prefectures (Italia, Gallia,
		# Illyricum); warlord kingdoms stay where they are until their diocese is restored
		LR_sort_existing_western_dioceses_effect = yes
	}

	every_vassal = {
		custom = custom.every_admin_vassal
		limit = { government_has_mechanic = administrative }
		add_opinion = {
			target = scope:root_scope
			modifier = admiration_opinion
			opinion = 20
		}
	}
}

restore_western_roman_provinces_system_decision_effect = {
	#Sort the diocese
	if = {
		limit = {
			OR = {
				has_title = title:h_western_roman_empire
				has_title = title:h_roman_empire
			}
		}
		LR_sort_existing_western_dioceses_effect = yes # LR 2026-11-06: the three Theodosian prefectures
	}
	else_if = { 
		limit = { has_title = title:e_spqr }
		title:k_diocese_illyricum = { set_de_jure_liege_title = title:e_spqr }
		title:k_diocese_dardania = { set_de_jure_liege_title = title:e_spqr }
		title:k_diocese_italia_annonaria = { set_de_jure_liege_title = title:e_spqr }
		title:k_diocese_italia_suburbicaria = { set_de_jure_liege_title = title:e_spqr }
	}
	else_if = { 
		limit = { has_title = title:e_western_roman_empire }
		title:k_diocese_illyricum = { set_de_jure_liege_title = title:e_western_roman_empire }
		title:k_diocese_dardania = { set_de_jure_liege_title = title:e_western_roman_empire }
		title:k_diocese_italia_annonaria = { set_de_jure_liege_title = title:e_western_roman_empire }
		title:k_diocese_italia_suburbicaria = { set_de_jure_liege_title = title:e_western_roman_empire }
	}
	# LR: no automatic de jure shift. Taking the administration decision only makes you the restored empire and sorts
	# any dioceses that already exist under it; each diocese (and its provinces, Commands and prefecture) is restored
	# by its own prefecture decision. The old duchy move and kingdom destruction were removed here (backup: pre_admin_no_shift).
}

	if = {
		limit = {
			title:k_italy = { is_titular = yes }
		}
	}

	# LR 2026-11-06: once a hegemony or Rome exists, the diocese goes to its prefecture under it

	# whose capital is now de jure in the diocese is destroyed; independent rivals keep their crowns.
	LR_destroy_kingdoms_in_restored_diocese_effect = { DIOCESE = k_diocese_italia_suburbicaria }
}

### Region-based diocese restoration
# Makes a region de jure part of a diocese kingdom: its province counties, or (without provinces) every duchy in it.
# Usage: LR_move_region_duchies_to_diocese_effect = { REGION = custom_roman_macedonia DIOCESE = k_diocese_macedonia }
LR_move_region_duchies_to_diocese_effect = {
	# Dioceses with late-Roman provinces take exactly their provinces' counties (below), so non-Roman counties stay
	# in their own duchies; the others still take every duchy with a county in the region
	if = {
		limit = { NOT = { LR_diocese_has_roman_provinces_trigger = { DIOCESE = $DIOCESE$ } } }
		every_county_in_region = {
			region = $REGION$
			duchy = {
				if = {
					limit = {
						NOT = { is_in_list = $REGION$_duchies }
					}
					add_to_temporary_list = $REGION$_duchies
				}
			}
		}
		every_in_list = {
			list = $REGION$_duchies
			limit = {
				NOT = { de_jure_liege = title:$DIOCESE$ }
			}
			set_de_jure_liege_title = title:$DIOCESE$
		}
	}
	# Late-Roman provinces: counties move from the old duchies into the diocese's province duchies (where defined)
	LR_assign_roman_provinces_effect = { DIOCESE = $DIOCESE$ }
}

# Under a Roman hegemony, puts an eastern diocese in its prefecture: Pontica for Pontus, Armenia and Gothia once
# LR_split_pontic_prefecture_decision has been taken, otherwise Macedonia. Root holds the hegemony.
# Usage: LR_sort_macedonian_diocese_prefecture_effect = { DIOCESE = k_diocese_pontus }
LR_sort_macedonian_diocese_prefecture_effect = {
	if = {
		limit = {
			OR = {
				has_title = title:h_eastern_roman_empire
				has_title = title:h_roman_empire
			}
		}
		if = {
			limit = { has_title = title:h_eastern_roman_empire }
			title:h_eastern_roman_empire = { save_scope_as = prefecture_hegemony }
		}
		else = {
			title:h_roman_empire = { save_scope_as = prefecture_hegemony }
		}
		if = {
			limit = {
				has_global_variable = LR_pontic_prefecture_split
				title:$DIOCESE$ = {
					OR = {
						this = title:k_diocese_pontus
						this = title:k_diocese_armenia
						this = title:k_diocese_gothia
					}
				}
			}
			title:$DIOCESE$ = { set_de_jure_liege_title = title:e_praetorian_prefecture_pontica }
			title:e_praetorian_prefecture_pontica = { set_de_jure_liege_title = scope:prefecture_hegemony }
		}
		else = {
			title:$DIOCESE$ = { set_de_jure_liege_title = title:e_praetorian_prefecture_macedonia }
			title:e_praetorian_prefecture_macedonia = { set_de_jure_liege_title = scope:prefecture_hegemony }
		}
	}
}

# LR 2026-11-06 (user): after a diocese is restored, destroy every landed kingdom held by the restorer or anyone in
# his realm whose capital county is now de jure in that diocese (its heartland became Roman). Independent rulers keep
# their kingdoms; landless titles are spared. Character scope (the restorer). DIOCESE = k_diocese_*
LR_destroy_kingdoms_in_restored_diocese_effect = {
	every_held_title = {
		limit = { LR_kingdom_absorbed_by_diocese_trigger = { DIOCESE = $DIOCESE$ } }
		add_to_temporary_list = LR_absorbed_kingdoms
	}
	every_sub_realm_title = {
		limit = { LR_kingdom_absorbed_by_diocese_trigger = { DIOCESE = $DIOCESE$ } }
		add_to_temporary_list = LR_absorbed_kingdoms
	}
	every_in_list = {
		list = LR_absorbed_kingdoms
		save_temporary_scope_as = absorbed_kingdom
		holder ?= { destroy_title = scope:absorbed_kingdom }
	}
}
