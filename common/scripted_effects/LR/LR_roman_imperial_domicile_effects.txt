# LR 2026-11-19 (user): root = a character (LR_roman_imperial_domicile.0001). Rebuilds their family seat as the type of their
# government: the domus for the Principate and the Dominate; when leaving them, the seat of the government they fell back to.
# A government not listed keeps the domus (add it here if one turns up).
LR_convert_roman_imperial_domicile_effect = {
	switch = {
		trigger = has_government
		principate_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = roman_imperial_domicile NEW_GOVERNMENT_TYPE = principate_government }
		}
		roman_dominate_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = roman_imperial_domicile NEW_GOVERNMENT_TYPE = roman_dominate_government }
		}
		# the classical group's estate
		roman_autocratic_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = roman_autocratic_government }
		}
		roman_administrative_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = roman_administrative_government }
		}
		roman_dynastic_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = roman_dynastic_government }
		}
		alexandrian_administrative_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = alexandrian_administrative_government }
		}
		alexandrian_dynastic_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = alexandrian_dynastic_government }
		}
		hellenistic_autocratic_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = hellenistic_autocratic_government }
		}
		hellenistic_imperial_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = classical_group_estate NEW_GOVERNMENT_TYPE = hellenistic_imperial_government }
		}
		# vanilla seats
		administrative_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = estate NEW_GOVERNMENT_TYPE = administrative_government }
		}
		republic_government = {
			tgp_domicile_conversion_when_changing_government_type = { NEW_DOMICILE_TYPE = merchant_company NEW_GOVERNMENT_TYPE = republic_government }
		}
	}
}
