# LR 2026-11-06 (user): the HRE as a hegemony of translatio imperii, after the Lux Invicta / Lux Renata lore
# FUTURE (user 2026-11-06): the Sacred Roman Empire stays separate from the Roman World for now; a translatio imperii
# mechanic tying it to the Roman World (and later its own title set) is to come.

# The Carolingian Empire becomes the Holy Roman Empire (e_hre). Character scope (the emperor).
LR_proclaim_holy_roman_empire_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:e_hre = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
		copy_title_history = title:e_carolingian_empire
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = {
		title:e_carolingian_empire = {
			every_in_de_jure_hierarchy = {
				continue = { tier >= tier_empire }
				limit = { tier = tier_kingdom }
				set_de_jure_liege_title = title:e_hre
			}
		}
		set_primary_title_to = title:e_hre
	}
	if = {
		limit = { has_title = title:e_carolingian_empire }
		destroy_title = title:e_carolingian_empire
	}
}

# The Holy Roman Empire as a hegemony (h_hre). e_hre is dissolved into it: its Carolingian kingdoms go to the
# Carolingian Empire, its other Gaulish kingdoms to Gaul (the Sacral Kingdom or the High Kingdom of the Franks) and the
# rest to Germany: three empires under the hegemony (LR 2026-11-22; before, the Carolingian Empire was dissolved too).
# Character scope (the new emperor).
LR_found_holy_roman_hegemony_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:h_hre = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
		if = {
			limit = { exists = title:e_hre.holder }
			copy_title_history = title:e_hre
		}
		else_if = { # LR 2026-11-22: Reform the Empire, the Sacral Empire becomes the HRE
			limit = { root = { has_title = title:h_sacral_empire } }
			copy_title_history = title:h_sacral_empire
		}
		add_title_law = princely_elective_succession_law # the Electors, as the old e_hre had
	}
	resolve_title_and_vassal_change = scope:change
	custom_tooltip = LR_found_holy_roman_hegemony_dejure_tt
	hidden_effect = {
		# LR 2026-11-22 (Frankish title sets, agreed): the Carolingian Empire stays, as the third empire under the
		# hegemony beside Gaul and Germany (the Treaty of Verdun's three parts). Its regna (and the warlord kingdoms they
		# replace) go back to it from e_hre; e_hre's other kingdoms are sorted to Gaul or Germany as before.
		title:e_hre = {
			every_in_de_jure_hierarchy = {
				continue = { tier >= tier_empire }
				limit = { tier = tier_kingdom }
				if = {
					limit = { LR_carolingian_kingdom_trigger = yes }
					set_de_jure_liege_title = title:e_carolingian_empire
				}
				else = {
					LR_hre_sort_kingdom_effect = yes
				}
			}
		}
		title:e_carolingian_empire = { set_de_jure_liege_title = title:h_hre }
		# LR 2026-11-22: after the Mayor's Crown, the High Kingdom of the Franks stands in for the Sacral Kingdom
		if = {
			limit = { LR_tgp_sacral_gaul_frankish_crown_rules_gaul_trigger = yes }
			title:e_frankish_empire = { set_de_jure_liege_title = title:h_hre }
		}
		else = {
			title:e_sacral_kingdom_gaul = { set_de_jure_liege_title = title:h_hre }
		}
		title:e_germany = { set_de_jure_liege_title = title:h_hre }
		set_primary_title_to = title:h_hre
	}
	if = {
		limit = { has_title = title:e_hre }
		destroy_title = title:e_hre
	}
	# LR 2026-11-22 (one crown at a time): the Sacral Empire gives way, and the regna already organised are laid
	# out over their old duchies (the princely layout)
	if = {
		limit = { has_title = title:h_sacral_empire }
		destroy_title = title:h_sacral_empire
	}
	hidden_effect = {
		LR_frankish_crown_relayout_effect = { SET = merovingian GRANT = no }
		LR_title_set_reorganise_realm_effect = yes
	}
	# (LR 2026-11-22: e_carolingian_empire is no longer destroyed here; it stays under the hegemony)
	# The Electors: keep the ones the papal route chose, or pick them now (the Karling route chooses none), and
	# write them onto h_hre for the tooltip (the old variables were on the destroyed e_hre)
	if = {
		limit = { NOT = { has_global_variable_list = hre_elector_list } }
		hre_assign_dynamic_prince_electors_effect = yes
	}
	else = {
		hre_elector_list_save_effect = yes
	}
}

# LR 2026-11-06: saves the Holy Roman emperor (holder of h_hre, else of e_hre) as scope:$NAME$. Any scope.
LR_save_holy_roman_emperor_effect = {
	if = {
		limit = { exists = title:h_hre.holder }
		title:h_hre.holder = { save_scope_as = $NAME$ }
	}
	else_if = {
		limit = { exists = title:e_hre.holder }
		title:e_hre.holder = { save_scope_as = $NAME$ }
	}
}

# Title scope (a kingdom): Gaulish kingdoms to the Sacral Kingdom of Gaul, the rest to Germany
LR_hre_sort_kingdom_effect = {
	if = {
		limit = {
			title_capital_county ?= { title_province = { geographical_region = LR_world_europe_gaul } }
		}
		if = {
			limit = { LR_tgp_sacral_gaul_frankish_crown_rules_gaul_trigger = yes }
			set_de_jure_liege_title = title:e_frankish_empire
		}
		else = {
			set_de_jure_liege_title = title:e_sacral_kingdom_gaul
		}
	}
	else = {
		set_de_jure_liege_title = title:e_germany
	}
}

# LR 2026-11-06: scope:LR_hre_crown is the live Holy Roman crown, h_hre if held, else e_hre. Any scope.
LR_save_holy_roman_crown_effect = {
	if = {
		limit = { exists = title:h_hre.holder }
		title:h_hre = { save_scope_as = LR_hre_crown }
	}
	else = {
		title:e_hre = { save_scope_as = LR_hre_crown }
	}
}
