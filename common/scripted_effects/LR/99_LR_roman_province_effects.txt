# Late-Roman provinces: move a restored diocese's counties into its province duchies (see 01_LR_roman_provinces.txt)

# Root is the restoring ruler. Called at the end of LR_move_region_duchies_to_diocese_effect.
LR_assign_roman_provinces_effect = {
	if = {
		limit = { title:$DIOCESE$ = { this = title:k_diocese_italia_suburbicaria } }
		LR_roman_provinces_diocese_italia_suburbicaria_effect = yes
	}
	else_if = {
		limit = { title:$DIOCESE$ = { this = title:k_diocese_italia_annonaria } }
		LR_roman_provinces_diocese_italia_annonaria_effect = yes
	}
	else_if = {
		limit = { title:$DIOCESE$ = { this = title:k_diocese_africa } }
		LR_roman_provinces_diocese_africa_effect = yes
	}
	else_if = {
		limit = { title:$DIOCESE$ = { this = title:k_diocese_mauretania } }
		LR_roman_provinces_diocese_mauretania_effect = yes
	}
}
