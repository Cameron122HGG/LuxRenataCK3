# LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
# root = a ruler. Called from LR_normalize_dynastic_government_effect (every government change, title gain and yearly).
# Universum Regnum: the holder of h_hre; otherwise western_imperial
LR_western_universum_government_effect = {
	if = {
		limit = {
			LR_gradus_has_auh_trigger = no
			has_government = western_universum_government
		}
		change_government = western_imperial_government
	}
	else_if = {
		limit = {
			LR_gradus_has_auh_trigger = yes
			top_liege = this
			has_title = title:h_hre
			OR = {
				has_government = western_imperial_government
				has_government = western_administrative_government
			}
		}
		change_government = western_universum_government
	}
	else_if = {
		limit = {
			has_government = western_universum_government
			NOR = {
				has_title = title:h_hre
			}
		}
		change_government = western_imperial_government
	}
}

# Alma: the holder of h_mamlekhet_yisrael or after Proclaim the Everlasting Kingdom while at hegemony tier; otherwise mamlakah_imperial
LR_mamlakah_alma_government_effect = {
	if = {
		limit = {
			LR_gradus_has_auh_trigger = no
			has_government = mamlakah_alma_government
		}
		change_government = mamlakah_imperial_government
	}
	else_if = {
		limit = {
			LR_gradus_has_auh_trigger = yes
			top_liege = this
			has_title = title:h_mamlekhet_yisrael
			OR = {
				has_government = mamlakah_imperial_government
				has_government = mamlakah_administrative_government
			}
		}
		change_government = mamlakah_alma_government
	}
	else_if = {
		limit = {
			has_government = mamlakah_alma_government
			NOR = {
				has_title = title:h_mamlekhet_yisrael
				AND = {
					has_character_flag = LR_mamlakah_alma_proclaimed
					highest_held_title_tier >= tier_hegemony
				}
			}
		}
		change_government = mamlakah_imperial_government
	}
}
