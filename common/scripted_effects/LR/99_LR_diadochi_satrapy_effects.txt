# Antigonids and Antipatrids found a kingdom; their satrapies need the empire above it.
# Usage: LR_diadochi_ensure_successor_empire_effect = { KINGDOM = k_antigonid_kingdom EMPIRE = e_antigonid_empire }
LR_diadochi_ensure_successor_empire_effect = {
	if = {
		limit = { NOT = { has_title = title:$EMPIRE$ } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = empire_change
			add_claim_on_loss = no
		}
		title:$EMPIRE$ = {
			change_title_holder = {
				holder = root
				change = scope:empire_change
			}
		}
		resolve_title_and_vassal_change = scope:empire_change
		hidden_effect = { set_primary_title_to = title:$EMPIRE$ }
	}
	if = {
		limit = { has_title = title:$KINGDOM$ }
		title:$KINGDOM$ = { set_de_jure_liege_title = title:$EMPIRE$ }
	}
}

# Root is the restoring ruler: creates the satrapy and lays out its hyparchies. The satrapy goes into its strategia
# under h_hellenistic_empire if root holds it, an Indian satrapy into its strategia under h_greek_india if root holds that,
# and otherwise under EMPIRE (the successor empire).
# Usage: LR_restore_satrapy_effect = { SATRAPY = k_satrapy_syria EMPIRE = e_seleukid_empire }
LR_restore_satrapy_effect = {
	title:$SATRAPY$ = { save_scope_as = restored_satrapy }
	title:$EMPIRE$ = { save_scope_as = satrapy_empire }
	if = {
		limit = { has_title = title:h_hellenistic_empire }
		LR_satrapy_strategia_$SATRAPY$_effect = { HEGEMONY = h_hellenistic_empire }
	}
	else_if = {
		limit = {
			has_title = title:h_greek_india
			LR_is_indian_satrapy_trigger = { SATRAPY = $SATRAPY$ }
		}
		LR_satrapy_strategia_$SATRAPY$_effect = { HEGEMONY = h_greek_india }
	}
	else = {
		title:$SATRAPY$ = { set_de_jure_liege_title = title:$EMPIRE$ }
	}
	LR_satrapy_hyparchies_$SATRAPY$_effect = yes
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:$SATRAPY$ = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change
}

# Debug (LR_debug_take_satrapy_lands_decision): root takes every county of a satrapy that is outside their realm
# Usage: LR_debug_take_satrapy_lands_effect = { REGION = custom_satrapy_syria }
LR_debug_take_satrapy_lands_effect = {
	every_county_in_region = {
		region = $REGION$
		limit = {
			NOR = {
				holder = root
				holder.top_liege = root
			}
		}
		add_to_temporary_list = LR_debug_$REGION$_counties
	}
	every_in_list = {
		list = LR_debug_$REGION$_counties
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = change
			add_claim_on_loss = no
		}
		change_title_holder_include_vassals = {
			holder = root
			change = scope:change
		}
		resolve_title_and_vassal_change = scope:change
	}
}
