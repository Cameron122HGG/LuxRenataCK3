# Successor satrapies (see 03_LR_diadochi_hyparchies.txt and 01_LR_diadochi_satrapy_decisions.txt)

# Antigonids and Antipatrids found a kingdom; their satrapies need the empire above it.
# Usage: LR_diadochi_ensure_successor_empire_effect = { KINGDOM = k_antigonid_kingdom EMPIRE = e_antigonid_empire }
LR_diadochi_ensure_successor_empire_effect = {
	if = {
		limit = { NOT = { has_title = title:$EMPIRE$ } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = empire_change
			add_claim_on_loss = no
		}
		title:$EMPIRE$ = {
			change_title_holder = {
				holder = root
				change = scope:empire_change
			}
		}
		resolve_title_and_vassal_change = scope:empire_change
		hidden_effect = { set_primary_title_to = title:$EMPIRE$ }
	}
	if = {
		limit = { has_title = title:$KINGDOM$ }
		title:$KINGDOM$ = { set_de_jure_liege_title = title:$EMPIRE$ }
	}
}

# Root is the restoring ruler: creates the satrapy under the successor empire and lays out its hyparchies.
# Usage: LR_restore_satrapy_effect = { SATRAPY = k_satrapy_syria EMPIRE = e_seleukid_empire }
LR_restore_satrapy_effect = {
	title:$SATRAPY$ = { save_scope_as = restored_satrapy }
	title:$EMPIRE$ = { save_scope_as = satrapy_empire }
	title:$SATRAPY$ = { set_de_jure_liege_title = title:$EMPIRE$ }
	LR_satrapy_hyparchies_$SATRAPY$_effect = yes
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:$SATRAPY$ = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change
}
