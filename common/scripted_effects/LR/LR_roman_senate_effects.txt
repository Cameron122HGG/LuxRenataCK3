# LR 2026-11-25 Reconvene the Senate (tools/roman_senate/build_roman_senate.py)

# root = the Republic's head. HALF = western / eastern, SEAT = the Senate's county
LR_roman_senate_establish_effect = {
	title:e_$HALF$_roman_republic = {
		set_variable = LR_roman_senate
		LR_roman_title_names_refresh_effect = yes # named by its seat (LR_roman_senate_name_effect)
	}
	title:$SEAT$ = {
		if = {
			limit = { NOT = { has_county_modifier = LR_roman_curia_modifier } }
			add_county_modifier = LR_roman_curia_modifier
		}
	}
	if = {
		limit = { NOT = { has_character_modifier = LR_princeps_senatus_modifier } }
		add_character_modifier = LR_princeps_senatus_modifier
	}
	add_prestige = major_prestige_gain
	add_legitimacy = major_legitimacy_gain
	# Senatorial provinces: the governors of the dioceses and prefectures sit in the Senate
	every_vassal_or_below = {
		limit = {
			is_landed = yes
			any_held_title = { LR_roman_senate_province_title_trigger = yes }
		}
		add_character_flag = republic_appointed_senator
	}
	custom_tooltip = LR_roman_senate_provinces_tt
	if = {
		limit = { has_realm_law = republic_senate_size_1 }
		add_realm_law_skip_effects = republic_senate_size_2
	}
	else_if = {
		limit = { has_realm_law = republic_senate_size_2 }
		add_realm_law_skip_effects = republic_senate_size_3
	}
	LR_roman_senate_update_effect = yes
}

# Title: name a Senate title after its seat, in the Roman title-name style (var:LR_rtn_pick, set by LR_roman_title_names_refresh_effect)
LR_roman_senate_name_effect = {
	if = {
		limit = { has_variable = LR_roman_title_name }
		remove_variable = LR_roman_title_name
	}
	if = {
		limit = { OR = { this = title:e_western_roman_republic this = title:e_western_roman_empire } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_western }
	}
	else_if = {
		limit = { OR = { this = title:e_eastern_roman_republic this = title:e_eastern_roman_empire } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_eastern }
	}
	else_if = {
		limit = { OR = { this = title:e_gallic_empire this = title:e_gallic_republic } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_gallic }
	}
	else_if = {
		limit = { OR = { this = title:e_palmyrene_empire this = title:e_palmyrene_republic } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_palmyrene }
	}
	else_if = {
		limit = { OR = { this = title:e_brittanic_empire this = title:e_brittanic_republic } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_brittanic }
	}
	else_if = {
		limit = { OR = { this = title:e_southern_roman_empire this = title:e_southern_roman_republic } }
		LR_roman_senate_set_name_effect = { KEY = LR_senate_southern }
	}
}

LR_roman_senate_set_name_effect = {
	if = {
		limit = { LR_roman_court_style_latin_trigger = yes }
		set_title_name = $KEY$_latin
	}
	else_if = {
		limit = { LR_roman_court_style_greek_trigger = yes }
		set_title_name = $KEY$_greek
	}
	else = {
		set_title_name = $KEY$_foreign
	}
}

# root = a ruler. The seat modifier while the capital is the seat of a Senate the ruler holds (normalizer, every change and yearly)
LR_roman_senate_update_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_western_roman_republic this = title:e_western_roman_empire }
					}
					capital_county ?= title:c_roma_urbs
				}
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_eastern_roman_republic this = title:e_eastern_roman_empire }
					}
					capital_county ?= title:c_byzantion
				}
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_gallic_empire this = title:e_gallic_republic }
					}
					capital_county ?= title:c_trier
				}
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_palmyrene_empire this = title:e_palmyrene_republic }
					}
					capital_county ?= title:c_palmyra
				}
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_brittanic_empire this = title:e_brittanic_republic }
					}
					capital_county ?= title:c_middlesex
				}
				AND = {
					any_held_title = {
						has_variable = LR_roman_senate
						OR = { this = title:e_southern_roman_empire this = title:e_southern_roman_republic }
					}
					capital_county ?= title:c_tunis
				}
			}
		}
		if = {
			limit = { NOT = { has_character_modifier = LR_roman_senate_seat_modifier } }
			add_character_modifier = LR_roman_senate_seat_modifier
		}
	}
	else_if = {
		limit = { has_character_modifier = LR_roman_senate_seat_modifier }
		remove_character_modifier = LR_roman_senate_seat_modifier
	}
}

# root = a ruler leaving the Republic for the Dominate: the Senate is no longer the half's government
LR_roman_senate_clear_effect = {
	every_held_title = {
		limit = { has_variable = LR_roman_senate }
		remove_variable = LR_roman_senate
		if = {
			limit = { has_variable = LR_regional_senate }
			remove_variable = LR_regional_senate
		}
		LR_roman_title_names_refresh_effect = yes
	}
	if = {
		limit = { has_character_modifier = LR_roman_senate_seat_modifier }
		remove_character_modifier = LR_roman_senate_seat_modifier
	}
}

# root = a regional emperor (Convene the Senate). Claims the unheld shadow, founds the hegemony if missing (as the court does),
# and makes the shadow the line's Senate. SHADOW HEGEMONY REGION SORT = as LR_regional_reorganize_effect; SEAT = county; LINE = flag
LR_regional_senate_establish_effect = {
	if = {
		limit = {
			NOT = { has_title = title:$SHADOW$ }
			NOT = { exists = title:$SHADOW$.holder }
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:$SHADOW$ = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = {
			if = {
				limit = { has_title = title:$HEGEMONY$ }
				title:$SHADOW$ = { set_de_jure_liege_title = title:$HEGEMONY$ }
			}
		}
	}
	LR_found_regional_hegemony_with_shadow_effect = { SHADOW = $SHADOW$ HEGEMONY = $HEGEMONY$ REGION = $REGION$ SORT = $SORT$ }
	title:$SHADOW$ = {
		set_variable = LR_roman_senate
		set_variable = { name = LR_regional_senate value = flag:$LINE$ }
		LR_roman_title_names_refresh_effect = yes # named by its seat (LR_roman_senate_name_effect)
	}
	title:$SEAT$ = {
		if = {
			limit = { NOT = { has_county_modifier = LR_roman_curia_modifier } }
			add_county_modifier = LR_roman_curia_modifier
		}
	}
	if = { # 2026-11-25: a Republic's head is Princeps Senatus, an emperor Consul
		limit = { government_has_flag = government_is_republic NOT = { has_character_modifier = LR_princeps_senatus_modifier } }
		add_character_modifier = LR_princeps_senatus_modifier
	}
	else_if = {
		limit = { NOT = { government_has_flag = government_is_republic } NOT = { has_character_modifier = LR_regional_consul_modifier } }
		add_character_modifier = LR_regional_consul_modifier
	}
	add_prestige = major_prestige_gain
	add_legitimacy = major_legitimacy_gain
	dynasty ?= { add_dynasty_prestige = 1000 }
	LR_roman_senate_enrol_effect = yes
	LR_roman_senate_update_effect = yes
}

# root = the head of a regional Senate. Senators of the provinces: under a government with an assembly the governors sit in it;
# otherwise they are seated in the Senate as an honour (opinion)
LR_roman_senate_enrol_effect = {
	if = {
		limit = { government_has_flag = government_is_republic }
		every_vassal_or_below = {
			limit = {
				is_landed = yes
				any_held_title = { LR_roman_senate_province_title_trigger = yes }
			}
			add_character_flag = republic_appointed_senator
		}
	}
	else = {
		every_vassal_or_below = {
			limit = {
				is_landed = yes
				any_held_title = { LR_roman_senate_province_title_trigger = yes }
			}
			add_opinion = {
				target = root
				modifier = LR_seated_in_senate_opinion
			}
		}
	}
	custom_tooltip = LR_roman_senate_provinces_regional_tt
}
