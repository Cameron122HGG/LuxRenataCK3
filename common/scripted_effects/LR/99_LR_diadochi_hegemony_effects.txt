# Diadochi hegemonies (gen_diadochi_hegemonies.py)

# A satrapy goes into its strategia under HEGEMONY, unless a rival hegemony holds the strategia: then under EMPIRE
LR_satrapy_diadochi_k_satrapy_macedonia_effect = {
	if = {
		limit = {
			title:e_strategia_europe = {
				OR = {
					NOT = { exists = de_jure_liege }
					de_jure_liege = title:$HEGEMONY$
					NOT = { exists = de_jure_liege.holder }
				}
			}
		}
		title:k_satrapy_macedonia = { set_de_jure_liege_title = title:e_strategia_europe }
		title:e_strategia_europe = { set_de_jure_liege_title = title:$HEGEMONY$ }
	}
	else = {
		title:k_satrapy_macedonia = { set_de_jure_liege_title = title:$EMPIRE$ }
	}
}
