# LR 2026-11-22 (user): the republic tier law's rank-up gives the region's title-set title of the next tier.
# LR_rank_up_republic_effect is a copy of vanilla gdp_rank_up_republic_effect (common/scripted_effects/gdp_republic_effects.txt,
# CK3 1.21.0.3; RE-COPY ON PATCH DAY, keeping the four '# LR' changes) and is called by LR's copy of common/laws/04_republic_laws.txt.
# The LR changes:
#   1. first it looks at the founder's capital county's de jure title of the next tier (duchy / kingdom / empire): that is
#      the title of the region's current title set. It is taken when it is unheld, has no can_create of its own
#      (LR_title_has_own_creation_gate_trigger, generated by tools/title_types_from_registry.py) and passes every
#      can_create_title gate (LR_title_creation_rules_trigger: set natives, ages, warlord dates, Archontes cap...)
#   2. otherwise vanilla's dynamic "Republic of ..." title (title_type_republic; LR's set types have no key prefix, which
#      dynamic titles need)
#   3. a set title keeps its own arms, colour, history and capital
#   4. the old title is destroyed only when it is a dynamic one (no de jure counties)

# Root = the founder. Saves scope:lr_rank_up_set_title when the set title can be taken; needs scope:old_republic_title.
LR_republic_rank_up_find_set_title_effect = {
	capital_county ?= {
		if = {
			limit = { scope:old_republic_title.tier = tier_county }
			duchy ?= { save_scope_as = lr_rank_up_candidate }
		}
		else_if = {
			limit = { scope:old_republic_title.tier = tier_duchy }
			kingdom ?= { save_scope_as = lr_rank_up_candidate }
		}
		else_if = {
			limit = { scope:old_republic_title.tier = tier_kingdom }
			empire ?= { save_scope_as = lr_rank_up_candidate }
		}
	}
	if = {
		limit = {
			exists = scope:lr_rank_up_candidate
			scope:lr_rank_up_candidate = {
				NOT = { exists = holder }
				LR_title_has_own_creation_gate_trigger = no
			}
			LR_title_creation_rules_trigger = { TITLE = scope:lr_rank_up_candidate }
		}
		scope:lr_rank_up_candidate = { save_scope_as = lr_rank_up_set_title }
	}
}

LR_rank_up_republic_effect = {
	save_scope_as = republican_founder
	primary_title = {
		save_scope_as = old_republic_title
		title_confederation ?= {
			if = {
				limit = { confederation_type = confederation_type:trade_league }
				save_scope_as = trade_league
				leading_title = { save_scope_as = old_league_leader }
			}
		}
	}

	# LR: the next tier's title of the region's title set, when it is free and the founder may create it
	LR_republic_rank_up_find_set_title_effect = yes
	if = {
		limit = { exists = scope:lr_rank_up_set_title }
		get_title = scope:lr_rank_up_set_title
		scope:lr_rank_up_set_title = { save_scope_as = new_republic_title }
		custom_tooltip = LR_republic_rank_up_set_title_desc
	}
	# Else vanilla: a dynamic title of your next tier
	else_if = {
		limit = {
			primary_title.tier = tier_kingdom
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = empire
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_empire_title_desc
	}
	else_if = {
		limit = {
			primary_title.tier = tier_duchy
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = kingdom
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_kingdom_title_desc
	}
	else_if = { # I.e. county
		limit = {
			primary_title.tier = tier_county
		}
		create_dynamic_title = {
			type = title_type_republic
			tier = duchy
			holder = scope:republican_founder
			name = NEW_REPUBLIC_TITLE_TIER_NAME
			adj = NEW_REPUBLIC_TITLE_TIER_ADJ
			save_scope_as = new_republic_title
		}
		custom_tooltip = republic_rank_up_duchy_title_desc
	}

	# We'll hide the following effects for clarity - the important part above use already custom tooltips
	hidden_effect = {
		if = {
			limit = { exists = scope:new_republic_title }
			# Set up the new title
			if = {
				limit = { NOT = { exists = scope:lr_rank_up_set_title } } # LR: a set title keeps its own arms, colour, history and capital
				scope:new_republic_title ?= {
					# CoA and color
					set_coa = scope:old_republic_title
					set_color_from_title = scope:old_republic_title
					# Copy old title history
					copy_title_history = scope:old_republic_title
					# Set capital
					set_capital_county = scope:old_republic_title.title_capital_county
					set_no_automatic_claims = yes
				}
			}
			# Move Title MaAs to new title
			if = {
				limit = {
					scope:old_republic_title = {
						any_title_maa_regiment = { }
					}
				}
				scope:old_republic_title = { transfer_title_maa_ownership = scope:new_republic_title }
			}
			# Update Trade League member/leader if relevant
			scope:trade_league ?= {
				# For blocking on_member_join/leave effects
				add_to_variable_list = {
					name = rank_change_members
					target = scope:republican_founder
				}
				add_to_variable_list = {
					name = rank_change_titles
					target = scope:old_republic_title
				}
				add_to_variable_list = {
					name = rank_change_titles
					target = scope:new_republic_title
				}
				# Must remove first or code blocks holder from 'joining' again
				remove_confederation_member_title = scope:old_republic_title
				add_confederation_member_title = scope:new_republic_title
				if = {
					limit = { scope:old_league_leader ?= scope:old_republic_title }
					scope:republican_founder = { save_scope_as = league_creator }
					set_leading_title = scope:new_republic_title
					if = { # For safety, as title member changes can mess things up
						limit = { exists = var:is_hanseatic_league }
						set_coa = k_hansa
						set_confederation_name = HANSEATIC_LEAGUE_NAME
					}
					else = {
						set_coa = scope:new_republic_title
						set_confederation_name = TRADE_LEAGUE_BASE_NAME
					}
				}
				clear_variable_list = rank_change_members
				clear_variable_list = rank_change_titles
			}
			# Transfer existing monopolies to the new title
			if = {
				limit = {
					scope:old_republic_title = {
						title_has_trade_monopoly = yes
					}
				}
				scope:old_republic_title = {
					every_trade_monopoly_for_title = {
						# Set up what we need for creating a new monopoly
						# Trade Good
						if = {
							limit = { is_complete_monopoly = no }
							monopoly_trade_good_category = { save_scope_as = trade_good_category_type }
							scope:republican_founder = {
								set_variable = { name = new_monopoly_trade_good_category_type value = scope:trade_good_category_type }
							}
						}
						# Complete Monopoly
						else = {
							scope:republican_founder = {
								set_variable = { name = is_complete_monopoly value = yes }
							}
						}
						# Exploitative Monopoly
						if = {
							limit = {
								monopoly_type = exploitative
							}
							scope:republican_founder = {
								set_variable = { name = is_exploitative_monopoly value = yes }
							}
						}
						# Trade Market
						monopoly_trade_market = {
							save_scope_as = trade_market
							scope:republican_founder = {
								set_variable = { name = new_monopoly_trade_market value = scope:trade_market }
							}
						}

						# First, destroy the old monopoly
						destroy_trade_monopoly = yes

						# Then create the new monopoly
						scope:republican_founder = {
							if = {
								limit = {
									exists = var:is_complete_monopoly
									exists = var:is_exploitative_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									type = exploitative
									is_complete = yes
								}
							}
							else_if = {
								limit = {
									exists = var:is_complete_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									type = normal
									is_complete = yes
								}
							}
							else_if = {
								limit = {
									exists = var:is_exploitative_monopoly
								}
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									trade_good_category = var:new_monopoly_trade_good_category_type
									type = exploitative
								}
							}
							else = {
								create_trade_monopoly = {
									owning_title = scope:new_republic_title
									trade_market = var:new_monopoly_trade_market
									trade_good_category = var:new_monopoly_trade_good_category_type
									type = normal
								}
							}
							scope:new_republic_title = {
								random_trade_monopoly_for_title = {
									limit = {
										NOT = { is_in_list = new_title_monopolies }
									}
									save_scope_as = new_monopoly_scope
									if = {
										limit = { is_complete_monopoly = yes }
										set_trade_monopoly_name = new_complete_title_monopoly_name
									}
									else = {
										set_trade_monopoly_name = new_good_title_monopoly_name
									}
									add_to_list = new_title_monopolies
								}
							}

							# Finally, clear the variables
							remove_variable = new_monopoly_trade_good_category_type
							remove_variable = new_monopoly_trade_market
							remove_variable = is_complete_monopoly
							remove_variable = is_exploitative_monopoly
						}
					}
				}
			}
			# Destroy old title if needed
			if = {
				limit = {
					scope:old_republic_title.tier > tier_county
					# LR: only a dynamic republic title; a real (set) title with de jure land is kept
					scope:old_republic_title = { NOT = { any_de_jure_county = { } } }
				}
				scope:republican_founder = { destroy_title = scope:old_republic_title }
			}
		}
	}
}
