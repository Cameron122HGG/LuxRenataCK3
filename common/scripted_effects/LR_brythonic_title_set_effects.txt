# Brythonic title set (gen_brythonic_title_set.py): root restores a kingdom of the Britons, laying out its duchies
# over its counties, and receives it. The kingdom goes under its empire, the empire under the Celtic Empire if root holds it.

LR_restore_brython_gwynedd_effect = { # Gwynedd
	title:k_brython_gwynedd = { set_de_jure_liege_title = title:e_brython_cymru }
	if = {
		limit = { has_title = title:h_western_celtic_empire }
		title:e_brython_cymru = { set_de_jure_liege_title = title:h_western_celtic_empire }
	}
	title:d_brython_gwynedd_uwch_conwy = { set_de_jure_liege_title = title:k_brython_gwynedd }
	title:c_anglesey = { set_de_jure_liege_title = title:d_brython_gwynedd_uwch_conwy }
	title:c_caernarfonshire = { set_de_jure_liege_title = title:d_brython_gwynedd_uwch_conwy }
	title:c_merioneth = { set_de_jure_liege_title = title:d_brython_gwynedd_uwch_conwy }
	title:d_brython_perfeddwlad = { set_de_jure_liege_title = title:k_brython_gwynedd }
	title:c_denbighshire = { set_de_jure_liege_title = title:d_brython_perfeddwlad }
	title:d_brython_manaw = { set_de_jure_liege_title = title:k_brython_gwynedd }
	title:c_isle_of_man = { set_de_jure_liege_title = title:d_brython_manaw }
	create_title_and_vassal_change = {
		type = created
		save_scope_as = LR_brython_change
		add_claim_on_loss = no
	}
	title:k_brython_gwynedd = {
		change_title_holder = {
			holder = root
			change = scope:LR_brython_change
		}
	}
	resolve_title_and_vassal_change = scope:LR_brython_change
	LR_brython_end_briton_high_kingship_effect = yes
}
