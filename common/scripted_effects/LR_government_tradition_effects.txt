# Culture scope: keep only one government tradition. Keeps the one scope:founder's government belongs to
# (for a new hybrid culture), otherwise the first in group order.
LR_keep_one_government_tradition_effect = {
	if = {
		limit = { LR_government_tradition_count_value > 1 }
		if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_classical
				scope:founder ?= { government_has_flag = government_is_classical_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_classical }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_western
				scope:founder ?= { government_has_flag = government_is_western_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_western }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_despotic
				scope:founder ?= { government_has_flag = government_is_despotic_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_despotic }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_indic
				scope:founder ?= { government_has_flag = government_is_indic_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_indic }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_islamic
				scope:founder ?= { government_has_flag = government_is_islamic_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_islamic }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_sinic
				scope:founder ?= { government_has_flag = government_is_sinic_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_sinic }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_japanese
				scope:founder ?= { government_has_flag = government_is_japanese_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_japanese }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_barbarian
				scope:founder ?= { government_has_flag = government_is_barbarian_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_barbarian }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_classical }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_classical }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_western }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_western }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_despotic }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_despotic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_indic }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_indic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_islamic }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_islamic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_sinic }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_sinic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_japanese }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_japanese }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_barbarian }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_barbarian }
		}
		LR_drop_other_government_traditions_effect = yes
		remove_variable = LR_gov_tradition_keep
	}
}

# Culture scope, var:LR_gov_tradition_keep = flag of the government tradition to keep: remove the others.
LR_drop_other_government_traditions_effect = {
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_classical
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_classical }
		}
		remove_culture_tradition = lux_tradition_gov_classical
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_western
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_western }
		}
		remove_culture_tradition = lux_tradition_gov_western
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_despotic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_despotic }
		}
		remove_culture_tradition = lux_tradition_gov_despotic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_indic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_indic }
		}
		remove_culture_tradition = lux_tradition_gov_indic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_islamic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_islamic }
		}
		remove_culture_tradition = lux_tradition_gov_islamic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_sinic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_sinic }
		}
		remove_culture_tradition = lux_tradition_gov_sinic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_japanese
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_japanese }
		}
		remove_culture_tradition = lux_tradition_gov_japanese
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_barbarian
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_barbarian }
		}
		remove_culture_tradition = lux_tradition_gov_barbarian
	}
}

# Culture scope, scope:tradition = flag of the government tradition just added: drop the culture's other one.
LR_replace_government_tradition_effect = {
	set_variable = { name = LR_gov_tradition_keep value = scope:tradition }
	LR_drop_other_government_traditions_effect = yes
	remove_variable = LR_gov_tradition_keep
}

# Character scope: move a ruler whose government is on another group's ladder to the same tier of the group
# their culture's government tradition names. Plain and dynastic tiers go to the group's plain government
# (LR_normalize_dynastic_government_effect then picks the dynastic twin); an independent administrative ruler
# goes to the group's administrative government. Administrative vassals follow their realm, top tiers
# (Roman Imperial, Persianate Autocratic and the like) keep their government, and so do sacral rulers in the realm of
# e_sacral_kingdom_gaul (the sacral group is tied to that title, not to a culture).
LR_apply_culture_government_group_effect = {
	if = {
		limit = {
			is_ruler = yes
			is_landed = yes
			LR_has_group_ladder_government_trigger = yes
			culture = { LR_culture_has_government_tradition_trigger = yes }
			LR_government_matches_culture_group_trigger = no
			NAND = {
				government_has_mechanic = administrative
				is_independent_ruler = no
			}
			# The sacral group comes from the Gaul title: its realm keeps its sacral governments
			NAND = {
				government_has_flag = government_is_sacral_group
				LR_in_sacral_kingdom_gaul_realm_trigger = yes
			}
		}
		if = {
			limit = { government_has_mechanic = administrative }
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				if = {
					limit = { LR_culture_is_hellenic_trigger = yes }
					change_government = alexandrian_administrative_government
				}
				else = {
					change_government = roman_administrative_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = persianate_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = gupta_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_islamic } }
				change_government = iqta_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				change_government = chinese_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					change_government = japan_administrative_government
				}
				else = {
					change_government = chinese_feudal_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_barbarian } }
				change_government = great_tribal_government
			}
		}
		else = {
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				change_government = classical_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = despotic_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = indic_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_islamic } }
				change_government = iqta_dynastic_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				change_government = chinese_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					change_government = japan_feudal_government
				}
				else = {
					change_government = chinese_feudal_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_barbarian } }
				change_government = great_tribal_government
			}
		}
	}
}
