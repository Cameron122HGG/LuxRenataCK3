# Lux Renata government traditions

# Culture scope: give a culture with no government tradition the one its heritage points to.
# (Sacral Kingship is never seeded: cultures gain it by restoring Gaul's sacral monarchy.)
LR_seed_government_tradition_effect = {
	if = {
		limit = { LR_culture_has_government_tradition_trigger = no }
		if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_latin
						has_cultural_pillar = heritage_byzantine
						has_cultural_pillar = heritage_hellenistic
						has_cultural_pillar = heritage_indo_hellenic
						has_cultural_pillar = heritage_sino_hellenic
						has_cultural_pillar = heritage_caucasian
						has_cultural_pillar = heritage_ostrogothic
						has_cultural_pillar = heritage_vandalic
						has_cultural_pillar = heritage_visigothic
						has_cultural_pillar = heritage_suevic
						has_cultural_pillar = heritage_vlach
						has_cultural_pillar = heritage_brythonic
						has_cultural_pillar = heritage_ancient_greek
						has_cultural_pillar = heritage_albanian
					}
				}
				add_culture_tradition = lux_tradition_gov_classical
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_frankish
						has_cultural_pillar = heritage_central_germanic
						has_cultural_pillar = heritage_north_germanic
						has_cultural_pillar = heritage_west_germanic
						has_cultural_pillar = heritage_iberian
						has_cultural_pillar = heritage_south_slavic
						has_cultural_pillar = heritage_east_slavic
						has_cultural_pillar = heritage_illyrian
						has_cultural_pillar = heritage_goidelic
						has_cultural_pillar = heritage_west_slavic
						has_cultural_pillar = heritage_baltic
						has_cultural_pillar = heritage_balto_finnic
					}
				}
				add_culture_tradition = lux_tradition_gov_western
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_iranian
						has_cultural_pillar = heritage_sarmatian
					}
				}
				add_culture_tradition = lux_tradition_gov_despotic
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_dravidian
						has_cultural_pillar = heritage_indo_aryan
						has_cultural_pillar = heritage_burman
						has_cultural_pillar = heritage_mon_khmer
					}
				}
				add_culture_tradition = lux_tradition_gov_indic
			}
			else_if = {
				limit = {
					OR = { # LR 2026-11-19: the Semitic heritages (syriac was classical, phoenician despotic) and the Berbers (were despotic)
						has_cultural_pillar = heritage_berber
						has_cultural_pillar = heritage_arabic
						has_cultural_pillar = heritage_israelite
						has_cultural_pillar = heritage_syriac
						has_cultural_pillar = heritage_phoenician
					}
				}
				add_culture_tradition = lux_tradition_gov_mamlakah # mamlakah-only (seed by heritage)
			}
			else_if = { # LR 2026-11-19: Turkic cultures are Persianate (user: "keep turkic persianate")
				limit = { has_cultural_pillar = heritage_turkic }
				add_culture_tradition = lux_tradition_gov_despotic
			}
			else_if = { # LR 2026-11-19: the Nile and the Horn are Nilotic (user: "Nilotic, with Ethiopia")
				limit = {
					OR = {
						has_cultural_pillar = heritage_east_african
						has_cultural_pillar = heritage_egyptian
						has_cultural_pillar = heritage_somalian
					}
				}
				add_culture_tradition = lux_tradition_gov_nilotic
			}
			else_if = { # LR 2026-11-19: settled sub-Saharan Africa is Sudanic (user: "settled subsaharan africans get their own group")
				limit = {
					OR = {
						has_cultural_pillar = heritage_akan
						has_cultural_pillar = heritage_bantu
						has_cultural_pillar = heritage_central_african
						has_cultural_pillar = heritage_sahelian
						has_cultural_pillar = heritage_senegambian
						has_cultural_pillar = heritage_west_african
						has_cultural_pillar = heritage_yoruba
					}
				}
				add_culture_tradition = lux_tradition_gov_sudanic
			}
			else_if = { # LR 2026-11-19: Magyar cultures are Barbarian (user, after Persianate felt too common)
				limit = { has_cultural_pillar = heritage_magyar }
				add_culture_tradition = lux_tradition_gov_barbarian
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_chinese
						has_cultural_pillar = heritage_korean
						has_cultural_pillar = heritage_mongolic
						has_cultural_pillar = heritage_qiangic
						has_cultural_pillar = heritage_tai
						has_cultural_pillar = heritage_tibetan
						has_cultural_pillar = heritage_tocharian
						has_cultural_pillar = heritage_buyeo
						has_cultural_pillar = heritage_hmongic
						has_cultural_pillar = heritage_viet
						has_cultural_pillar = heritage_tungusic
						has_cultural_pillar = heritage_nivkh
					}
				}
				add_culture_tradition = lux_tradition_gov_sinic
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_japonic
						has_cultural_pillar = heritage_ainuic
					}
				}
				add_culture_tradition = lux_tradition_gov_japanese
			}
			else_if = {
				limit = {
					OR = {
						has_cultural_pillar = heritage_hunnic
						has_cultural_pillar = heritage_gothic
						has_cultural_pillar = heritage_volga_finnic
						has_cultural_pillar = heritage_ugro_permian
					}
				}
				add_culture_tradition = lux_tradition_gov_barbarian
			}
			# No heritage match: the culture has no way of government until it chooses one
			else_if = {
				limit = { NOT = { has_cultural_tradition = lux_tradition_gov_anarchy } }
				add_culture_tradition = lux_tradition_gov_anarchy
			}
	}
	else_if = {
		limit = { has_cultural_tradition = lux_tradition_gov_anarchy }
		remove_culture_tradition = lux_tradition_gov_anarchy # e.g. a hybrid of an anarchic culture and a settled one
	}
}

# Culture scope: keep only one government tradition. Keeps the one scope:founder's government belongs to
# (for a new hybrid culture), otherwise the first in group order.
LR_keep_one_government_tradition_effect = {
	if = {
		limit = { LR_government_tradition_count_value > 1 }
		if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_classical
				scope:founder ?= { government_has_flag = government_is_classical_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_classical }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_western
				scope:founder ?= { government_has_flag = government_is_western_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_western }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_despotic

			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_despotic }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_indic
				scope:founder ?= { government_has_flag = government_is_indic_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_indic }

					}
				}
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_japanese }
		}
		else_if = {
			limit = {
				has_cultural_tradition = lux_tradition_gov_barbarian
				scope:founder ?= { government_has_flag = government_is_barbarian_group }
			}
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_barbarian }
		}

			limit = { has_cultural_tradition = lux_tradition_gov_classical }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_classical }
		}

			limit = { has_cultural_tradition = lux_tradition_gov_western }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_western }
		}

			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_despotic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_indic }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_indic }
		}
		else_if = {
			limit = { has_cultural_tradition = lux_tradition_gov_mamlakah }
			set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_mamlakah }

# Culture scope, var:LR_gov_tradition_keep = flag of the government tradition to keep: remove the others
# (and Anarchy, which any government tradition ends).
LR_drop_other_government_traditions_effect = {
	if = {
		limit = { has_cultural_tradition = lux_tradition_gov_anarchy }
		remove_culture_tradition = lux_tradition_gov_anarchy
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_classical
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_classical }
		}
		remove_culture_tradition = lux_tradition_gov_classical
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_western
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_western }
		}
		remove_culture_tradition = lux_tradition_gov_western
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_despotic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_despotic }
		}
		remove_culture_tradition = lux_tradition_gov_despotic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_indic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_indic }
		}
		remove_culture_tradition = lux_tradition_gov_indic
	}
	if = {
		limit = {

			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_mamlakah }

	}
	if = {

		}

			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_nilotic }
		}
		remove_culture_tradition = lux_tradition_gov_nilotic
	}
	if = {
		limit = {
			has_cultural_tradition = lux_tradition_gov_sinic
			NOT = { var:LR_gov_tradition_keep = flag:lux_tradition_gov_sinic }

# Culture scope, scope:tradition = the government tradition just added (a culture_tradition, not a flag):

		limit = { scope:tradition = culture_tradition:lux_tradition_gov_classical }
		set_variable = { name = LR_gov_tradition_keep value = flag:lux_tradition_gov_classical }

# Character scope: move a ruler whose government is on another group's ladder to the same tier of the group
# their culture's government tradition names. Plain and dynastic tiers go to the group's plain government
# (LR_normalize_dynastic_government_effect then picks the dynastic twin); an independent administrative ruler
# goes to the group's administrative government. Administrative vassals follow their realm, top tiers
# (Roman Imperial, Persianate Autocratic and the like) keep their government, and so do sacral rulers in the realm of
# e_sacral_kingdom_gaul (the sacral group is tied to that title, not to a culture).
LR_apply_culture_government_group_effect = {
	if = {
		limit = {
			is_ruler = yes
			is_landed = yes
			LR_has_group_ladder_government_trigger = yes
			culture = { LR_culture_has_government_tradition_trigger = yes }
			LR_government_matches_culture_group_trigger = no
			NAND = {
				government_has_mechanic = administrative
				is_independent_ruler = no
			}
			# The sacral group comes from the Gaul title: its realm keeps its sacral governments
			NAND = {
				government_has_flag = government_is_sacral_group
				LR_in_sacral_kingdom_gaul_realm_trigger = yes
			}
		}
		if = {
			limit = { government_has_mechanic = administrative }
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				if = {
					limit = { LR_culture_is_hellenic_trigger = yes }
					change_government = alexandrian_administrative_government
				}
				else = {
					change_government = roman_administrative_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = persianate_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = gupta_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_mamlakah } }
				change_government = mamlakah_administrative_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sudanic } }
				change_government = sudanic_administrative_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_nilotic } }
				change_government = nilotic_administrative_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					change_government = meritocratic_government
				}
				else = {
					change_government = chinese_feudal_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					change_government = japan_administrative_government
				}
				else = {
					change_government = japanese_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_barbarian } }
				change_government = great_tribal_government
			}
		}
		else = {
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				change_government = classical_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = despotic_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = indic_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_mamlakah } }
				change_government = mamlakah_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sudanic } }
				change_government = sudanic_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_nilotic } }
				change_government = nilotic_government # LR 2026-11-19 (Muslims are moved on to Iqta by normalizing)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				change_government = chinese_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					add_character_flag = { flag = can_become_japanese_feudal days = 1 } # Soryo is otherwise only allowed under a Japanese realm (tgp_can_become_japanese_feudal_trigger)
					change_government = japan_feudal_government
				}
				else = {
					change_government = japanese_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_barbarian } }
				change_government = great_tribal_government
			}
		}
	}
}

# Culture scope, scope:LR_gov_model = a ruler: end Anarchy by adopting the government tradition of the model's government.
# A government outside the groups goes to the nearest one: Mandala to the Indic group, tribal, nomadic and herder
# governments to the great tribal confederations, anything else (vanilla Feudal, Clan, Administrative) to Feudal Custom.
LR_adopt_government_tradition_of_model_effect = {
	scope:LR_gov_model = {
		if = {
			limit = { government_has_flag = government_is_classical_group }
			prev = { add_culture_tradition = lux_tradition_gov_classical }
		}
		else_if = {
			limit = { government_has_flag = government_is_western_group }
			prev = { add_culture_tradition = lux_tradition_gov_western }
		}
		else_if = {
			limit = { government_has_flag = government_is_despotic_group }
			prev = { add_culture_tradition = lux_tradition_gov_despotic }
		}
		else_if = {
			limit = {
				OR = {
					government_has_flag = government_is_indic_group
					government_has_flag = government_is_mandala
				}
			}
			prev = { add_culture_tradition = lux_tradition_gov_indic }
		}
		else_if = {
			limit = { government_has_flag = government_is_mamlakah_group }
			prev = { add_culture_tradition = lux_tradition_gov_mamlakah }
		}
		else_if = {
			limit = { government_has_flag = government_is_sudanic_group }
			prev = { add_culture_tradition = lux_tradition_gov_sudanic }
		}
		else_if = {
			limit = { government_has_flag = government_is_nilotic_group }
			prev = { add_culture_tradition = lux_tradition_gov_nilotic }
		}
		else_if = { # LR 2026-11-19: Iqta is faith-based, not a tradition: the culture takes the one its language or heritage points to
			limit = { government_has_flag = government_is_islamic_group }
			prev = { LR_seed_government_tradition_effect = yes }
		}
		else_if = {
			limit = {
				OR = {
					government_has_flag = government_is_japanese_group
					government_has_flag = government_is_japanese_plain
				}
			}
			prev = { add_culture_tradition = lux_tradition_gov_japanese }
		}
		else_if = {
			limit = { government_has_flag = government_is_sinic_group }
			prev = { add_culture_tradition = lux_tradition_gov_sinic }
		}
		else_if = {
			limit = {
				OR = {
					government_has_flag = government_is_barbarian_group
					government_has_flag = government_is_tribal
					government_has_flag = government_is_nomadic
					government_has_flag = government_is_herder
				}
			}
			prev = { add_culture_tradition = lux_tradition_gov_barbarian }
		}
		else = {
			prev = { add_culture_tradition = lux_tradition_gov_western }
		}
	}
}

# Culture scope: an AI-led culture in Anarchy settles on the government of its head, or failing one its highest-ranked ruler.
LR_resolve_government_anarchy_effect = {
	if = {
		limit = { exists = culture_head }
		culture_head = { save_scope_as = LR_gov_model }
	}
	else = {
		save_scope_as = LR_anarchic_culture
		ordered_ruler = {
			limit = {
				culture = scope:LR_anarchic_culture
				is_landed = yes
			}
			order_by = highest_held_title_tier
			save_scope_as = LR_gov_model
		}
	}
	if = {
		limit = { exists = scope:LR_gov_model }
		LR_adopt_government_tradition_of_model_effect = yes
	}
}
