# LR 2026-11-06: the Solar Imperial West/East split (LR_solar_imperial_schism_prep.md in the working folder)

# Gives k_helios_autokrator to the senior Eastern emperor who follows the Eastern Solar rite: the emperor of a restored
# Rome, else the Eastern hegemon, else the Eastern Roman emperor, else decayed Byzantium. None scope.
LR_sync_helios_autokrator_effect = {
	if = {
		limit = { title:h_roman_empire.holder ?= { LR_is_sun_emperor_candidate_trigger = yes } }
		title:h_roman_empire.holder = { save_scope_as = sun_emperor }
	}
	else_if = {
		limit = { title:h_eastern_roman_empire.holder ?= { LR_is_sun_emperor_candidate_trigger = yes } }
		title:h_eastern_roman_empire.holder = { save_scope_as = sun_emperor }
	}
	else_if = {
		limit = { title:e_eastern_roman_empire.holder ?= { LR_is_sun_emperor_candidate_trigger = yes } }
		title:e_eastern_roman_empire.holder = { save_scope_as = sun_emperor }
	}
	else_if = {
		limit = { title:e_byzantium.holder ?= { LR_is_sun_emperor_candidate_trigger = yes } }
		title:e_byzantium.holder = { save_scope_as = sun_emperor }
	}
	if = {
		limit = {
			exists = scope:sun_emperor
			NOT = { title:k_helios_autokrator.holder ?= scope:sun_emperor }
		}
		scope:sun_emperor = {
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			title:k_helios_autokrator = {
				change_title_holder = {
					holder = scope:sun_emperor
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}

# The Great Solar Schism: the rites move into the waiting faiths (as pam_apply_great_schism_creation_effect). None scope.
LR_solar_schism_split_effect = {
	set_global_variable = { name = LR_solar_schism_happened value = yes }
	remove_global_variable = LR_solar_schism_progress
	rite:solar_imperial_rite = {
		set_can_convert_to_rite = yes
		set_parent_faith = {
			target = faith:helios_autokrator_faith
			main = yes
		}
	}
	rite:sol_invictus_rite = {
		set_can_convert_to_rite = yes
		set_parent_faith = {
			target = faith:sol_invictus_faith
			main = yes
		}
	}
	faith:helios_autokrator_faith = { set_religious_head_title = title:k_helios_autokrator }
	faith:sol_invictus_faith = { set_religious_head_title = title:d_pontifex_solis_invicti }
}

# History-time split (LR_solar_church_history.txt, 6000.1.1). The guard variable stops on_clerical_region_faith_mismatch
# (pam_on_actions.txt) from firing pam_clergy_events mid-history, which crashed the game (2026-11-06). None scope.
LR_solar_schism_history_split_effect = {
	set_global_variable = { name = LR_solar_schism_history_split value = yes }
	LR_solar_schism_split_effect = yes
	remove_global_variable = LR_solar_schism_history_split
}

# Healing the schism: both rites return to Solar Imperial, headed by the incarnate Sun. None scope.
LR_solar_schism_mend_effect = {
	remove_global_variable = LR_solar_schism_happened
	set_global_variable = { name = LR_solar_schism_progress value = 0 }
	set_global_variable = { name = LR_solar_schism_mended value = yes }
	LR_sc_reopen_schism_effect = yes # LR: the Solar Church may brew a new schism
	rite:solar_imperial_rite = {
		set_parent_faith = {
			target = faith:solar_imperial_faith
			main = yes
		}
	}
	rite:sol_invictus_rite = {
		set_parent_faith = {
			target = faith:solar_imperial_faith
			main = no
		}
	}
	faith:solar_imperial_faith = { set_religious_head_title = title:k_helios_autokrator }
}

# Yearly, before the split: two emperors and a Western-rite Rome facing an Eastern-rite Constantinople drive the
# schism on; one restored Rome draws it back. At 100 the faith splits, or, with the Solar Church, the Eclipse decides
# (LR_sc_change_schism_progress_effect, LR_solar_church_effects.txt). None scope.
LR_solar_schism_yearly_effect = {
	if = {
		limit = { LR_sc_progress_can_change_trigger = yes }
		if = {
			limit = { LR_solar_two_emperors_trigger = yes }
			LR_sc_change_schism_progress_effect = { VALUE = 3 }
		}
		if = {
			limit = {
				title:c_roma.holder.top_liege ?= { rite ?= rite:sol_invictus_rite }
				title:c_byzantion.holder.top_liege ?= { rite ?= rite:solar_imperial_rite }
			}
			LR_sc_change_schism_progress_effect = { VALUE = 2 }
		}
		if = {
			limit = { exists = title:h_roman_empire.holder }
			LR_sc_change_schism_progress_effect = { VALUE = -5 }
		}
	}
}
