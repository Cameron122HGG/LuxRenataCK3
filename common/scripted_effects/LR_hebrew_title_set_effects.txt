# Proclaim Mamlekhet Yisra'el: root takes the hegemony; every eretz goes under it, and every mamlakhah that
# already has land goes under its eretz.
LR_proclaim_mamlekhet_yisrael_effect = {
	if = {
		limit = { title:k_mamlakhah_phoenicia = { any_de_jure_county = { always = yes } } }
		title:k_mamlakhah_phoenicia = { set_de_jure_liege_title = title:e_eretz_yisrael }
	}
	if = {
		limit = { title:k_mamlakhah_syria = { any_de_jure_county = { always = yes } } }
		title:k_mamlakhah_syria = { set_de_jure_liege_title = title:e_eretz_yisrael }
	}
	if = {
		limit = { title:k_mamlakhah_mesopotamia = { any_de_jure_county = { always = yes } } }
		title:k_mamlakhah_mesopotamia = { set_de_jure_liege_title = title:e_eretz_ever_hanahar }
	}
	if = {
		limit = { title:k_mamlakhah_babylon = { any_de_jure_county = { always = yes } } }
		title:k_mamlakhah_babylon = { set_de_jure_liege_title = title:e_eretz_ever_hanahar }
	}
	if = {
		limit = { title:k_mamlakhah_egypt = { any_de_jure_county = { always = yes } } }
		title:k_mamlakhah_egypt = { set_de_jure_liege_title = title:e_eretz_ham }
	}
	title:e_eretz_yisrael = { set_de_jure_liege_title = title:h_mamlekhet_yisrael }
	title:e_eretz_ever_hanahar = { set_de_jure_liege_title = title:h_mamlekhet_yisrael }
	title:e_eretz_ham = { set_de_jure_liege_title = title:h_mamlekhet_yisrael }
	create_title_and_vassal_change = {
		type = created
		save_scope_as = LR_mamlekhet_change
		add_claim_on_loss = no
	}
	title:h_mamlekhet_yisrael = {
		change_title_holder = {
			holder = root
			change = scope:LR_mamlekhet_change
		}
	}
	resolve_title_and_vassal_change = scope:LR_mamlekhet_change
	dynasty ?= { add_dynasty_prestige = major_dynasty_prestige_gain }
}
