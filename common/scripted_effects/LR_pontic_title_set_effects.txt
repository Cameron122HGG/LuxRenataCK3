# Scope: a title left behind by a Pontic layout. If it has a holder but no de jure land left, the holder gets the
# Pontic title of the same tier over its capital county (if unheld and theirs to use) and the old title is dissolved.
LR_pontic_dissolve_if_empty_effect = {
	if = {
		limit = {
			exists = holder
			NOT = { LR_title_set_pontic_title_trigger = yes }
			NOT = { this = title:e_mithridatid_empire }
			NOT = { any_de_jure_county = { always = yes } }
		}
		save_temporary_scope_as = LR_pontic_old
		holder = { save_temporary_scope_as = LR_pontic_old_holder }
		title_capital_county ?= {
			if = {
				limit = { scope:LR_pontic_old.tier = tier_duchy }
				duchy ?= { save_temporary_scope_as = LR_pontic_new }
			}
			else_if = {
				limit = { scope:LR_pontic_old.tier = tier_kingdom }
				kingdom ?= { save_temporary_scope_as = LR_pontic_new }
			}
		}
		if = {
			limit = {
				exists = scope:LR_pontic_new
				scope:LR_pontic_new = {
					LR_title_set_pontic_title_trigger = yes
					NOT = { exists = holder }
				}
				scope:LR_pontic_old_holder = { LR_title_set_pontic_native_trigger = yes }
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = LR_pontic_change
				add_claim_on_loss = no
			}
			scope:LR_pontic_new = {
				change_title_holder = {
					holder = scope:LR_pontic_old_holder
					change = scope:LR_pontic_change
				}
			}
			resolve_title_and_vassal_change = scope:LR_pontic_change
		}
		scope:LR_pontic_old_holder = { destroy_title = scope:LR_pontic_old }
		clear_saved_scope = LR_pontic_new
	}
}
