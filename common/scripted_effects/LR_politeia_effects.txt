# LR 2026-11-25 Politeia (tools/politeia/build_politeia.py)
# Normalizer (LR_normalize_dynastic_government_effect): a Hellenistic-group republic gets the Hellenic law first, then the
# Politeia (on_government_change runs inside change_government, so the law must be there before); a Politeia that is no
# longer one goes back to the plain republic.
LR_hellenic_republic_government_effect = {
	if = {
		limit = {
			has_government = republic_government
			LR_hellenic_group_republic_trigger = yes
		}
		change_government = hellenic_republic_government # 2026-11-25: the government first: the republic_type_hellenic law is exclusive to it
		if = {
			limit = { NOT = { has_realm_law_flag = LR_hellenic_republic_type } }
			add_realm_law_skip_effects = republic_type_hellenic
		}
	}
	else_if = {
		limit = {
			has_government = hellenic_republic_government
			LR_hellenic_group_republic_trigger = no
			NOT = { has_realm_law_flag = LR_hellenic_republic_type }
		}
		change_government = republic_government
	}
}

# character scope (the archon): appoint the most martial adult vassal of another house, else the most martial courtier
LR_appoint_strategos_effect = {
	if = {
		limit = {
			has_government = hellenic_republic_government
			has_realm_law_flag = LR_strategos_autokrator
		}
		save_scope_as = LR_archon
		if = {
			limit = { any_vassal = { LR_strategos_candidate_trigger = yes } }
			ordered_vassal = {
				limit = { LR_strategos_candidate_trigger = yes }
				order_by = martial
				save_scope_as = LR_strategos
			}
		}
		else_if = {
			limit = { any_courtier = { LR_strategos_candidate_trigger = yes } }
			ordered_courtier = {
				limit = { LR_strategos_candidate_trigger = yes }
				order_by = martial
				save_scope_as = LR_strategos
			}
		}
		if = {
			limit = { exists = scope:LR_strategos }
			try_start_diarchy = strategos_diarchy
			set_diarch = scope:LR_strategos
		}
	}
}

LR_end_strategia_effect = {
	if = {
		limit = { has_diarchy_type = strategos_diarchy }
		end_diarchy = yes
	}
}

LR_strategia_upkeep_effect = {
	if = {
		limit = {
			has_government = hellenic_republic_government
			has_realm_law_flag = LR_strategos_autokrator
			NOT = { exists = diarch }
		}
		LR_appoint_strategos_effect = yes
	}
}
