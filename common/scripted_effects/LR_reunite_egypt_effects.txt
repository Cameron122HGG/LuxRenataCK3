# Reunite the Two Lands (LR_reunite_egypt_decision)
# The duchies of Upper and Lower Egypt return to k_egypt, which goes under e_egypt and is granted to root.
# Root's own Upper and Lower Egypt titles, now without land, are destroyed.
LR_reunite_egypt_effect = {
	custom_tooltip = LR_reunite_egypt_dejure_tt
	hidden_effect = {
		title:d_alexandria = { set_de_jure_liege_title = title:k_egypt }
		title:d_delta = { set_de_jure_liege_title = title:k_egypt }
		title:d_cairo = { set_de_jure_liege_title = title:k_egypt }
		title:d_sinai = { set_de_jure_liege_title = title:k_egypt }
		title:d_al-said = { set_de_jure_liege_title = title:k_egypt }
		title:d_al-wahat = { set_de_jure_liege_title = title:k_egypt }
		title:d_eastern_desert = { set_de_jure_liege_title = title:k_egypt }
		title:d_nobatia = { set_de_jure_liege_title = title:k_egypt }
		title:k_egypt = { set_de_jure_liege_title = title:e_egypt }
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:k_egypt = {
		change_title_holder = {
			holder = root
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change

	hidden_effect = {
		if = {
			limit = { has_title = title:k_lower_egypt }
			destroy_title = title:k_lower_egypt
		}
		if = {
			limit = { has_title = title:k_upper_egypt }
			destroy_title = title:k_upper_egypt
		}
	}

	add_prestige = major_prestige_gain
}
