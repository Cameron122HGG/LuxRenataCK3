# Lux Invicta Graeco-Roman religion content (gen_graeco_roman_religion.py)

# Root holds d_vestal_college and is leaving it: it passes to the eldest serving Vestal, or is destroyed.
LR_pass_vestal_college_effect = {
	save_scope_as = LR_old_maxima
	if = {
		limit = { any_living_character = { has_trait = LR_vestalis NOT = { this = scope:LR_old_maxima } } }
		ordered_living_character = {
			limit = { has_trait = LR_vestalis NOT = { this = scope:LR_old_maxima } }
			order_by = age
			save_scope_as = LR_new_maxima
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = LR_vestal_change
			add_claim_on_loss = no
		}
		title:d_vestal_college = {
			change_title_holder = {
				holder = scope:LR_new_maxima
				change = scope:LR_vestal_change
			}
		}
		resolve_title_and_vassal_change = scope:LR_vestal_change
	}
	else = {
		destroy_title = title:d_vestal_college
	}
}

# Root (a Roman-faith ruler holding Rome and Constantinople) becomes Pontifex Maximus, head of the Roman faith
LR_create_pontifex_maximus_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = LR_pontifex_change
		add_claim_on_loss = no
	}
	title:d_pontifex_maximus = {
		change_title_holder = {
			holder = root
			change = scope:LR_pontifex_change
		}
	}
	resolve_title_and_vassal_change = scope:LR_pontifex_change
	faith:roman_faith = {
		every_faith_rite = {
			limit = { rite_has_doctrine = doctrine_no_head }
			change_rite_doctrine = doctrine_temporal_head
		}
		set_religious_head_title = title:d_pontifex_maximus
	}
}
