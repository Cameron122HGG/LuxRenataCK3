# Scope: a title left behind by an ajnad layout. If it has a holder but no de jure land left, the holder gets the
# ajnad title of the same tier over its capital county (if unheld and theirs to use) and the old title is dissolved.
LR_ajnad_dissolve_if_empty_effect = {
	if = {
		limit = {
			exists = holder
			NOT = { LR_title_set_ajnad_title_trigger = yes }
			NOT = { any_de_jure_county = { always = yes } }
		}
		save_temporary_scope_as = LR_ajnad_old
		holder = { save_temporary_scope_as = LR_ajnad_old_holder }
		title_capital_county ?= {
			if = {
				limit = { scope:LR_ajnad_old.tier = tier_duchy }
				duchy ?= { save_temporary_scope_as = LR_ajnad_new }
			}
			else_if = {
				limit = { scope:LR_ajnad_old.tier = tier_kingdom }
				kingdom ?= { save_temporary_scope_as = LR_ajnad_new }
			}
		}
		if = {
			limit = {
				exists = scope:LR_ajnad_new
				scope:LR_ajnad_new = {
					LR_title_set_ajnad_title_trigger = yes
					NOT = { exists = holder }
				}
				scope:LR_ajnad_old_holder = { LR_title_set_ajnad_native_trigger = yes }
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = LR_ajnad_change
				add_claim_on_loss = no
			}
			scope:LR_ajnad_new = {
				change_title_holder = {
					holder = scope:LR_ajnad_old_holder
					change = scope:LR_ajnad_change
				}
			}
			resolve_title_and_vassal_change = scope:LR_ajnad_change
		}
		scope:LR_ajnad_old_holder = { destroy_title = scope:LR_ajnad_old }
		clear_saved_scope = LR_ajnad_new
	}
}

# The ruler (root) restores the Caliphate: creates al-Khilafa over the wilayat
LR_ajnad_restore_khilafa_effect = {
	if = {
		limit = { NOT = { exists = title:e_khilafa.holder } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = LR_ajnad_change
			add_claim_on_loss = no
		}
		title:e_khilafa = {
			change_title_holder = {
				holder = root
				change = scope:LR_ajnad_change
			}
		}

	}
	title:k_wilaya_sham = { set_de_jure_liege_title = title:e_khilafa }

	title:k_wilaya_hijaz = { set_de_jure_liege_title = title:e_khilafa }
	set_primary_title_to = title:e_khilafa

		limit = { LR_ajnad_holds_caliph_title_trigger = yes }
		add_character_modifier = { modifier = LR_ajnad_commander_of_the_faithful_modifier }
	}
	else = {
		add_character_modifier = { modifier = LR_ajnad_guardian_of_the_holy_cities_modifier }
	}
	LR_title_set_reorganise_realm_effect = yes
}
