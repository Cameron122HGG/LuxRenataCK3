# Dynastic tier of the Lux Renata government groups

# Root is a ruler. The game hands out the vanilla administrative and feudal governments as the defaults of their mechanic
# (e.g. on becoming someone's vassal); Lux Renata history never uses them. Replace them with the same tier of the realm's
# group (the top liege's), or failing that the group of the ruler's culture (its government tradition).
LR_replace_vanilla_default_government_effect = {
	if = {
		limit = { has_government = administrative_government }
		# LR 2026-11-18 mixed-group realms (option B): no administrative vassal in a realm whose top liege is not
		# administrative. It goes feudal and is normalized again (the feudal branch picks its culture's group).
		if = {
			limit = {
				top_liege != this
				top_liege = { NOT = { government_has_mechanic = administrative } }
			}
			change_government = feudal_government # re-normalized by on_government_change
		}
		else_if = { # LR 2026-11-18: Muslims take the Islamicate group's administration (user: "all Muslims Iqta")
			limit = { faith = { religion_tag = islam_religion } }
			change_government = iqta_administrative_government
		}
		else_if = {
			limit = {
				top_liege != this
				top_liege = { government_has_mechanic = administrative }
				# LR 2026-11-18 mixed-group realms (option B, user: "lets do B"): administrative vassals always take
				# their top liege's administration, whatever their culture: each group's admin contract has its own
				# province types and flags, which do not mix (see LR_align_admin_vassal_government_effect).
				# Muslims stay Iqta (above). Feudal and clan vassals keep their own group (feudal branch below).
				OR = {
					top_liege = { LR_in_classical_group_trigger = yes }
					top_liege = { LR_in_western_group_trigger = yes }
					top_liege = { LR_in_despotic_group_trigger = yes }
					top_liege = { LR_in_mamlakah_group_trigger = yes }
					top_liege = { LR_in_sudanic_group_trigger = yes }
					top_liege = { LR_in_nilotic_group_trigger = yes }
					top_liege = { LR_in_indic_group_trigger = yes }
					top_liege = { LR_in_islamic_group_trigger = yes }
					top_liege = { LR_in_sinic_group_trigger = yes }
					top_liege = { LR_in_japanese_group_trigger = yes }
					top_liege = { LR_in_sacral_group_trigger = yes }
				}
			}
			if = {
				limit = { top_liege = { LR_in_classical_group_trigger = yes } }
				if = {
					limit = { top_liege = { government_has_flag = government_is_hellenistic_government } }
					change_government = alexandrian_administrative_government
				}
				else = {
					change_government = roman_administrative_government
				}
			}
			else_if = {
				limit = { top_liege = { LR_in_western_group_trigger = yes } }
				change_government = western_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_despotic_group_trigger = yes } }
				change_government = persianate_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_indic_group_trigger = yes } }
				change_government = gupta_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_mamlakah_group_trigger = yes } }
				change_government = mamlakah_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_sudanic_group_trigger = yes } }
				change_government = sudanic_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_nilotic_group_trigger = yes } }
				change_government = nilotic_administrative_government
			}
			else_if = { # mamlakah-only. LR 2026-11-19: Iqta is for Muslims (handled above); a non-Muslim in an Iqta realm takes the secular Mamlakah
				limit = { top_liege = { LR_in_islamic_group_trigger = yes } }
				change_government = mamlakah_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_sinic_group_trigger = yes } }
				if = {
					limit = { top_liege = { government_has_flag = government_is_celestial } }
					change_government = celestial_government
				}
				else_if = {
					limit = { top_liege = { has_government = meritocratic_government } }
					change_government = meritocratic_government
				}
				else = { # LR 2026-11-25 Sinic twins: Junxian, the Sinic administrative twin
					change_government = sinic_administrative_government
				}
			}
			else_if = {
				limit = { top_liege = { LR_in_japanese_group_trigger = yes } }
				change_government = japan_administrative_government
			}
			else_if = {
				limit = { top_liege = { LR_in_sacral_group_trigger = yes } }
				change_government = sacral_administrative_government
			}
		}
		else_if = {
			limit = { has_title = title:e_sacral_kingdom_gaul }
			change_government = sacral_administrative_government # normalized to the DLC tier
		}
		else = {
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				if = {
					limit = { LR_culture_is_hellenic_trigger = yes }
					change_government = alexandrian_administrative_government
				}
				else = {
					change_government = roman_administrative_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = persianate_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = gupta_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_mamlakah } } # LR 2026-11-19: was the Islamicate tradition
				change_government = mamlakah_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sudanic } } # LR 2026-11-19: was the Islamicate tradition
				change_government = sudanic_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_nilotic } } # LR 2026-11-19: was the Islamicate tradition
				change_government = nilotic_administrative_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				change_government = sinic_administrative_government # LR 2026-11-25 Sinic twins: Junxian, normalized to the DLC tier
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					change_government = japan_administrative_government
				}
				else = {
					change_government = japanese_government
				}
			}
			else = { # LR 2026-11-19: no tradition, Anarchy or Barbarian (no administrative tier): Feudal Custom's administration, never vanilla
				change_government = western_administrative_government
			}
		}
	}
	else_if = {
		limit = { has_government = feudal_government }
		if = { # LR 2026-11-18: every Muslim takes the Islamicate group, whatever the culture or the liege's group
			limit = { faith = { religion_tag = islam_religion } }
			change_government = iqta_government # iqta_dynastic_government with the DLC (LR_normalize_dynastic_government_effect)
		}
		else_if = {
			limit = {
				top_liege != this
				# LR 2026-11-18 (user: "vassals should keep their own group"): only a vassal whose culture has no LR
				# government tradition takes its top liege's group; the others fall through to their culture's group below.
				culture = {
					NOR = {
						has_cultural_tradition = lux_tradition_gov_classical
						has_cultural_tradition = lux_tradition_gov_western
						has_cultural_tradition = lux_tradition_gov_despotic
						has_cultural_tradition = lux_tradition_gov_indic
						has_cultural_tradition = lux_tradition_gov_mamlakah
						has_cultural_tradition = lux_tradition_gov_sudanic
						has_cultural_tradition = lux_tradition_gov_nilotic
						has_cultural_tradition = lux_tradition_gov_sinic
						has_cultural_tradition = lux_tradition_gov_japanese
						has_cultural_tradition = lux_tradition_gov_barbarian
					}
				}
				OR = {
					top_liege = { LR_in_classical_group_trigger = yes }
					top_liege = { LR_in_western_group_trigger = yes }
					top_liege = { LR_in_despotic_group_trigger = yes }
					top_liege = { LR_in_mamlakah_group_trigger = yes }
					top_liege = { LR_in_sudanic_group_trigger = yes }
					top_liege = { LR_in_nilotic_group_trigger = yes }
					top_liege = { LR_in_indic_group_trigger = yes }
					top_liege = { LR_in_islamic_group_trigger = yes }
					top_liege = { LR_in_sinic_group_trigger = yes }
					top_liege = { LR_in_japanese_group_trigger = yes }
					top_liege = { LR_in_sacral_group_trigger = yes }
					top_liege = { LR_in_barbarian_group_trigger = yes }
				}
			}
			LR_take_top_liege_feudal_group_effect = yes
		}
		else_if = {
			limit = { has_title = title:e_sacral_kingdom_gaul }
			change_government = sacral_government # normalized to the DLC tier
		}
		else = {
			if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_classical } }
				change_government = classical_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_western } }
				change_government = western_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_despotic } }
				change_government = despotic_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_indic } }
				change_government = indic_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_mamlakah } } # LR 2026-11-19: was the Islamicate tradition
				change_government = mamlakah_government # mamlakah_dynastic_government with the DLC (LR_normalize_dynastic_government_effect)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sudanic } } # LR 2026-11-19: was the Islamicate tradition
				change_government = sudanic_government # sudanic_dynastic_government with the DLC (LR_normalize_dynastic_government_effect)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_nilotic } } # LR 2026-11-19: was the Islamicate tradition
				change_government = nilotic_government # nilotic_dynastic_government with the DLC (LR_normalize_dynastic_government_effect)
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_sinic } }
				change_government = chinese_feudal_government
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_japanese } }
				if = {
					limit = { has_tgp_dlc_trigger = yes }
					add_character_flag = { flag = can_become_japanese_feudal days = 1 } # Soryo is otherwise only allowed under a Japanese realm (tgp_can_become_japanese_feudal_trigger)
					change_government = japan_feudal_government
				}
				else = {
					change_government = japanese_government
				}
			}
			else_if = {
				limit = { culture = { has_cultural_tradition = lux_tradition_gov_barbarian } }
				change_government = great_tribal_government
			}
			else = { # LR 2026-11-19: no tradition or Anarchy, outside an LR realm: Feudal Custom, never vanilla feudal (as Anarchy's fallback)
				change_government = western_feudal_government
			}
		}
	}
	else_if = {
		limit = {
			has_government = clan_government
			faith = { religion_tag = islam_religion } # LR 2026-11-18: every Muslim clan ruler becomes Iqta (user: "all Muslims Iqta")
		}
		change_government = iqta_government # iqta_dynastic_government with the DLC (LR_normalize_dynastic_government_effect)
	}
	# LR 2026-11-18/19: every non-Muslim clan ruler is set to feudal and normalized again (user: "make sure vanilla feudal, clan,
	# and administrative are always superseded"): the feudal branch above then picks the liege's group, the culture's
	# tradition (e.g. Zoroastrian Iranians -> Persianate) or Feudal Custom.
	else_if = {
		limit = { has_government = clan_government }
		change_government = feudal_government # re-normalized by on_government_change
	}
}

# Root is a ruler. With the DLC, the plain starting governments (classical, western feudal, despotic, Indic feudal, sacral) become
# their dynastic version; without it, the dynastic governments fall back to the plain ones.
# LR 2026-11-19 (user) DLC ladder of the 9 administrative groups: Roads to Power + All Under Heaven = dynastic and administrative
# twins (+ Imperial tiers); Roads to Power alone = the RtP-only tier (Roman, Alexandrian, Latin, Persianate, Mamlakah Diwan,
# Sudanic Court, Nilotic Court, Gupta, Iqta Diwan) for the administrative twins and Imperial tiers, the plain government for the
# dynastic twins; neither = the plain group government.
# Run at game start and whenever a government changes, so decisions and events can keep setting the plain government.
LR_normalize_dynastic_government_effect = {
	LR_downgrade_upgraded_roman_government_effect = yes # LR 2026-11-19: no Dominate/Principate for decayed Roman empires
	LR_kosmokratoria_government_effect = yes
	LR_celestial_government_effect = yes # LR 2026-11-25 Sinic twins: Celestial follows h_china (LR_celestial_effects.txt) # LR 2026-11-19: the Kosmokratoria follows h_hellenistic_empire and the DLCs (LR_kosmokratoria_effects.txt)
	LR_japanese_court_government_effect = yes # LR 2026-11-25 Japanese ranks (tools/japan_ranks/build_japan_ranks.py): Chotei, Bakufu, Tenka (LR_japan_ranks_effects.txt)
	LR_sacral_unctio_government_effect = yes # LR 2026-11-25 Unctio (tools/sacral_unctio/build_sacral_unctio.py): the restored Sacral King (LR_sacral_unctio_effects.txt)
	LR_persianate_farr_government_effect = yes # LR 2026-11-25 Farr (tools/persianate_farr/build_persianate_farr.py): the holder of Eranshahr (LR_persianate_farr_effects.txt)
	LR_gupta_sarvabhauma_government_effect = yes # LR 2026-11-25 Sarvabhauma (tools/gupta_sarvabhauma/build_gupta_sarvabhauma.py): h_india or the Ashvamedha (LR_gupta_sarvabhauma_effects.txt)
	LR_western_universum_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_sacral_maiestas_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_iqta_imama_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_mamlakah_alma_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_nilotic_kebra_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_sudanic_kurufaba_government_effect = yes # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
	LR_signature_modifiers_effect = yes # LR 2026-11-25 GRADUS signature: Mamlakah, Nilotic, Sudanic standing modifiers (LR_gradus_signature_effects.txt)
	LR_roman_republic_government_effect = yes # LR 2026-11-25 regional republics (tools/regional_republics/build_regional_republics.py): the Roman Republic government
	LR_roman_senate_update_effect = yes # LR 2026-11-25 Reconvene the Senate (tools/roman_senate/build_roman_senate.py): the Senate's seat modifier
	LR_personality_modifiers_effect = yes # LR 2026-11-25 GRADUS personality: one per group on every rank (LR_gradus_personality_effects.txt)
	LR_dynastic_signature_effect = yes # LR 2026-11-25 GRADUS dynastic signature: the dynastic ranks' house modifiers (LR_gradus_dynastic_signature_effects.txt)
	LR_schedule_roman_title_twins_effect = yes # LR 2026-11-19: Republic titles for the Principate, Empire titles (a day later; swapping inside change_government crashed) otherwise (LR_roman_title_twins_effects.txt)
	LR_principate_align_vassal_effect = yes # LR 2026-11-19: Principate governors (LR_principate_effects.txt)
	LR_align_admin_vassal_government_effect = yes # LR 2026-11-18 mixed-group realms (LR_mixed_group_realm_effects.txt)
	LR_replace_vanilla_default_government_effect = yes
	# LR 2026-11-18 (user: "make despotic Muslims Iqta too"): Muslims in the despotic (Persianate) group take the
	# Islamicate government of the same tier. Iqta becomes Iqta Dynastic with the DLC further down.
	if = {
		limit = {
			faith = { religion_tag = islam_religion }
			OR = {
				has_government = despotic_government
				has_government = persianate_dynastic_government
				has_government = persianate_administrative_government
				has_government = persianate_imperial_government
				has_government = persianate_farr_government # LR 2026-11-25 Farr (tools/persianate_farr/build_persianate_farr.py)
				has_government = mamlakah_government # LR 2026-11-19: the Mamlakah group too
				has_government = sudanic_government # LR 2026-11-19: the Sudanic group too
				has_government = nilotic_government # LR 2026-11-19: the Nilotic group too
				has_government = mamlakah_dynastic_government
				has_government = sudanic_dynastic_government
				has_government = nilotic_dynastic_government
				has_government = mamlakah_administrative_government
				has_government = sudanic_administrative_government
				has_government = nilotic_administrative_government
				has_government = mamlakah_imperial_government
				has_government = mamlakah_alma_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
				has_government = sudanic_imperial_government
				has_government = sudanic_kurufaba_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
				has_government = nilotic_imperial_government
				has_government = nilotic_kebra_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
			}
		}
		if = {
			limit = {
				OR = {
					has_government = persianate_administrative_government
					has_government = mamlakah_administrative_government
					has_government = sudanic_administrative_government
					has_government = nilotic_administrative_government
				}
			}
			change_government = iqta_administrative_government
		}
		else_if = { # LR 2026-11-25: the imperial ranks of the group become the Sultanate, the Iqta imperial rank
			limit = {
				OR = {
					has_government = persianate_imperial_government
					has_government = persianate_farr_government # LR 2026-11-25 Farr (tools/persianate_farr/build_persianate_farr.py)
					has_government = mamlakah_imperial_government
					has_government = mamlakah_alma_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
					has_government = sudanic_imperial_government
					has_government = sudanic_kurufaba_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
					has_government = nilotic_imperial_government
					has_government = nilotic_kebra_government # LR 2026-11-25 universal ranks (tools/universal_ranks/build_universal_ranks.py)
				}
			}
			change_government = iqta_imperial_government
		}
		else_if = {
			limit = {
				OR = {
					has_government = persianate_dynastic_government
					has_government = mamlakah_dynastic_government
					has_government = sudanic_dynastic_government
					has_government = nilotic_dynastic_government
				}
			}
			change_government = iqta_dynastic_government
		}
		else = {
			change_government = iqta_government
		}
	}
	# LR 2026-11-19: Iqta is only for Muslims (user: "Iqta group can be a faith based subcategory for any ruler practicing
	# islam"). A non-Muslim Iqta ruler goes back to the vanilla default of the same mechanic, which
	# LR_replace_vanilla_default_government_effect maps to the liege's or the culture's group on re-normalizing.
	else_if = {
		limit = {
			NOT = { faith = { religion_tag = islam_religion } }
			government_has_flag = government_is_islamic_group
		}
		if = {
			limit = { government_has_mechanic = administrative }
			change_government = administrative_government # re-normalized by on_government_change
		}
		else = {
			change_government = feudal_government # re-normalized by on_government_change
		}
	}
	# Japanese realms: Soryo and Ritsuryo need All Under Heaven; without it they use Japanese Feudal (sinic group)
	if = {
		limit = {
			has_tgp_dlc_trigger = no
			OR = {
				has_government = japan_feudal_government
				AND = { # LR 2026-11-25: with Roads to Power alone Ritsuryo becomes Autocratic (LR_admin_to_rtp_government_effect)
					has_government = japan_administrative_government
					LR_has_rtp_governments_dlc_trigger = no
				}
			}
		}
		change_government = japanese_government
	}
	else_if = {
		limit = {
			has_tgp_dlc_trigger = yes
			has_government = japanese_government
		}
		add_character_flag = { flag = can_become_japanese_feudal days = 1 } # Soryo is otherwise only allowed under a Japanese realm (tgp_can_become_japanese_feudal_trigger)
		change_government = japan_feudal_government
	}
	# Sinic group: Meritocratic and Celestial need All Under Heaven; without it they use Chinese Feudal
	else_if = {
		limit = {
			has_tgp_dlc_trigger = no
			LR_has_rtp_governments_dlc_trigger = no # LR 2026-11-25: with Roads to Power alone they become Autocratic (below)
			OR = {
				has_government = meritocratic_government
				has_government = celestial_government
			}
		}
		change_government = chinese_feudal_government
	}
	# Sacral group: its DLC tiers (Soryo, Ritsuryo) need only the All Under Heaven DLC
	if = {
		limit = { government_has_flag = government_is_sacral_group }
		if = {
			limit = {
				has_tgp_dlc_trigger = yes
				has_government = sacral_government
			}
			change_government = sacral_feudal_government
		}
		else_if = {
			limit = {
				has_tgp_dlc_trigger = no
				OR = {
					has_government = sacral_feudal_government
					AND = {
						has_government = sacral_administrative_government
						LR_has_rtp_governments_dlc_trigger = no
					}
				}
			}
			change_government = sacral_government
		}
		else_if = { # LR 2026-11-25: with Roads to Power alone Palatine becomes Autocratic
			limit = {
				has_government = sacral_administrative_government
				LR_has_rtp_governments_dlc_trigger = yes
			}
			change_government = rtp_autocratic_government
		}
	}
	else_if = {
		limit = { LR_has_dynastic_governments_dlc_trigger = yes }
		if = {
			limit = { has_government = classical_government }
			if = {
				limit = { LR_culture_is_hellenic_trigger = yes }
				change_government = alexandrian_dynastic_government
			}
			else = {
				change_government = roman_dynastic_government
			}
