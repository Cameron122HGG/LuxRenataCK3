# Galatian title set (gen_galatian_title_set.py): root restores a Galatian kingdom, laying out its duchies over its
# counties, and receives it. The kingdom goes under its empire, the empire under Reixath Galáthon if root holds it.

LR_restore_galat_tolistobogioi_effect = { # Tolistobogioi
	title:k_galat_tolistobogioi = { set_de_jure_liege_title = title:e_galat_trigalatia }
	if = {
		limit = { has_title = title:h_reixath_galathon }
		title:e_galat_trigalatia = { set_de_jure_liege_title = title:h_reixath_galathon }
	}
	title:d_galat_pessinunta = { set_de_jure_liege_title = title:k_galat_tolistobogioi }
	title:c_gordium = { set_de_jure_liege_title = title:d_galat_pessinunta }
	title:d_galat_uoturoi = { set_de_jure_liege_title = title:k_galat_tolistobogioi }
	title:c_galatia_salutaris = { set_de_jure_liege_title = title:d_galat_uoturoi }
	title:d_galat_ambitoutoi = { set_de_jure_liege_title = title:k_galat_tolistobogioi }
	title:c_amorion = { set_de_jure_liege_title = title:d_galat_ambitoutoi }
	title:d_galat_blukion = { set_de_jure_liege_title = title:k_galat_tolistobogioi }
	title:c_hadrianopolis = { set_de_jure_liege_title = title:d_galat_blukion }
	create_title_and_vassal_change = {
		type = created
		save_scope_as = LR_galat_change
		add_claim_on_loss = no
	}
	title:k_galat_tolistobogioi = {
		change_title_holder = {
			holder = root
			change = scope:LR_galat_change
		}
	}
	resolve_title_and_vassal_change = scope:LR_galat_change
}

# Used by LR_form_reixath_galathon_effect: root restores every Galatian kingdom whose lands it completely controls and
# that is not yet restored.
LR_galat_restore_all_controlled_effect = {
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_tolistobogioi
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_tolistobogioi }
		}
		LR_restore_galat_tolistobogioi_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_tektosages
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_tektosages }
		}
		LR_restore_galat_tektosages_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_trokmoi
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_trokmoi }
		}
		LR_restore_galat_trokmoi_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_pontogalatia
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_pontogalatia }
		}
		LR_restore_galat_pontogalatia_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_tulis
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_tulis }
		}
		LR_restore_galat_tulis_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_skordiskoi
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_skordiskoi }
		}
		LR_restore_galat_skordiskoi_effect = yes
	}
	if = {
		limit = {
			completely_controls_region = LR_area_k_galat_makedunia
			LR_satrapy_restorable_trigger = { SATRAPY = k_galat_makedunia }
		}
		LR_restore_galat_makedunia_effect = yes
	}
}

# Used by LR_form_reixath_galathon_effect: both high-kingships go under Reixath Galáthon, and every restored kingdom
# under its high-kingship.
LR_galat_set_under_reixath_effect = {
	if = {
		limit = { title:k_galat_tolistobogioi = { any_de_jure_county = { always = yes } } }
		title:k_galat_tolistobogioi = { set_de_jure_liege_title = title:e_galat_trigalatia }
	}
	if = {
		limit = { title:k_galat_tektosages = { any_de_jure_county = { always = yes } } }
		title:k_galat_tektosages = { set_de_jure_liege_title = title:e_galat_trigalatia }
	}
	if = {
		limit = { title:k_galat_trokmoi = { any_de_jure_county = { always = yes } } }
		title:k_galat_trokmoi = { set_de_jure_liege_title = title:e_galat_trigalatia }
	}
	if = {
		limit = { title:k_galat_pontogalatia = { any_de_jure_county = { always = yes } } }
		title:k_galat_pontogalatia = { set_de_jure_liege_title = title:e_galat_trigalatia }
	}
	if = {
		limit = { title:k_galat_tulis = { any_de_jure_county = { always = yes } } }
		title:k_galat_tulis = { set_de_jure_liege_title = title:e_galat_danuuia }
	}
	if = {
		limit = { title:k_galat_skordiskoi = { any_de_jure_county = { always = yes } } }
		title:k_galat_skordiskoi = { set_de_jure_liege_title = title:e_galat_danuuia }
	}
	if = {
		limit = { title:k_galat_makedunia = { any_de_jure_county = { always = yes } } }
		title:k_galat_makedunia = { set_de_jure_liege_title = title:e_galat_danuuia }
	}
	title:e_galat_trigalatia = { set_de_jure_liege_title = title:h_reixath_galathon }
	title:e_galat_danuuia = { set_de_jure_liege_title = title:h_reixath_galathon }
}
