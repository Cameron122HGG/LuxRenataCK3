# Can this character be appointed as a dux by $EMPEROR$?
LR_can_be_roman_dux_trigger = {
	is_adult = yes
	is_imprisoned = no
	NOT = { this = $EMPEROR$ }
	NOT = { is_heir_of = $EMPEROR$ }
	LR_holds_roman_office_trigger = no
	can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = $EMPEROR$ }
	# Like the Grand Marshal: a courtier, or a Roman Imperial ruler below kingdom tier
	OR = {
		is_ruler = no
		AND = {
			government_has_flag = government_is_roman_imperial
			highest_held_title_tier <= tier_duchy
		}
	}
}

# Is the diocese $DIOCESE$ held within this emperor's realm, and its command not yet filled?
LR_roman_dux_command_available_trigger = {
	exists = title:$DIOCESE$.holder
	title:$DIOCESE$.holder.top_liege = root
	NOT = { exists = title:$TITLE$.holder }
}
