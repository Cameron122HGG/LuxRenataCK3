# Lux Renata (11-18): Roman and Western Roman heritage portraits.
# The roman and western_roman cultures and their Latin-heritage descendants (Italians, Gallo-Romans, Vlachs,
# Romano-Britons...) shave more often than the bearded Greek East (Byzantine-heritage children are excluded).
# Used by has_no_beard in gfx/portraits/portrait_modifiers/01_beards_base.txt.
# scope:culture is the portrait's culture (also set when no character exists, e.g. culture previews).
portrait_western_roman_heritage_trigger = {
	scope:culture ?= {
		OR = {
			this = culture:roman
			this = culture:western_roman
			AND = {
				any_parent_culture_or_above = {
					OR = {
						this = culture:roman
						this = culture:western_roman
					}
				}
				has_cultural_pillar = heritage_latin
			}
		}
	}
}
