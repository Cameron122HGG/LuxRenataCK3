# Solar (radiate) crown (11-20): Roman emperors of the solar religion wear it.
# Used by the lr_solar_crown_* entries in gfx/portraits/portrait_modifiers/renov_clothes.txt
# (group renov_roman_crown_modifiers, where it outweighs the golden laurel).

# Character scope: holds a Roman imperial title (not the Republic) and follows a faith of the solar religion
lr_solar_roman_emperor_trigger = {
	exists = this
	exists = this.faith
	faith = { religion = religion:solar_religion }
	OR = {
		has_title = title:h_roman_empire
		has_title = title:h_eastern_roman_empire
		has_title = title:h_western_roman_empire
		has_title = title:e_eastern_roman_empire
		has_title = title:e_western_roman_empire
		has_title = title:e_byzantium
		has_title = title:e_spqr
	}
}
