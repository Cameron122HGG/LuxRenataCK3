# Dynastic tier of the Lux Renata government groups (twins of the administrative governments)

# The dynastic governments need the DLC of their administrative twin; without it realms use the plain government (classical etc.)
LR_has_dynastic_governments_dlc_trigger = {
	has_ep3_dlc_trigger = yes
	has_tgp_dlc_trigger = yes
}

# Hellenistic-heritage cultures get the Alexandrian rather than the Roman version of the classical group
LR_culture_is_hellenic_trigger = {
	culture = {
		OR = {
			has_cultural_pillar = heritage_hellenistic
			has_cultural_pillar = heritage_indo_hellenic
			has_cultural_pillar = heritage_sino_hellenic
		}
	}
}

# Does this administrative (or top-tier) government have a dynastic twin to fall back to?
LR_has_dynastic_twin_trigger = {
	OR = {
		has_government = roman_administrative_government
		has_government = roman_autocratic_government
		has_government = roman_imperial_government
		has_government = decayed_principate_government
		has_government = alexandrian_administrative_government
		has_government = western_administrative_government
		has_government = persianate_administrative_government
		has_government = persianate_autocratic_government
		has_government = gupta_administrative_government
		has_government = iqta_administrative_government
		has_government = japan_administrative_government
		has_government = sacral_administrative_government
	}
}
