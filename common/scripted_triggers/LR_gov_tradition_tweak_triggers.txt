# Lux Renata 2026-11-18: government tradition tweaks (LR_gov_traditions.txt): realm drift discount, pick requirements, AI adoption

# Character scope: the government belongs to the classical group
LR_government_in_classical_group_trigger = {
	government_has_flag = government_is_classical_group
}

# Culture scope: the culture knows the classical group: some of its counties lie in a realm of that group, or its head's
# realm, or a realm bordering it, is of that group
LR_culture_knows_classical_group_trigger = {
	OR = {
		any_culture_county = {
			holder ?= { top_liege = { LR_government_in_classical_group_trigger = yes } }
		}
		culture_head ?= {
			OR = {
				top_liege = { LR_government_in_classical_group_trigger = yes }
				any_neighboring_top_liege_realm_owner = { LR_government_in_classical_group_trigger = yes }
			}
		}
	}
}
