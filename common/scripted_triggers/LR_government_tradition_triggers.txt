# Culture scope: does the culture have any government tradition?
LR_culture_has_government_tradition_trigger = {
	OR = {
		has_cultural_tradition = lux_tradition_gov_classical
		has_cultural_tradition = lux_tradition_gov_western
		has_cultural_tradition = lux_tradition_gov_despotic
		has_cultural_tradition = lux_tradition_gov_indic
		has_cultural_tradition = lux_tradition_gov_mamlakah
		has_cultural_tradition = lux_tradition_gov_sudanic
		has_cultural_tradition = lux_tradition_gov_nilotic
		has_cultural_tradition = lux_tradition_gov_sinic
		has_cultural_tradition = lux_tradition_gov_japanese
		has_cultural_tradition = lux_tradition_gov_barbarian
	}
}

# Character scope: does the government belong to the group the culture's government tradition names?
LR_government_matches_culture_group_trigger = {
	OR = {
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_classical }
			government_has_flag = government_is_classical_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_western }
			government_has_flag = government_is_western_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_despotic }
			government_has_flag = government_is_despotic_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_indic }
			government_has_flag = government_is_indic_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_mamlakah }
			government_has_flag = government_is_mamlakah_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_sudanic }
			government_has_flag = government_is_sudanic_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_nilotic }
			government_has_flag = government_is_nilotic_group
		}
		AND = { # LR 2026-11-19: Iqta is the faith-based group of every Muslim ruler, whatever the culture's tradition
			faith = { religion_tag = islam_religion }
			government_has_flag = government_is_islamic_group
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_sinic }
			government_has_flag = government_is_sinic_group
			NOT = { government_has_flag = government_is_japanese_plain } # Japanese Feudal belongs to the Japanese tradition
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_japanese }
			OR = {
				government_has_flag = government_is_japanese_group
				government_has_flag = government_is_japanese_plain # Japanese Feudal: the Japanese tradition's government without All Under Heaven
			}
		}
		AND = {
			culture = { has_cultural_tradition = lux_tradition_gov_barbarian }
			government_has_flag = government_is_barbarian_group
		}
	}
}

# Character scope: the sacral group belongs to the realm of e_sacral_kingdom_gaul, not to a culture.
# A sacral ruler there keeps their government whatever their culture's government tradition says.
LR_in_sacral_kingdom_gaul_realm_trigger = {
	OR = {
		has_title = title:e_sacral_kingdom_gaul
		top_liege ?= { has_title = title:e_sacral_kingdom_gaul }
	}
}

# Character scope, $OTHER$ = another character: do both governments belong to the same government group?
# (Japanese Feudal counts as Japanese, not sinic.) Used by the Anarchy event (LR_gov_anarchy_events.txt).
LR_government_shares_tradition_group_trigger = {
	OR = {
		AND = {
			government_has_flag = government_is_classical_group
			$OTHER$ = { government_has_flag = government_is_classical_group }
		}
		AND = {
			government_has_flag = government_is_western_group
			$OTHER$ = { government_has_flag = government_is_western_group }
		}
		AND = {
			government_has_flag = government_is_despotic_group
			$OTHER$ = { government_has_flag = government_is_despotic_group }
		}
		AND = {
			government_has_flag = government_is_indic_group
			$OTHER$ = { government_has_flag = government_is_indic_group }
		}
		AND = {
			government_has_flag = government_is_islamic_group
			$OTHER$ = { government_has_flag = government_is_islamic_group }
		}
		AND = {
			government_has_flag = government_is_mamlakah_group
			$OTHER$ = { government_has_flag = government_is_mamlakah_group }
		}
		AND = {
			government_has_flag = government_is_sudanic_group
			$OTHER$ = { government_has_flag = government_is_sudanic_group }
		}
		AND = {
			government_has_flag = government_is_nilotic_group
			$OTHER$ = { government_has_flag = government_is_nilotic_group }
		}
		AND = {
			government_has_flag = government_is_barbarian_group
			$OTHER$ = { government_has_flag = government_is_barbarian_group }
		}
		AND = {
			government_has_flag = government_is_sacral_group
			$OTHER$ = { government_has_flag = government_is_sacral_group }
		}
		AND = {
			government_has_flag = government_is_sinic_group
			NOT = { government_has_flag = government_is_japanese_plain }
			$OTHER$ = {
				government_has_flag = government_is_sinic_group
				NOT = { government_has_flag = government_is_japanese_plain }
			}
		}
		AND = {
			OR = {
				government_has_flag = government_is_japanese_group
				government_has_flag = government_is_japanese_plain
			}
			$OTHER$ = {
				OR = {
					government_has_flag = government_is_japanese_group
					government_has_flag = government_is_japanese_plain
				}
			}
		}
	}
}
