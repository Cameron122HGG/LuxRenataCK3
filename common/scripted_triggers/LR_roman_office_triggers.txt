# Landless Roman offices: Praetorian Prefects and ministries

LR_is_roman_prefect_title_trigger = {
	OR = {
		this = title:k_praefectus_praetorio_italia
		this = title:k_praefectus_praetorio_gallia
		this = title:k_praefectus_praetorio_illyricum
		this = title:k_praefectus_praetorio_macedonia
		this = title:k_praefectus_praetorio_oriens
	}
}

LR_is_roman_minister_title_trigger = {
	OR = {
		this = title:k_lr_ministry_east_cubiculi
		this = title:k_lr_ministry_west_cubiculi
		this = title:k_lr_ministry_east_agentes
		this = title:k_lr_ministry_west_agentes
		this = title:k_lr_ministry_east_officiorum
		this = title:k_lr_ministry_west_officiorum
		this = title:k_lr_ministry_east_memoriae
		this = title:k_lr_ministry_west_memoriae
		this = title:k_lr_ministry_east_urbi
		this = title:k_lr_ministry_west_urbi
		this = title:k_lr_ministry_east_militum
		this = title:k_lr_ministry_west_militum
	}
}

	OR = {

		this = title:k_lr_ministry_west_militum
	}
}

# Does this character hold any landless Roman office (dux, legion, prefect or minister)?
LR_holds_roman_office_trigger = {
	any_held_title = {
		OR = {
			LR_is_roman_command_title_trigger = yes
			LR_is_roman_prefect_title_trigger = yes
			LR_is_roman_minister_title_trigger = yes
		}
	}
}

# Can this character be appointed Praetorian Prefect by $EMPEROR$?
LR_can_be_roman_prefect_trigger = {
	is_adult = yes
	is_imprisoned = no
	NOT = { this = $EMPEROR$ }
	NOT = { is_heir_of = $EMPEROR$ }
	LR_holds_roman_office_trigger = no
	LR_can_be_roman_minister_trigger = { COURT_OWNER = $EMPEROR$ }
	OR = {
		is_ruler = no
		AND = {
			government_has_flag = government_is_roman_imperial
			highest_held_title_tier <= tier_duchy
		}
	}
}

# Every diocese of the prefecture of Italy and Africa is held within root's realm, and the prefecture has no prefect
LR_roman_prefecture_italia_available_trigger = {
	exists = title:k_diocese_italia_annonaria.holder
	title:k_diocese_italia_annonaria.holder.top_liege = root
	exists = title:k_diocese_italia_suburbicaria.holder
	title:k_diocese_italia_suburbicaria.holder.top_liege = root
	exists = title:k_diocese_africa.holder
	title:k_diocese_africa.holder.top_liege = root
	NOT = { exists = title:k_praefectus_praetorio_italia.holder }
}
