# Gate for every Renovatio Roman portrait modifier (togas, centurion armour, helmets...).
# Since 11-20 the Roman Nostalgia tradition alone decides it. Which cultures start with the
# tradition is set by the game rule lr_roman_nostalgia_start (common/on_action/LR_roman_nostalgia_on_actions.txt).
renov_roman_portrait_trigger = {
	exists = this
	exists = this.culture
	this.culture = { has_cultural_tradition = lux_tradition_roman_nostalgia }
}
