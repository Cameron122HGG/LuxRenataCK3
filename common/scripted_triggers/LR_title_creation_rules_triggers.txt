# LR 2026-11-22 (user): the body of the can_create_title rule (common/scripted_rules/00_rules.txt), moved here so
# script can ask the same question before giving a title (the republic rank-up, LR_rank_up_republic_effect).
# Character scope (the creator); TITLE = the title. Edit the gates here; tools/register_frankish_sets.py and
# tools/register_punic_set.py add their set gates here too.
LR_title_creation_rules_trigger = {
	# LR: the de jure base list (gen_base_titles.py): its duchies, kingdoms and empires only come from decisions and events
	trigger_if = {
		limit = { $TITLE$ = { LR_is_base_list_title_trigger = yes } }
		custom_tooltip = {
			text = LR_base_list_creation_tooltip
			always = no
		}
	}
	# LR: warlord titles (gen_warlord_titles.py) open for creation by region (user, 2026-11-22): from the year their capital
	# county joins the warlord map (936 / 986 / 1036, the Eastern Roman core 1060; tools/warlord_open_dates.py). Before that only decisions and events give them.
	trigger_if = {
		limit = { $TITLE$ = { LR_is_warlord_title_trigger = yes } }
		custom_tooltip = {
			text = LR_warlord_title_creation_tooltip
			LR_warlord_title_is_open_trigger = { TITLE = $TITLE$ }
		}
	}
	# LR 2026-11-22 (user): the Archontes (city lords risen from the Potentes) create duchies, and kingdoms only from the
	# warlord titles; nothing higher until they settle into their culture's government (LR_settle_archontes_decision)
	trigger_if = {
		limit = { government_has_flag = government_is_archontes }
		custom_tooltip = {
			text = LR_archontes_duchy_cap_tooltip
			OR = {
				$TITLE$.tier <= tier_duchy
				AND = {
					$TITLE$.tier = tier_kingdom
					$TITLE$ = { LR_is_warlord_title_trigger = yes }
				}
			}
		}
	}
	# LR: the Bronze Age and Classical Age titles (gen_bronze_classical.py) can only be created in their own age
	trigger_if = {
		limit = { $TITLE$ = { LR_is_bronze_title_trigger = yes } }
		custom_tooltip = {
			text = LR_bronze_title_creation_tooltip
			OR = {
				AND = {
					current_date >= 1700.1.1
					current_date < 3800.1.1
				}
				# LR: revived Hatti (LR_revive_hatti_decision): Luwian-faith rulers may create its titles in any age
				AND = {
					$TITLE$ = { has_variable = LR_revived_hatti_title }
					faith.religion ?= religion:anatolian_religion
				}
			}
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_is_classical_title_trigger = yes } }
		custom_tooltip = {
			text = LR_classical_title_creation_tooltip
			current_date >= 3800.1.1
			current_date < 5641.1.1 # LR: the Classical layer lasts to the medieval switch (gen_layers.py)
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_is_medieval_title_trigger = yes } }
		custom_tooltip = {
			text = LR_medieval_title_creation_tooltip
			current_date >= 4677.1.1
			current_date < 6036.1.1 # LR: the medieval set lasts until the warlord map has reached every region (gen_medieval.py)
		}
	}
	# LR: the Eastern Roman Themes (tiers_936.py) can only be created by restored Roman emperors and their vassals
	trigger_if = {
		limit = { $TITLE$ = { LR_is_eastern_roman_theme_title_trigger = yes } }
		custom_tooltip = {
			text = LR_eastern_roman_theme_creation_tooltip
			LR_title_set_roman_native_trigger = yes
		}
	}
	# LR: Roman vassal tiers (user, 2026-11-05): no kingdom-tier vassals under a decayed Roman empire
	trigger_if = {
		limit = {
			$TITLE$ = { LR_is_landed_kingdom_title_trigger = yes }
			is_independent_ruler = no
		}
		custom_tooltip = {
			text = LR_decayed_roman_no_vassal_kingdoms_tooltip
			NOT = { top_liege = { LR_is_decayed_roman_realm_trigger = yes } }
		}
	}
	# LR: title sets (gen_title_sets.py, LR_title_set_rules.md): only a set's natives may create its titles
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_hellenistic_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_hellenistic_creation_tooltip
			LR_title_set_hellenistic_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_roman_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_roman_creation_tooltip
			LR_title_set_roman_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_persian_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_persian_creation_tooltip
			LR_title_set_persian_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_hebrew_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_hebrew_creation_tooltip
			LR_title_set_hebrew_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_brythonic_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_brythonic_creation_tooltip
			LR_title_set_brythonic_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_galatian_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_galatian_creation_tooltip
			LR_title_set_galatian_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_punic_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_punic_creation_tooltip
			LR_title_set_punic_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_pontic_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_pontic_creation_tooltip
			LR_title_set_pontic_native_trigger = yes
			LR_pontic_set_unlocked_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_kemet_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_kemet_creation_tooltip
			LR_title_set_kemet_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_frankish_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_frankish_creation_tooltip
			LR_title_set_frankish_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_missatica_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_missatica_creation_tooltip
			LR_title_set_missatica_native_trigger = yes
		}
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_german_title_trigger = yes } }
		custom_tooltip = {
			text = LR_title_set_german_creation_tooltip
			LR_title_set_german_native_trigger = yes
		}
	}
}
