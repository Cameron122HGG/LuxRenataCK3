# Successor satrapies

# Character: holds one of the successor kingdoms or empires
LR_is_diadochi_ruler_trigger = {
	OR = {
		has_title = title:e_antipatrid_empire
		has_title = title:k_antipatrid_kingdom
		has_title = title:e_antigonid_empire
		has_title = title:k_antigonid_kingdom
		has_title = title:e_seleukid_empire
		has_title = title:e_ptolemaic_empire
	}
}

# Satrapy can be (re)restored: it has no land, or none of its de jure hierarchy is held
# Usage: LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_syria }
LR_satrapy_restorable_trigger = {
	OR = {
		title:$SATRAPY$ = { is_titular = yes }
		title:$SATRAPY$ = {
			NOT = {
				any_in_de_jure_hierarchy = { is_titular = no }
			}
		}
	}
}
