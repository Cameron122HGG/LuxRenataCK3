# LR 2026-11-25 group republics (tools/group_republics/build_group_republics.py)
# A republic of Phoenician heritage (Carthage, Tyre, Gades): a landed assembly head (the Suffetate).
LR_punic_group_republic_trigger = {
	government_has_flag = government_is_republic
	NOT = { government_has_flag = government_is_principate }
	is_landed = yes
	gdp_is_assembly_head_trigger = yes
	culture = { has_cultural_pillar = heritage_phoenician }
}

LR_co_suffete_candidate_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	NOT = { this = scope:LR_first_suffete }
	NOT = { house ?= scope:LR_first_suffete.house }
}
