# Character (an Augustus, as root): can command CMD (over REGION) be given to a Caesar? (2026-11-25: a dynamic
# title is made for it.) No living Caesar holds it, your capital is outside it, a county of your own domain is in
# it, and at least half its counties are in your realm.
LR_caesar_can_command_trigger = {
	NOT = { any_in_global_list = { variable = LR_caesar_commands var:LR_caesar_region ?= flag:$CMD$ } }
	NOT = { capital_province = { geographical_region = $REGION$ } }
	any_held_title = {
		tier = tier_county
		is_landless_type_title = no
		title_province = { geographical_region = $REGION$ }
	}
	any_county_in_region = {
		region = $REGION$
		percent >= 0.5
		holder.top_liege ?= root
	}
}
