# LR Roman Republic triggers (Silk & Silver / 1.21), written 2026-11-17.
# Restoration checks reuse LR_roman_restoration_triggers.txt (county in realm, prefectures).

# Title scope
LR_is_roman_republic_title_trigger = {
	OR = {
		this = title:e_western_roman_republic
		this = title:e_eastern_roman_republic
		this = title:h_roman_republic
		this = title:h_western_roman_republic # LR 2026-11-19
		this = title:h_eastern_roman_republic # LR 2026-11-19
	}
}

# Character scope
LR_is_roman_republic_consul_trigger = {
	OR = {
		has_title = title:e_western_roman_republic
		has_title = title:e_eastern_roman_republic
		has_title = title:h_roman_republic
		has_title = title:h_western_roman_republic # LR 2026-11-19
		has_title = title:h_eastern_roman_republic # LR 2026-11-19
	}
}

# Character scope: who may see a restore-the-republic decision. HALF = western / eastern
LR_roman_republic_restorer_shown_trigger = {
	has_gdp_dlc_trigger = yes
	is_ruler = yes
	is_playable_character = yes
	government_has_flag = government_is_republic
	NOT = { exists = title:e_$HALF$_roman_republic.holder }
	NOT = { exists = title:h_roman_republic.holder } # LR 11-18: no separate half once the Res Publica is united
	NOT = { has_global_variable = flag_restored_$HALF$_roman_republic }
	NOT = {
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_restored_$HALF$_roman_republic
		}
	}
}

# Character scope: the republic is big enough for an empire-tier title (Republic Tier Law kingdom or higher)
LR_roman_republic_tier_ready_trigger = {
	OR = {
		has_realm_law = republic_tier_kingdom
		has_realm_law = republic_tier_empire
	}
}

# LR 11-18 (user): republics cannot create titles, so a diocese counts when its land is controlled,
# not only when the k_diocese title is held. The restore effect then grants the dioceses.
# Character scope. DIOCESE = diocese key (region custom_roman_<DIOCESE>, title k_diocese_<DIOCESE>)
LR_roman_republic_diocese_ready_trigger = {
	OR = {
		LR_roman_restoration_diocese_held_trigger = { DIOCESE = $DIOCESE$ }
		completely_controls_region = custom_roman_$DIOCESE$
	}
}

# Character scope: the land of the Prefecture of Italia
LR_roman_republic_prefecture_italia_trigger = {
	LR_roman_republic_diocese_ready_trigger = { DIOCESE = italia_annonaria }
	LR_roman_republic_diocese_ready_trigger = { DIOCESE = italia_suburbicaria }
	LR_roman_republic_diocese_ready_trigger = { DIOCESE = africa }
	LR_roman_republic_diocese_ready_trigger = { DIOCESE = mauretania }
}
