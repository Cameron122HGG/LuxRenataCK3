# A republic of the Hellenistic group: a landed assembly head whose culture keeps Classical Statecraft and is of Hellenistic
# heritage (the Roman group's twin, LR_roman_group_republic_trigger).
LR_hellenic_group_republic_trigger = {
	government_has_flag = government_is_republic
	NOT = { government_has_flag = government_is_principate }
	is_landed = yes
	gdp_is_assembly_head_trigger = yes
	culture = { has_cultural_tradition = lux_tradition_gov_classical }
	LR_culture_is_hellenic_trigger = yes
}

	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	NOT = { this = scope:LR_archon }
	NOT = { house ?= scope:LR_archon.house }
}
