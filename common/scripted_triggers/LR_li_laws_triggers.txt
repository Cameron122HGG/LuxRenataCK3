# a ruler's soul: their rite's soul doctrine if it has one, else their faith's
LR_li_rite_has_soul_trigger = {
	exists = rite
	rite = {
		OR = {
			rite_has_doctrine = lux_doctrine_populist_soul
			rite_has_doctrine = lux_doctrine_clerical_soul
			rite_has_doctrine = lux_doctrine_statist_soul
			rite_has_doctrine = lux_doctrine_traditional_soul
			rite_has_doctrine = lux_doctrine_scholarly_soul
			rite_has_doctrine = lux_doctrine_martial_soul
			rite_has_doctrine = lux_doctrine_messianic_soul
		}
	}
}

LR_li_soul_populist_trigger = {
	trigger_if = {
		limit = { LR_li_rite_has_soul_trigger = yes }
		rite = { rite_has_doctrine = lux_doctrine_populist_soul }
	}
	trigger_else = {
		faith = { has_doctrine = lux_doctrine_populist_soul }
	}
}

# the faith lets women sit on the council like men (neither male- nor female-dominated)
LR_li_women_council_open_trigger = {
	rite = {
		NOT = { rite_has_parameter = male_dominated_council }
		NOT = { rite_has_parameter = female_dominated_council }
	}
}

# $COURT_OWNER$'s Status of Women law (LR_women_status) for a woman on the council; men, rulers and overrides pass
LR_li_women_council_law_trigger = {
	OR = {
		is_female = no
		is_ruler = yes
		has_variable = can_be_councillor_override
		$COURT_OWNER$ = { NOT = { has_realm_law_flag = LR_women_status_0 has_realm_law_flag = LR_women_status_1 has_realm_law_flag = LR_women_status_2 has_realm_law_flag = LR_women_status_3 } }
		AND = {
			$COURT_OWNER$ = { OR = { has_realm_law_flag = LR_women_status_1 has_realm_law_flag = LR_women_status_2 has_realm_law_flag = LR_women_status_3 } }
			OR = { is_close_family_of = $COURT_OWNER$ is_spouse_of = $COURT_OWNER$ }
		}
		AND = {
			$COURT_OWNER$ = { has_realm_law_flag = LR_women_status_2 }
			OR = { diplomacy >= 12 martial >= 12 stewardship >= 12 intrigue >= 12 learning >= 12 }
		}
		AND = {
			$COURT_OWNER$ = { has_realm_law_flag = LR_women_status_3 }
			OR = { diplomacy >= 8 martial >= 8 stewardship >= 8 intrigue >= 8 learning >= 8 }
		}
	}
}

# the culture lets women fight like men (Royal Court martial custom), or without Royal Court the faith does
LR_li_women_combat_open_trigger = {
	trigger_if = {
		limit = { has_dlc_feature = diverge_culture }
		culture ?= {
			NOT = { has_cultural_parameter = martial_custom_male_only_combatant }
			NOT = { has_cultural_parameter = martial_custom_female_only_combatant }
		}
	}
	trigger_else = {
		rite = { rite_has_parameter = combatant_can_be_either_gender_if_no_roco }
	}
}

# $ARMY_OWNER$'s Female Marshals law (LR_female_marshals) for a woman commander or knight; men and the owner pass
LR_li_female_marshals_law_trigger = {
	OR = {
		is_female = no
		this = $ARMY_OWNER$
		$ARMY_OWNER$ = { NOT = { has_realm_law_flag = LR_female_marshals_0 has_realm_law_flag = LR_female_marshals_1 has_realm_law_flag = LR_female_marshals_2 } }
		AND = {
			$ARMY_OWNER$ = { has_realm_law_flag = LR_female_marshals_1 }
			OR = { martial >= 16 prowess >= 16 }
		}
		AND = {
			$ARMY_OWNER$ = { has_realm_law_flag = LR_female_marshals_2 }
			OR = { martial >= 12 prowess >= 12 }
		}
	}
}

# GRADUS gates (tools/li_laws): rulers without the Administration law (unlanded) pass
LR_li_has_legal_administration_trigger = {
	custom_tooltip = {
		text = LR_li_has_legal_administration_tt
		OR = {
			has_realm_law_flag = LR_admin_legal
			is_landed = no
		}
	}
}
