# LR 2026-11-19: Principate triggers (see the header of common/scripted_effects/LR/LR_principate_effects.txt)

# Character: the Princeps, head of a Principate's Senate
LR_is_princeps_trigger = {
	has_government = principate_government
	gdp_is_assembly_head_trigger = yes
}

# Character: should be a Principate governor (principate_government with Senatorial or Imperial province contracts):
# a landed direct vassal of duchy tier or above of a Princeps, of a settled lay government, not Muslim (Muslims are
# always Iqta, LR_replace_vanilla_default_government_effect)
LR_principate_governor_trigger = {
	exists = liege
	liege = {
		this = top_liege
		has_government = principate_government
	}
	is_landed = yes
	highest_held_title_tier >= tier_duchy
	NOT = { faith = { religion_tag = islam_religion } }
	LR_can_have_upgraded_roman_government_trigger = yes # LR 2026-11-19: not a decayed Roman empire's ruler
	government_has_flag = government_is_settled_non_tribal
	NOR = {
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_holy_order
		government_has_flag = government_is_mercenary
		government_has_flag = government_is_landless_adventurer
		government_has_flag = government_is_ecclesiastical
	}
}

# Character (the recipient): can be named successor by $PRINCEPS$. Must be in a candidate set of principate_elective
# (common/succession_election/LR_principate_elective.txt): close or extended family, a direct vassal or a councillor.
LR_principate_successor_candidate_trigger = {
	NOT = { this = $PRINCEPS$ }
	is_adult = yes
	is_imprisoned = no
	is_incapable = no
	is_clergy = no
	OR = {
		is_close_or_extended_family_of = $PRINCEPS$
		AND = {
			is_ruler = yes
			liege = $PRINCEPS$
		}
		is_councillor_of = $PRINCEPS$
	}
}

# Character: has a Roman imperial government: Roman Autocratic, Roman Dominate or the Principate.
# LR 2026-11-19 (user): "principate should be considered an imperial type even though it uses republic features".
# Mechanics tied to the administrative mechanic (offices, ministries, title legions) still check their own flags.
LR_has_roman_imperial_government_trigger = {
	government_has_flag = government_is_roman_imperial_type
}

# Character (root = the ruler taking a diocese decision): the restored diocese title:$DIOCESE$ is held by root or anyone in root's realm.
# LR 2026-11-19: the diocese decisions checked only root and direct vassals, so a diocese held under a prefecture did not count.
LR_roman_diocese_held_in_realm_trigger = {
	exists = title:$DIOCESE$.holder
	title:$DIOCESE$.holder = {
		OR = {
			this = root
			top_liege = root
		}
	}
}

# LR 2026-11-19 (user): the Roman Dominate and the Principate are "special, very complex evolutions of roman imperial": both need
# Roads to Power, Silk & Silver and All Under Heaven. Used in both governments' can_get_government and their four decisions.
LR_has_roman_evolution_dlc_trigger = {
	custom_tooltip = {
		text = LR_has_roman_evolution_dlc_tt
		has_ep3_dlc_trigger = yes
		has_gdp_dlc_trigger = yes
		has_tgp_dlc_trigger = yes
	}
}

# LR 2026-11-19: the Princeps' ministers (LR_principate_ministry_positions.txt); a copy of LR_can_be_roman_minister_trigger
LR_can_be_principate_minister_trigger = {
	can_be_councillor_basics_trigger = yes
	# Has to be a Principate ruler (a governor or legate) or a non-ruler
	OR = {
		is_ruler = no
		government_has_flag = government_is_principate
	}
	# Gender is based on laws of the liege
	OR = {
		has_variable = can_be_councillor_override
		AND = {
			is_female = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = male_only_law
					has_realm_law = male_preference_law
				}
			}
		}
		AND = {
			is_male = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = female_only_law
					has_realm_law = female_preference_law
				}
			}
		}
	}
	# Players are only allowed if they have enough merit or is a councillor already, unless they are friends or disciples/elders
	# We do this to prevent the AI from "randomly" requesting the player to be on their council
	trigger_if = {
		limit = {
			is_ai = no
			$COURT_OWNER$ = { is_ai = yes }
			NOT = { is_councillor_of = $COURT_OWNER$ }
		}
		OR = {
			merit_level >= merit_level_expected_empire_value
			has_relation_friend = $COURT_OWNER$
			has_relation_disciple = $COURT_OWNER$
			has_relation_elder = $COURT_OWNER$
		}
	}
}

# Character (the emperor): may raise another Roman command of kind OFFICE (legion / dux). LR 2026-11-19 (user): Roman Imperial
# gets "fewer of each": at most 3 legions and 3 duces in the realm; the Principate (legions) and the Dominate (duces) have no cap.
LR_roman_imperial_office_cap_trigger = {
	trigger_if = {
		limit = { has_government = roman_autocratic_government }
		custom_tooltip = {
			text = LR_roman_imperial_$OFFICE$_cap_tt
			NOT = {
				any_vassal = {
					count >= 3
					any_held_title = { LR_is_roman_$OFFICE$_title_trigger = yes }
				}
			}
		}
	}
}

# Character: holds a Roman title on the wrong side for their government, with its twin free (LR_schedule_roman_title_twins_effect).
# LR 2026-11-19 (user): Empire side for every Roman imperial government (the Principate too), Republic side for a plain republic.
LR_needs_roman_title_twin_swap_trigger = {
	OR = {
		AND = {
			LR_has_roman_imperial_government_trigger = yes
			OR = {
				AND = {
					has_title = title:e_western_roman_republic
					NOT = { exists = title:e_western_roman_empire.holder }
				}
				AND = {
					has_title = title:e_eastern_roman_republic
					NOT = { exists = title:e_eastern_roman_empire.holder }
				}
				AND = {
					has_title = title:h_western_roman_republic
					NOT = { exists = title:h_western_roman_empire.holder }
				}
				AND = {
					has_title = title:h_eastern_roman_republic
					NOT = { exists = title:h_eastern_roman_empire.holder }
				}
				AND = {
					has_title = title:h_roman_republic
					NOT = { exists = title:h_roman_empire.holder }
				}
			}
		}
		AND = {
			has_government = republic_government
			OR = {
				AND = {
					has_title = title:e_western_roman_empire
					NOT = { exists = title:e_western_roman_republic.holder }
				}
				AND = {
					has_title = title:e_eastern_roman_empire
					NOT = { exists = title:e_eastern_roman_republic.holder }
				}
				AND = {
					has_title = title:h_western_roman_empire
					NOT = { exists = title:h_western_roman_republic.holder }
				}
				AND = {
					has_title = title:h_eastern_roman_empire
					NOT = { exists = title:h_eastern_roman_republic.holder }
				}
				AND = {
					has_title = title:h_roman_empire
					NOT = { exists = title:h_roman_republic.holder }
				}
			}
		}
	}
}
