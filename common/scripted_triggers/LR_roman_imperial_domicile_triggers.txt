# LR 2026-11-19 (user): root = a character. Their family seat doesn't match roman_imperial_domicile: they are on the Roman
# Principate or the Roman Dominate with another seat type, or they left both and still have the domus. Only family-title
# holders, since the conversion (vanilla tgp_domicile_conversion_when_changing_government_type) rebuilds that title.
LR_needs_roman_imperial_domicile_conversion_trigger = {
	any_held_title = { is_noble_family_title = yes }
	OR = {
		AND = {
			OR = {
				has_government = roman_principate_government
				has_government = roman_dominate_government
			}
			NOT = { domicile ?= { is_domicile_type = roman_imperial_domicile } }
		}
		AND = {
			NOT = { has_government = roman_principate_government }
			NOT = { has_government = roman_dominate_government }
			domicile ?= { is_domicile_type = roman_imperial_domicile }
		}
		AND = { # LR 2026-11-25 Roman villa (tools/roman_villa/build_roman_villa.py): the Roman ranks live in a villa
			OR = { has_government = roman_dynastic_government has_government = roman_administrative_government has_government = roman_imperial_government }
			NOT = { domicile ?= { is_domicile_type = roman_villa } }
		}
		AND = { # LR 2026-11-25 Roman villa (tools/roman_villa/build_roman_villa.py): a villa left behind (another government)
			NOR = { has_government = roman_dynastic_government has_government = roman_administrative_government has_government = roman_imperial_government }
			domicile ?= { is_domicile_type = roman_villa }
		}
		AND = { # LR 2026-11-25: Autocratic keeps vanilla's estate
			has_government = rtp_autocratic_government
			NOT = { domicile ?= { is_domicile_type = estate } }
		}
		AND = { # LR 2026-11-25: from Autocratic (or vanilla administrative) to a group administration: that group's seat
			domicile ?= { is_domicile_type = estate }
			LR_has_group_estate_administration_trigger = yes
		}
	}
}
