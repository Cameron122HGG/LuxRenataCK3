# Governments that never get a house-named realm in the "all realms" mode
LR_dynastic_realm_name_excluded_government_trigger = {
	OR = {
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_republic
		government_has_flag = government_is_landless_adventurer
		government_has_flag = government_is_nomadic
		government_has_flag = government_is_herder
		government_has_flag = government_is_holy_order
		government_has_flag = government_is_mercenary
	}
}

# root = a ruler. Are they the head of their own faith (a temporal head like the caliph)?
# Their realm is then named after the head of faith title's word ("Abbasid Caliphate"), even though
# the head of faith title (e.g. d_sunni) is usually lower tier than their secular primary title.
LR_dynastic_realm_name_temporal_head_trigger = {
	faith = {
		exists = religious_head
		religious_head = root
	}
}

# root = a ruler. Does the realm name follow the house ("Abbasid Caliphate")?
# Otherwise (only in the "all realms" mode) it is "<Tier> of <Title>" ("Kingdom of France").
LR_dynastic_realm_name_house_form_trigger = {
	government_allows = dynasty_named_realms
	LR_dynastic_realm_name_culture_trigger = yes
}

# root = a ruler. Should their primary title get a dynastic or tiered realm name?
LR_dynastic_realm_name_valid_trigger = {
	NOT = { has_game_rule = lr_dynastic_realm_names_vanilla }
	is_alive = yes
	exists = house
	exists = primary_title
	primary_title = {
		tier >= tier_kingdom
		# Roman titles are named by LR_roman_title_names_effects.txt
		NOT = { LR_roman_named_title_trigger = yes }
	}
	OR = {
		AND = {
			has_game_rule = lr_dynastic_realm_names_dynastic
			LR_dynastic_realm_name_house_form_trigger = yes
		}
		AND = {
			has_game_rule = lr_dynastic_realm_names_all
			NOT = { LR_dynastic_realm_name_excluded_government_trigger = yes }
		}
	}
}

# scope = a title we named, root = its ruler. Is the stored name still right?
LR_dynastic_realm_name_up_to_date_trigger = {
	has_variable = LR_dynastic_realm_name
	has_variable = LR_dynastic_realm_name_culture
	has_variable = LR_dynastic_realm_name_faith
	var:LR_dynastic_realm_name = root.house
	var:LR_dynastic_realm_name_culture = root.culture
	var:LR_dynastic_realm_name_faith = root.faith
	# named as head of faith <=> still head of faith
	OR = {
		AND = {
			has_variable = LR_dynastic_realm_name_hof
			root = { LR_dynastic_realm_name_temporal_head_trigger = yes }
		}
		AND = {
			NOT = { has_variable = LR_dynastic_realm_name_hof }
			root = { NOT = { LR_dynastic_realm_name_temporal_head_trigger = yes } }
		}
	}
}
