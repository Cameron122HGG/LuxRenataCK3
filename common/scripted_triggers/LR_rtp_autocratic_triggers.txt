# Lux Renata 2026-11-25 (user): the group of an Autocratic ruler (rtp_autocratic_government has no group flag).
# Muslims are Iqta, rulers in the Sacral Kingdom of Gaul's realm are Sacral, everyone else follows their culture's
# government tradition (no tradition or Anarchy = Western, as Feudal Custom is the fallback). Character scope.
# tools/rtp_tier/consolidate_autocratic.py

LR_rtp_autocratic_iqta_trigger = {
	has_government = rtp_autocratic_government
	faith = { religion_tag = islam_religion }
}

LR_rtp_autocratic_sacral_trigger = {
	has_government = rtp_autocratic_government
	NOT = { faith = { religion_tag = islam_religion } }
	LR_in_sacral_kingdom_gaul_realm_trigger = yes
}

# an Autocratic ruler whose group comes from their culture
LR_rtp_autocratic_by_culture_trigger = {
	has_government = rtp_autocratic_government
	NOT = { faith = { religion_tag = islam_religion } }
	NOT = { LR_in_sacral_kingdom_gaul_realm_trigger = yes }
}

		culture = { LR_culture_has_government_tradition_trigger = no }
	}
}
