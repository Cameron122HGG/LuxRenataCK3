coa_switch_to_mena_trigger = {
	holder = {
		NOT = {
			faith.religion = religion:christianity_religion
		}
		OR = {
			faith.religion = religion:islam_religion
			OR = {			
				culture = { has_coa_gfx = arabic_group_coa_gfx }
				culture = { has_coa_gfx = berber_group_coa_gfx }
			}
		}
	}
}

# Mamluk runuk designs (probably overcomplicated this)
ps_coa_mamluk_runuk_trigger = {
    OR = { # Core cultures
        AND = { # character
            exists = culture
            culture = {
                OR = {
                    this = culture:egyptian
                    this = culture:maghrebi
                    this = culture:levantine
                }
            }
        }
        AND = { # title
            exists = holder
            holder = {
                culture = {
                    OR = {
                        this = culture:egyptian
                        this = culture:maghrebi
                        this = culture:levantine
                    }
                }
            }
        }

        # Vassal of an above core culture & kingdom tier at least
        AND = {
            exists = culture
            culture = {
                OR = { # Kipchak and Cuman could probably be added also
                    this = culture:bedouin
                    this = culture:yemeni
                    this = culture:beja
                    this = culture:turkish
                    this = culture:kurdish
					# nubian probably
                }
            }
            OR = {
                AND = {
                    exists = top_liege
                    top_liege = {
                        highest_held_title_tier >= tier_kingdom
                        culture = {
                            OR = {
                                this = culture:egyptian
                                this = culture:maghrebi
                                this = culture:levantine
                            }
                        }
                    }
                }
                AND = {
                    exists = liege
                    liege = {
                        highest_held_title_tier >= tier_kingdom
                        culture = {
                            OR = {
                                this = culture:egyptian
                                this = culture:maghrebi
                                this = culture:levantine
                            }
                        }
                    }
                }
            }
        }

        AND = { # title
            exists = holder
            holder = {
                culture = {
                    OR = {
                        this = culture:bedouin
                        this = culture:yemeni
                        this = culture:beja
                        this = culture:turkish
                        this = culture:kurdish
                    }
                }
                OR = {
                    AND = {
                        exists = top_liege
                        top_liege = {
                            highest_held_title_tier >= tier_kingdom
                            culture = {
                                OR = {
                                    this = culture:egyptian
                                    this = culture:maghrebi
                                    this = culture:levantine
                                }
                            }
                        }
                    }
                    AND = {
                        exists = liege
                        liege = {
                            highest_held_title_tier >= tier_kingdom
                            culture = {
                                OR = {
                                    this = culture:egyptian
                                    this = culture:maghrebi
                                    this = culture:levantine
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
