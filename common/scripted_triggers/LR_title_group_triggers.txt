# Title groups: each groups the title sets of one age, so an age keeps one shared group as its sets are divided
# (the Roman group holds the Principate provinces and the Late Roman dioceses; the Medieval group holds the medieval
# set and the Eastern Roman Themes, and a Caliphal set would join it). Title scope.

# Bronze Age: the Bronze Age set (1700) and the Hebrew set (Mamlekhet Yisra'el; user, 2026-11-06)
LR_title_group_bronze_trigger = {
	OR = {
		LR_is_bronze_title_trigger = yes
		LR_title_set_hebrew_title_trigger = yes
	}
}

# Classical: the Classical areas (3800)
LR_title_group_classical_trigger = {
	LR_is_classical_title_trigger = yes
	NOT = { LR_title_set_roman_title_trigger = yes }
}

# Hellenistic: the hyparchies, satrapies and strategiai (4677)
LR_title_group_hellenistic_trigger = {
	LR_title_set_hellenistic_title_trigger = yes
}

# Roman: the Principate provinces (4974) and the Late Roman provinces, dioceses and prefectures (5286)
LR_title_group_roman_trigger = {
	LR_title_set_roman_title_trigger = yes
}

# Medieval: the medieval set (5476/5641) and the Eastern Roman Themes (5641/5668, and the 936 restoration)
LR_title_group_medieval_trigger = {
	OR = {
		LR_is_medieval_title_trigger = yes
		LR_is_eastern_roman_theme_title_trigger = yes
		LR_title_set_brythonic_title_trigger = yes
		LR_title_set_galatian_title_trigger = yes
		LR_title_set_punic_title_trigger = yes
	}
}

# Warlord: the warlord map (from 936 by region)
LR_title_group_warlord_trigger = {
	LR_is_warlord_title_trigger = yes
}

# Base list: the de jure base list the other sets are drawn from
LR_title_group_base_list_trigger = {
	LR_is_base_list_title_trigger = yes
}
