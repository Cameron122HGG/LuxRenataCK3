# Character: may use the Hellenistic set (rulers of a Successor realm, rulers inside one and rulers of Hellenistic heritage)
LR_title_set_hellenistic_native_trigger = {
	OR = {
		LR_is_diadochi_ruler_trigger = yes
		top_liege ?= { LR_is_diadochi_ruler_trigger = yes }
		culture ?= { has_cultural_pillar = heritage_hellenistic }
	}
}

# Character: holds a Hellenistic set title without being native to the set
LR_title_set_hellenistic_outsider_trigger = {
	NOT = { LR_title_set_hellenistic_native_trigger = yes }
	any_held_title = { LR_title_set_hellenistic_title_trigger = yes }
}

# Character: may use the Roman set (restored Roman emperors and rulers inside a restored Roman empire)
LR_title_set_roman_native_trigger = {
	OR = {
		is_restored_roman_emperor_trigger = yes
		top_liege ?= { is_restored_roman_emperor_trigger = yes }
	}
}

# Character: holds a Roman set title without being native to the set

	NOT = { LR_title_set_roman_native_trigger = yes }
	any_held_title = { LR_title_set_roman_title_trigger = yes }
}

# Title: one of the Eastern Roman Themes set's titles (themes, catepanates, exarchates and client kingdoms, tiers_936.py)
LR_title_set_themes_title_trigger = {
	LR_is_eastern_roman_theme_title_trigger = yes
}

# Character: may use the Themes set (restored Roman emperors and rulers inside a restored Roman empire; user's choice
# 2026-11-05, so the 1020 rebels who reject Julian VII hold their themes under Foreign Administration)
LR_title_set_themes_native_trigger = {
	LR_title_set_roman_native_trigger = yes
}

# Character: holds a Themes set title without being native to the set
LR_title_set_themes_outsider_trigger = {
	NOT = { LR_title_set_themes_native_trigger = yes }
	any_held_title = { LR_title_set_themes_title_trigger = yes }
}

# Title: one of the Persia set's titles. The Persian warlord titles don't count (user, 2026-11-05); the set's own
# formable satrapies are still being designed, so for now it holds no titles.
LR_title_set_persian_title_trigger = {
	always = no
}

# Character: may use the Persia set (rulers of Iranian heritage; the user chose culture only, 2026-11-05, so
# Perso-Hellenistic, Yavanan and other outsiders hold Persian titles under Foreign Administration)
LR_title_set_persian_native_trigger = {
	culture ?= { has_cultural_pillar = heritage_iranian }
}

# Character: holds a Persia set title without being native to the set
LR_title_set_persian_outsider_trigger = {
	NOT = { LR_title_set_persian_native_trigger = yes }
	any_held_title = { LR_title_set_persian_title_trigger = yes }
}

# Title: belongs to any title set
LR_is_set_title_trigger = {
	OR = {
		LR_title_set_hellenistic_title_trigger = yes
		LR_title_set_roman_title_trigger = yes
		LR_title_set_themes_title_trigger = yes
		LR_title_set_persian_title_trigger = yes
	}
}

# Character: may use the set of title $TITLE$ (always true for a title in no set)
LR_may_use_title_set_trigger = {
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_hellenistic_title_trigger = yes } }
		LR_title_set_hellenistic_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_roman_title_trigger = yes } }
		LR_title_set_roman_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_themes_title_trigger = yes } }
		LR_title_set_themes_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_persian_title_trigger = yes } }
		LR_title_set_persian_native_trigger = yes
	}
}
