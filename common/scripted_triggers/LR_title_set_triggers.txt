# Character: may use the Hellenistic set (rulers of a Successor realm, rulers inside one and rulers of Hellenistic heritage)
LR_title_set_hellenistic_native_trigger = {
	OR = {
		LR_is_diadochi_ruler_trigger = yes
		top_liege ?= { LR_is_diadochi_ruler_trigger = yes }
		culture ?= { has_cultural_pillar = heritage_hellenistic }
	}
}

# Character: holds a Hellenistic set title without being native to the set
LR_title_set_hellenistic_outsider_trigger = {
	NOT = { LR_title_set_hellenistic_native_trigger = yes }
	any_held_title = { LR_title_set_hellenistic_title_trigger = yes }
}

# Title: belongs to any title set
LR_is_set_title_trigger = {
	OR = {
		LR_title_set_hellenistic_title_trigger = yes
		LR_title_set_roman_title_trigger = yes
		LR_title_set_themes_title_trigger = yes
		LR_title_set_persian_title_trigger = yes
		LR_title_set_hebrew_title_trigger = yes
	}
}

# Character: may use the set of title $TITLE$ (always true for a title in no set)
LR_may_use_title_set_trigger = {
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_hellenistic_title_trigger = yes } }
		LR_title_set_hellenistic_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_roman_title_trigger = yes } }
		LR_title_set_roman_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_themes_title_trigger = yes } }
		LR_title_set_themes_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_persian_title_trigger = yes } }
		LR_title_set_persian_native_trigger = yes
	}
	trigger_if = {
		limit = { $TITLE$ = { LR_title_set_hebrew_title_trigger = yes } }
		LR_title_set_hebrew_native_trigger = yes
	}
}
