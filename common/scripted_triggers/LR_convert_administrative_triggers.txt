# Lux Renata: which options of convert_to_administrative_decision a ruler sees (00_LR_government_decisions.txt).
# An independent ruler sees the administrative governments of their own group; a vassal sees only the government of
# their administrative top liege (they join the realm's administration). In the classical group, Hellenistic-heritage
# cultures see Alexandrian and everyone else Roman, as in LR_replace_vanilla_default_government_effect.

# Character scope: the government is in none of the Lux Renata groups (vanilla Feudal or Clan)
LR_has_no_government_group_trigger = {
	NOR = {
		LR_in_classical_group_trigger = yes
		LR_in_western_group_trigger = yes
		LR_in_despotic_group_trigger = yes
		LR_in_mamlakah_group_trigger = yes
		LR_in_sudanic_group_trigger = yes
		LR_in_nilotic_group_trigger = yes
		LR_in_indic_group_trigger = yes
		LR_in_islamic_group_trigger = yes
		LR_in_sinic_group_trigger = yes
		LR_in_japanese_group_trigger = yes
		LR_in_sacral_group_trigger = yes
		LR_in_barbarian_group_trigger = yes
	}
}

# Character scope
LR_convert_admin_sees_administrative_trigger = {
	has_ep3_dlc_trigger = yes
	LR_has_rtp_governments_dlc_trigger = no # LR 2026-11-25: with Roads to Power alone the only option is Autocratic
	trigger_if = {
		limit = { top_liege = this }
		LR_has_no_government_group_trigger = yes
	}
	trigger_else = {
		top_liege = {
			has_government = administrative_government
		}
	}
}

# Character scope
LR_convert_admin_sees_roman_administrative_trigger = {
	has_ep3_dlc_trigger = yes
	LR_has_rtp_governments_dlc_trigger = no # LR 2026-11-25: with Roads to Power alone the only option is Autocratic
	trigger_if = {
		limit = { top_liege = this }
		LR_in_classical_group_trigger = yes
		LR_culture_is_hellenic_trigger = no # Hellenistic heritages take the Alexandrian branch
	}
	trigger_else = {
		top_liege = {
			OR = {
				has_government = roman_administrative_government
				LR_rtp_autocratic_roman_trigger = yes # LR 2026-11-25 Autocratic of this group
				has_government = roman_imperial_government
				has_government = roman_dominate_government
			}
		}
	}
}

# LR 2026-11-25 (user): with Roads to Power alone every group but Barbarian converts to Autocratic, the one option shown.
# Character scope
LR_convert_admin_sees_rtp_autocratic_trigger = {
	LR_has_rtp_governments_dlc_trigger = yes
	trigger_if = {
		limit = { top_liege = this }
		NOT = { government_has_flag = government_is_barbarian_group }
	}
	trigger_else = {
		top_liege = {
			has_government = rtp_autocratic_government
		}
	}
}
