# Diadochi lines: a family's kingdom, empire and hegemony titles (gen_diadochi_lines.py)

# Title scope: is this a title of a diadochi line?
LR_is_diadochi_line_title_trigger = {
	OR = {
		this = title:k_antipatrid_kingdom
		this = title:e_antipatrid_empire

	OR = {
