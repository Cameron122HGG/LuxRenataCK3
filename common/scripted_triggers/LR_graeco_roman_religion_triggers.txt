# Lux Invicta Graeco-Roman religion content (gen_graeco_roman_religion.py)
# Character: follows one of the Graeco-Roman pagan faiths (LI: graeco_roman, jupiter, roman_imperial, alexandros, sol...)
LR_is_graeco_roman_pagan_trigger = {
	faith ?= {
		OR = {
			this = faith:roman_faith
			this = faith:hellenic_pagan
			this = faith:alexandrian_faith
			this = faith:solar_imperial_faith
		}
	}
}

# Character: has been initiated into one of the mystery cults
LR_is_mystery_initiate_trigger = {
	OR = {
		has_trait = LR_cult_of_orpheus
		has_trait = LR_cult_of_serapis
		has_trait = LR_cult_of_eleusis
		has_trait = LR_cult_of_trophonius
	}
}
