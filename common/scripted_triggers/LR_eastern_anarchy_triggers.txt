# The Eastern Anarchy (gen_eastern_anarchy.py)
# Is an Eastern Roman empire held by an independent ruler?
LR_eastern_roman_empire_restored_trigger = {
	OR = {
		title:e_byzantium = {
			exists = holder
			holder = { is_independent_ruler = yes }
		}
		title:e_eastern_roman_empire = {
			exists = holder
			holder = { is_independent_ruler = yes }
		}
		title:h_eastern_roman_empire = {
			exists = holder
			holder = { is_independent_ruler = yes }
		}
	}
}
