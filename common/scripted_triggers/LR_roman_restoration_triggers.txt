# Restore the Roman Empire (restore_roman_empire_decision), reworked 2026-11-06.
# A Roman diocese counts toward the restoration if its kingdom title is held in your realm,
# or if you completely control its lands (dioceses are often unheld, e.g. the East under the Themes).

# Character scope. DIOCESE = the diocese key without k_diocese_ (its region is custom_roman_<DIOCESE>)
LR_roman_restoration_diocese_trigger = {
	OR = {
		title:k_diocese_$DIOCESE$ = {
			exists = holder
			holder = {
				OR = {
					this = root
					target_is_liege_or_above = root
				}
			}
		}
		completely_controls_region = custom_roman_$DIOCESE$
	}
}

# Character scope: at least 6 of the 8 dioceses of the West (the ones Restore the Hegemony of the West checks)
LR_roman_restoration_western_dioceses_trigger = {
	calc_true_if = {
		amount >= 6
		LR_roman_restoration_diocese_trigger = { DIOCESE = italia_annonaria }
		LR_roman_restoration_diocese_trigger = { DIOCESE = italia_suburbicaria }
		LR_roman_restoration_diocese_trigger = { DIOCESE = illyricum }
		LR_roman_restoration_diocese_trigger = { DIOCESE = africa }
		LR_roman_restoration_diocese_trigger = { DIOCESE = mauretania }
		LR_roman_restoration_diocese_trigger = { DIOCESE = hispania }
		LR_roman_restoration_diocese_trigger = { DIOCESE = viennensis }
		LR_roman_restoration_diocese_trigger = { DIOCESE = lugdunensis }
	}
}

# Character scope: the county is held by you or inside your realm
LR_roman_restoration_county_in_realm_trigger = {
	exists = title:$COUNTY$.holder
	title:$COUNTY$.holder = {
		OR = {
			this = root
			target_is_liege_or_above = root
		}
	}
}


# Cleave the Empire (split_roman_empire_decision), 2026-11-06. Character scope: your capital lies in the Theodosian East
LR_cleave_keeps_east_trigger = {
	capital_province = { geographical_region = custom_theodosian_ere }
}

# Character scope: your capital lies in either half of the Theodosian Empire
LR_cleave_capital_in_empire_trigger = {
	capital_province = {
		OR = {
			geographical_region = custom_theodosian_ere
			geographical_region = custom_theodosian_wre
		}
	}
}

# Character scope: the other half's empire title can go to a junior Caesar (unheld, yours, or already theirs)
LR_cleave_caesar_title_free_trigger = {
	OR = {
		NOT = { exists = title:$TITLE$.holder }
		title:$TITLE$.holder = root
		title:$TITLE$.holder = scope:new_holder
	}
}


# Restore the Hegemony of the West / East (2026-11-06): the half's core prefecture plus one bordering prefecture.
# A prefecture counts when every one of its dioceses (as Appoint a Prefect defines them) is held in your realm.
# Borders: Gallia - Italia - Illyricum - Macedonia - Oriens. Illyricum is contested and counts for either half.

# Character scope: the diocese title is held by you or inside your realm
LR_roman_restoration_diocese_held_trigger = {
	exists = title:k_diocese_$DIOCESE$.holder
	title:k_diocese_$DIOCESE$.holder.top_liege = root
}
