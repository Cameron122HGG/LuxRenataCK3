############################################
# Extracted saint relics - artifact template
# pam_saint_relic_template
############################################

# var:pam_saint_relic_saint is set at extraction
pam_saint_relic_template = {
	can_equip = {
		always = yes
	}

	can_benefit = {
		# One readable line; the has_variable guard stays inside so the raw variable check isn't shown.
		custom_description = {
			text = pam_saint_relic_benefit_faith_tt
			# subject exposes the buried saint to the loc as CHARACTER.
			subject = scope:artifact.var:pam_saint_relic_saint
			scope:artifact = { has_variable = pam_saint_relic_saint }
			faith.religion = scope:artifact.var:pam_saint_relic_saint.religion
		}
	}

	is_holy_relic = {
		custom_description = {
			subject = scope:artifact.var:pam_saint_relic_saint
			text = pam_saint_relic_benefit_faith_tt

			scope:artifact = { has_variable = pam_saint_relic_saint	}
			religion = scope:artifact.var:pam_saint_relic_saint.religion
		}
	}

	fallback = {
		court_grandeur_baseline_add = 3
	}

	unique = yes
}
