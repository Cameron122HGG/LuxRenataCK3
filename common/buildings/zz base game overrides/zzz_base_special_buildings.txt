the_tower_of_london_01 = {

	effect_desc = {
		triggered_desc = {

			desc = tower_of_london_pacification_effect
		}
	}

	asset = {
		type = pdxmesh
		name = "sprawl_london_tower_mesh"
	}

		if = {
			limit = {

			limit = {
				scope:character.culture ?= culture:norman

		dread_gain_mult = 0.3
	}
