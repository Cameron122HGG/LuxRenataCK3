# LR 2026-11-22 (user): the Tower House, the merchant house's fortified seat inside its trade post.
# Each level adds to the post's entrenchment (LR_entrenchment_value, LR_entrenchment_values.txt); at high
# entrenchment the house can take the county (Raise the House Banner, Seize the Stronghold) as a vassal of its holder.
# Root is the province. Only merchants that may entrench can build it (LR_tower_house_owner_trigger).
# Its name changes with the county's culture (LR_tower_house_l_english.yml).

lr_tower_house_01 = {
	type_icon = regular_buildings/watchtowers.dds

	cost = { gold = trade_post_building_tier_1_cost_value }
	construction_time = trade_post_building_construction_time_value
	next_building = lr_tower_house_02

	can_construct_potential = {
		trigger_if = {
			limit = { exists = trade_post_district }
			trade_post_owner ?= { LR_tower_house_owner_trigger = yes }
		}
	}

	is_enabled = {
		trigger_if = {
			limit = { exists = trade_post_district }
			trade_post_owner ?= { LR_tower_house_owner_trigger = yes }
		}
	}

	effect_desc = {
		desc = lr_tower_house_effect_desc
	}

	character_modifier = {
		monthly_prestige = 0.05
		trade_competitiveness_add = 1
	}
	county_modifier = {
		control_baseline_add = -1
	}
	province_modifier = {
		fort_level = 1
		garrison_size = 0.05
	}

	flag = lr_tower_house_flag

	ai_value = {
		base = 12
		ai_trade_post_new_building_modifier = yes
	}
}
