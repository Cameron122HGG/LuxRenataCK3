### Roman Republics
#
# LR 2026-11-19: no longer given to any title. The static Roman Republic titles (e_western/e_eastern_roman_republic,
# h_western/h_eastern_roman_republic, h_roman_republic) showed the type's default white colour instead of their own, so they
# set the same properties directly. Kept for create_dynamic_title and the planned naming dynamics.
# Their keys do not start with the prefix, so the prefix is only used for titles made by create_dynamic_title.
# Based on vanilla's title_type_republic, plus the naming the Roman Republic titles had set one by one.
# Naming dynamics for these titles will be built on this type later.
#

	prefix = lrromrep

	properties = {
		definite_form = yes
		no_automatic_claims = yes
		can_be_named_after_dynasty = no
		ruler_uses_title_name = yes
	}
}
