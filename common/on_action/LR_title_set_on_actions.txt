# Title sets (gen_title_sets.py): Foreign Administration follows set titles on gain, loss and yearly

on_title_gain = {
	on_actions = { LR_title_set_on_title_change }
}
on_title_lost = {
	on_actions = { LR_title_set_on_title_change }
}
yearly_playable_pulse = {
	on_actions = { LR_title_set_yearly }
}

# root = the new or old holder, scope:title = the title
LR_title_set_on_title_change = {
	trigger = { scope:title = { LR_is_set_title_trigger = yes } }
	effect = { LR_title_set_refresh_effect = yes }
}

LR_title_set_yearly = {
	trigger = { is_landed = yes }
	effect = { LR_title_set_refresh_effect = yes }
}
