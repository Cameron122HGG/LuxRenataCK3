on_title_gain = {
	on_actions = { LR_roman_dux_on_title_gain }
}

on_title_lost = {
	on_actions = { LR_roman_dux_on_title_lost }
}

# root = new holder, scope:title = the title
# Covers both the Appoint a Dux decision and later appointments by succession.
LR_roman_dux_on_title_gain = {
	trigger = {
		scope:title = { LR_is_roman_dux_title_trigger = yes }
	}
	effect = {
		LR_roman_dux_setup_effect = yes
	}
}

# root = old holder, scope:title = the title
LR_roman_dux_on_title_lost = {
	trigger = {
		is_alive = yes
		scope:title = { LR_is_roman_dux_title_trigger = yes }
		has_character_modifier = LR_roman_dux_modifier
	}
	effect = {
		if = {
			limit = {
				NOT = {
					any_held_title = {
						this != scope:title
						LR_is_roman_dux_title_trigger = yes
					}
				}
			}
			remove_character_modifier = LR_roman_dux_modifier
		}
	}
}
