# TEST: at the 100.1.1 nomad start (bm_test_nomad_100) each de jure duchy's counties are gathered under one nomad
# lord - the generated ruler of the duchy's capital county - and every landed independent ruler becomes a nomad.
# Done at game start rather than in title history, so no later bookmark inherits these lords. Needs Khans of the Steppe.
# Not for release.
on_game_start = {
	on_actions = { LR_test_nomad_start }
}

LR_test_nomad_start = {
	trigger = {
		current_date < 5101.1.1
		has_mpo_dlc_trigger = yes
	}
	effect = {
		# One lord per duchy: the generated count of the duchy's capital county takes its other counties; the counts
		# who lose their only county join his court
		every_county = {
			limit = {
				exists = holder
				holder = {
					is_independent_ruler = yes
					highest_held_title_tier = tier_county
				}
				exists = de_jure_liege
				de_jure_liege = { tier = tier_duchy }
				exists = de_jure_liege.title_capital_county.holder
				NOT = { holder = de_jure_liege.title_capital_county.holder }
				de_jure_liege.title_capital_county.holder = {
					is_independent_ruler = yes
					highest_held_title_tier <= tier_county
				}
			}
			de_jure_liege.title_capital_county.holder = { save_scope_as = LR_duchy_lord }
			holder = { save_scope_as = LR_old_count }
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = LR_change
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = scope:LR_duchy_lord
				change = scope:LR_change
			}
			resolve_title_and_vassal_change = scope:LR_change
			if = {
				limit = { scope:LR_old_count = { is_landed = no } }
				scope:LR_duchy_lord = { add_courtier = scope:LR_old_count }
			}
		}
		every_independent_ruler = {
			limit = {
				is_landed = yes
				NOT = { government_has_flag = government_is_nomadic }
				NOT = { has_government = republic_government } # city and temple rulers can't be nomads
				NOT = { has_government = theocracy_government }
			}
			change_government = nomad_government
		}
	}
}
