# An Augustus dies (2026-11-24, Imperial College). His hegemony remembers him (LR_college_prev_augustus).
# At Harmonia (lr_college_tetrarchic_succession) and unless his heir is a Caesar, his senior Caesar is marked
# for promotion (LR_college_promote, carried out on the heir's title gain). Otherwise his Caesars, except his
# heir, get a pressed claim on the hegemony.
LR_caesar_on_death = {
	trigger = {
		LR_caesar_is_augustus_trigger = yes
		primary_title.title_confederation ?= { any_confederation_member = { LR_caesar_of_root_trigger = yes } }
	}
	effect = {
		primary_title = { save_scope_as = caesar_hegemony }
		primary_title.title_confederation = { save_scope_as = lr_college }
		scope:caesar_hegemony = { set_variable = { name = LR_college_prev_augustus value = root } }
		if = {
			limit = {
				scope:lr_college = { has_cohesion_level_parameter = lr_college_tetrarchic_succession }
				NOT = { primary_heir ?= { has_trait = caesar } }
			}
			scope:lr_college = {
				ordered_confederation_member = {
					limit = { LR_caesar_of_root_trigger = yes }
					order_by = { value = 0 subtract = var:LR_caesar_seq } # the senior (first named)
					save_scope_as = lr_promoted
				}
			}
			scope:caesar_hegemony = { set_variable = { name = LR_college_promote value = scope:lr_promoted } }
		}
		else = {
			scope:lr_college = {
				every_confederation_member = {
					limit = { LR_caesar_of_root_trigger = yes NOT = { this = root.primary_heir } }
					add_pressed_claim = scope:caesar_hegemony
				}
			}
		}
	}
}

# Gaining Rome or a half: a Caesar is Augustus now; the new holder of a dead Augustus's hegemony takes over
# his Caesars, or hands the purple to the senior Caesar marked for promotion and becomes that Augustus's Caesar.

				limit = {
					scope:title = { exists = var:LR_college_promote }

				scope:title.var:LR_college_promote = { save_scope_as = lr_promoted }

				# the Caesars of the dead Augustus serve the promoted one

						limit = { exists = var:LR_caesar_augustus var:LR_caesar_augustus = scope:lr_dead_augustus }

				trigger_event = LR_caesar.0005
			}
			else = {
				scope:title.title_confederation ?= {
					every_confederation_member = {
						limit = { exists = var:LR_caesar_augustus var:LR_caesar_augustus = scope:lr_dead_augustus }
						set_variable = { name = LR_caesar_augustus value = root }
					}
				}
