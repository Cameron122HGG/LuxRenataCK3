on_game_start_after_lobby = {
	on_actions = {
		romans_check_gamerule_eastern_roman
		romans_check_gamerule_western_roman
	}
}

romans_check_gamerule_eastern_roman = {
	effect = {
		if = {
			limit = {
				has_game_rule = eastern_roman
			}
			culture:greek = {
				set_culture_name = {
					noun = eastern_roman_name
					collective_noun = eastern_roman_collective
					prefix = eastern_roman_prefix
				}
			}
		}
		else_if = {
			limit = {
				has_game_rule = rhomaion
			}
			culture:roman = {
				set_culture_name = {
					noun = rhomaion_name
					collective_noun = rhomaion_collective
					prefix = rhomaion_prefix
				}
			}
		}
	}
}

romans_check_gamerule_western_roman = {
	effect = {
		if = {
			limit = {
				has_game_rule = western_roman
			}
			culture:roman = {
				set_culture_name = {
					noun = western_roman_name
					collective_noun = western_roman_collective
					prefix = western_roman_prefix
				}
			}
		}
		else_if = {
			limit = {
				has_game_rule = romae
			}
			culture:roman = {
				set_culture_name = {
					noun = romae_name
					collective_noun = romae_collective
					prefix = romae_prefix
				}
			}
		}
		else_if = {
			limit = {
				has_game_rule = roman_default
			}
			culture:roman = {
				set_culture_name = {
					noun = roman
					collective_noun = roman_collective_noun
					prefix = roman_prefix
				}
			}
		}
	}
}
