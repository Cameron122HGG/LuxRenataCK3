on_title_lost = {
	on_actions = { LR_roman_comes_on_title_lost }
}

on_game_start = {
	on_actions = { LR_roman_comes_on_game_start }
}

# root = old holder, scope:title = the title. (Gaining a comes title is handled by LR_roman_dux_on_title_gain.)
LR_roman_comes_on_title_lost = {
	trigger = {
		is_alive = yes
		scope:title = { LR_is_roman_comes_title_trigger = yes }
		has_character_modifier = LR_roman_comes_modifier
	}
	effect = {
		if = {
			limit = {
				NOT = {
					any_held_title = {
						this != scope:title
						LR_is_roman_comes_title_trigger = yes
					}
				}
			}
			remove_character_modifier = LR_roman_comes_modifier
		}
	}
}
