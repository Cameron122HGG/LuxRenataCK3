# Age 2 names for the de jure base list (gen_proto_names.py): from 1700.1.1 to 3800.1.1, the end of the Bronze Age.
# Games started from 3800.1.1 on keep the prehistory (Age 1) names.

on_game_start = {
	on_actions = { LR_bronze_age_names_check }
}
yearly_global_pulse = {
	on_actions = { LR_bronze_age_names_check LR_bronze_age_names_end }
}

LR_bronze_age_names_check = {
	trigger = {
		current_date >= 1700.1.1
		current_date < 3800.1.1
		NOT = { has_global_variable = LR_bronze_age_names }
	}
	effect = {
		set_global_variable = LR_bronze_age_names
		LR_apply_bronze_age_names_effect = yes
	}
}

LR_bronze_age_names_end = {
	trigger = {
		current_date >= 3800.1.1
		has_global_variable = LR_bronze_age_names
		NOT = { has_global_variable = LR_bronze_age_names_ended }
	}
	effect = {
		set_global_variable = LR_bronze_age_names_ended
		LR_reset_bronze_age_names_effect = yes
	}
}
