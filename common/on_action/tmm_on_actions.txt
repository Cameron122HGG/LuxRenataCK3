# Trait Map Mode - starts the situation behind the map mode.
# The quarterly pulse (players only) catches saves started without the mod and
# situations from an older layout of the mod (ended, then started again).
# The map mode button can also start it at once (scripted GUI tmm_start_situation).

on_game_start_after_lobby = {
	on_actions = { tmm_ensure_situation }
}

quarterly_playable_pulse = {
	on_actions = { tmm_ensure_situation_player }
}

tmm_ensure_situation = {
	effect = {
		tmm_ensure_situation_effect = yes
	}
}

tmm_ensure_situation_player = {
	trigger = {
		is_ai = no
	}
	effect = {
		tmm_ensure_situation_effect = yes
	}
}
