# Lux Renata 2026-11-18: government tradition tweaks (LR_gov_traditions.txt). Culture scope.

# Counties of the culture (at least 1, as a divisor)
LR_culture_county_count_value = {
	value = 0
	every_culture_county = { add = 1 }
	min = 1
}

# Share (0 to 1) of the culture's counties held in a realm whose top liege is of the classical group
LR_culture_share_under_classical_value = {
	value = 0
	every_culture_county = {
		limit = { holder ?= { top_liege = { LR_government_in_classical_group_trigger = yes } } }
		add = 1
	}
	divide = LR_culture_county_count_value
}
