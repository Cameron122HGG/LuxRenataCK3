gdp_trade_league_monthly_change_base_value = {
	value = 0
	# Base value per league member
	add = {
		value = 0
		every_confederation_member = {
			limit = { exists = primary_title.title_confederation }
			add = 0.01
		}
		desc = NUMBER_OF_TRADE_LEAGUE_MEMBERS
	}
	# If we have very few members, add a small penalty
	if = {
		limit = {
			any_confederation_member = {
				exists = primary_title.title_confederation
				count < gdp_trade_league_min_member_penalty_value
			}
		}
		add = {
			value = -0.1
			if = {
				limit = {
					cohesion <= 20
				}
				add = {
					value = -0.06
					multiply = gdp_confederation_title_member_count_value
				}
			}
			if = {
				limit = {
					cohesion > 20
				}
				add = {
					value = -0.12
					multiply = gdp_confederation_title_member_count_value
				}
			}
			if = {
				limit = {
					cohesion > 45
				}
				add = {
					value = -0.06
					multiply = gdp_confederation_title_member_count_value
				}
			}
			if = {
				limit = {
					cohesion > 70
				}
				add = {
					value = -0.06
					multiply = gdp_confederation_title_member_count_value
				}
			}
			desc = LEAGUE_HAS_TOO_FEW_MEMBERS
		}
	}
	# If we have too many members, add a scaling penalty
	if = {
		limit = {
			any_confederation_member = {
				exists = primary_title.title_confederation
				count > gdp_trade_league_max_member_penalty_value
			}
		}
		add = {
			value = -0.1
			every_confederation_member = {
				limit = { exists = primary_title.title_confederation }
				add = {
					value = -0.06
					multiply = { # Multiply by the number of members above the soft limit
						value = prev.gdp_confederation_title_member_count_value
						subtract = gdp_trade_league_max_member_penalty_value
					}
				}
			}
			desc = LEAGUE_HAS_TOO_MANY_MEMBERS
		}
	}
	# Add a bonus for every owned monopoly
	if = {
		limit = {
			any_trade_monopoly_for_confederation = {
				count > 0
			}
		}
		add = {
			value = 0
			every_trade_monopoly_for_confederation = {
				if = {
					limit = { is_complete_monopoly = yes }
					add = 0.04
				}
				else = {
					add = 0.02
				}
			}
			desc = LEAGUE_HAS_TRADE_MONOPOLIES
		}
	}
}

					NOT = { has_same_house_aspiration_as = scope:assembly_ruler.primary_heir.house }
				}
			}
			add = {
				value = -15
				desc = LAW_CANDIDATE_NOT_SIGNORIAL_DESC
			}
		}
		if = {
			limit = {
				house ?= {
					has_house_aspiration_parameter = republic_communal
				}
			}
			add = {
				value = 10
				desc = REPUBLIC_COMMUNAL_AMBITION_DESC
			}
		}

		# Ruler specific acceptance
		if = {
			limit = {
				this != scope:assembly_ruler # To avoid weird acceptance in relation to themself
			}
			scope:assembly_ruler = {
				# Criminal
				if = {
					limit = {
						gdp_is_criminal_for_republic_punishment_laws_trigger = yes
					}
					add = {
						value = 25
						desc = LAW_LIEGE_CRIMINAL_DESC
					}
				}

				# House considerations
				if = {
					limit = {
						house ?= root.house
						this != scope:proposer
					}
					add = {
						value = -20
						if = {
							limit = {
								root.house ?= {
									OR = {
										has_house_aspiration_parameter = republic_aggrandizement
										has_house_aspiration_parameter = republic_signorial
									}
								}
							}
							multiply = 2
						}
						desc = LAW_LIEGE_SAME_HOUSE_DESC
					}
				}

				# Popular opinion
				if = {
					limit = {
						capital_county.popular_opinion < 0
					}
					add = {
						value = capital_county.popular_opinion
						if = {
							limit = {
								root.house ?= {
									has_house_aspiration_parameter = republic_communal
								}
							}
							multiply = -0.5
						}
						else = {
							multiply = -0.25
						}
						desc = LAW_RULER_POPULAR_OPINION_DESC
					}
				}
			}
		}

		# Ruler self acceptance (i.e. ruler is voting on themself)
		if = {
			limit = {
				this = scope:assembly_ruler
				# Ruler is not the primary heir, otherwise this can just be a tactical re-election
				primary_heir ?= { this != root }
			}
			# The ruler wants to stay in office
			if = {
				limit = { has_trait = ambitious }
				add = {
					value = -50
					desc = LAW_TRAIT_AMBITIOUS_DESC
				}
			}
			if = {
				limit = { has_trait = stubborn }
				add = {
					value = -20
					desc = LAW_TRAIT_STUBBORN_DESC
				}
			}
			if = {
				limit = { has_trait = arrogant }
				add = {
					value = -10
					desc = LAW_TRAIT_ARROGANT_DESC
				}
			}
			if = {
				limit = { has_trait = greedy }
				add = {
					value = -10
					desc = LAW_TRAIT_GREEDY_DESC
				}
			}

			# The ruler is willing to admit defeat
			if = {
				limit = { has_trait = humble }
				add = {
					value = 20
					desc = LAW_TRAIT_HUMBLE_DESC
				}
			}
			if = {
				limit = { has_trait = fickle }
				add = {
					value = 10
					desc = LAW_TRAIT_FICKLE_DESC
				}
			}
		}
