choose_patron_god_generic = {
	scope = character
	saved_scopes = {
		patron_god_name
	}

	is_shown = {

	}

	is_valid = {
		trigger_if = {
			limit = {
				has_perk = zealous_proselytizer_perk
			}
			piety >= 20
		}
		trigger_else = {
			piety >= minor_piety_value
		}
		var:selected_faith = faith
	}	
	
	

	effect = {
		parse_patron_god_name_flag_complex = { PATRON_GOD = scope:patron_god_name APPLY = agot_set_patron_god_effect }
	}
}
