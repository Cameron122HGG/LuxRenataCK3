choose_patron_god_generic = {
	scope = character
	saved_scopes = {
		patron_god_name
	}

	is_shown = {

	}

	is_valid = {
		trigger_if = {
			limit = {
				has_perk = zealous_proselytizer_perk
			}
			piety >= 20
		}
		trigger_else = {
			piety >= minor_piety_value
		}
		var:selected_faith = faith
	}	
	
	

	effect = {
		# The whole card in the Patron Gods tab is now the button (2026-11-24), so check is_valid again here,
		# and never re-pick (and pay for) the god that is already the patron.
		if = {
			limit = {
				trigger_if = {
					limit = {
						has_perk = zealous_proselytizer_perk
					}
					piety >= 20
				}
				trigger_else = {
					piety >= minor_piety_value
				}
				var:selected_faith = faith
				NOT = { var:patron_god ?= scope:patron_god_name }
			}
			parse_patron_god_name_flag_complex = { PATRON_GOD = scope:patron_god_name APPLY = agot_set_patron_god_effect }
		}
	}
}

	scope = character
	saved_scopes = { faith }
	effect = {
		if = {
			limit = { NOT = { var:selected_faith ?= scope:faith } }

				name = selected_faith
				value = scope:faith
			}
		}
	}
}
