# Title Set Map (LR_title_set_map situation)

# Map mode button fallback when the situation doesn't exist yet (a save started before it existed)
LR_tsm_start_situation = {
	scope = character

	is_valid = {
		NOT = { exists = situation:LR_title_set_map }
	}

	effect = {
		LR_tsm_ensure_situation_effect = yes
	}
}

# View buttons of the window. is_valid = not the current view, which also gives the 'selected' state of the button.
LR_tsm_view_land = {
	scope = character

	is_valid = {
		situation:LR_title_set_map ?= {
			NOT = { LR_tsm_view_is_land_trigger = yes }
		}
	}

	effect = {
		LR_tsm_set_view_effect = { VIEW = land }
	}
}

LR_tsm_view_rule = {
	scope = character

	is_valid = {
		situation:LR_title_set_map ?= {
			NOT = { LR_tsm_view_is_rule_trigger = yes }
		}
	}

	effect = {
		LR_tsm_set_view_effect = { VIEW = rule }
	}
}

# Visibility checks for the window (no effect): is this view shown?
LR_tsm_is_land = {
	scope = character
	is_valid = { situation:LR_title_set_map ?= { LR_tsm_view_is_land_trigger = yes } }
}

LR_tsm_is_rule = {
	scope = character
	is_valid = { situation:LR_title_set_map ?= { LR_tsm_view_is_rule_trigger = yes } }
}

LR_tsm_is_preview = {
	scope = character
	is_valid = { situation:LR_title_set_map ?= { LR_tsm_view_is_preview_trigger = yes } }
}
