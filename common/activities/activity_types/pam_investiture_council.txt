############################################################################################
# Investiture Council: a slim variant of the Ecumenical Council convened at the end of the Investiture
# Controversy to settle the Clerical Appointment doctrine. Not player-plannable (is_shown gated on the
# nomination flag), started via an AI-host nudge. Summons + one Debate phase (no tenet debates); reuses the
# ecumenical-council helpers with its own stance-based sort/tally, leaving the shipped one untouched. Host
# champions Spiritual; the vote decides (innovators >= conservatives -> Spiritual, else Temporal), removing
# Conditional either way.
############################################################################################

activity_investiture_council = {

	notify_player_can_join_activity = yes

	# Only visible to the AI host the situation nominates, so a player never sees it in the planner.
	is_shown = {
		has_pam_dlc_trigger = yes
		has_character_flag = investiture_council_nominated
	}

	# The pending-council flag limits starting to the nominated host, not every eligible cleric
	can_start_showing_failures_only = {
		is_available_adult = yes
		has_pam_dlc_trigger = yes
		can_hold_ecumenical_council_trigger = yes
		has_character_flag = investiture_council_nominated
		faith = { has_variable = investiture_council_pending }
		custom_tooltip = {
			text = can_host_ecumenical_council_tt
			OR = {
				any_held_title = { is_head_of_faith = yes }
				top_participant_group:the_christian_church ?= { participant_group_type = christian_clerical_main_power_rulers }
				is_clergy = yes
				has_theological_agent_puppet_trigger = yes
				# We match pam_valid_investiture_council_host_trigger because clergy or group can change after nomination
				government_has_flag = government_is_ecclesiastical
				any_held_title = {
					tier >= tier_duchy
					has_clerical_region = yes
				}
			}
		}
	}

	is_valid = {
		scope:host = {
			is_physically_able = yes
			custom_tooltip = {
				text = can_host_ecumenical_council_tt
				OR = {
					any_held_title = { is_head_of_faith = yes }
					top_participant_group:the_christian_church ?= { participant_group_type = christian_clerical_main_power_rulers }
					is_clergy = yes
					has_theological_agent_puppet_trigger = yes
					# Same drift extension as can_start so a running council does not invalidate
					government_has_flag = government_is_ecclesiastical
					any_held_title = {
						tier >= tier_duchy
						has_clerical_region = yes
					}
				}
			}
		}
	}

	on_invalidated = {
		if = {
			limit = {
				scope:host = { is_landed = no }
			}
			every_attending_character = { trigger_event = activity_system.0321 }
		}
		else = {
			invalidation_incapable_effect = yes
			invalidation_imprisoned_effect = yes
			invalidation_nobody_present_effect = yes
			invalidation_host_isolated_effect = yes
		}
		# We prompt again at once so the crisis does not wait for the two-year in-absentia fallback
		if = {
			limit = {
				scope:host.faith = {
					has_variable = investiture_council_pending
					has_doctrine = doctrine_clerical_succession_conditional
				}
			}
			scope:host.faith = { pam_nudge_investiture_council_host_effect = yes }
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes
		# Same reconvene guarantee when the host dies mid-council.
		if = {
			limit = {
				scope:host.faith = {
					has_variable = investiture_council_pending
					has_doctrine = doctrine_clerical_succession_conditional
				}
			}
			scope:host.faith = { pam_nudge_investiture_council_host_effect = yes }
		}
	}

	is_location_valid = {
		custom_tooltip = {
			text = ecumenical_council_not_valid_province
			OR = {
				# Preferred: the host's own capital/seat. Always allowed so a forced council never lacks a site.
				this = scope:host.capital_province
				# Falls back to a faith holy site rather than another Head of Faith's seat
				AND = {
					barony ?= { is_holy_site_of = scope:host.faith }
					county.holder.faith ?= scope:host.faith
					# is_head_of_faith is a landed_title trigger and errors here
					NOT = { county.holder ?= { faith.religious_head = this } }
				}
			}
		}
	}

	province_score = {
		add = ecumenical_council_province_reference_value
	}

	# We avoid all here because it warns for AI province filters
	province_filter = top_realm
	ai_province_filter = top_realm

	# Forced council, and ai_will_do doubles as the chance to host once picked
	ai_will_do = {
		value = 1000000
	}

	ai_check_interval_by_tier = {
		barony = 1
		county = 1
		duchy = 1
		kingdom = 1
		empire = 1
		hegemony = 1
	}

	ai_will_select_province = {
		add = {
			value = 5
			desc = "base"
		}
		# Strongly prefer the host's own capital/seat
		if = {
			limit = {
				this = scope:host.capital_province
			}
			add = {
				value = 10000
				desc = activity_ecumenical_capital_province
			}
		}
		# Fallback only (capital not a valid site): the NEAREST holy site of the faith - nearer scores higher.
		else = {
			add = {
				value = 50
				desc = activity_ecumenical_council_holy_site_province
			}
			if = {
				limit = { squared_distance = { target = scope:host.capital_province value < squared_distance_small } }
				add = { value = 40 }
			}
			else_if = {
				limit = { squared_distance = { target = scope:host.capital_province value < squared_distance_medium } }
				add = { value = 25 }
			}
			else_if = {
				limit = { squared_distance = { target = scope:host.capital_province value < squared_distance_large } }
				add = { value = 10 }
			}
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	# No player-facing cooldown: convened by the situation, not planned.
	cost = {
		gold = { value = 0 }
	}

	conclusion_description = {
		desc = pam_investiture_council_conclusion.intro
		first_valid = {
			triggered_desc = {
				trigger = { scope:activity.var:third_debate_resolution ?= flag:passed }
				desc = pam_investiture_council_conclusion.spiritual
			}
			desc = pam_investiture_council_conclusion.temporal
		}
	}

	###################
	# PHASES
	###################

	phases = {
		# Summons Phase: gather the faithful, flavor intro.
		investiture_council_phase_summons = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					ecumenical_council_gather_clergy_effect = yes
					ecumenical_council_gather_non_clergy_effect = yes
					trigger_event = pam_investiture_council_events.0002
				}
				# Longer than the routine council so distant force-gathered prelates can arrive
				if = {
					limit = {
						this = scope:host
					}
					scope:activity = { progress_activity_phase_after = { months = 3 } }
				}
			}

			on_weekly_pulse = {
				trigger_event = { on_action = ecumenical_council_summons_events }
			}
		}

		# Debate Phase: the single Clerical Appointment debate.
		investiture_council_phase_debate = {
			order = 2
			is_predefined = yes

			on_enter_phase = {
				scope:activity = { progress_activity_phase_after = { days = 3 } }
			}

			on_phase_active = {
				# Force the debated topic to the faith's current Clerical Appointment doctrine (Conditional during IC).
				if = {
					limit = { this = scope:host }
					scope:host.faith = {
						if = {
							limit = { has_doctrine = doctrine_clerical_succession_conditional }
							doctrine:doctrine_clerical_succession_conditional = { save_scope_as = ec_debate_topic_3 }
						}
						else_if = {
							limit = { has_doctrine = doctrine_clerical_succession_temporal_appointment }
							doctrine:doctrine_clerical_succession_temporal_appointment = { save_scope_as = ec_debate_topic_3 }
						}
						else_if = {
							limit = { has_doctrine = doctrine_clerical_succession_spiritual_appointment }
							doctrine:doctrine_clerical_succession_spiritual_appointment = { save_scope_as = ec_debate_topic_3 }
						}
						else_if = {
							limit = { has_doctrine = doctrine_clerical_succession_temporal_fixed_appointment }
							doctrine:doctrine_clerical_succession_temporal_fixed_appointment = { save_scope_as = ec_debate_topic_3 }
						}
						else = {
							doctrine:doctrine_clerical_succession_spiritual_fixed_appointment = { save_scope_as = ec_debate_topic_3 }
						}
					}
					scope:activity = {
						set_variable = {
							name = current_topic
							value = scope:ec_debate_topic_3
						}
					}
					# Host debate starting event (host champions Spiritual).
					trigger_event = pam_investiture_council_events.0010
					scope:activity = { progress_activity_phase_after = { days = 3 } }
				}
			}

			# No sway pulse: shared sway events ignore lock lists and can overturn kowtow/war

			# Resolve the vote (tally innovators vs conservatives, apply Spiritual/Temporal outcome).
			on_end = {
				if = {
					limit = { this = scope:host }
					pam_investiture_council_tally_effect = yes
				}
			}
		}
	}

	###################
	# GUEST HANDLING
	###################

	open_invite = yes

	# A forced council must be able to start even if few/no guests can be invited, so it never stalls.
	allow_zero_guest_invites = yes

	max_guests = 100

	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		NOT = { government_has_flag = government_is_republic }
		NOT = { government_has_flag = government_is_landless_adventurer }
		NOT = { government_has_flag = government_is_mercenary }
		faith = scope:host.faith
		OR = {
			# The faith's clergy attend their own synod regardless of distance or diplomatic range.
			pam_is_investiture_clergy_trigger = yes
			# Heads of rite and kowtow/war voters attend at any distance so locked votes land
			rite.head_of_rite ?= this
			has_variable = investiture_stance
			# Secular attendees keep the routine-council rule: kings+, or nearby/pious dukes, within diplomatic range.
			AND = {
				in_diplomatic_range = scope:host
				OR = {
					highest_held_title_tier >= tier_kingdom
					AND = {
						highest_held_title_tier >= tier_duchy
						OR = {
							piety_level >= 3
							location = { squared_distance = { target = scope:host.capital_province value < squared_distance_medium } }
						}
					}
				}
			}
		}
	}

	host_intents = {
		intents = { spiritual_intent }
		default = spiritual_intent
	}

	guest_intents = {
		intents = { spiritual_intent clerical_appointment_intent }
		default = spiritual_intent
	}

	guest_join_chance = {
		base = 5
		base_activity_modifier = yes
		activity_guest_shared_ai_accept_modifier = yes
		activity_guest_health_ai_accept_modifier = yes
		# We accept landed prelates automatically
		modifier = {
			faith = scope:host.faith
			pam_is_investiture_landed_prelate_trigger = yes
			add = 1000
			desc = pam_investiture_council_summons_clergy_value
		}
	}

	guest_subsets = {
		clergy
		non_clergy
		innovator
		conservative
	}

	wait_time_before_start = {
		months = ecumenical_council_wait_time_before_start_months
	}

	max_guest_arrival_delay_time = { months = 5 }

	guest_invite_rules = {
		defaults = {
			1 = activity_invite_hof_and_hor
			1 = activity_invite_cardinals
			1 = activity_invite_archbishops
			1 = activity_invite_clergy
			2 = activity_invite_neighbouring_rulers
			2 = activity_invite_rule_all_lieges
			2 = activity_invite_rule_vassals
			3 = activity_invite_rule_fellow_vassals
			3 = activity_invite_rule_councillors
			4 = activity_invite_rule_courtiers
			5 = activity_invite_rule_guests
		}
	}

	###################
	# ACTIVITY WINDOW
	###################

	window_characters = {
		player_traveling = {
			camera = camera_activity_horse
			effect = {
				scope:player ?= {
					if = {
						limit = {
							NOT = { location = scope:activity.activity_location }
						}
						add_to_list = characters
					}
				}
			}
			scripted_animation = {
				triggered_animation = {
					trigger = {
						scope:character.location = { is_sea_province = yes }
					}
					animation = { idle }
				}
				triggered_animation = {
					trigger = { scope:character = { has_trait = impatient } }
					animation = { jockey_gallop }
				}
				animation = jockey_walk
			}
		}
		player_guest = {
			camera = camera_event_center
			effect = {
				scope:player ?= {
					if = {
						limit = {
							this != scope:activity.activity_host
							location = scope:activity.activity_location
						}
						add_to_list = characters
					}
				}
			}
			# Calmer than the shared arrival animation, whose zealous branch picks condemnation poses
			# Inline animation values have to be single names
			scripted_animation = {
				triggered_animation = {
					trigger = { scope:character = { pam_is_investiture_clergy_trigger = yes } }
					animation = prayer_standing
				}
				animation = personality_content
			}
		}
		host = {
			camera = camera_event_center
			# Left unconditional so the host still appears on the conclusion page after leaving the venue
			effect = {
				scope:host ?= { add_to_list = characters }
			}
			scripted_animation = {
				animation = acknowledging
			}
		}
		# Conclusion page uses the tally's camp representatives because the attending list is already dispersing
		conclusion_church_rep = {
			camera = camera_event_left
			effect = {
				scope:activity.var:investiture_conclusion_innovator ?= { add_to_list = characters }
			}
			scripted_animation = {
				triggered_animation = {
					trigger = { scope:activity.var:third_debate_resolution ?= flag:passed }
					animation = personality_content
				}
				animation = sadness
			}
		}
		conclusion_crown_rep = {
			camera = camera_event_right
			effect = {
				scope:activity.var:investiture_conclusion_conservative ?= { add_to_list = characters }
			}
			scripted_animation = {
				triggered_animation = {
					trigger = { scope:activity.var:third_debate_resolution ?= flag:passed }
					animation = sadness
				}
				animation = personality_content
			}
		}
		travel_host = {
			camera = camera_event_horse_left
			effect = {
				if = {
					limit = {
						NOT = { scope:host.location = scope:activity.activity_location }
					}
					scope:host = { add_to_list = characters }
				}
			}
			scripted_animation = {
				triggered_animation = {
					trigger = {
						scope:character.location = { is_sea_province = yes }
					}
					animation = { idle }
				}
				triggered_animation = {
					trigger = { scope:character = { has_trait = impatient } }
					animation = { jockey_gallop }
				}
				animation = jockey_walk
			}
		}
		prelate_guest_arrived = {
			camera = camera_event_left
			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					ordered_attending_character = {
						limit = {
							this != scope:host
							location = scope:activity.activity_location
							pam_is_investiture_clergy_trigger = yes
						}
						order_by = {
							value = 1
							add = {
								value = highest_held_title_tier
								multiply = 15
							}
						}
						add_to_list = characters
					}
				}
			}
			scripted_animation = ecumenical_council_clergy_guest_animation_alt
		}
		secular_guest_arrived = {
			camera = camera_event_right
			effect = {
				if = {
					limit = { scope:host.location = scope:activity.activity_location }
					ordered_attending_character = {
						limit = {
							this != scope:host
							location = scope:activity.activity_location
							NOT = { pam_is_investiture_clergy_trigger = yes }
						}
						order_by = {
							value = 1
							add = {
								value = highest_held_title_tier
								multiply = 15
							}
							add = {
								value = domain_size
								multiply = 3
							}
							add = realm_size
						}
						add_to_list = characters
					}
				}
			}
			# The shared arrival animation is prayer-led and feels wrong on secular lords
			scripted_animation = {
				triggered_animation = {
					trigger = { scope:character = { has_trait = zealous } }
					animation = prayer_standing
				}
				triggered_animation = {
					trigger = {
						scope:character = {
							OR = {
								has_trait = cynical
								has_trait = lazy
							}
						}
					}
					animation = boredom
				}
				animation = personality_content
			}
		}
	}

	###################
	# ON ACTIONS
	###################

	pulse_actions = {
		entries = {
			apa_friendly_discussions
			apa_studying_scripture
			apa_prayer
			apa_communal_prayer
			apa_passive_religious_learning_host
			apa_passive_religious_learning_guest
		}
		chance_of_no_event = 5
	}

	on_start = {
		add_activity_log_entry = {
			key = pam_investiture_council_start_log
			character = scope:host
			score = 10
		}
		# Clear every nominee so an old reminder cannot start a second council
		scope:host.faith = {
			every_faith_character = {
				limit = { has_character_flag = investiture_council_nominated }
				remove_character_flag = investiture_council_nominated
			}
		}
		# Distinct from the timer-expiry announcement (0058)
		# We save host and venue again because the on_start scopes are unreliable inside the event
		scope:host = { save_scope_as = investiture_convene_host }
		scope:activity.activity_location = { save_scope_as = investiture_convene_location }
		pam_investiture_fire_event_for_participant_players_effect = {
			EVENT = pam_the_christian_church.0059
			FAITH = scope:host.faith
		}
		# We wait a month to add guests, because adding during the invite window errors on a pending accept
		scope:host = { trigger_event = { id = pam_investiture_council_events.0092 days = 30 } }
	}

	on_enter_passive_state = {
		# This state fires on arrival and again before debate
		if = {
			limit = {
				scope:activity = { has_current_phase = investiture_council_phase_summons }
				NOT = { this = scope:activity.activity_host }
			}
			trigger_event = pam_investiture_council_events.0001
		}
		else_if = {
			limit = {
				scope:activity = { has_current_phase = investiture_council_phase_debate }
				NOT = { this = scope:activity.activity_host }
			}
			trigger_event = pam_investiture_council_events.0003
		}
	}
