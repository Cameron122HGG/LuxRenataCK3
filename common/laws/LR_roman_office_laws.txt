# Appointment succession for the landless Roman offices (after celestial_ministry_appointment_succession_law)

# Praetorian Prefects
LR_roman_prefect_appointment_succession_law = {
	law_group_type = title_succession_laws
	index = 21

	can_have = {
		government_has_flag = government_is_roman_imperial
	}
	can_pass = {
	}
	can_title_have = {
		LR_is_roman_prefect_title_trigger = yes
	}
	can_realm_have = {
		always = no
	}
	can_remove_from_title = {
		custom_tooltip = {
			text = celestial_ministry_appointment_cannot_remove_desc
			always = no
		}
	}
	succession = {
		order_of_succession = appointment
		appointment_type = LR_roman_civic_governor
	}
	flag = appointment_type_succession
	flag = appointment_with_level_requirement
}
