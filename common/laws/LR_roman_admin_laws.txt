# Roman army command and retirement laws, adapted from the celestial ones (02_admininistrative_laws.txt).
# They keep the celestial law flags (admin_title_troop_cost_reduction*, celestial_retirement_law, celestial_retirement_age_*)
# so the existing troop-cost values, retirement checks and retirement interactions work for Rome too.

@admin_law_cooldown_value = 5

roman_army_liege_law_0 = {
	law_group_type = roman_army_liege_law
	index = 0

	modifier = {
		roman_dominate_government_tax_contribution_add = -0.05
	}

	can_keep = {
		realm_law_use_roman_bureaucracy = yes
		is_independent_ruler = yes
	}

	can_pass = {
		custom_tooltip = {
			text = admin_law_cooldown_desc
			NOT = { has_variable = admin_law_cooldown }
		}
	}

	on_pass = {
		## Set cooldown
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				NOR = { # Rulers should keep any higher levels
					has_realm_law = roman_army_liege_law_1
					has_realm_law = roman_army_liege_law_2
					has_realm_law = roman_army_liege_law_3
				}
			}
			add = 1
		}
	}
}

roman_army_liege_law_1 = {
	law_group_type = roman_army_liege_law
	index = 1

	modifier = {
		roman_dominate_government_vassal_opinion = -5
		men_at_arms_title_cap = 1
	}
	flag = admin_title_troop_cost_reduction

	can_keep = {
		realm_law_use_roman_bureaucracy = yes
		is_independent_ruler = yes
		custom_tooltip = {
			text = have_roman_bureaucracy_1_or_higher
			has_realm_law_flag = authority_level_1
		}
	}

	can_pass = {
		is_independent_ruler = yes
		custom_tooltip = {
			text = admin_law_cooldown_desc
			NOT = { has_variable = admin_law_cooldown }
		}
	}

	pass_cost = {
		prestige = {
			add = admin_law_prestige_cost
		}
	}

	on_pass = {
		## Set cooldown
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				NOR = { # Rulers should keep any higher levels
					has_realm_law = roman_army_liege_law_2
					has_realm_law = roman_army_liege_law_3
				}
				OR = {
					ai_has_cautious_personality = yes
					ai_has_warlike_personality = yes
				}
			}
			add = 2
		}
	}
}

roman_army_vassal_law_0 = {
	law_group_type = roman_army_vassal_law
	index = 0

	modifier = {
		roman_dominate_government_tax_contribution_add = -0.05
		men_at_arms_cap = 2
	}

	can_keep = {
		realm_law_use_roman_bureaucracy = yes
		is_independent_ruler = no
		trigger_if = {
			limit = {
				liege = { is_independent_ruler = yes }
			}
			liege = { has_realm_law = roman_army_liege_law_0 }
		}
		trigger_else = {
			liege = { has_realm_law = roman_army_vassal_law_0 }
		}
	}

	can_pass = {
		trigger_if = {
			limit = {
				liege = {
					OR = {
						has_realm_law = roman_army_liege_law_0
						has_realm_law = roman_army_vassal_law_0
					}
				}
			}
			custom_tooltip = {
				text = imperial_bureaucracy_pending_law_change
				always = no
			}
		}
	}

	on_pass = {
		## Set cooldown
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
	}

	ai_will_do = {
		value = 0
	}
}
