# Realm law of Roman Autocratic (gen_roman_laws.py): the vanilla meritocratic bureaucracy (00_realm_laws.txt), with
# Roman Autocratic's tax bonus and third-century names. Keep the mechanics in step with the vanilla laws.

@imperial_bureaucracy_cooldown_years_1 = 10
@imperial_bureaucracy_cooldown_years_2 = 20
@imperial_bureaucracy_cooldown_years_3 = 30

roman_autocratic_bureaucracy_0 = {
	law_group_type = roman_autocratic_bureaucracy
	index = 0

	modifier = {
		belligerent_opinion = -10
		barons_and_minor_landholders_opinion = 20
		glory_hound_opinion = 10
		parochial_opinion = 10
		courtly_opinion = 5
	}
	flag = uses_roman_autocratic_bureaucracy
	flag = titles_cannot_leave_realm_on_succession # Hardcoded flag
	flag = can_have_tributaries
	flag = authority_level_0
	flag = has_requirement_for_authority_level

	can_keep = {
		realm_law_use_roman_autocratic_bureaucracy = yes
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			liege = { has_realm_law = roman_autocratic_bureaucracy_0 }
		}
	}

	can_pass = {
		trigger_if = {
			limit = {
				should_synchronize_liege_law = yes
				liege = { has_realm_law = roman_autocratic_bureaucracy_0 }
			}
			custom_tooltip = {
				text = imperial_bureaucracy_pending_law_change
				always = no
			}
		}
	}

	on_pass = {
		# Remove modifiers.
		remove_law_related_modifiers_effect = yes
	}
}

roman_autocratic_bureaucracy_1 = {
	law_group_type = roman_autocratic_bureaucracy
	index = 1

	modifier = {
		belligerent_opinion = 5
		barons_and_minor_landholders_opinion = -30
		glory_hound_opinion = -15
		parochial_opinion = -15
		courtly_opinion = -5
		minority_opinion = -10
		roman_autocratic_government_tax_contribution_add = 0.05
	}
	flag = uses_roman_autocratic_bureaucracy
	flag = title_revocation_allowed
	flag = vassal_retraction_allowed
	flag = can_change_partition_succession_laws
	flag = vassal_internal_wars_banned
	flag = diarchs_want_to_subsidise_without_this_flag
	flag = can_have_tributaries
	flag = authority_level_1

	can_keep = {
		realm_law_use_roman_autocratic_bureaucracy = yes
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			liege = { has_realm_law = roman_autocratic_bureaucracy_1 }
		}
	}

	can_pass = {
		trigger_if = {
			limit = {
				has_realm_law = roman_autocratic_bureaucracy_0
				should_synchronize_liege_law = no
			}
			custom_description = {
				subject = root
				text = "has_roman_autocratic_bureaucracy_cooldown"
				NAND = {
					has_variable = imperial_bureaucracy_cooldown
					NOT = {
						culture = { has_innovation = innovation_all_things }
					}
				}
			}
		}
		trigger_if = {
			limit = {
				should_synchronize_liege_law = yes
				liege = { has_realm_law = roman_autocratic_bureaucracy_1 }
			}
			custom_tooltip = {
				text = imperial_bureaucracy_pending_law_change
				always = no
			}
		}
		can_pass_law_ca1_trigger = yes
	}

	pass_cost = {
		prestige = {
			if = {
				limit = {
					has_realm_law = roman_autocratic_bureaucracy_0
				}
				add = increase_crown_authority_prestige_cost
			}
		}
	}

	on_pass = {
		# Increment cooldown breaker tally, if appropriate.
		if = {
			limit = {
				has_realm_law = roman_autocratic_bureaucracy_0
			}
			calculate_authority_cooldown_break_effect = yes
		}
		# Set regular cooldown.
		set_variable = {
			name = imperial_bureaucracy_cooldown
			years = @imperial_bureaucracy_cooldown_years_1
		}
		# Remove modifiers.
		remove_law_related_modifiers_effect = yes
	}

	ai_will_do = {
		if = {
			limit = { has_realm_law = roman_autocratic_bureaucracy_0 }
			value = 1
		}
	}
}
