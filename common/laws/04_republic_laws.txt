### Republic Types

@republic_type_prestige_cost = 2000
@republic_type_treasury_cost = 1000

# Realm size penalty is X% of base cost per county
@republic_type_realm_size_prestige_cost = @[republic_type_prestige_cost * 0.01]
@republic_type_realm_size_treasury_cost = @[republic_type_treasury_cost * 0.01]
@republic_type_realm_size_popular_prestige_cost = @[republic_type_prestige_cost * 0.1]
@republic_type_realm_size_popular_treasury_cost = @[republic_type_treasury_cost * 0.1]

republic_type_communal = {
	law_group_type = republic_type
	index = 1

	pass_phrase =  republic_type_pass_phrase
	confirmation_title = republic_type_pass_phrase
	confirmation_button_text = republic_type_confirmation_button_text

	modifier = {
		prosperity_growth_factor = small_prosperity_growth_gain
		character_capital_popular_opinion_add = 10
		character_capital_county_control_baseline_add = 10
		domicile_build_speed = -0.1
		domicile_build_gold_cost = -0.1
		domicile_monthly_gold_add = 0.25
	}

	flag = uses_republic_type
	flag = can_change_succession_laws

	can_keep = {
		trigger_if = {
			limit = {
				gdp_is_assembly_head_trigger = no
			}
			liege ?= {
				has_realm_law = republic_type_communal
			}
		}
	}

	pass_cost = {
		prestige = {
			add = {
				value = @republic_type_prestige_cost
				desc = BASE
			}
			if = {
				limit = { sub_realm_size > 1 }
				add = {
					value = @republic_type_realm_size_prestige_cost
					multiply = sub_realm_size # Per county within the (sub)realm
					desc = realm_size
				}
			}
		}
		treasury = {
			add = {
				value = @republic_type_treasury_cost
				desc = BASE
			}
			if = {
				limit = { sub_realm_size > 1 }
				add = {
					value = @republic_type_realm_size_treasury_cost
					multiply = sub_realm_size # Per county within the (sub)realm
					desc = realm_size
				}
			}
		}
	}
	
	on_pass = {
		# Remove modifiers.
		remove_law_related_modifiers_effect = yes
	}
	
	requires_approve = { always = no }

	ai_will_do = {
		value = ai_will_do_republic_type_communal_value
	}
}

					value = @republic_type_realm_size_popular_prestige_cost

					desc = realm_size
				}
			}
			# Popular opinion - add/subtract 0.5% of base cost per opinion
			if = {
				limit = {
					capital_county ?= { popular_opinion > 0 }
					gdp_is_assembly_head_trigger = yes
				}
				add = {
					value = @republic_type_prestige_cost
					multiply = {
						value = capital_county.popular_opinion
						divide = 200
					}
					desc = popular_opinion_concept_desc
				}
			}
			else_if = {
				limit = {
					capital_county ?= { popular_opinion < 0 }
					gdp_is_assembly_head_trigger = yes
				}
				subtract = {
					value = @republic_type_prestige_cost
					multiply = {
						value = capital_county.popular_opinion
						divide = 200
					}
					desc = popular_opinion_concept_desc
				}
			}
			# Title tier
			if = {
				limit = {
					highest_held_title_tier >= tier_kingdom
				}
				add = {
					value = @republic_type_prestige_cost
					switch = {
						trigger = primary_title.tier
						tier_hegemony = { multiply = 7 }
						tier_empire = { multiply = 3 }
					}

				}

				}
			}
		}
		treasury = {

				limit = { sub_realm_size > 1 }

					value = @republic_type_realm_size_popular_treasury_cost
					multiply = sub_realm_size # Per county within the (sub)realm

				}
			}
			# Popular opinion - add/subtract 0.5% of base cost per opinion
			if = {
				limit = {
					capital_county ?= { popular_opinion > 0 }
					gdp_is_assembly_head_trigger = yes
				}
				add = {
					value = @republic_type_treasury_cost
					multiply = {

					}
					desc = popular_opinion_concept_desc
				}
			}
			else_if = {

				}
				subtract = {
