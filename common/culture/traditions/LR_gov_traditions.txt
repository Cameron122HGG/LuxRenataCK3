# Lux Renata government traditions: each culture has exactly one, and it decides which group of governments
# (plain -> dynastic -> administrative -> top tier) its rulers use. Seeded in the culture files by heritage.
# Adopting another one replaces it and moves the culture's rulers to the same tier of the new group
# (LR_government_tradition_on_actions.txt). The AI never adopts them on its own.
# A culture left without one falls into Anarchy (lux_tradition_gov_anarchy, at the end of this file) until it picks one.

lux_tradition_gov_classical = {
	category = realm

	layers = {
		0 = steward
		1 = mediterranean
		4 = roman_ceremonies.dds
	}

	parameters = {
		lux_government_group_classical = yes
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			multiply = 2
			if = { # Anarchy: settling on a way of government is the culture's first need
				limit = { has_cultural_tradition = lux_tradition_gov_anarchy }
				multiply = {
					value = 0.25
					desc = LR_tradition_cost_in_anarchy_desc
				}
			}
		}
	}

	ai_will_do = { value = 0 }
}

# Anarchy: a culture without a government tradition. It cannot be picked; it is given to any culture that has no
# government tradition (no heritage match, or its tradition was removed) and goes as soon as one is adopted.
# While it lasts, government traditions cost a quarter as much and ignore the tradition cooldown (00_rules.txt), the
# player culture head is asked every year to settle on one (LR_gov_anarchy_events.txt), and an AI-led culture settles
# on its leading ruler's government after five years (LR_resolve_government_anarchy_effect).
lux_tradition_gov_anarchy = {
	category = realm

	layers = {
		0 = intrigue
		1 = default_todo.dds
		4 = crown.dds
	}

	is_shown = {
		has_cultural_tradition = lux_tradition_gov_anarchy
	}
	can_pick = {
		always = no
	}

	parameters = {
		lux_government_anarchy = yes
	}

	character_modifier = {
		vassal_opinion = -15
		legitimacy_gain_mult = -0.25
		domain_tax_mult = -0.15
		levy_size = -0.15
		monthly_prestige_gain_mult = -0.2
	}

	county_modifier = {
		county_opinion_add = -15
		development_growth_factor = -0.25
	}

	cost = {
		prestige = 0
	}

	ai_will_do = { value = 0 }
}
