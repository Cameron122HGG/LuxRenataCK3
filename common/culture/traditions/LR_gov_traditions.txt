# Lux Renata government traditions: each culture has exactly one, and it decides which group of governments
# (plain -> dynastic -> administrative -> top tier) its rulers use. Seeded in the culture files by heritage.
# Adopting another one replaces it and moves the culture's rulers to the same tier of the new group
# (LR_government_tradition_on_actions.txt). 2026-11-18: a culture can only pick a group it knows, a group much of the
# culture already lives under is cheaper, the AI takes up the group most of it lives under, and each has a small bonus.
# A culture left without one falls into Anarchy (lux_tradition_gov_anarchy, at the end of this file) until it picks one.

lux_tradition_gov_classical = {
	category = realm

	layers = {
		0 = steward
		1 = mediterranean
		4 = roman_ceremonies.dds
	}

	parameters = {
		lux_government_group_classical = yes
	}

	# LR 2026-11-18: a culture can only take up a way of government it knows (its own lands, its head's realm or a neighbour)
	can_pick = {
		custom_tooltip = {
			text = LR_tradition_gov_classical_known_tt
			LR_culture_knows_classical_group_trigger = yes
		}
		# LR 2026-11-18: a full culture may only replace its old government tradition (or Anarchy); replacing another
		# tradition would lose it as well as the old government tradition (LR_drop_other_government_traditions_effect)
		trigger_if = {
			limit = { exists = scope:replacing }
			custom_tooltip = {
				text = LR_tradition_gov_replaces_gov_tt
				OR = {
					scope:replacing = culture_tradition:lux_tradition_gov_classical
					scope:replacing = culture_tradition:lux_tradition_gov_western
					scope:replacing = culture_tradition:lux_tradition_gov_despotic
					scope:replacing = culture_tradition:lux_tradition_gov_indic
					scope:replacing = culture_tradition:lux_tradition_gov_mamlakah
					scope:replacing = culture_tradition:lux_tradition_gov_nilotic
					scope:replacing = culture_tradition:lux_tradition_gov_sudanic
					scope:replacing = culture_tradition:lux_tradition_gov_sinic
					scope:replacing = culture_tradition:lux_tradition_gov_japanese
					scope:replacing = culture_tradition:lux_tradition_gov_barbarian
					scope:replacing = culture_tradition:lux_tradition_gov_anarchy
				}
			}
		}
	}

	# LR 2026-11-18, after Lux Invicta's classical_government (demesne -2, pikemen defence +50%, court +10, cities built with prestige)
	character_modifier = {
		domain_limit = -1
		pikemen_toughness_mult = 0.1
		court_grandeur_baseline_add = 2
		city_holding_build_gold_cost = -0.1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			multiply = 2
			# LR 2026-11-18 realm drift: cheaper when much of the culture already lives in realms of this group
			if = {
				limit = { LR_culture_share_under_classical_value >= 0.5 }
				multiply = {
					value = 0.5
					desc = LR_tradition_cost_realm_drift_desc
				}
			}
			else_if = {
				limit = { LR_culture_share_under_classical_value >= 0.25 }
				multiply = {
					value = 0.75
					desc = LR_tradition_cost_realm_drift_desc
				}
			}
			if = { # Anarchy: settling on a way of government is the culture's first need
				limit = { has_cultural_tradition = lux_tradition_gov_anarchy }
				multiply = {
					value = 0.25
					desc = LR_tradition_cost_in_anarchy_desc
				}
			}
		}
	}

	# LR 2026-11-18: an AI culture takes up the group most of it lives under, if its head's realm is of that group too
	ai_will_do = {
		value = 0
		if = {
			limit = {
				NOT = { has_cultural_tradition = lux_tradition_gov_classical }
				LR_culture_share_under_classical_value >= 0.6
				culture_head ?= { top_liege = { LR_government_in_classical_group_trigger = yes } }
			}
			add = 100
		}
	}
}

# Anarchy: a culture without a government tradition. It cannot be picked; it is given to any culture that has no
# government tradition (no heritage match, or its tradition was removed) and goes as soon as one is adopted.
# While it lasts, government traditions cost a quarter as much and ignore the tradition cooldown (00_rules.txt), the
# player culture head is asked every year to settle on one (LR_gov_anarchy_events.txt), and an AI-led culture settles
# on its leading ruler's government after five years (LR_resolve_government_anarchy_effect).
lux_tradition_gov_anarchy = {
	category = realm

	layers = {
		0 = intrigue
		1 = default_todo.dds
		4 = crown.dds
	}

	is_shown = {
		has_cultural_tradition = lux_tradition_gov_anarchy
	}
	can_pick = {
		always = no
	}

	parameters = {
		lux_government_anarchy = yes
	}

	character_modifier = {
		vassal_opinion = -15
		legitimacy_gain_mult = -0.25
		domain_tax_mult = -0.15
		levy_size = -0.15
		monthly_prestige_gain_mult = -0.2
	}

	county_modifier = {
		popular_opinion_add = -15
		prosperity_growth_factor = -0.25
	}

	cost = {
		prestige = 0
	}

	ai_will_do = { value = 0 }
}
