# Hellenistic Nostalgia (11-20)
# Cultures that hold this tradition will dress in the Hellenistic clothes (Macedonian and Spartan
# armour, Phrygian and Pilos helmets...) once those models are in. The portrait gate is
# lr_hellenistic_portrait_trigger in common/scripted_triggers/LR_hellenistic_portrait_triggers.txt.
# Which cultures start with it: game rule lr_hellenistic_nostalgia_start, given out by
# common/on_action/LR_hellenistic_nostalgia_on_actions.txt.

lux_tradition_hellenistic_nostalgia = {
	category = regional

	layers = {
		0 = learning
		1 = mediterranean
		4 = philosopher.dds
	}

	# LR 2026-11-25: PARKED until the rework - hidden and unpickable, and the game rule that handed it out is gone, so no
	# culture has it and the Hellenistic clothes, helmets and armour (all gated on it) stay dormant. Originals in
	# parked/nostalgia_rollback_2026-11-25/. To restore: put back the two eligible-trigger blocks below.
	is_shown = {
		always = no	# was: lr_hellenistic_nostalgia_eligible_trigger = yes
	}
	can_pick = {
		always = no	# was: lr_hellenistic_nostalgia_eligible_trigger = yes
	}

	character_modifier = {
		monthly_court_grandeur_change_mult = 0.1
		monthly_prestige_gain_mult = 0.05
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = lux_tradition_hellenistic_nostalgia }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
		}
	}

	ai_will_do = {
		value = 25
		if = {
			limit = {
				OR = {
					has_cultural_pillar = heritage_ancient_greek
					has_cultural_pillar = heritage_hellenistic
				}
			}
			add = 75
		}
	}
}
