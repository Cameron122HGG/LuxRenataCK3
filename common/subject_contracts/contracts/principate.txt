# LR 2026-11-19: Principate governors (vassals with principate_government; contract group principate_vassal).
# Replaces the Decayed Principate's administrative themes. Augustus' settlement of 27 BC split the provinces:
#   - Senatorial provinces (provinciae populi Romani): the governor is a senator (proconsul) and sits in the Senate
#     (republic_assembly_composition_roman_senate checks the flag principate_senatorial_province); he sends taxes
#     to Rome and keeps few troops.
#   - Imperial provinces (provinciae Caesaris): the frontier provinces, held by a legate of the Princeps, who holds
#     the legions; more levies and men-at-arms, less tax, and no seat in the Senate.

principate_obligations = {
	obligation_levels = {
		default = {
			levies = 0.2
			tax = 0.4
		}
	}
}

principate_provinces = {
	display_mode = radiobutton
	is_shown = {
		scope:subject.primary_title.tier >= tier_duchy
	}
	obligation_levels = {
		principate_province_senatorial = {
			default = yes
			position = { 0 0 }
			icon = "gfx/interface/icons/theme_administration_types/icon_civilian_administration.dds"

			ai_liege_desire = {
				value = 1
				if = { # The Princeps keeps the frontier for himself
					limit = {
						scope:subject = {
							any_sub_realm_county = {
								any_neighboring_county = {
									holder = { NOT = { top_liege = scope:subject.top_liege } }
								}
							}
						}
					}
					add = -2
				}
			}
			ai_subject_desire = 2 # Senators prefer a seat in the Senate to the toil of the frontier

			score = 1
			color = { 0.55 0.15 0.35 1.0 }

			tax_factor = 0.25
			levies_factor = -0.5

			subject_modifier = {
				prosperity_growth_factor = 0.1
				monthly_prestige_gain_mult = 0.1
				men_at_arms_title_cap = -1
				monthly_merit = 0.5 # LR 2026-11-19: the cursus honorum
			}

			flag = principate_senatorial_province
			flag = obligation_high_taxes
		}
		principate_province_imperial = {
			position = { 1 0 }
			icon = "gfx/interface/icons/theme_administration_types/icon_military_administration.dds"

			is_valid = {
				custom_tooltip = { # LR 2026-11-19: a legate of the Princeps has risen on the cursus honorum
					text = principate_province_imperial_rank_desc
					scope:subject = { merit_level >= 2 }
				}
				custom_tooltip = {
					text = principate_province_imperial_valid_desc
					scope:subject = {
						any_sub_realm_county = {
							any_neighboring_county = {
								holder = {
									NOT = { top_liege = scope:subject.top_liege }
								}
							}
						}
					}
				}
			}

			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							OR = {
								has_trait = education_martial
								martial >= high_skill_rating
							}
						}
					}
					add = 2
				}
				if = {
					limit = {
						scope:liege = { is_at_war = yes }
					}
					add = 1
				}
			}
			ai_subject_desire = 0

			score = 0
			color = { 0.7 0.2 0.2 1.0 }

			tax_factor = -0.4
			levies_factor = 1

			subject_modifier = {
				men_at_arms_title_cap = 2
				men_at_arms_title_limit = 2
				fort_level = 1
				defender_advantage = 4
				levy_reinforcement_rate = 0.25
				monthly_merit = 0.5 # LR 2026-11-19: the cursus honorum
			}

			flag = principate_imperial_province
		}
	}
}
