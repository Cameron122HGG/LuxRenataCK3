			gui_tags = { military }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes

						limit = { scope:liege = { ai_has_cautious_personality = yes } }
						add = 1

				}
			}
			ai_subject_desire = 0
			score = 0

		}
		roman_autocratic_province_praeses = {
			position = { 1 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_metropolitan_administration.dds"
			gui_tags = { civilian }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							OR = {
								has_trait = education_stewardship
								has_trait = education_diplomacy
								stewardship >= high_skill_rating
							}
						}
					}
					add = 2
				}
			}
			ai_subject_desire = 0
			score = 0
			color = { 0.75 0.6 0.2 1.0 }
			tax_factor = 0.15
			levies_factor = -0.3
			enable_title_maa = no
			subject_modifier = {
				monthly_merit_per_liege_tax = 0.9

			}
			flag = roman_autocratic_province_praeses
			flag = meritocratic_civil_appointment
			flag = province_civilian
			flag = roman_autocratic_civil_appointment
			appointment_trait_flag = civilian_province
		}
		roman_autocratic_province_procurator = {
			position = { 2 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_industrial_administration.dds"
			gui_tags = { civilian }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							OR = {
								has_trait = education_stewardship
								stewardship >= very_high_skill_rating
							}
						}
					}
					add = 2
					if = {
						limit = { scope:liege = { ai_has_builder_or_pious_builder_personality = yes } }
						add = 1
					}
				}
			}
			ai_subject_desire = 0
			score = 0
			color = { 0.2 0.2 0.7 1.0 }
			is_valid = {
				custom_tooltip = {
					text = roman_autocratic_province_procurator_amount_desc
					OR = {
						scope:liege = {
							any_vassal = {
								count < roman_autocratic_province_procurator_max_value
								vassal_contract_has_flag = roman_autocratic_province_procurator
							}
						}
						scope:subject = { vassal_contract_has_flag = roman_autocratic_province_procurator }
					}
				}
			}
			tax_factor = 0.35
			levies_factor = -0.3
			subject_opinion = -5
			enable_title_maa = no
			subject_modifier = {
				monthly_merit_per_liege_tax = 1.2

			}

			flag = province_civilian

		}
		roman_autocratic_province_dux = {
			position = { 0 1 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_military_administration.dds"
			gui_tags = { military }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							OR = {
								has_trait = education_martial
								martial >= very_high_skill_rating
							}
							any_sub_realm_county = {
								any_neighboring_county = {
									holder = { top_liege != scope:subject.top_liege }
								}
							}
						}
					}
					add = 2
					if = {
						limit = { scope:liege = { ai_has_warlike_personality = yes } }

					}
				}

			ai_subject_desire = 0
			score = 0
			color = { 0.7 0.2 0.2 1.0 }
			is_valid = {
				custom_tooltip = {
					text = roman_autocratic_province_dux_amount_desc
					OR = {
						scope:liege = {
							any_vassal = {
								count < roman_autocratic_province_dux_max_value
								vassal_contract_has_flag = roman_autocratic_province_dux
							}
						}
						scope:subject = { vassal_contract_has_flag = roman_autocratic_province_dux }
					}
				}
				custom_tooltip = {
					text = roman_autocratic_province_dux_border_desc
					scope:subject = {
						any_sub_realm_county = {
							any_neighboring_county = {
								holder = { top_liege != scope:subject.top_liege }
							}
						}
					}
				}
			}
			levies_factor = 0.5
			subject_modifier = {
				development_growth_factor = -0.2
				monthly_county_control_growth_factor = 0.1
				additional_fort_level = 2

			flag = meritocratic_military_appointment
			flag = province_military
			flag = roman_autocratic_military_appointment
			appointment_trait_flag = military_province

	}
