great_tribal_government_obligations = {
	uses_opinion_of_liege = yes
	obligation_levels = {
		default = {
			levies = {
				scope:vassal = {
					if = {
						limit = {
							is_ai = yes
						}
						value = 0.30
						if = {
							limit = {
								scope:opinion_of_liege > 0
							}
							add =
							{
								value = scope:opinion_of_liege
								multiply = 0.001
							}
						}
						else = {
							add =
							{
								value = scope:opinion_of_liege
								multiply = 0.0005
							}
						}
					}
					else = { # is a player
						value = 0.20
					}
				}
			}
			min_levies = {
				scope:liege = {
					if = {
						limit = {
							has_realm_law = crown_authority_1
						}
						value = 0.10
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_2
						}
						value = 0.20
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_3
						}
						value = 0.30
					}
					else = {
						value = 0.0
					}
				}
			}
			tax = {
				scope:vassal = {
					if = {
						limit = {
							is_ai = yes
						}
						value = 0.10
						if = { 
							limit = {
								scope:opinion_of_liege > 0
							}
							add = {
								value = scope:opinion_of_liege
								multiply = 0.0004
							}
						}
						else = {
							add = {
								value = scope:opinion_of_liege
								multiply = 0.0002
							}
						}
					}
					else = { # is a player
						value = 0.08
					}
				}
			}
			min_tax = {
				scope:liege = {
					if = {
						limit = {
							has_realm_law = crown_authority_1
						}
						value = 0.05
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_2
						}
						value = 0.10
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_3
						}
						value = 0.15
					}
					else_if = {
						value = 0.0
					}
				}
			}
			tax_contribution_postfix = "clan_government_tax_contribution_postfix"
			levies_contribution_postfix = "clan_government_levies_contribution_postfix"
			
			unclamped_contribution_label = "clan_government_unclamped_contribution_label"
			min_contribution_label = "clan_government_min_contribution_label"
		}
	}
}
