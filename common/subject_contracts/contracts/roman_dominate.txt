# Governor obligations of the Roman Dominate (gen_roman_dominate_contracts.py)

# Governor ranks of a province (duchy): civil only, but for a few combined commands. Vicarii (dioceses) and
# prefects are not provincial governors, so only governors whose highest title is a duchy hold a rank

	display_mode = radiobutton
	is_shown = {
		scope:subject.highest_held_title_tier = tier_duchy
	}
	obligation_levels = {
		roman_dominate_province_praeses = {
			default = yes
			position = { 0 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_standard_administration.dds"
			gui_tags = { civilian }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
						}
					}
					add = 1
					if = {
						limit = { scope:liege = { ai_has_cautious_personality = yes } }
						add = 1
					}
				}
			}
			ai_subject_desire = 0
			score = 0
			color = { 0.7 0.7 0.7 1.0 }
			enable_title_maa = no

				monthly_merit_per_liege_tax = 0.8

						scope:subject = {
							is_ai = yes
							OR = {

								diplomacy >= very_high_skill_rating
							}
						}
					}
					add = 2
				}
			}
			ai_subject_desire = 0
			score = 0
			color = { 0.75 0.6 0.2 1.0 }
			is_valid = {
				custom_tooltip = {
					text = liege_is_not_civilian_desc

						trigger_if = {
							limit = {
								this != top_liege
								highest_held_title_tier = tier_duchy
							}
							vassal_contract_has_flag = meritocratic_civil_appointment
						}
					}
				}
				custom_tooltip = {
					text = roman_dominate_province_consularis_amount_desc
					OR = {
						scope:liege = {
							any_vassal = {
								count < roman_dominate_province_consularis_max_value
								vassal_contract_has_flag = roman_dominate_province_consularis
							}
						}
						scope:subject = { vassal_contract_has_flag = roman_dominate_province_consularis }
					}
				}
			}
			tax_factor = 0.1
			subject_opinion = 5
			enable_title_maa = no
			subject_modifier = {
				monthly_merit_per_liege_tax = 1.0
				monthly_prestige = 0.5

			flag = roman_dominate_province_consularis
			flag = meritocratic_civil_appointment
			flag = province_civilian
			flag = roman_dominate_civil_appointment
			appointment_trait_flag = civilian_province
		}
		roman_dominate_province_corrector = {
			position = { 2 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_industrial_administration.dds"
			gui_tags = { civilian }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							OR = {
								has_trait = education_stewardship
								stewardship >= very_high_skill_rating
							}
						}
					}
					add = 2
					if = {
						limit = { scope:liege = { ai_has_builder_or_pious_builder_personality = yes } }
						add = 1
					}
				}
			}
			ai_subject_desire = 0
			score = 0
			color = { 0.2 0.2 0.7 1.0 }
			is_valid = {
				custom_tooltip = {
					text = liege_is_not_civilian_desc
					scope:liege = {
						trigger_if = {
							limit = {
								this != top_liege
								highest_held_title_tier = tier_duchy
							}
							vassal_contract_has_flag = meritocratic_civil_appointment
						}

					text = roman_dominate_province_corrector_amount_desc

							any_vassal = {
								count < roman_dominate_province_corrector_max_value

			flag = meritocratic_civil_appointment
			flag = province_civilian
			flag = roman_dominate_civil_appointment
			appointment_trait_flag = civilian_province
		}
		roman_dominate_province_proconsul = {
			position = { 3 0 }
			icon = "gfx/interface/icons/celestial_administration_types/icon_protectorate_administration.dds"
			gui_tags = { civilian }
			ai_liege_desire = {
				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							prestige_level >= 3
							OR = {
								diplomacy >= very_high_skill_rating

			ai_subject_desire = 0

				value = 0
				if = {
					limit = {
						scope:subject = {
							is_ai = yes
							OR = {
								has_trait = education_martial
								martial >= very_high_skill_rating

								any_neighboring_county = {
									holder = { top_liege != scope:subject.top_liege }

	}
}

# Senatorial grade of the office (hidden; the salary it draws). County: Perfectissimus, duchy (province): Clarissimus,
# kingdom (diocese): Spectabilis, empire (prefecture): Illustris
roman_dominate_salary_rank = {
	display_mode = hidden
	defaults_to_highest_valid_level = yes
	can_be_changed = {
		always = no
	}
	obligation_levels = {
		roman_dominate_salary_rank_none = {
			position = { 0 0 }
			default = yes
			ai_liege_desire = 0
			ai_subject_desire = 0
		}
		roman_dominate_salary_rank_perfectissimus = { # county
			position = { 1 0 }
			parent = roman_dominate_salary_rank_none
			is_valid = {
				scope:subject = {
					highest_held_title_tier = tier_county
					primary_title = {
						is_noble_family_title = no
					}
				}
			}
			ai_liege_desire = 2
			ai_subject_desire = 2
			subject_modifier = {
				monthly_treasury_from_salary_budget_base = 0.3
			}
		}
		roman_dominate_salary_rank_clarissimus = { # duchy
			position = { 2 0 }
			parent = roman_dominate_salary_rank_perfectissimus
			is_valid = {
				scope:subject = {
					highest_held_title_tier = tier_duchy
				}
			}
			ai_liege_desire = 3
			ai_subject_desire = 3
			subject_modifier = {
				monthly_treasury_from_salary_budget_base = 0.75
			}
		}
		roman_dominate_salary_rank_spectabilis = { # kingdom
			position = { 3 0 }
			parent = roman_dominate_salary_rank_clarissimus
			is_valid = {
				scope:subject = {
					highest_held_title_tier = tier_kingdom
				}
			}
			ai_liege_desire = 4
			ai_subject_desire = 4
			subject_modifier = {
				monthly_treasury_from_salary_budget_base = 1.0
			}
		}
		roman_dominate_salary_rank_illustris = { # empire
			position = { 4 0 }
			parent = roman_dominate_salary_rank_spectabilis
			is_valid = {
				scope:subject = {
					highest_held_title_tier = tier_empire
					is_landless_ruler = no
				}
			}
			ai_liege_desire = 5
			ai_subject_desire = 5
			subject_modifier = {
				monthly_treasury_from_salary_budget_base = 2.0
			}
		}
	}
}
