classical_government = {
	government_rules = {
		create_cadet_branches = yes
		rulers_should_have_dynasty = yes
		dynasty_named_realms = yes
		legitimacy = yes
		allow_accolades = yes
	}

	royal_court = any

	fallback = 5

	primary_holding = castle_holding
	valid_holdings = { temple_citadel_holding }
	required_county_holdings = { castle_holding city_holding church_holding }
	primary_heritages = { heritage_ostrogothic heritage_latin heritage_byzantine heritage_hellenistic heritage_indo_hellenic heritage_sino_hellenic heritage_caucasian heritage_vandalic heritage_visigothic heritage_suevic heritage_brythonic heritage_vlach }

	vassal_contract_group = classical_vassal

	ai = {
		use_legends = yes
	}

	# Use flags instead of has_government for moddability if possible (i.e., wherever not visible to the player).
	flags = {
		government_is_feudal
		government_is_classical_group
		government_is_classical
		government_is_settled
		may_elevate_co_monarch
		government_uses_crown_authority
		government_uses_domain_limit
	}

	mechanic_type = feudal
	is_mechanic_type_default = no

	possible_grant_vassal_governments = { theocracy_government ecclesiastical_government }

	# Grant candidate for nomads and herders who want to settle
	grant_vassal_ai_will_do = {
		value = 0
		if = {
			limit = {
				pam_nomad_grant_settled_government_trigger = yes
				faith = {
					NOT = {
						religion = religion:islam_religion
						#has_doctrine_parameter = unreformed
					}
				}
			}
			add = 10
		}
	}

	color = rgb { 238 238 238 }
	realm_mask_offset = { 0.0 0.01 }
	realm_mask_scale = { 1 1 }
}

roman_administrative_government = {
	government_rules = {
		create_cadet_branches = yes
		rulers_should_have_dynasty = yes
		dynasty_named_realms = no
		landless_playable = yes
		legitimacy = yes
		state_faith = yes
		use_as_base_on_landed = yes
		use_as_base_on_rank_up = yes
		allow_as_base_for_baronies = no
		inherit_from_dynastic_government = no
		sticky_government = yes
		house_aspirations = yes
		noble_families = yes
		admin_allows_holding_multiple_primary_tier_titles = yes
		treasury = yes
		treasury_vassal_development = yes
		replace_gold_cost_by_treasury = yes
		allow_accolades = yes
	}

	royal_court = top_liege
	#blocked_subject_courts = { meritocratic_government celestial_government japan_administrative_government administrative_government steppe_admin_government }

	domicile_type = classical_group_estate
	main_administrative_tier = duchy
	min_appointment_tier = county
	minimum_provincial_maa_tier = duchy

	fallback = 6

	primary_holding = castle_holding
	valid_holdings = { city_holding temple_citadel_holding }
	required_county_holdings = { castle_holding city_holding church_holding }
	primary_heritages = { heritage_latin heritage_byzantine heritage_caucasian heritage_vandalic heritage_visigothic heritage_suevic heritage_ostrogothic }

	vassal_contract_group = roman_administrative_vassal

	can_get_government = {
		#The LAAMP Construct Holding decision
		trigger_if = {
			limit = {
				has_variable = {
					name = city_construction_is_temple_citadel
					name = city_construction_is_castle
				}
			}
			always = no
		}
	}

	character_modifier = {
		#levy_size = -0.5
		#feudal_government_tax_contribution_mult = -0.5
		#feudal_government_levy_contribution_mult = -0.5
		#clan_government_tax_contribution_mult = -0.5
		#clan_government_levy_contribution_mult = -0.5
		men_at_arms_cap = -1
		men_at_arms_limit = -1
		knight_limit = -2
		vassal_limit = 100
		accolades = -1
		title_creation_cost_mult = -0.5
		#mercenary_hire_cost_mult = 2
		men_at_arms_maintenance = 0.25

		# Vassals get Treasury income that is this percentage less than their taxes to their liege
		monthly_treasury_from_liege_mult = -0.15
	}

	top_liege_character_modifier = {
		# 85% of vassal income goes to the treasury
		monthly_treasury_from_vassals = 0.85
		#men_at_arms_maintenance = 1.5
		#men_at_arms_recruitment_cost = 1
	}

	# Use flags instead of has_government for moddability if possible (i.e., wherever not visible to the player).
	flags = {
		government_is_administrative
		government_has_influence
		government_has_treasury
		government_has_powerful_families
		government_uses_administrative_house_aspirations
		# These flags only exist for loc and in-game descriptions
		government_has_estate
		government_has_title_men_at_arms
		government_is_settled
		government_uses_domicile_but_not_adventurer
		government_uses_domain_limit
		government_uses_admin_province_obligations
		government_uses_imperial_bureaucracy
		government_uses_salary_budget
		government_uses_military_budget
		
		government_is_classical_group
		government_is_roman_administrative
		government_is_roman_government
		may_elevate_co_monarch
		government_has_county_tier_noble_families
	}

	ai = {
		use_legends = yes
	}

	mechanic_type = administrative
	is_mechanic_type_default = no

	possible_grant_vassal_governments = { theocracy_government ecclesiastical_government }

	color = { 119 38 38 }
	realm_mask_offset = { 0.0 0.01 }
	realm_mask_scale = { 1 1 }
}

	government_rules = {

		clan_government_levy_contribution_mult = -0.5

		# Vassals get Treasury income that is this percentage less than their taxes to their liege

	}

		# 95% of vassal income goes to the treasury
		monthly_treasury_from_vassals = 0.95
	}

	ai_ruler_desired_kingdom_titles = {
		value = -1 # Keep all
		if = {
			limit = {
				highest_held_title_tier >= tier_empire
			}
			add = 2 # Is 1 == keep one kingdom only!
		}
	}
	ai_ruler_desired_empire_titles = {
		value = -1 # Keep all
		if = {
			limit = {
				highest_held_title_tier = tier_hegemony
			}
			add = 1 # Is 0 == Don't keep any Empires - they should all be 'great circuits'
		}
	}

	can_get_government = {
		has_tgp_dlc_trigger = yes
	}

	# Use flags instead of has_government for moddability if possible (i.e., wherever not visible to the player).
	flags = {
		government_is_meritocratic
		# These flags only exist for loc and in-game descriptions
		government_has_merit
		government_has_influence
		government_has_treasury
		government_has_title_men_at_arms
		government_has_powerful_families
		government_has_county_tier_noble_families
		government_has_east_asian_estate
		government_is_settled
		government_uses_domicile_but_not_adventurer
		government_uses_domain_limit
		government_uses_admin_province_obligations
		government_uses_merit_family_aspirations
		has_special_house_aspirations
		government_uses_meritocratic_bureaucracy
		government_uses_salary_budget
		government_uses_military_budget
		
		government_is_roman_autocratic
		government_is_roman_government
		government_is_classical_group
	}

	ai = {
		use_legends = yes
	}

	currency_levels_cap = {
		merit = 5
	}

	possible_grant_vassal_governments = { theocracy_government ecclesiastical_government }
