# Sinic group: Chinese Feudal (plain tier) -> Meritocratic (administrative, All Under Heaven) -> Celestial (top tier,
# All Under Heaven). Meritocratic and Celestial live in 00_government_types.txt and carry government_is_sinic_group.
# Japanese Feudal is the plain tier for Japanese realms without All Under Heaven (with it they use the Japanese group).

	government_rules = {
		create_cadet_branches = yes
		rulers_should_have_dynasty = yes
		dynasty_named_realms = yes
		legitimacy = yes
		allow_accolades = yes
	}

	royal_court = any

	fallback = 5

	primary_holding = castle_holding
	valid_holdings = { temple_citadel_holding }
	required_county_holdings = { castle_holding city_holding church_holding }
	primary_heritages = { heritage_tibetan heritage_chinese heritage_mongolic heritage_qiangic heritage_korean heritage_tocharian heritage_tai }

	vassal_contract_group = feudal_vassal

	ai = {
		use_legends = yes
	}

	# Use flags instead of has_government for moddability if possible (i.e., wherever not visible to the player).
	flags = {
		government_is_feudal
		government_is_sinic_group
		government_is_chinese_feudal
		government_is_sinic_plain
		government_is_settled
		may_elevate_co_monarch
		government_uses_crown_authority
		government_uses_domain_limit
	}

	mechanic_type = feudal
	is_mechanic_type_default = no

	possible_grant_vassal_governments = { theocracy_government ecclesiastical_government }

	# Grant candidate for nomads and herders who want to settle
	grant_vassal_ai_will_do = {
		value = 0
		if = {
			limit = {
				pam_nomad_grant_settled_government_trigger = yes
				faith = {
					NOT = {
						religion = religion:islam_religion
						#has_doctrine_parameter = unreformed
					}
				}
			}
			add = 10
		}
	}

	color = rgb { 17 156 70 }
	realm_mask_offset = { 0.0 0.01 }
	realm_mask_scale = { 1 1 }
}
