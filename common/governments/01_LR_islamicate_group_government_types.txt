iqta_dynastic_government = {
	government_rules = {
		create_cadet_branches = yes
		rulers_should_have_dynasty = yes
		legitimacy = yes
		dynasty_named_realms = yes
		always_use_patronym = yes
		allow_accolades = yes
	}

	royal_court = any

	primary_holding = castle_holding
	primary_heritages = { heritage_arabic heritage_turkic }

	fallback = 2

	preferred_religions = { islam_religion }

	valid_holdings = { temple_citadel_holding }
	required_county_holdings = { castle_holding city_holding church_holding }

	house_unity = clan_house_unity
	tax_slot_type = clan_tax_slot

	vassal_contract_group = clan_vassal

	ai = {
		use_legends = yes
	}

	opinion_of_liege = {
		scope:vassal = {
			if = {
				limit = {
					is_powerful_vassal = yes
					NOT = {
						is_allied_to = scope:liege
					}
				}
				value = clan_powerful_vassal_no_alliance_opinion_penalty_value
			}
		}
	}
	opinion_of_liege_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:vassal = {
						NOT = {
							is_allied_to = scope:liege
						}
						is_powerful_vassal = yes
					}
				}
				desc = "GOVERNMENT_CLAN_NOT_ALLIED_POWERFUL"
			}
		}
	}

	# Use flags instead of has_government for moddability if possible (i.e., wherever not visible to the player).
	flags = {
		government_is_clan
		government_has_tax_slot
		may_appoint_viziers
		government_is_settled
		government_uses_crown_authority
		government_uses_domain_limit
		government_is_iqta
	}

	mechanic_type = clan
	is_mechanic_type_default = yes

	possible_grant_vassal_governments = { theocracy_government }

	# Grant candidate for nomads and herders who want to settle
	grant_vassal_ai_will_do = {
		value = 0
		if = {
			limit = {
				pam_nomad_grant_settled_government_trigger = yes
				faith = { religion = religion:islam_religion }
			}
			add = 10
		}
	}

	color = hsv{ 0.39 0.93 0.54 }
	realm_mask_offset = { 0.0 0.03 }
	realm_mask_scale = { 1 1 }
}
