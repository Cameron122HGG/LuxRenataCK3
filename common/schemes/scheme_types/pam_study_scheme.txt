				}

			add = {
				value = ecclesiastical_domicile_study_scripture_success_value_1
				desc = STUDY_SCRIPTURE_ESTATE_CELLS

			add = {

		modifier = {
			add = scope:owner.learning
			desc = "SCHEME_MY_LEARNING"
		}
		# Faith
		modifier = {
			add = 15
			desc = "STUDY_SCRIPTURE_PURSUIT_OF_KNOWLEDGE"

		}
		modifier = {
			add = 10
			desc = "STUDY_SCRIPTURE_SCHOLASTICISM"
			scope:owner = { pam_study_scripture_scholasticism_trigger = yes }
		}
		# Ecclesiastical estate buildings
		modifier = {
			add = ecclesiastical_domicile_study_scripture_success_value_1
			desc = "STUDY_SCRIPTURE_ESTATE_SCHOOL"
			scope:owner = {
				domicile ?= { has_domicile_parameter = estate_study_scripture_success_1 }
			}
		}
		modifier = {
			add = ecclesiastical_domicile_study_scripture_success_value_2
			desc = "STUDY_SCRIPTURE_ESTATE_SCHOOL"

				domicile ?= { has_domicile_parameter = estate_study_scripture_focus_2 }
			}
		}

	# On Actions
	on_phase_completed = {
		# Save scopes
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		# Stressed AI sets the books aside instead of studying on
		if = {
			limit = {
				scope:owner = {
					is_ai = yes
					stress_level >= 1
				}
			}
			end_scheme = yes
		}
		else = {
			# Let's see if we made progress
			random = {
				chance = scope:scheme.scheme_success_chance
				save_scope_value_as = {
					name = scheme_successful
					value = yes
				}
			}
			# Determine the outcome
			scope:owner = {
				if = {
					limit = { exists = scope:scheme_successful }
					trigger_event = { on_action = study_scripture_success }
				}
				else = {
					trigger_event = { on_action = study_scripture_failure }
				}
			}
			reset_scheme_progress = yes
		}
	}

	## Invalidation
	on_invalidated = {
	}
