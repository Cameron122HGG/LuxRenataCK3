# LR 2026-11-25: GRADUS imperial rise decisions (tools/imperial_rise/build_imperial_rise.py, re-runnable).
# Same shape for every group (see LR_GRADUS_government_system.md, Uniqueness): independent, empire tier, small empire,
# prestige level 3, peace; the group's own rise currency; the realm's governors rise too. The Sinic one is

LR_proclaim_king_of_kings_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_cerimonial_decision.dds"
	}
	desc = LR_proclaim_king_of_kings_decision_desc
	selection_tooltip = LR_proclaim_king_of_kings_decision_tooltip
	confirm_text = LR_proclaim_king_of_kings_decision_confirm
	decision_group_type = realm
	sort_order = 42

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 60
		hegemony = 60
	}

	is_shown = {
		LR_gradus_has_auh_trigger = yes # the imperial ranks need All Under Heaven (merit)
		has_government = persianate_administrative_government
		is_independent_ruler = yes
	}

	is_valid = {
		highest_held_title_tier >= tier_empire
		prestige_level >= 3
		realm_size >= small_empire_size_value
		custom_tooltip = {
			text = LR_proclaim_king_of_kings_decision_houses_tt
			any_powerful_vassal = {
				count = all
				opinion = {
					target = root
					value >= 20
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	cost = {
		prestige = 750
		treasury = 500
	}

	effect = {
		change_government = persianate_imperial_government
		custom_tooltip = LR_imperial_rise_governors_tt
		hidden_effect = {
			LR_raise_realm_administrations_effect = { FROM = persianate_administrative_government TO = persianate_imperial_government }
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 25
		modifier = {
			add = 25
			OR = {
				has_trait = arrogant
				has_trait = ambitious
			}
		}
	}
}
