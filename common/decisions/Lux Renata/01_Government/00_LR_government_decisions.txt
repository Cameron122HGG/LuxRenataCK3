convert_high_tribal_to_civ_group_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	desc = convert_high_tribal_to_civ_group_decision_desc

	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	is_shown = {
		government_has_flag = government_is_high_tribal
		OR = {
			liege = this
			AND = {
				liege != this
				liege = {
					NOR = {
						government_has_flag = government_is_tribal
						government_has_flag = government_is_nomadic
					}
				}
			}
		}
	}

	is_valid = {
		has_realm_law = crown_authority_2
		custom_description = {
			text = faith_is_reformed
			subject = root
			faith = { NOT = { has_doctrine_parameter = unreformed } }
		}
		prestige_level >= 1
		custom_description = { #You need to have 70% of all (non-regional/unique) tribal innovations!
			text = has_all_tribal_innovations
			subject = root.culture
			culture = {
				calc_true_if = {
					amount >= 9
					has_innovation = innovation_motte
					has_innovation = innovation_catapult
					has_innovation = innovation_barracks
					has_innovation = innovation_mustering_grounds
					has_innovation = innovation_bannus
					has_innovation = innovation_quilted_armor
					has_innovation = innovation_gavelkind
					has_innovation = innovation_crop_rotation
					has_innovation = innovation_city_planning
					has_innovation = innovation_casus_belli
					has_innovation = innovation_plenary_assemblies
					has_innovation = innovation_ledger
					has_innovation = innovation_development_01
					has_innovation = innovation_currency_01
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = permafrost_feudal_ban
			NOT = { culture = { has_cultural_parameter = permafrost_cannot_become_feudal } }
		}
	}

	cost = {
		prestige = 1000
	}

	effect = {
		#gain_legitimizing_legend_seed_tooltip_effect = yes		
		if = {
			limit = {
				liege != this
				OR = {
					liege = { government_has_flag = government_is_classical }
					AND = {
						liege = { government_has_flag = government_is_classical }
						faith.religion = { is_in_family = rf_graeco_roman }
					}
				}
			}
			change_government = classical_government
		}
		else_if = {
			limit = {
				liege != this
				OR = {
					liege = { government_has_flag = government_is_despotic }
					AND = {
						liege = { government_has_flag = government_is_despotic }
						faith.religion = { is_in_family = rf_archaic }
					}
				}
			}
			change_government = despotic_government
		}
		else_if = {
			limit = {
				liege != this
				OR = {
					liege = { has_government = iqta_government }
					AND = {
						liege = { has_government = iqta_government }
						religion = religion:islam_religion
					}
				}
			}
			change_government = iqta_government
		}		
		##Independent
		else_if = {
			limit = {
				liege = this
				faith.religion = { is_in_family = rf_graeco_roman }
			}
			change_government = classical_government
		}
		else_if = {
			limit = {
				liege = this
				faith.religion = { is_in_family = rf_archaic }
			}
			change_government = despotic_government
		}
		else_if = {
			limit = {
				liege = this
				faith = { religion_tag = islam_religion }
			}
			change_government = iqta_government
		}		
		else = { change_government = western_feudal_government }
		
		##Convert Vassals
		custom_tooltip = {
			text = LR_high_tribal_vassal_adoption
			if = {
				limit = {
					government_has_flag = government_is_classical
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = classical_government
				}
			}
			else_if = {
				limit = {
					government_has_flag = government_is_despotic
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = despotic_government
				}
			}
			else_if = {
				limit = {
					has_government = iqta_government
				}
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = iqta_government
				}
			}		
			else = {
				every_vassal_or_below = {
					limit = { government_has_flag = government_is_high_tribal }
					change_government = western_feudal_government
				}
			}
		}
		
		save_scope_as = actor
		trigger_event = LR_gov_events.0001
		hidden_effect = {
			every_vassal = {
				trigger_event = {
					id = LR_gov_events.0002
					days = 3
				}
			}
		}
	}

	ai_potential = {
		NOT = { mpo_has_gok_mongol_empire_trigger = yes } #Stop the Mongols from slowing their conquest. 
	}

	ai_will_do = {
		base = 100
	}
}

			}
			desc = convert_to_administrative_decision_confirm
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 36
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 0
	}

	is_shown = {
		OR = {
			has_ep3_dlc_trigger = yes
			has_tgp_dlc_trigger = yes
		}
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_japan_feudal
			government_has_flag = government_is_classical
			government_has_flag = government_is_despotic
			government_has_flag = government_is_western_feudal
			government_has_flag = government_is_iqta
			government_has_flag = government_is_chinese_feudal
			government_has_flag = government_is_indic
		}
		trigger_if = {
			limit = { top_liege != this }
			liege = { government_allows = administrative }
		}
	}

	is_valid = {
		trigger_if = {
			limit = { top_liege != this }
			liege = { government_allows = administrative }
		}
		trigger_else = {
			OR = {
				custom_tooltip = {
					text = HAS_SEAL_OF_INVESTITURE
					any_character_artifact = {
						OR = {
							artifact_type = seal_of_investiture
							artifact_type = seal_of_investiture_court
						}
					}
				}
				trigger_if = {
					limit = {
						culture = { has_cultural_parameter = can_adopt_administrative_government_as_kingdom_tier }
					}
					custom_tooltip = {
						text = king_tier_tt
						primary_title.tier >= tier_kingdom
					}
				}

					AND = { # To force the correct tooltip in loc
						custom_tooltip = {
							text = empire_tier_tt
							primary_title.tier >= tier_empire
						}
						root = { save_temporary_scope_as = root_scope }
						custom_tooltip = {
							text = small_realm_size_tt
							realm_size >= small_empire_size_value
						}
					}
				}
			}
		}
		trigger_if = {
			limit = {
				top_liege = this
				is_ai = no
			}
			prestige_level >= 3
		}
		trigger_if = {
			limit = {
				top_liege = this
				is_ai = no
			}
			custom_tooltip = {
				text = all_powerful_vassals_required_tt
				any_powerful_vassal = {

					}

			}

				culture = { has_cultural_pillar = ethos_courtly }

			}
		}
	}

			has_ep3_dlc_trigger = yes

		}
		is_at_war = no
	}

				}

					}

				}

				}
				else_if = {
					limit = { root.monthly_character_income >= 60 }
					multiply = 2
				}

					limit = { root.monthly_character_income >= 40 }

				}

					limit = { root.monthly_character_income >= 20 }
					multiply = 1.1

						multiply = 2.5
					}

						limit = { root.realm_size >= 150 }

					}
