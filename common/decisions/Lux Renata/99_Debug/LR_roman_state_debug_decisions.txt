# LR 2026-11-25 Roman state debug decisions (tools/roman_consulship/build_roman_state_debug.py)

LR_debug_become_roman_republic_decision = {
	decision_group_type = debug
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	desc = LR_debug_become_roman_republic_decision_desc
	selection_tooltip = LR_debug_become_roman_republic_decision_tooltip
	confirm_text = LR_debug_become_roman_republic_decision_confirm

	is_shown = {
		debug_only = yes
		is_ruler = yes
		is_landed = yes
		NOT = { has_government = roman_republic_government }
	}

	is_valid = {
		LR_gradus_has_gdp_trigger = yes
		government_has_flag = government_is_republic
		custom_tooltip = { text = LR_debug_roman_republic_half_free_tt OR = { NOT = { exists = title:e_western_roman_republic.holder } NOT = { exists = title:e_eastern_roman_republic.holder } } }
		NOT = { exists = title:h_roman_republic.holder }
	}

	effect = {
		if = {
			limit = { NOT = { exists = title:e_western_roman_republic.holder } }
			LR_restore_roman_republic_effect = { HALF = western COUNT = wre }
		}
		else = {
			LR_restore_roman_republic_effect = { HALF = eastern COUNT = ere }
		}
		if = {
			limit = { has_government = republic_government has_realm_law = republic_type_roman }
			change_government = roman_republic_government
		}
		LR_appoint_co_consul_effect = yes
	}

	ai_check_interval = 0
	ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
}
