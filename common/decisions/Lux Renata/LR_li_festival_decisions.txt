# FUTURE (activities rework): consider turning this festival into a CK3 activity (a faith-specific feast with guests,
# invitations and phases) instead of a decision with a short event chain.
LR_hold_pagan_festival_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = religious
	ai_check_interval = 60
	cooldown = { years = 5 }

	desc = LR_hold_pagan_festival_decision_desc
	selection_tooltip = LR_hold_pagan_festival_decision_tooltip

	is_shown = {
		is_ruler = yes
		is_landed = yes
		OR = {
			faith.religion = { is_in_family = rf_pagan }
			faith.religion = { is_in_family = rf_eastern }
			faith.religion = religion:dualism_religion
		}
	}

	is_valid = {
		is_at_war = no
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
	}

	cost = {
		gold = minor_gold_value
	}

	effect = {
		custom_tooltip = LR_hold_pagan_festival_decision_effect_tt
		trigger_event = LR_festival.0101
	}

	ai_potential = { always = yes }
	ai_will_do = {
		base = 25
		modifier = { add = 25 has_trait = zealous }
		modifier = { add = -20 has_trait = cynical }
	}
}

# Sacrifice a prisoner at an Irminsul in your realm (Lux Invicta's sacrifice_irminsul)
LR_sacrifice_at_irminsul_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = religious
	ai_check_interval = 60
	cooldown = { years = 5 }

	desc = LR_sacrifice_at_irminsul_decision_desc
	selection_tooltip = LR_sacrifice_at_irminsul_decision_tooltip

	is_shown = {
		is_ruler = yes
		faith.religion = religion:germanic_religion
		any_realm_county = { has_variable = LR_irminsul }
	}

	is_valid = {
		custom_tooltip = {
			text = LR_sacrifice_at_irminsul_prisoner_tt
			any_prisoner = { NOT = { faith = root.faith } }
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
	}

	effect = {
		random_prisoner = {
			limit = { NOT = { faith = root.faith } }
			save_scope_as = LR_irminsul_victim
		}
		trigger_event = LR_festival.0950
	}

	ai_potential = { always = yes }
	ai_will_do = {
		base = 10
		modifier = { add = 20 has_trait = zealous }
		modifier = { add = -10 has_trait = compassionate }
	}
}
