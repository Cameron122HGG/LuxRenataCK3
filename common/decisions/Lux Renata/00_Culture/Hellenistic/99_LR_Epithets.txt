## Choose a Hellenistic Epithet
LR_choose_epithet = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	sort_order = 40
	decision_group_type = hellenic

	is_shown = {
		OR = {
			culture = { has_cultural_pillar = heritage_hellenistic }
		}
		has_any_nickname = no
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		is_available_healthy_adult = yes
		NOT = { exists = involved_activity }
		is_imprisoned = no
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "CHOOSE_EPITHET"
		show_from_start = yes

		item = { # Nicator
			is_shown = {
				num_wars_won > num_wars_lost
				num_battles_won >= 10
			}
			value = choose_epithet_nicator
			current_description = choose_epithet_nicator_desc
			localization = choose_epithet_nicator
			ai_chance = {
				value = 10

				if = {
					limit = { has_trait = brave }
					multiply = 1.1
				}
				if = {
					limit = { has_trait = wrathful }
					multiply = 1.05
				}
				if = {
					limit = { has_trait = ambitious }
					multiply = 1.05
				}
				if = {
					limit = { has_trait = education_martial }
					multiply = 1.2
				}
				if = {
					limit = { has_trait = strategist }
					multiply = 1.1
				}
			}
		}
		item = { # Soter
			is_shown = {
				OR = {
					has_trait = zealous
					has_trait = holy_warrior
					has_trait = paragon
					has_trait = savior
				}
			}
			value = choose_epithet_soter
			current_description = choose_epithet_soter_desc
			localization = choose_epithet_soter
			ai_chance = { value = 10 }
		}
		item = { # Theos
			value = choose_epithet_theos
			is_shown = {
				OR = {
					has_trait = zealous
					has_trait = arrogant
					has_trait = ambitious
					has_trait = lunatic_1
				}
				NOT = {
					faith = { has_doctrine = doctrine_monotheist }
				}
			}
			current_description = choose_epithet_theos_desc
			localization = choose_epithet_theos
			ai_chance = { value = 10 }
		}
		item = { # Eusebes
			value = choose_epithet_eusebes
			is_shown = {
				OR = {
					has_trait = zealous
					has_trait = humble
					has_trait = ambitious
				}
			}
			current_description = choose_epithet_eusebes_desc
			localization = choose_epithet_eusebes
			ai_chance = { value = 10 }
		}
		item = { # Euergetes
			value = choose_epithet_euergetes
			is_shown = {
				OR = {
					has_trait = generous
					has_trait = arrogant
					has_trait = compassionate
					has_trait = improvident
				}
			}
			current_description = choose_epithet_euergetes_desc
			localization = choose_epithet_euergetes
			ai_chance = { value = 10 }
		}
		item = { # Eleutherios
			value = choose_epithet_eleutherios
			is_shown = {
				OR = {
					has_trait = compassionate
					has_trait = just
					has_trait = peasant_leader
				}
			}
			current_description = choose_epithet_eleutherios_desc
			localization = choose_epithet_eleutherios
			ai_chance = { value = 10 }
		}
		item = { # Epiphanes
			value = choose_epithet_epiphanes
			is_shown = {
				OR = {
					has_trait = arrogant
					prestige_level >= 3
				}
			}
			current_description = choose_epithet_epiphanes_desc
			localization = choose_epithet_epiphanes
			ai_chance = { value = 10 }
		}
		item = { # Eupator
			value = choose_epithet_eupator
			is_shown = {
				OR = {
					has_trait = arrogant
					#var:parent_fame_lvl >= 3
					has_trait = pure_blooded
				}
			}
			current_description = choose_epithet_eupator_desc
			localization = choose_epithet_eupator
			ai_chance = { value = 10 }
		}
		item = { # Philadelphus
			value = choose_epithet_philadelphus
			is_shown = {
				OR = {
					any_sibling = {
						OR = {
							has_relation_friend = root
							has_relation_best_friend = root
							has_relation_lover = root
							has_relation_soulmate = root
							is_spouse_of = root
						}
					}
				}
			}
			current_description = choose_epithet_philadelphus_desc
			localization = choose_epithet_philadelphus
			ai_chance = { value = 10 }
		}
		item = { # Dikaios
			value = choose_epithet_dikaios
			is_shown = {
				OR = {
					has_trait = just
					has_trait = compassionate
				}
				NOR = {
					has_trait = callous
					has_trait = sadistic
					has_trait = arbitrary
				}
			}
			current_description = choose_epithet_dikaios_desc
			localization = choose_epithet_dikaios
			ai_chance = { value = 10 }
		}
		item = { # Ktistes
			value = choose_epithet_ktistes
			is_shown = {
				always = yes #Until I can think of something better.
			}
			current_description = choose_epithet_ktistes_desc
			localization = choose_epithet_ktistes
			ai_chance = { value = 10 }
		}
		#temp removal of frangopoulos and voulgaropoulos until I can figure out a better way to implement them.
	} # end widget

	effect = {
		switch = {
			trigger = yes
			scope:choose_epithet_nicator = {
				give_nickname = nick_nicator
			}
			scope:choose_epithet_soter = {
				give_nickname = nick_soter
			}
			scope:choose_epithet_theos = {
				give_nickname = nick_theos
			}
			scope:choose_epithet_eusebes = {
				give_nickname = nick_eusebes
			}
			scope:choose_epithet_dikaios = {
				give_nickname = nick_dikaios
			}
			scope:choose_epithet_ktistes = {
				give_nickname = nick_ktistes
			}
			scope:choose_epithet_euergetes = {
				give_nickname = nick_euergetes
			}
			scope:choose_epithet_eleutherios = {
				give_nickname = nick_eleutherios
			}
			scope:choose_epithet_epiphanes = {
				give_nickname = nick_epiphanes
			}
			scope:choose_epithet_eupator = {
				give_nickname = nick_eupator
			}
			scope:choose_epithet_philadelphus = {
				give_nickname = nick_philadelphus
			}
			#scope:choose_epithet_frangopoulos = {
			#	give_nickname = nick_frangopoulos
			#}
			#scope:choose_epithet_voulgaropoulos = {
			#	give_nickname = nick_voulgaropoulos
			#}
		}
	} # end effect

	ai_check_interval = 60
	ai_potential = {
		highest_held_title_tier > 3 # Only Kings and Emperors
		is_independent_ruler = yes
	}
}
