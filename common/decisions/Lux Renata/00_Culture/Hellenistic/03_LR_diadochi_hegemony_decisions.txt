# Reform a diadochi empire into its hegemony: completely control two whole strategiai (gen_diadochi_hegemonies.py)

LR_reform_antipatrid_hegemony_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_alexanders_restoration.dds"
	}
	desc = LR_reform_antipatrid_hegemony_decision_desc
	selection_tooltip = LR_reform_antipatrid_hegemony_decision_tooltip
	confirm_text = LR_reform_antipatrid_hegemony_decision_confirm
	decision_group_type = hellenic
	sort_order = 950
	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		has_title = title:e_antipatrid_empire
		NOT = { exists = title:h_antipatrid_hegemony.holder }
		NOT = { has_title = title:h_hellenistic_empire }
	}

	is_valid = {
		prestige_level >= 4
		custom_tooltip = {
			text = LR_reform_diadochi_hegemony_two_strategiai_tt
			calc_true_if = {
				amount >= 2
				completely_controls_region = custom_strategia_europe
				completely_controls_region = custom_strategia_asia
				completely_controls_region = custom_strategia_syria
				completely_controls_region = custom_strategia_persis
				completely_controls_region = custom_strategia_upper_satrapies
				completely_controls_region = custom_strategia_egypt
				completely_controls_region = custom_strategia_pentapotamia
				completely_controls_region = custom_strategia_gangaradia
				completely_controls_region = custom_strategia_ariake
				completely_controls_region = custom_strategia_dakinabades
			}
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		gold = 1500
	}

	effect = {
		custom_tooltip = LR_reform_antipatrid_hegemony_decision_effect_tooltip
		show_as_tooltip = { get_title = title:h_antipatrid_hegemony }
		custom_tooltip = LR_diadochi_megale_basileia_tt # Government changes next day (LR_diadochi_line_government_effect)
		hidden_effect = { LR_reform_diadochi_hegemony_effect = { EMPIRE = e_antipatrid_empire HEGEMONY = h_antipatrid_hegemony } }
		title:h_antipatrid_hegemony = { save_scope_as = founded_title }
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_warlord.0001
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
