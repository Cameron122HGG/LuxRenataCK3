# Successor satrapies: like the Roman diocese restoration, each successor picks one satrapy at a time

LR_restore_antipatrid_satrapies_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	ai_check_interval = 120
	decision_group_type = hellenic

	desc = LR_restore_antipatrid_satrapies_decision_desc
	selection_tooltip = LR_restore_antipatrid_satrapies_decision_tooltip
	confirm_text = LR_restore_antipatrid_satrapies_decision_confirm

	is_shown = {
		is_ruler = yes
		is_landed = yes
		OR = {
			has_title = title:e_antipatrid_empire
			AND = {
				has_title = title:k_antipatrid_kingdom
				NOT = { exists = title:e_antipatrid_empire.holder }
			}
		}
		OR = {
			LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_macedonia }
			LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_greece }
			LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_thrace }
		}
	}

	cost = {
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = 500
			}
		}
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = 500
			}
		}
		prestige = 500
	}

	is_valid = {
		prestige_level >= medium_prestige_level
		custom_description = {
			text = LR_restore_satrapies_decision_trigger
			subject = root
			OR = {
				AND = {
					completely_controls_region = custom_satrapy_macedonia
					LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_macedonia }
				}
				AND = {
					completely_controls_region = custom_satrapy_greece
					LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_greece }
				}
				AND = {
					completely_controls_region = custom_satrapy_thrace
					LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_thrace }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	widget = {
		gui = "decision_view_widget_restore_satrapy"
		controller = decision_option_list_controller
		decision_to_second_step_button = "LR_CHOOSE_SATRAPY_BUTTON"
		show_from_start = yes

		item = {
			value = custom_satrapy_macedonia
			current_description = {
				desc = custom_satrapy_macedonia_desc
			}
			localization = {
				desc = custom_satrapy_macedonia_desc
			}
			is_shown = {
				LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_macedonia }
			}
			is_valid = {
				completely_controls_region = custom_satrapy_macedonia
			}
			ai_chance = {
				value = 10
			}
		}
		item = {
			value = custom_satrapy_greece
			current_description = {
				desc = custom_satrapy_greece_desc
			}
			localization = {
				desc = custom_satrapy_greece_desc
			}
			is_shown = {
				LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_greece }
			}
			is_valid = {
				completely_controls_region = custom_satrapy_greece
			}
			ai_chance = {
				value = 10
			}
		}
		item = {
			value = custom_satrapy_thrace
			current_description = {
				desc = custom_satrapy_thrace_desc
			}
			localization = {
				desc = custom_satrapy_thrace_desc
			}
			is_shown = {
				LR_satrapy_restorable_trigger = { SATRAPY = k_satrapy_thrace }
			}
			is_valid = {
				completely_controls_region = custom_satrapy_thrace
			}
			ai_chance = {
				value = 10
			}
		}
	}

	effect = {
		custom_tooltip = LR_restore_satrapies_decision_effect_tt
		LR_diadochi_ensure_successor_empire_effect = { KINGDOM = k_antipatrid_kingdom EMPIRE = e_antipatrid_empire }
		hidden_effect = {
			if = {
				limit = { scope:custom_satrapy_macedonia = yes }
				LR_restore_satrapy_effect = { SATRAPY = k_satrapy_macedonia EMPIRE = e_antipatrid_empire }

	}

	ai_potential = {
