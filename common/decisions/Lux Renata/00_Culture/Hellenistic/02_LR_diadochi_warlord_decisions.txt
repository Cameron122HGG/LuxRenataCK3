# Formable kingdoms of post-Alexander warlords: each needs the lands of one or two satrapies

LR_found_lysimachid_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	desc = LR_found_lysimachid_decision_desc
	selection_tooltip = LR_found_lysimachid_decision_tooltip
	confirm_text = LR_found_lysimachid_decision_confirm
	decision_group_type = hellenic
	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		NOT = { exists = title:k_lysimachid_kingdom.holder }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_lysimachid_founded
			}
		}
		OR = {
			dynasty = dynasty:dynasty_hellenistic_lysimachus
			has_trait = trait:bloodline_lysimachus
		}
	}

	is_valid = {
		highest_held_title_tier >= tier_duchy
		prestige_level >= 2
		custom_tooltip = { # One diadochi line per ruler (gen_diadochi_lines.py)
			text = LR_not_other_diadochi_line_tt
			NOT = { LR_holds_other_diadochi_line_lysimachid_trigger = yes }
		}
		completely_controls_region = custom_satrapy_thrace
		custom_tooltip = {
			text = LR_found_lysimachid_decision_second_satrapy_tt
			OR = {
				completely_controls_region = custom_satrapy_bithynia
				completely_controls_region = custom_satrapy_ionia
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		gold = 500
		prestige = 500
	}

	effect = {
		custom_tooltip = LR_found_lysimachid_decision_effect_tooltip
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_lysimachid_founded
			}
			LR_create_diadochi_warlord_kingdom_effect = { TITLE = k_lysimachid_kingdom }
			change_government = alexandrian_administrative_government
		}
		title:k_lysimachid_kingdom = { save_scope_as = founded_title }
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_warlord.0001
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

LR_reform_lysimachid_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_alexanders_restoration.dds"
	}
	desc = LR_reform_lysimachid_empire_decision_desc
	selection_tooltip = LR_reform_lysimachid_empire_decision_tooltip
	confirm_text = LR_reform_lysimachid_empire_decision_confirm
	decision_group_type = hellenic
	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		has_title = title:k_lysimachid_kingdom
		NOT = { exists = title:e_lysimachid_empire.holder }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_lysimachid_empire
			}
		}
	}

	is_valid = {
		prestige_level >= 3
		completely_controls_region = custom_satrapy_thrace
		custom_tooltip = {
			text = LR_reform_lysimachid_empire_decision_two_satrapies_tt
			calc_true_if = {
				amount >= 2
				completely_controls_region = custom_satrapy_bithynia
				completely_controls_region = custom_satrapy_ionia
				completely_controls_region = custom_satrapy_lydia
			}
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		gold = 1000
	}

	effect = {
		custom_tooltip = LR_reform_lysimachid_empire_decision_effect_tooltip
		custom_tooltip = LR_diadochi_basileia_tt # Government changes next day (LR_diadochi_line_government_effect)
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_lysimachid_empire
			}
			LR_create_diadochi_warlord_empire_effect = { KINGDOM = k_lysimachid_kingdom EMPIRE = e_lysimachid_empire }
		}
		title:e_lysimachid_empire = { save_scope_as = founded_title }
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_warlord.0001
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

		reference = "gfx/interface/illustrations/decisions/decision_alexanders_restoration.dds"
	}
	desc = LR_reform_eumenid_empire_decision_desc
	selection_tooltip = LR_reform_eumenid_empire_decision_tooltip
	confirm_text = LR_reform_eumenid_empire_decision_confirm
	decision_group_type = hellenic
	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		has_title = title:k_eumenid_kingdom
		NOT = { exists = title:e_eumenid_empire.holder }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_eumenid_empire
			}
		}
	}

	is_valid = {
		prestige_level >= 3
		completely_controls_region = custom_satrapy_cappadocia
		custom_tooltip = {
			text = LR_reform_eumenid_empire_decision_two_satrapies_tt
			calc_true_if = {
				amount >= 2
				completely_controls_region = custom_satrapy_paphlagonia
				completely_controls_region = custom_satrapy_pontus
			}
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		gold = 1000
	}

	effect = {
		custom_tooltip = LR_reform_eumenid_empire_decision_effect_tooltip
		custom_tooltip = LR_diadochi_basileia_tt # Government changes next day (LR_diadochi_line_government_effect)
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_eumenid_empire
			}
			LR_create_diadochi_warlord_empire_effect = { KINGDOM = k_eumenid_kingdom EMPIRE = e_eumenid_empire }
		}
		title:e_eumenid_empire = { save_scope_as = founded_title }
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_warlord.0001
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
