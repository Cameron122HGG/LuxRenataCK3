LR_diadochi_seleukid_kingdom_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
    }
    desc = LR_diadochi_seleukid_kingdom_desc
    selection_tooltip = LR_diadochi_seleukid_kingdom_tooltip
	confirm_text = LR_diadochi_seleukid_kingdom_confirm
    decision_group_type = hellenic
    ai_check_interval = 120
    is_shown = {
        is_ruler = yes
        is_playable_character = yes
		NOT = { exists = title:k_seleukid_kingdom.holder }
		NOT = { has_primary_title = title:e_byzantium }
		NOT = { has_primary_title = title:e_antipatrid_empire }
		NOT = { has_primary_title = title:e_ptolemaic_empire }
		NOT = { dynasty = dynasty:dynasty_hellenistic_antigonid }
		NOT = { dynasty = dynasty:dynasty_hellenistic_antipater }
		NOT = { dynasty = dynasty:dynasty_hellenistic_ptolemaios}
		OR = {
			dynasty = dynasty:dynasty_hellenistic_balasid
			dynasty = dynasty:dynasty_hellenistic_seleukos
			has_trait = trait:bloodline_seleukos
		}
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_lux_seleukid_reborn
            }
        }
		custom_tooltip = {
			text = primary_title_is_kingdom_tooltip
			primary_title.tier = tier_kingdom
		}		
    }
    is_valid = {
        is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
		culture = { has_cultural_pillar = heritage_hellenistic }
		OR = {
		   NOT = { has_title = title:e_byzantium }
		   NOT = { exists = title:e_byzantium.holder }
		}
		OR = {
			dynasty = dynasty:dynasty_hellenistic_balasid
			dynasty = dynasty:dynasty_hellenistic_seleukos
			has_trait = trait:bloodline_seleukos
		}
		OR = {
			faith = { religion_tag = solar_religion }
			faith = { religion_tag = zoroastrianism_religion }
			faith = { religion_tag = hellenism_religion }
		}
		prestige_level >= 3

		completely_controls = title:k_syria
		completely_controls = title:k_jerusalem
		completely_controls = title:k_mesopotamia 
		completely_controls = title:k_nineveh
    }
    cost = {
		gold = 1000
    }
    effect = {
		custom_tooltip = LR_diadochi_seleukid_kingdom_effect_tooltip

		hidden_effect = {
			change_government = alexandrian_administrative_government
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_lux_seleukid_reborn
			}
			set_global_variable = {
				name = flag_lux_seleukid_reborn
				value = root
			}
			create_seleukid_kingdom_scripted_effect = yes
		}
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_kingdoms_events.0001
    }
    ai_potential = {
        always = yes
    }
    ai_will_do = {
        base = 1000000
    }
}

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_alexanders_restoration.dds"
	}
	decision_group_type = hellenic

	desc = LR_create_alexander_hegemony_decision_desc
	selection_tooltip = LR_create_alexander_hegemony_decision_tooltip
	
	sort_order = 1000 # Should be at the very top

	is_shown = {
		highest_held_title_tier = tier_empire
		OR = {
			culture = { has_cultural_pillar = heritage_hellenistic }
			culture = { has_cultural_pillar = heritage_indo_hellenic }
		}
		NOT = { has_global_variable = created_alexanders_hegemony }
		OR = {
			capital_province = { geographical_region = world_alexanders_empire }
			culture = { has_cultural_pillar = heritage_hellenistic }
			culture = { has_cultural_pillar = heritage_indo_hellenic }
		}
	}
	is_valid = {
		is_independent_ruler = yes
		
		custom_tooltip = {
			text = holds_imperial_title_tooltip
			primary_title.tier = tier_empire
		}
		prestige_level >= 4	
		completely_controls_region = world_alexanders_empire
		#calc_true_if = {
		#	amount >= 2
		#	completely_controls_region = world_alexanders_empire_1
		#	completely_controls_region = world_alexanders_empire_2
		#	completely_controls_region = world_alexanders_empire_3
		#}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		save_scope_as = new_hegemon
		show_as_tooltip = {
			get_title = title:h_hellenistic_empire
		}
		add_prestige = 5000
		dynasty ?= { add_dynasty_prestige = 5000 }		
		
		hidden_effect = {
			scope:new_hegemon = {
				get_title = title:h_hellenistic_empire
			}
			if = {
				limit = {  
					title:k_mesopotamia.de_jure_liege.title_capital_county.title_province = { 
						NOT = { geographical_region = world_alexanders_empire } 
					}
				}
				title:k_mesopotamia = { set_de_jure_liege_title = title:e_persia }
			}
			if = {
				limit = {
					title:k_nineveh.de_jure_liege.title_capital_county.title_province = { 
						NOT = { geographical_region = world_alexanders_empire } 
					}
				}
				title:k_nineveh = { set_de_jure_liege_title = title:e_persia }
			}
			every_empire = {
				limit = {
					#tier = tier_empire
					title_capital_county.title_province = { geographical_region = world_alexanders_empire }
				}
				set_de_jure_liege_title = title:h_hellenistic_empire
			}			
		}
		
		root = { save_scope_as = scoped_ruler }
		trigger_event = LR_diadochi_kingdoms_events.0003		
		set_global_variable = {
			name = created_alexanders_hegemony
			value = yes
		}
	}

	#cost = {
	#	gold = 2500
	#}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 24
		hegemony = 0
	}

	ai_potential = {
		culture = { has_cultural_pillar = heritage_hellenistic }
	}

	ai_will_do = {
		base = 100
	}
}

form_bosporan_kingdom_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_medi_estate.dds"
	}
	decision_group_type = major
	sort_order = 50

	is_shown = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_bosporan_kingdom
			}
		}
		culture = {
			OR = {
				has_cultural_pillar = heritage_central_germanic
				has_cultural_pillar = heritage_byzantine
			}
		}
		highest_held_title_tier <= tier_kingdom
		any_held_title = {
			OR = {
				de_jure_liege = title:d_crimea
				de_jure_liege = title:d_azov
				this = title:d_crimea
				this = title:d_azov
			}
		}
	}

	is_valid = {
		completely_controls = title:d_crimea
		completely_controls = title:d_azov
		top_liege = this
		prestige_level >= 3
	}
	
	is_valid_showing_failures_only = {
		is_landed = yes
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_bosporan_kingdom
		}

		house = {
			add_house_modifier = {
				modifier = bp3_bosporan_kingdom_modifier
				years = 100
			}
		}

		custom_tooltip = unlocks_black_sea_naval_conquest

		hidden_effect = {
			title:k_bosporan_kingdom = { set_de_jure_liege_title = title:d_crimea.empire }
		}

			type = created
			save_scope_as = title_change
			add_claim_on_loss = yes
		}

				holder = root
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change

		title:d_crimea = { set_de_jure_liege_title = title:k_bosporan_kingdom }

		adjust_de_jure_effect = {
			TITLE = title:d_bugeac
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_yedisan
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_levedia
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_don_valley
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_kizilyedisan
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_red_levedia
			DE_JURE = title:k_bosporan_kingdom
		}
		adjust_de_jure_effect = {
			TITLE = title:d_tana
			DE_JURE = title:k_bosporan_kingdom
		}
		if = {
			limit = {
				province:5330 = {
					has_holding_type = castle_holding
				}
				province:5277 = {
					has_holding = no
				}
			}
			province:5277 = {
				begin_create_holding = castle_holding
			}
		}		
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = 250
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = 250
			}
		}
		prestige = {
			value = 2000
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	ai_potential = {
		is_ruler = yes
		short_term_gold >= {
			value = major_gold_value
			multiply = 1.5
			round = yes
		}
		prestige >= {
			value = major_prestige_gain
			multiply = 3
			round = yes
		}
	}

	ai_will_do = {
		base = 100
	}
