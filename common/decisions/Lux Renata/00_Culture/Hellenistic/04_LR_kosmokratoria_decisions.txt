# Fusion policy: the Susa Weddings, once per reign

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = hellenic
	sort_order = 899

	desc = LR_kosmokratoria_susa_weddings_decision_desc
	selection_tooltip = LR_kosmokratoria_susa_weddings_decision_tooltip

	is_shown = {
		LR_is_kosmokrator_trigger = yes
		NOT = { has_character_flag = LR_held_susa_weddings }
	}

	is_valid = {
		custom_tooltip = {
			text = LR_kosmokratoria_susa_foreign_vassals_tt
			any_vassal = {
				count >= 3
				culture != root.culture
			}
		}
	}

		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = 500
		prestige = 500
	}

	effect = {
		add_character_flag = LR_held_susa_weddings
		add_character_modifier = {
			modifier = LR_susa_weddings_modifier
			years = 20
		}
		custom_tooltip = LR_kosmokratoria_susa_weddings_effect_tt
		hidden_effect = {
			LR_kosmokratoria_susa_weddings_effect = yes
		}
	}

	ai_check_interval = 60
	ai_potential = {
		LR_is_kosmokrator_trigger = yes
	}
	ai_will_do = {
		base = 50
	}
}
