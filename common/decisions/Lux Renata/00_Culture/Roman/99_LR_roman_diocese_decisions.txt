restore_western_roman_administration_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
    }
    ai_check_interval = 120

    sort_order = 120

    desc = restore_western_roman_administration_decision_desc
    selection_tooltip = restore_western_roman_administration_decision_tooltip
    
    decision_group_type = roman

    is_shown = {
        has_ep3_dlc_trigger = yes
        is_ruler = yes
        is_playable_character = yes
        is_minor_roman_emperor_excluding_byzantium_trigger = yes
		has_title = title:e_spqr
        NOT = { #Can only do it once.
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:western_roman_diocese_restored
            }
        }
    }

    is_valid = {
		completely_controls_region = custom_western_imperial_core
    }			

    effect = {
    		if = {
			limit = {
			has_dlc_feature = all_under_heaven
			}
			change_government = roman_autocratic_government
			save_scope_as = actor
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		every_vassal_or_below = {
			if = {
				limit = {
					has_dlc_feature = all_under_heaven
					OR = {
						government_has_flag = government_is_roman_administrative
						AND = {
							government_has_flag = government_is_roman_government
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_autocratic_government
			}
			else_if = {
				limit = {
					NOT = { has_dlc_feature = all_under_heaven }
					OR = {
						government_has_flag = government_is_classical
						AND = {
							government_has_flag = government_is_roman_government
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_administrative_government
			}
		}
        custom_tooltip = restore_western_diocese_system_effect_tooltip
        custom_tooltip = restore_western_diocese_system_unlock_tooltip
		
		###Beginning Variable for number of provinces restored	
		set_global_variable = {
			name = num_of_provinces_restored_wre
			value = 0
		}

        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:western_roman_diocese_restored
            }
        }

        trigger_event = LR_major_decisions.0007
    }   

    cost = {
        prestige = {
            value = 1000        
        }
    }

    is_valid_showing_failures_only = {
        is_at_war = no
        NOT = { exists = involved_activity }
        is_imprisoned = no
        has_contagious_deadly_disease_trigger = no
    }

    ai_potential = {
        OR = {
			#has_title = title:e_western_roman_empire
            has_title = title:e_spqr
            #has_title = title:h_western_roman_empire
			#has_title = title:h_roman_empire
        }
        
        is_at_war = no
        NOT = { exists = involved_activity }
        is_imprisoned = no
        has_contagious_deadly_disease_trigger = no
    }

    ai_will_do = {
        base = 100
    }
}
