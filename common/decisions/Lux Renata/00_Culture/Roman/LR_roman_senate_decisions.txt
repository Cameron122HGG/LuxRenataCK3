LR_hold_senatus_consultum_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	desc = LR_hold_senatus_consultum_decision_desc
	selection_tooltip = LR_hold_senatus_consultum_decision_tooltip
	confirm_text = LR_hold_senatus_consultum_decision_confirm
	decision_group_type = lr_ts_roman
	sort_order = 148
	cooldown = { years = 5 }
	ai_check_interval = 60

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		OR = {
			AND = {
				government_has_flag = government_is_republic
				any_held_title = { has_variable = LR_roman_senate }
			}
			any_held_title = { has_variable = LR_regional_senate }
		}
	}

	is_valid = {
		custom_tooltip = {
			text = LR_reconvene_senate_authority_tt
			NOT = { has_realm_law = republic_assembly_authority_ceremonial }
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
	}

	cost = {
		prestige = 200
	}

	effect = {
		if = {
			limit = { government_has_flag = government_is_republic }
			primary_title = {
				every_full_assembly_member = {
					limit = { this != root }
					add_opinion = {
						target = root
						modifier = LR_senatus_consultum_opinion
						years = 5
					}
				}
			}
		}
		else = { # a regional Senate under an emperor: the governors of the provinces
			every_vassal_or_below = {
				limit = {
					is_landed = yes
					any_held_title = { LR_roman_senate_province_title_trigger = yes }
				}
				add_opinion = {
					target = root
					modifier = LR_senatus_consultum_opinion
					years = 5
				}
			}
		}
		add_character_modifier = {
			modifier = LR_senatus_consultum_modifier
			years = 5
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 30
	}
}
