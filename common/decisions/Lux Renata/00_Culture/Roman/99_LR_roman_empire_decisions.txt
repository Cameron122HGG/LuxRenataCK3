#its boring right now, no point in choosing the pontic empire over the byzantines, so this is a decision to merge the two empires if both are held by the same person. It also serves as a way to prevent the decision from being used after the fact if the player takes Pontus as a vassal or something.
merge_pontus_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					primary_title = title:e_pontus
				}
				desc = merge_pontus_decision_alt
			}
			desc = merge_pontus_decision
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					primary_title = title:e_pontus
				}
				desc = merge_pontus_decision_desc_pontus
			}
			desc = merge_pontus_decision_desc
		}
		desc = merge_pontus_decision_desc_outro
	}
	decision_group_type = eastern_anarchy
	sort_order = 50

	is_shown = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:merged_pontus
			}
		}
		OR = {
        #culture = { has_cultural_pillar = heritage_byzantine }
        #culture = { has_cultural_pillar = heritage_hellenistic }
        has_title = title:e_pontus
		has_title = title:e_byzantium
		}
		NOR = {
			title:e_pontus = { is_titular = yes }
			title:e_byzantium = { is_titular = yes }
		}
	}

	is_valid = {
		has_title = title:e_pontus
		has_title = title:e_byzantium
		top_liege = this
		prestige_level >= 3
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:merged_pontus
		}

		save_scope_as = rightful_liege

		add_legitimacy_effect = { LEGITIMACY = massive_legitimacy_gain }
		dynasty ?= { add_dynasty_prestige = massive_dynasty_prestige_value }

		if = {
			limit = {
				primary_title = { this = title:e_pontus}
			}
			merge_pontus_effect = {
				PRIMARY = e_pontus
				SECONDARY = e_byzantium
				TEXT = alt
			}
		}
		else = {
			merge_pontus_effect = {
				PRIMARY = e_byzantium
				SECONDARY = e_pontus
				TEXT = original
			}
		}

		trigger_event = major_decisions.3200
		every_vassal_or_below = {
			trigger_event = major_decisions.3200
		}
	}

	cost = {
		prestige = {
			if = {
				limit = {
					is_ai = no
				}
				value = 1000
			}
			else = {
				value = 0
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	ai_potential = {
		is_ruler = yes
	}

	ai_will_do = {
		base = 100
	}
}
#Gallic Emperor Decision for when Romans secure power in the Sacral Kingdom of Gaul over the Franks.
restore_gallic_empire_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_restore_gallic_empire.dds"
	}
	decision_group_type = roman

	desc = restore_gallic_empire_decision_desc
	selection_tooltip = restore_gallic_empire_decision_tooltip

	ai_check_interval = 120

	is_shown = {
		is_ruler = yes
		is_landed_or_landless_administrative = yes
		OR = {
			government_has_flag = government_is_gaul_feudal
			government_has_flag = government_is_gaul_administrative
			#government_has_flag = government_is_roman_autocratic
			#government_has_flag = government_is_administrative
		}
		culture = { has_cultural_pillar = heritage_latin }
		any_realm_county = {
			title_province = { geographical_region = LR_world_europe_gaul }
		  }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:restore_gallic_empire_decision
			}
		}
		highest_held_title_tier <= tier_empire
	}

	is_valid = {
		highest_held_title_tier = tier_empire
		prestige_level >= high_prestige_level
		OR = {
			is_independent_ruler = yes
			AND = {
				highest_held_title_tier = tier_empire
				liege = {
					OR = {
						government_has_flag = government_is_roman_administrative
						government_has_flag = government_is_roman_autocratic
						government_has_flag = government_is_gaul_administrative
					}
				}
			}
		}
		completely_controls_region = LR_world_europe_gaul
		#completely_controls_region = custom_roman_britannia

	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	effect = {
		if = {
			limit = {
			has_dlc_feature = all_under_heaven
			}
			change_government = roman_autocratic_government
			save_scope_as = actor
			#restore_gallic_empire_provinces_viennensis_decision_effect = yes
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		every_vassal_or_below = {
			if = {
				limit = {
					has_dlc_feature = all_under_heaven
					OR = {
						government_has_flag = government_is_gaul_administrative
						AND = {
							government_has_flag = government_is_roman_government
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_autocratic_government
			}
			else_if = {
				limit = {
					NOT = { has_dlc_feature = all_under_heaven }
					OR = {
						government_has_flag = government_is_gaul_feudal
						AND = {
							government_has_flag = government_is_roman_government
							highest_held_title_tier > tier_county
						}
					}
				}
				change_government = roman_autocratic_government
			}
		}
		title:e_gallic_empire = {
			save_scope_as = title_e_gallic_empire
		}
		title:e_gallic_empire = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
			save_scope_as = title_e_gallic_empire
		}
		resolve_title_and_vassal_change = scope:title_change
		if = {
