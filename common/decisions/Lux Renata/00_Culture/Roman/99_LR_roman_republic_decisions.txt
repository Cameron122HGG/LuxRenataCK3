# LR Restore the Roman Republic: Western and Eastern versions (Silk & Silver / 1.21), written 2026-11-17.
# Plan: LR_republics_silk_silver_prep.md (in the Lux Renata 936 workspace).
# Vanilla's gdp_restore_roman_republic_decision is hidden in LR's shadow of
#   common/decisions/dlc_decisions/gdp/gdp_republic_decisions.txt.
#
# Requirements, mirroring Restore the Hegemony of the West / East:
#   - a republic (Republic Tier Law kingdom or higher), independent, not at war, prestige level 4
#   - the half's capital county (Rome / Constantinople) in the realm
#   - the half's core prefecture: Italia (West) / Macedonia (East): each diocese's land controlled, or the diocese held
#     in the realm (LR 11-18: republics cannot create titles); the effect grants the dioceses
# Effect: event LR_roman_republic.0001 / 0002 runs LR_restore_roman_republic_effect.
#
# United Res Publica (LR 11-18): LR_unite_roman_republic_decision, at the end of this file.

LR_restore_western_roman_republic_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	desc = LR_restore_western_roman_republic_decision_desc
	selection_tooltip = LR_restore_western_roman_republic_decision_tooltip
	confirm_text = LR_restore_western_roman_republic_decision_confirm
	decision_group_type = roman
	sort_order = 105

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 60
		empire = 36
		hegemony = 0
	}

	is_shown = {
		LR_roman_republic_restorer_shown_trigger = { HALF = western }
		culture = {
			OR = {
				has_cultural_tradition = tradition_republican_legacy
				has_cultural_pillar = heritage_latin
				any_parent_culture_or_above = { this = culture:roman } # LR 11-18: as vanilla's Restore the Roman Republic
			}
		}
	}

	is_valid = {
		LR_roman_republic_tier_ready_trigger = yes
		prestige_level >= 4
		custom_tooltip = {
			text = LR_restore_roman_republic_rome_in_realm_tt
			LR_roman_restoration_county_in_realm_trigger = { COUNTY = c_roma_urbs }
		}
		custom_tooltip = {
			text = LR_restore_roman_republic_prefecture_italia_tt
			LR_roman_republic_prefecture_italia_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		prestige = 1000
	}

	effect = {
		show_as_tooltip = {
			LR_restore_roman_republic_effect = { HALF = western }
		}
		if = {
			limit = { capital_county != title:c_roma_urbs }
			custom_tooltip = LR_restore_roman_republic_make_rome_capital_tt
		}
		trigger_event = LR_roman_republic.0001

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_western_roman_republic
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# LR 11-18: the united Res Publica. One republic holding both halves forms h_roman_republic (hegemony tier;
# vanilla 1.21 gives its own restored Roman Republic this hegemony, and republic_tier_empire keeps it).
# Effect: event LR_roman_republic.0010 runs LR_unite_roman_republic_effect.
LR_unite_roman_republic_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}
	desc = LR_unite_roman_republic_decision_desc
	selection_tooltip = LR_unite_roman_republic_decision_tooltip
	confirm_text = LR_unite_roman_republic_decision_confirm
	decision_group_type = roman
	sort_order = 106

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 36
		hegemony = 0
	}

	is_shown = {
		has_gdp_dlc_trigger = yes
		is_ruler = yes
		is_playable_character = yes
		government_has_flag = government_is_republic
		OR = {
			has_title = title:e_western_roman_republic
			has_title = title:e_eastern_roman_republic
		}
		NOT = { exists = title:h_roman_republic.holder }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_united_roman_republic
			}
		}
	}

	is_valid = {
		has_title = title:e_western_roman_republic
		has_title = title:e_eastern_roman_republic
		prestige_level >= max_prestige_level
		custom_tooltip = {
			text = LR_restore_roman_republic_rome_in_realm_tt
			LR_roman_restoration_county_in_realm_trigger = { COUNTY = c_roma_urbs }
		}
		custom_tooltip = {
			text = LR_restore_roman_republic_constantinople_in_realm_tt
			LR_roman_restoration_county_in_realm_trigger = { COUNTY = c_byzantion }
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		prestige = 2000
	}

	effect = {
		show_as_tooltip = {
			LR_unite_roman_republic_effect = yes
		}
		custom_tooltip = LR_unite_roman_republic_choose_seat_tt
		trigger_event = LR_roman_republic.0010

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_united_roman_republic
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
