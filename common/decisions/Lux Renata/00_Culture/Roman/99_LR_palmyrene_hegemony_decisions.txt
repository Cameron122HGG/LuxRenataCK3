# Reform the Palmyrene Empire into a hegemony
# Ported from Lux Invicta's create_e_palmyra (CK2): Zenobia's reach over Syria, Egypt and Anatolia. h_palmyrene_empire was
# defined (and used de jure in the 936-1060 history) but nothing created it.
LR_reform_palmyrene_hegemony_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"
	}
	decision_group_type = roman
	ai_check_interval = 120

	desc = LR_reform_palmyrene_hegemony_decision_desc
	selection_tooltip = LR_reform_palmyrene_hegemony_decision_tooltip
	confirm_text = LR_reform_palmyrene_hegemony_decision_confirm

	is_shown = {
		is_ruler = yes
		has_title = title:e_palmyrene_empire
		NOT = { exists = title:h_palmyrene_empire.holder }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_reformed_palmyrene_hegemony
			}
		}
	}

	is_valid = {
		prestige_level >= 4
		completely_controls = title:c_jerusalem
		completely_controls = title:c_alexandria
		completely_controls_region = LR_area_d_paphlagonia # its counties as at 1066 (LR_li_formable_area_regions.txt)
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	cost = {
		gold = 1500
	}

	effect = {
		custom_tooltip = LR_reform_palmyrene_hegemony_decision_effect_tt
		show_as_tooltip = { get_title = title:h_palmyrene_empire }
		hidden_effect = {
			title:e_palmyrene_empire = { set_de_jure_liege_title = title:h_palmyrene_empire }
			title:e_levant = { set_de_jure_liege_title = title:h_palmyrene_empire }
			title:e_egypt = { set_de_jure_liege_title = title:h_palmyrene_empire }
			create_title_and_vassal_change = {
				type = created
				save_scope_as = title_change
				add_claim_on_loss = no
			}
			title:h_palmyrene_empire = {
				change_title_holder = {
					holder = root
					change = scope:title_change
				}
			}
			resolve_title_and_vassal_change = scope:title_change
			set_primary_title_to = title:h_palmyrene_empire
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_reformed_palmyrene_hegemony
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
