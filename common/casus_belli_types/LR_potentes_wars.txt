# LR 2026-11-22 (user): Seize the Stronghold. A house of the Potentes takes by war a county where its Tower House is a House Seat
# (LR_house_banner_target_trigger), when the holder will not sell it (LR_raise_house_banner_decision). Vassal only: on
# victory the house splits and its head holds the county as a vassal of its old holder (LR_potentes_flip_effect).
# Copied from vanilla seize_realm_county_cb (gdp_merchant_wars.txt).

LR_seize_the_stronghold_cb = {
	icon = seize_land
	group = merchant_seize_realm

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no

	allowed_for_character = {
		scope:attacker = {
			LR_tower_house_owner_trigger = yes
			is_landed = no
		}
	}

	allowed_for_character_display_regardless = {
		scope:attacker = {
			has_gdp_dlc_trigger = yes
			is_at_war = no
		}
	}

	allowed_against_character = {
		scope:defender = { is_landed = yes }
	}

	target_titles = claim
	target_title_tier = county
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	ai_can_target_all_titles = { always = yes }

	ai_score_mult = {
		value = viking_conquest_ai_score_value
		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}

	valid_to_start = {
		scope:target = {
			any_county_province = {
				LR_house_banner_target_trigger = { HOUSE = scope:attacker }
			}
		}
	}

	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_county_province = {
					exists = trade_post_district
					trade_post_owner ?= scope:attacker
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	cost = {
		prestige = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			if = {
				limit = { scope:attacker = { has_character_flag = LR_house_banner_refused } }
				add = {
					value = -50
					desc = LR_house_banner_refused_cb_cost_desc
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = LR_seize_the_stronghold_victory_desc
	}

	on_victory = {
		on_victory_aggressive_war_effect = yes
		random_in_list = {
			list = target_titles
			save_scope_as = lr_banner_county
			holder = { save_scope_as = lr_banner_host }
			random_county_province = {
				limit = {
					exists = trade_post_district
					trade_post_owner ?= scope:attacker
				}
				save_scope_as = lr_banner_province
			}
		}
		scope:attacker = {
			if = {
				limit = { exists = scope:lr_banner_province }
				LR_potentes_flip_effect = yes
			}
			add_prestige_experience = minor_prestige_value
		}
	}

	on_white_peace_desc = {
		desc = county_conquest_cb_white_peace_desc
	}

	on_white_peace = {
		on_white_peace_effect = yes
		scope:attacker = { add_prestige = -50 }
	}

	on_defeat_desc = {
		desc = LR_seize_the_stronghold_defeat_desc
	}

	on_defeat = {
		on_defeat_aggressive_war_effect = yes
		scope:defender = {
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}
		# The defender pulls down the tower
		random_in_list = {
			list = target_titles
			save_scope_as = lr_entrench_county
			random_county_province = {
				limit = {
					exists = trade_post_district
					trade_post_owner ?= scope:attacker
				}
				save_scope_as = lr_entrench_province
			}
		}
		scope:attacker = { save_scope_as = lr_entrench_owner }
		if = {
			limit = { exists = scope:lr_entrench_province }
			scope:lr_entrench_county.holder ?= { LR_entrenchment_oust_effect = yes }
		}
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "LR_SEIZE_THE_STRONGHOLD_WAR_NAME"
	war_name_base = "LR_SEIZE_THE_STRONGHOLD_WAR_NAME_BASE"
	cb_name = "LR_SEIZE_THE_STRONGHOLD_CB_NAME"

	interface_priority = 83

	attacker_wargoal_percentage = 0.8

	max_attacker_score_from_battles = 75

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
