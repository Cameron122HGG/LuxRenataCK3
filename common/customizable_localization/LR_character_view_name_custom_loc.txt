# Character window header for rulers whose realm has a dynastic or tiered realm name (lr_dynastic_realm_names).
# The engine's ruler name is "$TIER$ $NAME$ of $TITLE$" with no article, which reads "Basileus X of Basileia of
# Pergamon". No script effect can give a title an article (set_title_prefix is the culture-prefix form, and
# definite_form is static), so the header is rebuilt here: "Basileus X of the Basileia of Pergamon". User, 11-19.
# Patronym cultures keep the engine name (no data function gives the patronym). Used by CHARACTER_VIEW_NAME
# and the character tooltip header (CHARACTER_TT_NAME, CHARACTER_TT_NAME_AGE), all in
# localization/replace/english/LR_dynastic_realm_names_replace_l_english.yml.
LR_character_view_name = {
	type = character

	# Roman imperial style (11-21): "Imperatore Cesare Evangelus Agosto of Roma" in the tooltip header too
	text = {
		trigger = {
			LR_has_imperial_style_trigger = yes
			exists = primary_title
		}
		localization_key = LR_character_view_name_styled
	}
	text = { # empress consort without a title: the name alone
		trigger = { LR_has_imperial_style_trigger = yes }
		localization_key = LR_character_view_name_top_styled
	}
	text = {
		trigger = {
			has_character_flag = LR_dynastic_realm_name_flag
			exists = primary_title
			exists = house
			NOT = { LR_character_name_patronym_trigger = yes }
			LR_character_name_house_first_trigger = yes
		}
		localization_key = LR_character_view_name_realm_house_first
	}
	text = {
		trigger = {
			has_character_flag = LR_dynastic_realm_name_flag
			exists = primary_title
			NOT = { LR_character_name_patronym_trigger = yes }
		}
		localization_key = LR_character_view_name_realm
	}
	text = {
		fallback = yes
		localization_key = LR_character_view_name_default
	}
}

# Two-line character window header (user, 2026-11-21): style of address and name on the top line, the held
# title under it with the age ("Imperator Caesar Marcus IV Augustus" / "of Rome, 21"), since both run long in LR.
# Rulers only; for everyone else LR_character_view_name_top is empty, the GUI hides the top line and the
# bottom line shows CHARACTER_VIEW_NAME as before. Patronym cultures keep the one-line engine name.
# GUI: gui/window_character.gui, vbox "name_health_and_relationship". Loc: LR_dynastic_realm_names_l_english.yml.
LR_character_view_name_top = {
	type = character

	text = {
		trigger = {
			exists = primary_title
			exists = house
			NOT = { LR_character_name_patronym_trigger = yes }
			LR_character_name_house_first_trigger = yes
		}
		localization_key = LR_character_view_name_top_house_first
	}
	# Roman imperial style, with or without a nickname: "Imperatore Cesare Evangelus Agosto" (LR_character_name_styled)
	text = {
		trigger = {
			LR_has_imperial_style_trigger = yes
			exists = primary_title
		}
		localization_key = LR_character_view_name_top_styled
	}
	text = {
		trigger = {
			exists = primary_title
			NOT = { LR_character_name_patronym_trigger = yes }
		}
		localization_key = LR_character_view_name_top_default
	}
	text = {
		fallback = yes
		localization_key = LR_character_view_name_top_none
	}
}

# Roman imperial style name (11-21, user): the plain titled name (CHARACTER_TITLED_FIRST_NAME, which has no
# suffix hook), then the suffix, then the nickname, so the suffix shows with or without a nickname and never twice.
LR_character_name_styled = {
	type = character

	text = {
		trigger = { has_any_nickname = yes }
		localization_key = LR_character_name_styled_nicknamed
	}
	text = {
		fallback = yes
		localization_key = LR_character_name_styled_plain
	}
}

# Bottom line for rulers who have a top line: "of the <realm>" for dynastic/tiered realm names, else "of <title>".
LR_character_view_name_title_line = {
	type = character

	text = {
		trigger = { has_character_flag = LR_dynastic_realm_name_flag }
		localization_key = LR_character_view_name_title_line_realm
	}
	text = {
		fallback = yes
		localization_key = LR_character_view_name_title_line_default
	}
}
