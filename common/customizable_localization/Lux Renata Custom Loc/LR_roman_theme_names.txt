# Roman Administrative province types named after the culture flavor sets (2026-11-05).
# A contract level has no culture of its own, so the name follows the culture of the player's top liege.
# Latin set: Latin heritage (Western Roman, Romano-Gothic, Romano-Hispanic, Romano-Briton) and the Goths.
# Greek set (the Byzantine flavor names) for everyone else.

LR_roman_theme_balanced = {
	type = character
	text = {
		trigger = {
			top_liege.culture ?= {
				OR = { has_cultural_pillar = heritage_latin has_cultural_pillar = heritage_ostrogothic has_cultural_pillar = heritage_visigothic has_cultural_pillar = heritage_vandalic has_cultural_pillar = heritage_suevic has_cultural_pillar = heritage_gothic }
			}
		}
		localization_key = balanced_duchy_western_roman_roman_administrative_land
	}
	text = {
		fallback = yes
		localization_key = balanced_duchy_byzantine_roman_administrative
	}
}
