# Piety level icons by religion, for GUI textures.
# CK3 ignores piety icon groups added to NGameIcons PIETY_GROUPS, so the GUI builds the path itself:
# gfx/interface/icons/modifiers/icon_piety_<LR_piety_icon_group>_<LR_piety_icon_level>.dds
# Keep the vanilla groups here in step with piety_icon_group in common/religion.

LR_piety_icon_group = {
	type = character
	random_valid = no

	text = {
		trigger = {
			OR = {
				faith.religion = religion:roman_religion
				faith.religion = religion:solar_religion
				faith.religion = religion:mithras_religion
			}
		}
		localization_key = LR_piety_icon_group_roman
	}
	text = {
		trigger = { faith.religion = { is_in_family = rf_graeco_roman } }
		localization_key = LR_piety_icon_group_hellenic
	}
	text = {
		trigger = { faith.religion = religion:islam_religion }
		localization_key = LR_piety_icon_group_islam
	}
	text = {
		trigger = { faith.religion = religion:judaism_religion }
		localization_key = LR_piety_icon_group_judaism
	}
	text = {
		trigger = {
			OR = {
				faith.religion = religion:dualism_religion
				faith.religion = religion:zoroastrianism_religion
			}
		}
		localization_key = LR_piety_icon_group_zoroastrian
	}
	text = {
		trigger = { faith.religion = religion:confucianism_religion }
		localization_key = LR_piety_icon_group_confucianism
	}
	text = {
		trigger = { faith.religion = { is_in_family = rf_sinitic } }
		localization_key = LR_piety_icon_group_taoism
	}
	text = {
		trigger = {
			faith.religion = {
				OR = {
					is_in_family = rf_eastern
					is_in_family = rf_graeco_dharma
				}
			}
		}
		localization_key = LR_piety_icon_group_eastern
	}
	text = {
		trigger = { faith.religion = { is_in_family = rf_abrahamic } }
		localization_key = LR_piety_icon_group_christian
	}
	text = {
		fallback = yes
		localization_key = LR_piety_icon_group_pagan
	}
}

	type = character
	random_valid = no

	text = { trigger = { piety_level <= 0 } localization_key = LR_piety_icon_level_0 }
	text = { trigger = { piety_level = 1 } localization_key = LR_piety_icon_level_1 }
	text = { trigger = { piety_level = 2 } localization_key = LR_piety_icon_level_2 }
	text = { trigger = { piety_level = 3 } localization_key = LR_piety_icon_level_3 }
	text = { trigger = { piety_level = 4 } localization_key = LR_piety_icon_level_4 }
	text = { trigger = { piety_level = 5 } localization_key = LR_piety_icon_level_5 }
	text = { trigger = { piety_level = 6 } localization_key = LR_piety_icon_level_6 }
	text = { trigger = { piety_level = 7 } localization_key = LR_piety_icon_level_7 }
	text = { trigger = { piety_level >= 8 } localization_key = LR_piety_icon_level_8 }
}
