# Galatian title set (Reixath Galáthon), gen_galatian_title_set.py, LR_galatian_title_set_proposal.md (user-approved 2026-11-06).
# Names are reconstructed Celtic forms. All titles start with no de jure land; LR_restore_galat_kingdoms_decision lays a
# kingdom out over its counties, and form_reixath_galathon_decision raises the hegemony. Only Brythonic cultures may create them.
# No warlord titles belong to this set (user 2026-11-06).

h_reixath_galathon = { # Reixath Galáthon, the eastern Celtic empire (name from a fan's custom Galatian empire campaign)
	# LR title set gate: also applies to usurping
	can_create = {
		custom_tooltip = {
			text = LR_title_set_galatian_creation_tooltip
			LR_title_set_galatian_native_trigger = yes
		}
	}
	color = { 20 150 70 }
	capital = c_galatia

	definite_form = yes
	can_be_named_after_dynasty = no

	ai_primary_priority = {
		if = {
			limit = { culture = culture:galatian }
			add = 100 # @correct_culture_primary_score
		}
	}
}
