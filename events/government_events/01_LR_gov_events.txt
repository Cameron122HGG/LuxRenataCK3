namespace = LR_gov_events

LR_gov_events.0001 = {
	type = character_event
	title = LR_gov_events.0001.t
	desc = LR_gov_events.0001.desc
	theme = realm
	override_background = { reference = courtyard }
	left_portrait = {
		character = root
		animation = storyteller
	}

	immediate = {
		custom_tooltip = {
			text = LR_gov_events_abandon_high_tribal
		}
		hidden_effect = {
			every_liege_or_above = {
				limit = {
					NOT = { top_liege = this }
				}
				send_interface_message = {
					type = event_stewardship_neutral_with_text
					title = msg_vassal_feudalised
					desc = msg_vassal_feudalised_desc
					right_icon = root
				}
			}
		}
	}

	option = {
		name = LR_gov_events.0001.a
	}
}

	override_background = { reference = courtyard }
	left_portrait = {

		animation = storyteller
	}

	immediate = {
		show_as_tooltip = {
			if = {
				limit = {
					has_government = clan_government
