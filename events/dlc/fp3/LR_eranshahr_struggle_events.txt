##### Crowning of the King of Kings #####
LR_eranshahr.0001 = {
	type = character_event
	window = fullscreen_event
	queue_icon = "gfx/interface/icons/splash_icons/splash_persia.dds"
	title = LR_eranshahr.0001.t
	desc = LR_eranshahr.0001.desc
	theme = court
	override_background = { reference = fp3_fullscreen_strength }
	override_sound = { reference = "event:/DLC/FP2/SFX/UI/fp2_struggle_ui_intro_animate" }

	immediate = {
		save_scope_as = eranshahr_ender
		LR_eranshahr_crowning_effect = yes
		play_music_cue = struggleend_cue
	}

	option = {
		name = LR_eranshahr.0001.a
		clicksound = "event:/DLC/FP3/SFX/UI/Notifications/Toasts/struggle_end_positive"
	}

	after = {
		tgp_silk_road_iranian_intermezzo_ending_effect = yes
		fp3_end_persian_struggle_effect = {
			ENDING = struggle_persia_ending_assertion_decision
			EVENT = LR_eranshahr.0002
			FLAG = flag:LR_eranshahr_crowning_ending
		}
	}
}
