# Manichaean alms
LR_manichaean.0101 = {
	type = character_event
	title = LR_manichaean.0101.t
	desc = LR_manichaean.0101.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		is_landed = yes
		gold >= medium_gold_value
		NOT = { faith.religious_head ?= root }
	}

	weight_multiplier = {
		base = 1
		modifier = { factor = 2 has_trait = generous }
		modifier = { factor = 0.5 has_trait = greedy }
	}

	immediate = {
		faith.religious_head ?= { save_scope_as = archegos }
	}

	option = {
		name = LR_manichaean.0101.a
		remove_short_term_gold = medium_gold_value
		add_piety = medium_piety_value
		if = {
			limit = { exists = scope:archegos }
			custom_tooltip = LR_manichaean.0101.a.tt
			trigger_event = {
				id = LR_manichaean.0102
				days = { 5 17 }
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 50 has_trait = generous }
			modifier = { add = 25 has_trait = zealous }
		}
	}

	option = {
		name = LR_manichaean.0101.b
		ai_chance = {
			base = 50
			modifier = { add = 50 has_trait = greedy }
		}
	}
}

# A courtier wishes to join the Elect
LR_manichaean.0201 = {
	type = character_event
	title = LR_manichaean.0201.t
	desc = LR_manichaean.0201.desc
	theme = faith

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:elect
		animation = worry
	}

	trigger = {
		is_landed = yes
		any_courtier = { LR_manichaean_elect_candidate_trigger = yes }
	}

	immediate = {
		random_courtier = {
			limit = { LR_manichaean_elect_candidate_trigger = yes }
			weight = {
				base = 1
				modifier = { add = 2 has_trait = zealous }
			}
			save_scope_as = elect
		}
	}

	option = {
		name = LR_manichaean.0201.a
		scope:elect = {
			add_trait = LR_manichaean_elect
			add_opinion = {
				target = root
				modifier = LR_manichaean_elect_allowed_opinion
			}
		}
		custom_tooltip = LR_manichaean.0201.a.tt
		random_list = {
			50 = {
				trigger_event = {
					id = LR_manichaean.0202
					years = { 2 4 }
				}
			}
			50 = {
				trigger_event = {
					id = LR_manichaean.0203
					years = { 2 4 }
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = zealous }
		}
	}

	option = {
		name = LR_manichaean.0201.b
		scope:elect = {
			add_opinion = {
				target = root
				modifier = LR_manichaean_elect_refused_opinion
				years = 3
			}
			add_stress = medium_stress_gain
		}
		ai_chance = { base = 70 }
	}
}
