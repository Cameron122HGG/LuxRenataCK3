	type = character_event
	title = LR_vestal.0001.t
	desc = LR_vestal.0001.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_content
	}

	trigger = { has_trait = LR_vestalis }

	immediate = {
		remove_trait = LR_vestalis
		add_trait = LR_retired_vestalis

	option = {
		name = LR_vestal.0001.a
		add_piety = 100

			limit = { has_title = title:d_vestal_college }
			custom_tooltip = LR_vestal_college_passes_tt
			hidden_effect = { LR_pass_vestal_college_effect = yes }
		}
	}
}
