# Ported from Lux Invicta events/celtic_changelings.txt (ancrel.211-244), 2026-11-06.
# Condensed: one event per step on the parent, instead of LI's selection/AI relay pairs.
#
# Child flags (all hidden):
#	LR_changeling_marked	the child has been picked for the chain (never picked again)
#	LR_changeling			the child really is a changeling (LI changeling_child)
#	LR_changeling_fake		the child is only thought to be one (LI changeling_child_fake)
#	LR_changeling_strange_1/2/3, LR_changeling_kills_1/2/3, LR_changeling_trade	chain steps done
#
# Start: LR_changeling_start_pulse (random_yearly_playable_pulse), on a Celtic pagan parent of a child aged 0-6.
# Steps: LR_changeling_chain_pulse (random_yearly_playable_pulse), on the parent the child lives with.
# End: the druid's trade with the Sidhe (LR_changeling.0030-0035), or the changeling's coming of age (LR_changeling.0040).

namespace = LR_changeling

### Druidic warning

# Starter (hidden): pick the child; a druid at court warns about it, otherwise the Sidhe take it unseen
LR_changeling.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		LR_is_celtic_pagan_trigger = yes
		NOT = { has_character_flag = LR_changeling_parent_cooldown }
		any_child = {
			age <= 6
			host = root
			NOT = { has_character_flag = LR_changeling_marked }
		}
	}

	immediate = {
		add_character_flag = { flag = LR_changeling_parent_cooldown years = 20 }
		random_child = {
			limit = {
				age <= 6
				host = root
				NOT = { has_character_flag = LR_changeling_marked }
			}
			save_scope_as = changeling_child
			add_character_flag = LR_changeling_marked
		}
		if = {
			limit = {
				any_courtier = {
					LR_is_druid_trigger = yes
					NOT = { this = scope:changeling_child }
				}
			}
			random_courtier = {
				limit = {
					LR_is_druid_trigger = yes
					NOT = { this = scope:changeling_child }
				}
				save_scope_as = changeling_druid
			}
			trigger_event = LR_changeling.0002
		}
		else = {
			# Child is stolen by the Sidhe (LI ancrel.215)
			scope:changeling_child = {
				random_list = {
					50 = { add_character_flag = LR_changeling }
					50 = { add_character_flag = LR_changeling_fake }
				}
			}
		}
	}
}

# Druid warns about child (LI ancrel.213)
LR_changeling.0002 = {
	type = character_event
	title = LR_changeling.0002.t
	desc = LR_changeling.0002.desc
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:changeling_druid
		animation = personality_zealous
	}
	lower_left_portrait = scope:changeling_child

	trigger = {
		scope:changeling_child = { is_alive = yes }
		scope:changeling_druid = { is_alive = yes }
	}

	# Dismiss the druid: the Sidhe take the child
	option = {
		name = LR_changeling.0002.a
		scope:changeling_druid = {
			add_opinion = { target = root modifier = LR_changeling_warning_dismissed_opinion years = 1 }
		}
		hidden_effect = {
			scope:changeling_child = {
				random_list = {
					50 = { add_character_flag = LR_changeling }
					50 = { add_character_flag = LR_changeling_fake }
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = cynical }
		}
	}

	# Try to appease the gods: offerings may keep the Sidhe away
	option = {
		name = LR_changeling.0002.b
		remove_short_term_gold = minor_gold_value
		add_piety = minor_piety_value
		custom_tooltip = LR_changeling.0002.b.tt
		hidden_effect = {
			scope:changeling_child = {
				random_list = {
					35 = { }
					35 = { add_character_flag = LR_changeling }
					30 = { add_character_flag = LR_changeling_fake }
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = zealous }
		}
	}
}

# Druid offers to save child (LI ancrel.233)
LR_changeling.0030 = {
	type = character_event
	title = LR_changeling.0030.t
	desc = LR_changeling.0030.desc
	theme = faith
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:changeling_druid
		animation = worry
	}
	lower_left_portrait = scope:changeling_child

	trigger = {
		any_child = {
			OR = {
				has_character_flag = LR_changeling
				has_character_flag = LR_changeling_fake
			}
			host = root
			age >= 3
			age < 16
			NOT = { has_character_flag = LR_changeling_trade }
		}
		any_courtier = {
			LR_is_druid_trigger = yes
			NOR = {
				has_character_flag = LR_changeling
				has_character_flag = LR_changeling_fake
			}
		}
	}

	immediate = {
		random_child = {
			limit = {
				OR = {
					has_character_flag = LR_changeling
					has_character_flag = LR_changeling_fake
				}
				host = root
				age >= 3
				age < 16
				NOT = { has_character_flag = LR_changeling_trade }
			}
			save_scope_as = changeling_child
			add_character_flag = LR_changeling_trade
		}
		random_courtier = {
			limit = {
				LR_is_druid_trigger = yes
				NOR = {
					has_character_flag = LR_changeling
					has_character_flag = LR_changeling_fake
				}
			}
			save_scope_as = changeling_druid
		}
	}

	# Perform the ritual
	option = {
		name = LR_changeling.0030.a
		custom_tooltip = LR_changeling_trade_approved_tt
		trigger_event = { id = LR_changeling.0031 days = 7 }
		ai_chance = {
			base = 60
			modifier = { add = 30 has_trait = zealous }
		}
	}

	# Refuse
	option = {
		name = LR_changeling.0030.b
		custom_tooltip = LR_changeling_trade_denied_tt
		scope:changeling_druid = {
			add_opinion = { target = root modifier = LR_changeling_trade_refused_opinion years = 2 }
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = cynical }
			modifier = { add = 20 has_trait = compassionate }
		}
	}
}
