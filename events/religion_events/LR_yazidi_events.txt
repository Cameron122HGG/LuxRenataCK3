# Dispatcher: root is the ruler holding Tawusgeran. Rolls a blessing for the ruler
# and for every adult vassal of the same faith (Lux Invicta rolled for each attendee).
LR_yazidi.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		trigger_event = LR_yazidi.0006
		every_vassal = {
			limit = {
				is_adult = yes
				is_imprisoned = no
				faith = root.faith
			}
			trigger_event = LR_yazidi.0006
		}
	}
}

# Per-character roll (Lux Invicta weights 10/10/10/5, plus a chance of nothing).
LR_yazidi.0006 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		random_list = {
			35 = {}
			10 = {
				trigger = {
					NOT = { has_trait = zealous }
					NOT = { has_trait = cynical }
				}
				trigger_event = LR_yazidi.0002
			}
			10 = {
				trigger_event = LR_yazidi.0003
			}
			10 = {
				trigger = {
					NOT = { has_trait = possessed_1 }
					NOT = { has_trait = possessed_genetic }
				}
				trigger_event = LR_yazidi.0004
			}
			5 = {
				trigger = {
					OR = {
						has_trait = ill
						has_trait = pneumonic
						has_trait = great_pox
						has_trait = early_great_pox
						has_trait = lovers_pox
						has_trait = leper
						has_trait = consumption
						has_trait = cancer
						has_trait = typhus
						has_trait = bubonic_plague
						has_trait = smallpox
						has_trait = measles
						has_trait = dysentery
						has_trait = ergotism
						has_trait = gout_ridden
						has_trait = wounded_1
						has_trait = wounded_2
						has_trait = wounded_3
					}
				}
				trigger_event = LR_yazidi.0005
			}
		}
	}
}

# Bandits (luxyazd.10169)
LR_yazidi.0111 = {
	type = character_event
	title = LR_yazidi.0111.t
	desc = LR_yazidi.0111.desc
	theme = faith
	left_portrait = {
		character = root
		animation = fear
	}

	option = { # Fight them!
		name = LR_yazidi.0111.a
		random_list = {
			50 = {
				trigger_event = LR_yazidi.0112
			}
			30 = {
				if = {
					limit = { NOT = { has_trait = craven } }
					add_trait = brave
				}
				trigger_event = LR_yazidi.0112
			}
			20 = {
				increase_wounds_effect = { REASON = battle }
				trigger_event = { id = LR_yazidi.0130 days = { 60 120 } }
			}
		}
		ai_chance = {
			base = 5
			modifier = { add = 20 has_trait = brave }
			modifier = { factor = 0 has_trait = craven }
		}
	}
	option = { # Run away!
		name = LR_yazidi.0111.b
		random_list = {
			70 = {}
			30 = {
				if = {
					limit = { NOT = { has_trait = brave } }
					add_trait = craven
				}
			}
		}
		trigger_event = LR_yazidi.0113
		ai_chance = {
			base = 5
			modifier = { factor = 0 has_trait = brave }
			modifier = { add = 20 has_trait = craven }
		}
	}
}
