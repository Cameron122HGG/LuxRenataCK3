	desc = LR_mysteries.0010.desc
	theme = faith
	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		is_adult = yes

		is_incapable = no
		LR_is_graeco_roman_pagan_trigger = yes
		NOT = { LR_is_mystery_initiate_trigger = yes }
		NOT = { has_character_flag = LR_refused_orpheus }
		NOR = {
			has_trait = intellect_bad
			has_trait = inbred

	weight_multiplier = {
		base = 1

		modifier = { factor = 1.5 has_trait = zealous }
		modifier = { factor = 0.5 has_trait = cynical }
	}

	option = {
		name = LR_mysteries.0010.a
		trigger_event = { id = LR_mysteries.0011 days = 3 }
		ai_chance = { base = 25 }
	}
	option = {
		name = LR_mysteries.0010.b
		add_character_flag = LR_refused_orpheus
		ai_chance = { base = 50 }
	}
}
