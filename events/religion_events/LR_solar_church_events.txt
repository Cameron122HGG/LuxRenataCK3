# 0020 - The Eclipse: take a stance on the schism (to Solar rulers in the Solar Church)
LR_solar_church.0020 = {
	type = character_event
	title = LR_solar_church.0020.t
	desc = LR_solar_church.0020.desc
	theme = faith
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = title:d_pontifex_solis_invicti.holder }
		character = title:d_pontifex_solis_invicti.holder
		animation = personality_zealous
	}
	lower_right_portrait = {
		trigger = { exists = title:k_helios_autokrator.holder }
		character = title:k_helios_autokrator.holder
	}

	# With the West: Sol Invictus and the priests of Rome
	option = {
		name = LR_solar_church.0020.a
		set_variable = { name = sc_schism_stance value = flag:west }
		custom_tooltip = LR_sc_stance_west_tt
		if = {
			limit = { LR_is_western_solar_trigger = yes }
			add_piety = 100
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				LR_is_western_solar_trigger = yes
			}
		}
	}

	# With the East: Helios Autokrator, the incarnate Sun
	option = {
		name = LR_solar_church.0020.b
		set_variable = { name = sc_schism_stance value = flag:east }
		custom_tooltip = LR_sc_stance_east_tt
		if = {
			limit = { LR_is_eastern_solar_trigger = yes }
			add_piety = 100
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				LR_is_eastern_solar_trigger = yes
			}
		}
	}

	# One Sun: no schism
	option = {
		name = LR_solar_church.0020.c
		set_variable = { name = sc_schism_stance value = flag:no_schism }
		custom_tooltip = LR_sc_stance_no_schism_tt
		add_prestige = 100
		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_zeal = -0.5
				ai_compassion = 0.5
			}
		}
	}
}
