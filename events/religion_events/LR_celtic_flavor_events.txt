# The Afanc
LR_celtic_flavor.0002 = {
	type = character_event
	title = LR_celtic_flavor.0002.t
	desc = LR_celtic_flavor.0002.desc
	theme = faith

	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		LR_is_celtic_pagan_trigger = yes
		is_travelling = no
		is_adult = yes
		is_imprisoned = no
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 2.5
			has_trait = zealous
		}
		modifier = {
			factor = 0.5
			has_trait = cynical
		}
		modifier = {
			factor = 3
			has_trait = paranoid
		}
		modifier = {
			factor = 1.67
			has_trait = brave
		}
		modifier = {
			factor = 2
			has_trait = arrogant
		}
		modifier = {
			factor = 0.55
			has_trait = content
		}
		modifier = {
			factor = 2
			has_any_illness = yes
		}
	}

	option = { # Stay and fight
		name = LR_celtic_flavor.0002.a
		add_piety = minor_piety_value
		add_prestige = minor_prestige_value
		random_list = {
			20 = {
				if = {
					limit = { has_trait = craven }
					remove_trait = craven
				}
				else = {
					add_trait = brave
				}
			}
			20 = {
				increase_wounds_effect = { REASON = wild_animal }
			}
			5 = {
				trigger = {
					NOT = { has_trait = maimed }
				}
				apply_maimed_trait_and_modifier_effect = yes
			}
			5 = {
				death = {
					death_reason = death_wild_animal
				}
			}
			50 = {
				add_prestige = minor_prestige_value
			}
		}
	}

	option = { # Run away!
		name = LR_celtic_flavor.0002.b
		add_piety = minor_piety_value
		add_prestige = -25
		if = {
			limit = { has_trait = brave }
			remove_trait = brave
		}
	}
}

# Character sick, rumour of Celts on St Kilda
LR_celtic_flavor.0005 = {
	type = character_event
	title = LR_celtic_flavor.0005.t
	desc = LR_celtic_flavor.0005.desc
	theme = faith

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		is_adult = yes
		age >= 16
		is_imprisoned = no
		is_travelling = no
		NOT = { LR_is_celtic_pagan_trigger = yes }
		NOT = { has_character_flag = LR_st_kilda_journey }
		culture = {
			OR = {
				has_cultural_pillar = heritage_goidelic
				has_cultural_pillar = heritage_brythonic
			}
		}
		OR = {
			has_trait = ill
			has_trait = pneumonic
			has_trait = bubonic_plague
			has_trait = smallpox
			has_trait = consumption
			has_trait = typhus
			has_trait = dysentery
			has_trait = measles
			has_trait = ergotism
			has_trait = cancer
			has_trait = leper
			has_trait = great_pox
			has_trait = early_great_pox
		}
		# Holds the Hebrides (St Kilda lies off the Outer Hebrides), or has them in the realm
		OR = {
			title:c_outer_hebrides.holder ?= {
				OR = {
					this = root
					target_is_liege_or_above = root
				}
			}
			title:c_inner_hebrides.holder ?= {
				OR = {
					this = root
					target_is_liege_or_above = root
				}
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 0.5
			has_trait = cynical
		}
		modifier = {
			factor = 0.5
			has_trait = zealous
		}
	}

	immediate = {
		add_character_flag = {
			flag = LR_st_kilda_journey
			years = 20
		}
	}

	option = { # Maybe these Celts can cure me
		name = LR_celtic_flavor.0005.a
		custom_tooltip = LR_celtic_flavor_travel_island_tt
		trigger_event = {
			id = LR_celtic_flavor.0006
			days = 5
		}
		ai_chance = {
			base = 50
		}
	}

	option = { # Don't these barbarians also collect the heads of their enemies?
		name = LR_celtic_flavor.0005.b
		custom_tooltip = LR_celtic_flavor_ignore_rumors_tt
		ai_chance = {
			base = 50
		}
	}
}

# Y Mab Darogan: the first Brythonic ruler to gain e_britannia or h_western_celtic_empire
LR_celtic_flavor.0010 = {
	type = character_event
	title = LR_celtic_flavor.0010.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { LR_is_celtic_pagan_trigger = yes }
				desc = LR_celtic_flavor.0010.desc_pagan
			}
			desc = LR_celtic_flavor.0010.desc
		}
	}
	theme = crown

	left_portrait = {
		character = root
		animation = happiness
	}

	trigger = {
		NOT = { has_global_variable = LR_first_brythonic_emperor }
		culture = { has_cultural_pillar = heritage_brythonic }
		OR = {
			has_title = title:e_britannia
			has_title = title:h_western_celtic_empire
		}
	}

	immediate = {
		set_global_variable = {
			name = LR_first_brythonic_emperor
			value = root
		}
		save_scope_as = celtic_emperor
		give_nickname = nick_pendragon
		add_prestige = major_prestige_value
		add_piety = medium_piety_value
		every_player = {
			limit = {
				NOT = { this = root }
			}
			trigger_event = LR_celtic_flavor.0011
		}
	}

	option = {
		name = LR_celtic_flavor.0010.a
	}
}
