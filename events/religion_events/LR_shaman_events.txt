# Ruler has no Court Shaman (shaman.35)
#################################################
LR_shaman.0001 = {
	type = character_event
	title = LR_shaman.0001.t
	desc = LR_shaman.0001.desc
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		is_ai = no
		LR_is_shamanic_trigger = yes
		highest_held_title_tier >= tier_county
		is_imprisoned = no
		NOT = { has_trait = LR_shaman }
		NOT = { has_character_flag = LR_shaman_reminder_cooldown }
		NOT = { has_character_flag = LR_shaman_invite_pending }
		NOT = { employs_court_position = LR_court_shaman_court_position }
	}

	immediate = {
		add_character_flag = {
			flag = LR_shaman_reminder_cooldown
			years = 5
		}
	}

	# There's a shaman at court already
	option = {
		name = LR_shaman.0001.a
		trigger = {
			can_employ_court_position_type = LR_court_shaman_court_position
			any_court_position_candidate = {
				court_position_type = LR_court_shaman_court_position
				has_trait = LR_shaman
			}
		}
		random_court_position_candidate = {
			court_position_type = LR_court_shaman_court_position
			limit = { has_trait = LR_shaman }
			save_scope_as = new_shaman
		}
		appoint_court_position = {
			recipient = scope:new_shaman
			court_position = LR_court_shaman_court_position
		}
	}

	# I must find a Shaman, and soon!
	option = {
		name = LR_shaman.0001.b
		trigger = { piety >= minor_piety_value }
		add_piety = {
			value = minor_piety_value
			multiply = -1
		}
		add_character_flag = {
			flag = LR_shaman_invite_pending
			years = 1
		}
		custom_tooltip = LR_shaman.0001.b.tt
		trigger_event = {
			id = LR_shaman.0002
			days = { 30 60 }
		}
	}

	# Not now
	option = {
		name = LR_shaman.0001.c
	}
}

LR_shaman.0002 = {
	type = character_event
	title = LR_shaman.0002.t
	desc = LR_shaman.0002.desc
	theme = faith
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:new_shaman
		animation = ecstasy
	}

	immediate = {
		remove_character_flag = LR_shaman_invite_pending
		create_character = {
			location = root.capital_province
			age = { 25 60 }
			gender_female_chance = 30
			culture = root.culture
			faith = root.faith
			trait = LR_shaman
			trait = education_learning_2
			learning = { 6 18 }
			save_scope_as = new_shaman
		}
		add_courtier = scope:new_shaman
		scope:new_shaman = {
			add_opinion = {
				target = root
				modifier = LR_shaman_patron_opinion
			}
		}
	}

	# Welcome to my court, as my Court Shaman
	option = {
		name = LR_shaman.0002.a
		trigger = {
			can_employ_court_position_type = LR_court_shaman_court_position
		}
		appoint_court_position = {
			recipient = scope:new_shaman
			court_position = LR_court_shaman_court_position
		}
	}

	# Welcome to my court.
	option = {
		name = LR_shaman.0002.b
	}
}

# The Shaman Returns (shaman.5); the advice is good or bad, a more learned Shaman reads the spirits better
LR_shaman.0105 = {
	type = character_event
	title = LR_shaman.0105.t
	desc = LR_shaman.0105.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_content
	}
	right_portrait = {
		trigger = { NOT = { scope:shaman = root } }
		character = scope:shaman
		animation = idle
	}

	option = {
		name = LR_shaman.0105.a
		if = {
			limit = { scope:shaman_topic = flag:age }
			trigger_event = {
				id = LR_shaman.0112
				days = 2
			}
		}
		else = {
			random_list = {
				40 = {
					modifier = {
						add = 10
						scope:shaman = { learning >= 8 }
					}
					modifier = {
						add = 15
						scope:shaman = { learning >= 16 }
					}
					modifier = {
						add = 10
						scope:shaman_topic = flag:health
						health >= 6
					}
					modifier = {
						add = -10
						scope:shaman_topic = flag:health
						health < 3
					}
					trigger_event = {
						id = LR_shaman.0110
						days = 2
					}
				}
				60 = {
					modifier = {
						add = -10
						scope:shaman = { learning >= 8 }
					}
					modifier = {
						add = -15
						scope:shaman = { learning >= 16 }
					}
					trigger_event = {
						id = LR_shaman.0111
						days = 2
					}
				}
			}
		}
	}
}
