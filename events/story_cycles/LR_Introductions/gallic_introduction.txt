namespace = french_introduction
french_introduction.0001 = {
    scope = none
    hidden = yes

    immediate = {
        every_player = {
            if = {
                limit = {
                    OR = {
                        has_title = title:k_sacral_king_gaul_throne
                        has_title = title:e_sacral_kingdom_gaul
					    capital_county.duchy = title:d_valois
                    }
                }
                trigger_event = french_introduction.0002
            }
            if = {
                limit = {
                    OR = {
                        has_title = title:k_belgica
					    capital_county.kingdom = title:k_belgica
                    }
                }
                trigger_event = french_introduction.0003
            }
            if = {
                limit = {
                    OR = {
                        has_title = title:k_lotharingia
					    capital_county.kingdom = title:k_lotharingia
                    }
                }
                trigger_event = french_introduction.0004
            }
            if = {
                limit = {
					    capital_county.kingdom = title:k_neustria
                }
                trigger_event = french_introduction.0005
            }
            if = {
                limit = {
					    capital_county.kingdom = title:k_normandy
                }
                trigger_event = french_introduction.0006
            }
            if = {
                limit = {
					    capital_county.kingdom = title:k_brittany
                }
                trigger_event = french_introduction.0007
            }
        }
    }
}
