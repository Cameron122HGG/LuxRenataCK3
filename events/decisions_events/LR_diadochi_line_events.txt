# Hidden: destroy downgraded titles, then choose a line if root holds more than one
LR_diadochi_line.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		LR_holds_diadochi_line_title_trigger = yes
	}

	immediate = {
		LR_destroy_downgraded_diadochi_titles_effect = yes
		if = {
			limit = { LR_holds_rival_diadochi_lines_trigger = yes }
			trigger_event = LR_diadochi_line.0002
		}
		else = {
			LR_diadochi_line_government_effect = yes
		}
	}
}
