		title:k_diocese_hibernia = {
			change_title_holder = {

			}
			if = {
				limit = {
					root.primary_title.tier = tier_empire
				}
				set_de_jure_liege_title = root.primary_title
			}
			else = {
		title:e_britannia = {
			set_de_jure_liege_title = root.primary_title
		}
			}
		}

			title:k_diocese_hibernia = {
			    set_de_jure_liege_title = title:e_praetorian_prefecture_gallia
		    }
		}
	}

LR_roman_diocese_event.0050 = {
	title = LR_roman_diocese_event.0050.t
	desc = LR_roman_diocese_event.0050.desc
	scope = character
	theme = diplomacy
	override_background = {
		reference = throne_room
	}
	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {
	}

	option = {
		name = LR_roman_diocese_event.0050.a
		#custom_tooltip = split_the_empire.custom.tooltip.1
		#custom_tooltip = split_the_empire.custom.tooltip.2
		
		#hidden_effect = { TFE_split_empire_effect = yes }

		ai_chance = { base = 10 }
	}
	option = {
		name = LR_roman_diocese_event.0050.b
		custom_tooltip = split_the_empire.custom.tooltip.3
		add_character_flag = split_empire_upon_death
		ai_chance = { base = 0 }
	}
	after = {		
		every_player = {
			limit = {
				#is_ai = no
				NOT = { this = scope:split_emperor }
				OR = {					
					is_within_diplo_range = { CHARACTER = scope:split_emperor }
					any_held_title = {
						tier = tier_county
						any_county_province = {
							OR = {
								geographical_region = world_europe
								geographical_region = world_africa_north
								geographical_region = world_africa_sahara
								geographical_region = world_middle_east
								geographical_region = world_middle_east_persia
								geographical_region = world_asia_minor
								geographical_region = world_steppe_east
							}		
						}
					}
					any_liege_or_above = {
						OR = {
							has_title = title:h_roman_empire
							has_title = title:e_byzantium
							has_title = title:h_eastern_roman_empire
							has_title = title:e_spqr
							has_title = title:h_western_roman_empire
						}
					}
				}
			}
			trigger_event = LR_roman_diocese_event.0051
		}
	}
}
