					limit = {
						main_rite = { is_rite_enabled = yes }
						NOT = { has_saint_character = root }

					add_saint = { character = root }
				}
			}
			if = {
				limit = {
					NOT = {
						title:c_jerusalem.holder.faith = {
							has_saint_character = character:saint_0001
						}
					}
				}
				title:c_jerusalem.holder.faith = {
					add_saint = { character = character:saint_0001 }
				}
			}
			title:c_jerusalem.holder = {
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001 #Assumption of Mary, avoid some random blasphemy against our beloved mother

			if = {
				limit = {
					title:c_jerusalem.holder.faith = {
						NOT = { religion = religion:christianity_religion }
					}
				}
				title:c_jerusalem.holder.faith = {
